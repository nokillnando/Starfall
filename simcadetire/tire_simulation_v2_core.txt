            if not cfg.AllowCheckVersion then return end

            local githubURL = "https://raw.githubusercontent.com/nokillnando/Starfall/main/simcadetire/tire_simulation_v2_core.txt"
            local localVersion = Version
            local maxRetries = 10

            local function doVersionCheck()
                if not http.canRequest() then
                    maxRetries = maxRetries - 1
                    if maxRetries <= 0 then return end
                    timer.simple(0.5, doVersionCheck)
                    return
                end

                http.get(githubURL,
                    function(data)
                        if not (data and data ~= "") then
                            print(W, "Tire Sim V2 version check failed - no data received")
                            return
                        end

                        local versionStart = data:find('Version = "')
                        if not versionStart then
                            print(W, "Tire Sim V2 version check failed - version string not found")
                            return
                        end

                        versionStart = versionStart + 11
                        local versionEnd = data:find('"', versionStart)
                        if not versionEnd or versionEnd <= versionStart then
                            print(W, "Tire Sim V2 version check failed - could not parse remote version")
                            return
                        end

                        local remoteVersion = data:sub(versionStart, versionEnd - 1)
                        local remoteParts = string.explode(".", remoteVersion)
                        local localParts = string.explode(".", localVersion)

                        local remoteMajor = tonumber(remoteParts[1]) or 0
                        local remoteMinor = tonumber(remoteParts[2]) or 0
                        local localMajor = tonumber(localParts[1]) or 0
                        local localMinor = tonumber(localParts[2]) or 0

                        if remoteMajor > localMajor or (remoteMajor == localMajor and remoteMinor > localMinor) then
                            print(W, "Check config file for updated default variables")
                            print(W, "Tire Sim V2 Update available! Local: ", G, localVersion, W, " | Remote: ", G, remoteVersion)
                        else
                            print(W, "Local Tire Sim V2 is up to date! Version: ", G, localVersion)
                        end
                    end,
                    function(err)
                        print(W, "Tire Sim V2 version check failed - could not connect to remote: ", R, err)
                    end
                )
            end

            timer.simple(1, doVersionCheck)
        end

    local function initServer(config)
        local Config = config
        Config.RightAxis = Config.ForwardAxis:cross(Config.UpAxis):getNormalized()
        
        local frontRatioNormalized = Config.WeightRatioFront / math.max(0.01, Config.WeightRatioFront + Config.WeightRatioRear)
        local massRatio = 1.0

        local InternalProperties = Config.InternalProperties or {}
        
        -- ============================================================================
        -- PACEJKA MAGIC FORMULA & PROPTIRE LONGITUDINAL CONSTANTS
        -- ============================================================================
        local U = 0.0254        -- Meters per Source inch
        local V0 = 40           -- PropTire low-speed slip denominator floor (in/s)
        local V_SLOW = 10       -- PropTire threshold to enter resting rubber state (in/s)
        local V_FAST = 20       -- PropTire threshold to enter active gmod_ice tire model (in/s)
        local V_HOLD = 2        -- PropTire locked-wheel threshold (in/s)
        local V_NOISE = 0.3     -- PropTire resting velocity noise floor (in/s)
        local RELAX = 0.5       -- PropTire impulse relaxation factor
        local HELD_SMOOTH = 0.5 -- PropTire brake-hold smoothing factor
        local WORLD_DOWN = Vector(0, 0, -1)
        local LOCAL_AXES = { Vector(1, 0, 0), Vector(0, 1, 0), Vector(0, 0, 1) }

        local function DetectWheelAxleAndRadius(wheelEntity, defaultRadius, vehicleRight)
            local mins, maxs = wheelEntity:obbMins(), wheelEntity:obbMaxs()
            local size = maxs - mins
            local s = { size.x, size.y, size.z }
            local axis = 1
            if math.abs(s[1] - s[2]) < 0.1 and math.abs(s[2] - s[3]) < 0.1 then
                -- Spherical / cubic OBB fallback: pick local axis most aligned with vehicleRight
                local bestDot = -1
                for i = 1, 3 do
                    local worldAx = wheelEntity:localToWorldVector(LOCAL_AXES[i])
                    local d = math.abs(worldAx:dot(vehicleRight))
                    if d > bestDot then
                        bestDot = d
                        axis = i
                    end
                end
            else
                for i = 2, 3 do
                    if s[i] < s[axis] then axis = i end
                end
            end
            local maxR = 0
            for i = 1, 3 do
                if i ~= axis then maxR = math.max(maxR, s[i] / 2) end
            end
            -- Use Config.TireRadius when set (> 0), falling back to wheel OBB radius detection
            local radius = (defaultRadius and defaultRadius > 0) and defaultRadius or ((maxR > 1) and math.clamp(maxR, 1, 200) or 16)
            return LOCAL_AXES[axis], radius
        end

        local function EvaluateMagicFormula(B, C, D, E, x)
            if (x ~= x) or (math.abs(x) > 1e6) then return 0 end
            local bx = B * x
            local atan_bx = math.atan(bx)
            return D * math.sin(C * math.atan(bx - E * (bx - atan_bx)))
        end

        local function isSafeNumber(n)
            return (n == n) and (n > -1e8) and (n < 1e8)
        end

        local function isSafeVector(v)
            if not v then return false end
            local x, y, z = v.x, v.y, v.z
            if (x ~= x) or (y ~= y) or (z ~= z) then return false end
            if (x > 1e8) or (x < -1e8) or (y > 1e8) or (y < -1e8) or (z > 1e8) or (z < -1e8) then return false end
            return true
        end

        local function safeApplyForceCenter(physObj, forceVec, maxMag)
            if not (physObj and physObj:isValid()) then return end
            if not isSafeVector(forceVec) then return end
            local lenSqr = forceVec:getLengthSqr()
            if lenSqr < 0.0001 then return end
            if maxMag and lenSqr > (maxMag * maxMag) then
                forceVec = forceVec:getNormalized() * maxMag
            end
            physObj:applyForceCenter(forceVec)
        end

        local function safeApplyForceOffset(physObj, forceVec, posVec, maxMag)
            if not (physObj and physObj:isValid()) then return end
            if not (isSafeVector(forceVec) and isSafeVector(posVec)) then return end
            local lenSqr = forceVec:getLengthSqr()
            if lenSqr < 0.0001 then return end
            if maxMag and lenSqr > (maxMag * maxMag) then
                forceVec = forceVec:getNormalized() * maxMag
            end
            physObj:applyForceOffset(forceVec, posVec)
        end

            local base = wire.ports.Base
            return (base and base:isValid()) and base or nil

        local VehicleFilter = {}
        local CheckedEntities = {}
        local Wheelbase = 0
        local TrackWidth = 0
        local TotalRealMass = 0

        local skipTicks = 0
        local lastAppliedForces = {}
        local cpuLoadPct = 0

        local function computeCleanUpVelocity(basePhys, vehicleUp)
            local baseMass = math.max(1, basePhys:getMass())
            local sysMass = baseMass
            local upMom = basePhys:getVelocity():dot(vehicleUp) * baseMass
            for _, wName in ipairs(OrderedWheelNames) do
                local wd = Wheels[wName]
                if wd and wd.PhysObj and wd.PhysObj:isValid() then
                    local wm = math.max(1, wd.PhysObj:getMass())
                    sysMass = sysMass + wm
                    upMom = upMom + wd.PhysObj:getVelocity():dot(vehicleUp) * wm
                end
            end
            return upMom / sysMass
        end

        local function traceFiltered(startPos, endPos)
            local st = trace.line(startPos, endPos, VehicleFilter)
            if st.Hit and not st.HitWorld and isValid(st.Entity) then
                local p = st.Entity:getParent()
                if isValid(p) and CheckedEntities[p] then
                    CheckedEntities[st.Entity] = true
                    VehicleFilter[#VehicleFilter + 1] = st.Entity
                    st = trace.line(startPos, endPos, VehicleFilter)
                end
            end
            return st
        end

        local function formatMass(value)
            return string.format("%.1f", value or 0)
        end

            Active = false,

            Seen = {},
            Result = {}

        local function performUnifiedVehicleScan(Base, isInitial)
            if not isValid(Base) then return 0, 0, 0 end
            local previousMass = TotalRealMass
            local previousFilterCount = lastReportedFilterCount or 0

            if isInitial then
                local fullTree = { Base }
                local seen = { [Base] = true }
                local queue = { Base }

                local extraStatic = wire.ports.ExtraFilterStatic
                if extraStatic and #extraStatic > 0 then
                    for _, ent in ipairs(extraStatic) do
                        if isValid(ent) and not seen[ent] then
                            seen[ent] = true
                            fullTree[#fullTree+1] = ent
                            queue[#queue+1] = ent
                        end
                    end
                end

                local extraSemi = wire.ports.ExtraFilterSemiDynamic
                if extraSemi and #extraSemi > 0 then
                    for _, ent in ipairs(extraSemi) do
                        if isValid(ent) and not seen[ent] then
                            seen[ent] = true
                            fullTree[#fullTree+1] = ent
                            queue[#queue+1] = ent
                        end
                    end
                end

                while #queue > 0 do
                    local current = table.remove(queue, 1)
                    if isValid(current) then
                        local parent = current:getParent()
                        if isValid(parent) and not seen[parent] then
                            seen[parent] = true
                            fullTree[#fullTree+1] = parent
                            queue[#queue+1] = parent
                        end
                        local okChildren, children = pcall(function() return current:getChildren() end)
                        if okChildren and type(children) == "table" then
                            for _, child in ipairs(children) do
                                if isValid(child) and not seen[child] then
                                    seen[child] = true
                                    fullTree[#fullTree+1] = child
                                    queue[#queue+1] = child
                                end
                            end
                        end
                        local constraints = constraint.getTable(current)
                        if constraints then
                            for i = 1, #constraints do
                                local c = constraints[i]
                                local other = (c.Ent1 == current) and c.Ent2 or c.Ent1
                                if isValid(other) and not seen[other] then
                                    seen[other] = true
                                    fullTree[#fullTree+1] = other
                                    queue[#queue+1] = other
                                end
                            end
                        end
                    end
                end

                VehicleFilter = {}
                CheckedEntities = {}
                local currentMass = 0
                local massChecked = {}

                for i = 1, #fullTree do
                    local ent = fullTree[i]
                    if isValid(ent) then
                        if not CheckedEntities[ent] then
                            CheckedEntities[ent] = true
                            VehicleFilter[#VehicleFilter+1] = ent
                        end
                        if not massChecked[ent] then
                            massChecked[ent] = true
                            if not isValid(ent:getParent()) then
                                local phys = ent:getPhysicsObject()
                                if isValid(phys) then
                                    currentMass = currentMass + phys:getMass()
                                end
                            end
                        end
                    end
                end

                if currentMass > ((previousMass and previousMass > 0) and previousMass + 10000 or Config.TotalVehicleMass + 10000) then
                    currentMass = (previousMass and previousMass > 0) and previousMass or Config.TotalVehicleMass
                end
                
                TotalRealMass = currentMass
                local filterCount = #VehicleFilter
                if Config.AllowPrintFilter then
                    print("[TireSimV2] Vehicle filter initialized with " .. filterCount .. " entities.")
                end
                if Config.AllowPrintMassRefresh then
                    print(
                        Config.W,
                        "[TireSimV2] Total Physical Mass detected: ",
                        Config.G,
                        formatMass(currentMass) .. "kg",
                        Config.W,
                        " (Config Virtual: ",
                        Config.G,
                        formatMass(Config.TotalVehicleMass) .. "kg",
                        Config.W,
                        ")"
                    )
                end
                lastReportedFilterCount = filterCount

                return filterCount, currentMass, previousMass
            else
                if TireScan.Active then return previousFilterCount, previousMass, previousMass end

                TireScan.Active = true
                TireScan.Queue = { Base }
                TireScan.Seen = { [Base] = true }
                TireScan.Result = {}

                local extraStatic = wire.ports.ExtraFilterStatic
                if extraStatic and #extraStatic > 0 then
                    for _, ent in ipairs(extraStatic) do
                        if isValid(ent) and not TireScan.Seen[ent] then
                            TireScan.Seen[ent] = true
                            table.insert(TireScan.Queue, ent)
                        end
                    end
                end

                local extraSemi = wire.ports.ExtraFilterSemiDynamic
                if extraSemi and #extraSemi > 0 then
                    for _, ent in ipairs(extraSemi) do
                        if isValid(ent) and not TireScan.Seen[ent] then
                            TireScan.Seen[ent] = true
                            table.insert(TireScan.Queue, ent)
                        end
                    end
                end

                return previousFilterCount, previousMass, previousMass
            end
        end

        local function tickTireScanner()
            if not TireScan.Active then return end

            local processed = 0
            local queue = TireScan.Queue
            local seen = TireScan.Seen
            local result = TireScan.Result

            while #queue > 0 and processed < 6 do
                local current = table.remove(queue, 1)
                if isValid(current) then
                    table.insert(result, current)
                    local parent = current:getParent()
                    if isValid(parent) and not seen[parent] then
                        seen[parent] = true
                        table.insert(queue, parent)
                    end
                    local okChildren, children = pcall(function() return current:getChildren() end)
                    if okChildren and type(children) == "table" then
                        for _, child in ipairs(children) do
                            if isValid(child) and not seen[child] then
                                seen[child] = true
                                table.insert(queue, child)
                            end
                        end
                    end
                    local constraints = constraint.getTable(current)
                    if constraints then
                        for i = 1, #constraints do
                            local c = constraints[i]
                            local other = (c.Ent1 == current) and c.Ent2 or c.Ent1
                            if isValid(other) and not seen[other] then
                                seen[other] = true
                                table.insert(queue, other)
                            end
                        end
                    end
                end
                processed = processed + 1
            end

            if #queue == 0 then
                TireScan.Active = false

                local previousFilterCount = lastReportedFilterCount or 0
                local previousMass = TotalRealMass

                VehicleFilter = {}
                CheckedEntities = {}
                local currentMass = 0
                local massChecked = {}

                for i = 1, #result do
                    local ent = result[i]
                    if isValid(ent) then
                        if not CheckedEntities[ent] then
                            CheckedEntities[ent] = true
                            VehicleFilter[#VehicleFilter+1] = ent
                        end
                        if not massChecked[ent] then
                            massChecked[ent] = true
                            if not isValid(ent:getParent()) then
                                local phys = ent:getPhysicsObject()
                                if isValid(phys) then
                                    currentMass = currentMass + phys:getMass()
                                end
                            end
                        end
                    end
                end

                if currentMass > ((previousMass and previousMass > 0) and previousMass + 10000 or Config.TotalVehicleMass + 10000) then
                    currentMass = (previousMass and previousMass > 0) and previousMass or Config.TotalVehicleMass
                end

                TotalRealMass = currentMass
                local filterCount = #VehicleFilter
                local massChanged = math.abs(currentMass - (previousMass or 0)) > 0.1
                local filterChanged = filterCount ~= previousFilterCount

                if Config.AllowPrintFilter and filterChanged then
                    print("[TireSimV2] Vehicle filter rebuilt with " .. filterCount .. " entities.")
                end
                if Config.AllowPrintMassRefresh and (massChanged or filterChanged) then
                    local prevMassText = (previousMass and previousMass >= 0) and formatMass(previousMass) or "?"
                    print(
                        Config.W,
                        "[TireSimV2] Refresh update: filter ",
                        Config.G,
                        tostring(previousFilterCount),
                        Config.W,
                        " -> ",
                        Config.G,
                        tostring(filterCount),
                        Config.W,
                        " | real mass ",
                        Config.G,
                        prevMassText .. "kg",
                        Config.W,
                        " -> ",
                        Config.G,
                        formatMass(currentMass) .. "kg"
                    )
                end
                lastReportedFilterCount = filterCount
                local effectiveVehicleMass = (TotalRealMass > 0) and TotalRealMass or Config.TotalVehicleMass
                massRatio = math.max(0.1, math.min(1.0, effectiveVehicleMass / Config.TotalVehicleMass))
            end
        end

        local function initializeWheels()
            local Base = getValidBase()
            if not Base then return end
    
            for name, holo in pairs(WheelHolos) do if holo and holo:isValid() then holo:remove() end end
            Wheels, WheelHolos, OrderedWheelNames, NumWheels, VehicleFilter, CheckedEntities, lastAppliedForces = {}, {}, {}, 0, {}, {}, {}
    
            local minFwd, maxFwd = 9e9, -9e9
            local minRight, maxRight = 9e9, -9e9
    
            local wheelCounts = {
                Front_Left = 0, Front_Right = 0, Rear_Left = 0, Rear_Right = 0,
                Front_Center = 0, Rear_Center = 0
            }
            local axleInputNames = {"FrontWheels", "RearWheels"}
            for _, inputName in ipairs(axleInputNames) do
                local wheelArray = wire.ports[inputName]
                local axleName = inputName:gsub("Wheels", "")
                if wheelArray and #wheelArray > 0 then
                    for i, wheelEntity in ipairs(wheelArray) do
                        if wheelEntity and wheelEntity:isValid() then
                            local localPos = Base:worldToLocal(wheelEntity:getPos())
                            minFwd = math.min(minFwd, localPos.x); maxFwd = math.max(maxFwd, localPos.x)
                            minRight = math.min(minRight, localPos.y); maxRight = math.max(maxRight, localPos.y)
    
                            local sideValue = localPos:dot(RightAxis)
                            local side = "Center"; if sideValue > 0.1 then side = "Right" elseif sideValue < -0.1 then side = "Left" end
                            
                            local countKey = axleName .. "_" .. side
                            wheelCounts[countKey] = (wheelCounts[countKey] or 0) + 1
                            local nameIndex = wheelCounts[countKey]
                            local wheelName = axleName .. "_" .. side .. "_" .. nameIndex
                            local tireProps = Config.TireSetup[axleName] or {}
                            local wheelPhysObj = wheelEntity:getPhysicsObject()
                            
                            if wheelPhysObj and wheelPhysObj:isValid() then
                                local initMat = (Config.WheelPhysicsMaterial and Config.WheelPhysicsMaterial ~= "jeeptire") and Config.WheelPhysicsMaterial or "rubber"
                                wheelPhysObj:setMaterial(initMat)
                            end
                            
                            local rightSign = (side == "Right") and 1 or ((side == "Left") and -1 or 0)
                            local isFrontWheel = (axleName == "Front")
                            local axleRadius = tireProps.Radius or tireProps.TireRadius or Config.TireRadius
                            local alocal, detectedRadius = DetectWheelAxleAndRadius(wheelEntity, axleRadius, Base:localToWorldVector(RightAxis))
                            
                            Wheels[wheelName] = {
                                Entity = wheelEntity, PhysObj = wheelPhysObj,
                                Axle = axleName,
                                Preset = tireProps.Preset,
                                CompoundName = tireProps.Compound or "Sport",
                                Width = tireProps.Width or 155,
                                AspectRatio = tireProps.AspectRatio or 70,
                                Load = 0,
                                RightSign = rightSign,
                                IsFrontWheel = isFrontWheel,
                                alocal = alocal,
                                Radius = detectedRadius,
                                held = 0,
                                wPrev = nil,
                                dwOurs = 0,
                                OnGround = false,
                            }
                            lastAppliedForces[wheelName] = { fx = 0, fy = 0, mag = 0 }
                            table.insert(OrderedWheelNames, wheelName)
                            NumWheels = NumWheels + 1
    
                            if Config.ShowWheelHolograms then
                                local rScale = (detectedRadius or 16) / 12.5
                                local wScale = ((tireProps.Width or 155) / 225) * 0.65
                                local holo = holograms.create(wheelEntity:getPos(), wheelEntity:getAngles(), "models/sprops/trans/wheel_d/t_wheel25.mdl", Vector(rScale, wScale, rScale))
                                holo:setParent(wheelEntity); holo:setColor(Color(200, 200, 200, 200)); holo:setMaterial("models/debug/debugwhite"); holo:suppressEngineLighting(true)
                                WheelHolos[wheelName] = holo
                            end
                        end
                    end
                end
            end
    
            if NumWheels > 1 then
                Wheelbase = math.abs(maxFwd - minFwd)
                TrackWidth = math.abs(maxRight - minRight)
            else
                Wheelbase = 0
                TrackWidth = 0
            end
            
            performUnifiedVehicleScan(Base, true)
            lastReportedFilterCount = #VehicleFilter
            
            local effectiveVehicleMass = (TotalRealMass > 0) and TotalRealMass or Config.TotalVehicleMass
            massRatio = math.max(0.1, math.min(1.0, effectiveVehicleMass / Config.TotalVehicleMass))
            
            print(Config.W, "[TireSimV2] Initialized with ", Config.G, NumWheels, Config.W, " wheels. Version " .. Version)
            wire.ports.NumWheels = NumWheels; wire.ports.WheelNames = OrderedWheelNames
        end
    
        timer.create("TireSimV2_ConstraintRefresh", math.max(0.25, Config.ConstraintRefreshInterval or 2), 0, function()
            local Base = getValidBase()
            if not Base then return end
            performUnifiedVehicleScan(Base, false)
        end)

        local lastVel = Vector(0,0,0)
        local thinkCounter = 0
        local lastTelemetrySend = 0
        local lastGroundedMask = -1
        local telemetryDirty = true
        local hasActiveForces = false
    
        hook.add("think", "TireSimV2_Main", function()
            local Base = getValidBase()
            if not Base or NumWheels == 0 then return end
            
            local basePhys = Base:getPhysicsObject()
            if not basePhys or not basePhys:isValid() then return end
            
            tickTireScanner()
            
            local isBaseMoveable = basePhys:isMoveable()
            if not isBaseMoveable then
                hasActiveForces = false
            end
            
            local throttleVal = wire.ports.Throttle or 0
            local brakeVal = wire.ports.Brake or 0
            local hasInput = math.abs(throttleVal) > Config.InputDeadzone or math.abs(brakeVal) > Config.InputDeadzone
            
            local currentVel = basePhys:getVelocity()
            local isMoving = currentVel:getLengthSqr() >= 1
            if not isMoving then
                for _, name in ipairs(OrderedWheelNames) do
                    local wData = Wheels[name]
                    if wData and wData.PhysObj and wData.PhysObj:isValid() then
                        if wData.PhysObj:getAngleVelocity():getLengthSqr() > 1 or wData.OnGround then
                            isMoving = true
                            break
                        end
                    end
                end
            end
            
            local needsTelemetryClear = (Config.ShowTelemetry3D or Config.ShowPacejkaGraph) and (lastGroundedMask ~= 0)
            if not isMoving and not hasInput and not hasActiveForces and not telemetryDirty and not needsTelemetryClear then
                return
            end
            
            thinkCounter = thinkCounter + 1
    
            local qUsed = quotaUsed()
            local qMax = quotaMax()
            if qMax > 0 then
                cpuLoadPct = (qUsed / qMax) * 100
            end
     
            local tickInterval = game.getTickInterval()
            local targetInterval = 1 / math.max(1, Config.UpdateRate)
            skipTicks = math.max(0, math.floor(targetInterval / tickInterval + 0.5) - 1)
            
            local shouldUpdateGrip = (thinkCounter % (skipTicks + 1) == 0)
            
            local vehicleForward = Base:localToWorldVector(Config.ForwardAxis)
            local vehicleRight = Base:localToWorldVector(RightAxis)
            local vehicleUp = Base:localToWorldVector(Config.UpAxis)
            local effectiveVehicleMass = math.max(50, (TotalRealMass > 0) and TotalRealMass or (Config.TotalVehicleMass or 1500))
            local sysUpVel = computeCleanUpVelocity(basePhys, vehicleUp)
            local chassisSpeed = currentVel:getLength()

            if shouldUpdateGrip then
                local dt = math.max(0.001, tickInterval * (skipTicks + 1))
                local accel = (currentVel - lastVel) / dt
                lastVel = currentVel
                local maxAccel = Config.Gravity * 15
                if accel:getLengthSqr() > (maxAccel * maxAccel) then
                    accel = accel:getNormalized() * maxAccel
                end
                local fwdAccel = accel:dot(vehicleForward)
                local rightAccel = accel:dot(vehicleRight)
                
                massRatio = math.max(0.1, math.min(1.0, effectiveVehicleMass / Config.TotalVehicleMass))
    
                local halfWheelsCount = math.max(1, NumWheels / 2)
                local weightTransferRight = 0
                if TrackWidth > 0 then
                    weightTransferRight = (-rightAccel / Config.Gravity) * effectiveVehicleMass * (Config.CenterOfMassHeight or 25) / TrackWidth
                end
    
                local throttleInput = 0
                local brakeInput = 0
                if wire.ports.Throttle then
                    local throttleRaw = wire.ports.Throttle or 0
                    local maxThrottle = math.max(Config.ThrottleMaxValue, math.abs(throttleRaw))
                    Config.ThrottleMaxValue = maxThrottle
                    throttleInput = (maxThrottle > 0) and math.clamp(math.abs(throttleRaw) / maxThrottle, 0, 1) or 0
                end
                if wire.ports.Brake then
                    local brakeRaw = wire.ports.Brake or 0
                    local maxBrake = math.max(Config.BrakeMaxValue, math.abs(brakeRaw))
                    Config.BrakeMaxValue = maxBrake
                    brakeInput = (maxBrake > 0) and math.clamp(math.abs(brakeRaw) / maxBrake, 0, 1) or 0
                end
                
                local hasThrottleNow = throttleInput > Config.InputDeadzone
                -- Fallback longitudinal G-force used ONLY for chassis load transfer when pedal wires are absent:
                local accelG_Fwd = fwdAccel / Config.Gravity
                local weightTransferThrottle = hasThrottleNow and throttleInput or math.clamp(accelG_Fwd, 0, 1)
                local weightTransferBrake = (brakeInput > Config.InputDeadzone) and brakeInput or math.clamp(-accelG_Fwd, 0, 1)
                
                local staticLoadFront = effectiveVehicleMass * Config.Gravity * frontRatioNormalized / halfWheelsCount
                local staticLoadRear = effectiveVehicleMass * Config.Gravity * (1 - frontRatioNormalized) / halfWheelsCount
                
                local chassisSpeedMph = chassisSpeed / 17.6
                local speedMphDiv100Sq = (chassisSpeedMph / 100) ^ 2
                local aeroLoadFront = Config.FrontAeroFactor * effectiveVehicleMass * speedMphDiv100Sq
                local aeroLoadRear = Config.RearAeroFactor * effectiveVehicleMass * speedMphDiv100Sq
                
                local transferForceBrake = 0
                local transferForceThrottle = 0
                if weightTransferBrake > Config.InputDeadzone then
                    transferForceBrake = weightTransferBrake * Config.BrakeTransferFactor * effectiveVehicleMass
                elseif weightTransferThrottle > Config.InputDeadzone then
                    transferForceThrottle = weightTransferThrottle * Config.ThrottleTransferFactor * effectiveVehicleMass
                end
                
                local loads_t, grip_pcts_t, forces_t, long_forces_t, lat_forces_t = {}, {}, {}, {}, {}
                local slips_t, slip_angles_t, damping_t = {}, {}, {}
                
                local baseAngVel = basePhys:getAngleVelocity()
                local SIDE_OFFSETS = {0, -1, 1}
                
                for j = #VehicleFilter, 1, -1 do
                     local e = VehicleFilter[j]
                     if not (e and e:isValid()) then
                          table.remove(VehicleFilter, j)
                     end
                end
    
                for i, name in ipairs(OrderedWheelNames) do
                    local wheelData = Wheels[name]
                    if wheelData.Entity and wheelData.Entity:isValid() and wheelData.PhysObj and wheelData.PhysObj:isValid() then
                        local compoundPresetName = wheelData.Preset or nil
                        local activeTireData = InternalProperties[wheelData.CompoundName] or InternalProperties.Sport
                        local isFrontWheel = wheelData.IsFrontWheel
                        
                        -- Load Calculation
                        local staticLoad = isFrontWheel and staticLoadFront or staticLoadRear
                        local aeroLoad = isFrontWheel and aeroLoadFront or aeroLoadRear
                        local transferLoad = 0
                        if transferForceBrake > 0 then
                            transferLoad = isFrontWheel and transferForceBrake or -transferForceBrake
                        elseif transferForceThrottle > 0 then
                            transferLoad = isFrontWheel and -transferForceThrottle or transferForceThrottle
                        end
                        
                        local rightSign = wheelData.RightSign
                        local maxWheelLoad = staticLoad * 6.0
                        wheelData.Load = math.clamp(staticLoad + aeroLoad + transferLoad + (weightTransferRight * rightSign), 50, maxWheelLoad)
    
                        local wheelPhys = wheelData.PhysObj
                        local wheelPos = wheelPhys:localToWorld(wheelPhys:getMassCenter())
                        local alocal = wheelData.alocal or Vector(0, 1, 0)
                        local aRaw = wheelPhys:localToWorldVector(alocal)
                        local a = (aRaw:getLengthSqr() > 0.0001) and aRaw:getNormalized() or vehicleRight
                        local wheelRight = a
                        if wheelRight:dot(vehicleRight) < 0 then wheelRight = -wheelRight end
                        wheelData.CurrentRight = wheelRight

                        -- Per-axle wheel radius (with global Config.TireRadius / OBB fallback)
                        local rs = wheelData.Radius or Config.TireRadius or 16
                        wheelData.EffectiveRadius = rs
                        local reach = rs + math.min(2, rs * 0.15)
                        local wheelTraceVec = -vehicleUp * reach

                        local hw = math.clamp((wheelData.Width or 155) * 0.03937 * 0.4, 2, 8)
                        local rawWheelFwd = a:cross(vehicleUp)
                        local wheelFwd = (rawWheelFwd:getLengthSqr() > 0.0001) and rawWheelFwd:getNormalized() or vehicleForward
                        local rayDirs = {
                            wheelTraceVec,
                            (-vehicleUp * 0.81915 + wheelFwd * 0.57358) * reach,
                            (-vehicleUp * 0.81915 - wheelFwd * 0.57358) * reach
                        }
                        local bestTr = nil
                        local bestFrac = 2.0
                        for _, rayVec in ipairs(rayDirs) do
                            for _, side in ipairs(SIDE_OFFSETS) do
                                local sPos = wheelPos + a * (hw * side)
                                local st = traceFiltered(sPos, sPos + rayVec)
                                if st.Hit and not st.StartSolid then
                                    local frac = st.Fraction or 1
                                    if frac < bestFrac then
                                        bestFrac = frac
                                        bestTr = st
                                    end
                                end
                            end
                        end
                        local tr = bestTr or { Hit = false, StartSolid = false }
                        
                        local latForceFinal = 0
                        local longForceFinal = 0
                        local totalCombinedMag = 0
                        local slipRatioMetric = 0
                        local slipAngleMetric = 0
                        local slipVel = 0
                        local gripPct = 100
                        local rollingResistance = 0
                        local yawMag = 0
                        local fwdGroundVel = 0
                        local wheelSpinLinear = 0
                        
                        if tr.Hit and not tr.StartSolid then
                            local hitNormal = tr.HitNormal
                            wheelData.HitNormal = hitNormal
                            
                            -- PropTire Longitudinal Basis (f = a x n, unflipped so linvel and fv share exact sign convention)
                            local fP = a:cross(hitNormal)
                            if fP:getLengthSqr() > 0.0001 then
                                fP = fP:getNormalized()
                            else
                                fP = vehicleForward
                            end
                            wheelData.CurrentFwd = fP
                            
                            local rawLat = hitNormal:cross(fP)
                            local wheelLat = (rawLat:getLengthSqr() > 0.0001) and rawLat:getNormalized() or vehicleRight
                            if wheelLat:dot(wheelRight) < 0 then wheelLat = -wheelLat end
                            wheelData.CurrentLat = wheelLat
                            
                            local rawRadial = a:cross(fP)
                            local radialDown = (rawRadial:getLengthSqr() > 0.0001) and rawRadial:getNormalized() or -hitNormal
                            local contactPoint = wheelPos + radialDown * rs
                            wheelData.CurrentContactPos = contactPoint
                            local hitMovableEnt = (not tr.HitWorld and tr.Entity and tr.Entity:isValid()) and tr.Entity or nil
                            wheelData.HitEntity = hitMovableEnt
                            wheelData.OnGround = true

                            local rawVel = wheelPhys:getVelocity()
                            local vP = rawVel + vehicleUp * (sysUpVel - rawVel:dot(vehicleUp))
                            local contactPointVel = vP
                            if hitMovableEnt then
                                local gp = hitMovableEnt:getPhysicsObject()
                                if gp and gp:isValid() then
                                    local vG = gp:getVelocityAtPoint(contactPoint)
                                    vP = vP - vG
                                    contactPointVel = contactPointVel - vG
                                end
                            end
                            
                            -- PropTire Wheel Tangential Velocity & Ground Forward Velocity
                            wheelSpinLinear = math.rad(wheelPhys:getAngleVelocity():dot(alocal)) * rs
                            fwdGroundVel = vP:dot(fP)
                            slipVel = wheelSpinLinear - fwdGroundVel
                            
                            -- Lateral Ground Velocity & Speed
                            local trueSpeed = contactPointVel:getLength()
                            local latGroundVel = vP:dot(wheelLat)
                            local moving = math.max(math.abs(fwdGroundVel), math.abs(wheelSpinLinear), math.abs(latGroundVel), trueSpeed)
                            
                            local aspectVal = wheelData.AspectRatio or 50
                            local aspectStiffnessMult = 50 / math.max(10, aspectVal)
                            
                            local rawSurfaceMult = activeTireData.SurfaceMultipliers[tr.MatType] or activeTireData.SurfaceMultipliers.Default
                            gripPct = math.clamp(100 * rawSurfaceMult, 0, 100)
                            wheelData.SurfaceCoF = (activeTireData.CoF or 1.0) * rawSurfaceMult
                            
                            local compoundPreset = (Config.CompoundPresets and compoundPresetName and Config.CompoundPresets[compoundPresetName]) or {}
                            local overallGripMult = compoundPreset.OverallGripMultiplier or Config.OverallGripMultiplier or 1.0
                            local lateralGripMult = compoundPreset.LateralGripMultiplier or Config.LateralGripMultiplier or 1.0
                            local longitudinalGripMult = compoundPreset.LongitudinalGripMultiplier or Config.LongitudinalGripMultiplier or 1.0
                            local frontGripBias = compoundPreset.FrontGripBias or Config.FrontGripBias or 0.5
                            local rearGripBias = compoundPreset.RearGripBias or Config.RearGripBias or 0.75
                            local widthFactor = wheelData.Width / 225
                            local gripBias = isFrontWheel and frontGripBias or rearGripBias
                            local finalGripMultiplier = rawSurfaceMult * widthFactor * gripBias
                            
                            -- Separate Peak Grip Capacities for Lateral & Longitudinal
                            local fz = wheelData.Load
                            local effectiveLoadA1 = activeTireData.Load_A1 / math.max(0.5, widthFactor)
                            local loadSens = math.max(0.2, (fz * effectiveLoadA1) + activeTireData.Load_A2)
                            local peakRaw = math.max(0, fz * loadSens * activeTireData.CoF)
                            local baseAvailableGrip = peakRaw * finalGripMultiplier * overallGripMult
                            local maxLatGrip  = baseAvailableGrip * (lateralGripMult or 1.0)
                            local maxLongGrip = baseAvailableGrip * (longitudinalGripMult or 1.0)
                            local maxAvailableGrip = math.max(maxLatGrip, maxLongGrip)
                            wheelData.PeakForce = maxAvailableGrip
                            wheelData.PeakLatForce = maxLatGrip
                            wheelData.PeakLongForce = maxLongGrip
                            
                            -- Aspect ratio + width stiffness scaling
                            local latStiffness  = aspectStiffnessMult
                            local longStiffness = (1.0 + (aspectStiffnessMult - 1.0) * 0.5) * math.sqrt(math.max(0.5, widthFactor))
                            
                            rollingResistance = activeTireData.BaseRollingResistance
                            wheelData.RollDragForce = rollingResistance
                            
                            -- ========================================================
                            -- PROPTIRE MATERIAL SWITCHING (gmod_ice WHEN MOVING, rubber WHEN SLOW)
                            -- Eliminates Source Engine's native "jeeptire" friction while driving so
                            -- 100% of longitudinal & lateral traction is governed by the tire model!
                            -- ========================================================
                            if wheelData.slow == nil then wheelData.slow = true end
                            if wheelData.slow and moving >= V_FAST and (hasThrottleNow or chassisSpeed >= 25) then
                                wheelData.slow = false
                                if wheelData.CurrentMat ~= "gmod_ice" then
                                    wheelPhys:setMaterial("gmod_ice")
                                    wheelData.CurrentMat = "gmod_ice"
                                end
                            elseif not wheelData.slow and (moving < V_SLOW or (not hasThrottleNow and chassisSpeed < 20)) then
                                wheelData.slow = true
                            end
                            if wheelData.slow then
                                local restMat = (Config.WheelPhysicsMaterial and Config.WheelPhysicsMaterial ~= "jeeptire") and Config.WheelPhysicsMaterial or "rubber"
                                if wheelData.CurrentMat ~= restMat then
                                    wheelPhys:setMaterial(restMat)
                                    wheelData.CurrentMat = restMat
                                end
                            end
                            wheelData.IsStatic = wheelData.slow
                            
                            -- ========================================================
                            -- SLIP & TRACTION CALCULATIONS (Exact PropTire + Pacejka Model)
                            -- ========================================================
                            if wheelData.slow then
                                longForceFinal = 0
                                latForceFinal = 0
                                wheelData.RawFx = 0
                                wheelData.RawFy = 0
                                totalCombinedMag = 0
                            else
                                -- PropTire True Lateral Slip Angle (alpha) with V0=40 floor (full lateral stiffness on curbs & low speeds)
                                local denomSpeed = math.max(math.abs(fwdGroundVel), trueSpeed, V0)
                                local lateralSlipAngle = math.clamp(math.atan(latGroundVel / denomSpeed), -1.55, 1.55)
                                slipAngleMetric = lateralSlipAngle
                                
                                -- Uncoupled Lateral Pacejka Force
                                local Lat_B = activeTireData.Lat_B * latStiffness
                                local Lat_E = activeTireData.Lat_E
                                local rawLat = EvaluateMagicFormula(Lat_B, activeTireData.Lat_C, maxLatGrip, Lat_E, lateralSlipAngle)
                                
                                -- PropTire Longitudinal Slip Ratio (s) (proptire.lua line 1239)
                                local longSlipRatio = math.clamp((wheelSpinLinear - fwdGroundVel) / math.max(math.abs(fwdGroundVel), math.abs(wheelSpinLinear), V0), -0.99, 0.99)
                                local fwdSign = (fP:dot(vehicleForward) >= 0) and 1 or -1
                                
                                -- Uncoupled Longitudinal Pacejka Force (proptire.lua line 904)
                                local Long_B = (activeTireData.Long_B or 12) * longStiffness
                                local Long_C = activeTireData.Long_C or 1.65
                                local Long_E = activeTireData.Long_E or 0.6
                                local rawLong = EvaluateMagicFormula(Long_B, Long_C, maxLongGrip, Long_E, longSlipRatio)
                                
                                -- ====================================================
                                -- [FRICTION ELLIPSE COMBINED SLIP COUPLING]
                                -- ====================================================
                                local normLat  = (maxLatGrip > 0.001) and (rawLat / maxLatGrip) or 0
                                local normLong = (maxLongGrip > 0.001) and (rawLong / maxLongGrip) or 0
                                local normCombined = math.sqrt(normLat * normLat + normLong * normLong)
                                local clampedLat = rawLat
                                local clampedLong = rawLong
                                
                                if normCombined > 1.0 then
                                    local scale = 1.0 / normCombined
                                    clampedLat = rawLat * scale
                                    clampedLong = rawLong * scale
                                end
                                
                                -- PropTire Effective Longitudinal Mass & Impulse Limit (jxMax) (proptire.lua lines 1187 & 1245)
                                local m = wheelPhys:getMass()
                                local I = math.abs(wheelPhys:getInertia():dot(alocal)) / (U * U)
                                local mx = (I > 0) and (1 / (1 / m + (1 - (wheelData.held or 0)) * (rs * rs) / I)) or m
                                local jxMax = RELAX * math.abs(wheelSpinLinear - fwdGroundVel) * mx
                                
                                -- Damping & Smoothing for Lateral Force (Reset on zero-crossing to prevent phase-lag wiggle)
                                local prevForces = lastAppliedForces[name] or { fx = 0, fy = 0 }
                                local prevFy = (prevForces.fy * clampedLat < 0) and 0 or prevForces.fy
                                local compoundGripDamping = (compoundPreset and compoundPreset.GripDampingFactor) or Config.GripDampingFactor or 1.0
                                local dampedLat = prevFy + (clampedLat - prevFy) / math.max(1, compoundGripDamping)
                                
                                local maxOneG = effectiveVehicleMass * Config.Gravity
                                local maxForceLim = maxOneG * 1.5
                                dampedLat = math.clamp(dampedLat, -maxForceLim, maxForceLim)
                                clampedLong = math.clamp(clampedLong, -maxForceLim, maxForceLim)
                                
                                wheelData.RawFx = clampedLong
                                wheelData.RawFy = dampedLat
                                longForceFinal = (massRatio > 0) and math.clamp(clampedLong * massRatio, -jxMax, jxMax) / massRatio or 0
                                latForceFinal  = dampedLat
                                totalCombinedMag = math.sqrt(longForceFinal * longForceFinal + latForceFinal * latForceFinal)
                                
                                -- Scale telemetry slip ratio by actual jxMax utilization so free-rolling wheels show ~0% slip on HUD
                                local utilScale = ( math.abs(clampedLong) > 0.01 ) and math.min(1.0, math.abs(longForceFinal) / math.abs(clampedLong)) or 0
                                slipRatioMetric = longSlipRatio * fwdSign * utilScale
                            end
                            
                            -- Yaw Stability Assist
                            local compoundYawStability = (compoundPreset and compoundPreset.YawStabilityFactor) or Config.YawStabilityFactor or 0.0
                            if compoundYawStability > 0 and not wheelData.slow then
                                local yawVel = baseAngVel.z
                                if math.abs(yawVel) > 1 and trueSpeed > 0.1 then
                                    local counterForce = -(contactPointVel / trueSpeed) * (math.abs(yawVel) * compoundYawStability * 50)
                                    yawMag = counterForce:dot(vehicleRight)
                                end
                            end
                        else
                            wheelData.PeakForce = 0
                            wheelData.PeakLatForce = 0
                            wheelData.PeakLongForce = 0
                            wheelData.RollDragForce = 0
                            wheelData.IsStatic = false
                            wheelData.OnGround = false
                            wheelData.HitEntity = nil
                            wheelData.HitNormal = nil
                            wheelData.RawFx = 0
                            wheelData.RawFy = 0
                            wheelData.wPrev = nil
                            wheelData.dwOurs = 0
                            wheelData.LongAnchorPos = nil
                            wheelData.LatAnchorPos = nil
                        end
                        
                        -- Rotational parking brake / BaseRollingResistance via angular-only damping (0 linear damping preserves susp_lib balance)
                        local targetDampAng = 0
                        if wheelData.OnGround then
                            if wheelData.slow and not hasThrottleNow and chassisSpeed < 15 then
                                targetDampAng = 500
                            else
                                targetDampAng = rollingResistance
                            end
                        end
                        if wheelData.LastDampAng ~= targetDampAng or wheelData.LastDampPos ~= 0 then
                            wheelData.PhysObj:setDamping(0, targetDampAng)
                            wheelData.LastDampPos = 0
                            wheelData.LastDampAng = targetDampAng
                        end
                        
                        lastAppliedForces[name] = { fx = longForceFinal, fy = latForceFinal, mag = totalCombinedMag }
                        wheelData.LastComponents = {
                            Fx = longForceFinal,
                            RawFx = wheelData.RawFx or 0,
                            Fy = latForceFinal,
                            RawFy = wheelData.RawFy or latForceFinal,
                            Mag = totalCombinedMag,
                            YawMag = yawMag,
                            SlipLat = math.abs(slipAngleMetric),
                            SlipLong = math.abs(slipRatioMetric),
                            SlipVel = slipVel
                        }
                        
                        loads_t[i] = wheelData.Load
                        grip_pcts_t[i] = gripPct
                        forces_t[i] = totalCombinedMag
                        long_forces_t[i] = longForceFinal
                        lat_forces_t[i] = latForceFinal
                        slips_t[i] = slipRatioMetric
                        slip_angles_t[i] = slipAngleMetric
                        damping_t[i] = rollingResistance
                    end
                end
                
                local anyForces = false
                local groundedMask = 0
                for j, name in ipairs(OrderedWheelNames) do
                    local wData = Wheels[name]
                    local comps = wData and wData.LastComponents
                    if wData and wData.OnGround then
                        anyForces = true
                        groundedMask = groundedMask + bit.lshift(1, (j - 1) % 30)
                    elseif comps and (comps.Mag > 0.05 or math.abs(comps.RawFx or 0) > 0.05 or math.abs(comps.RawFy or 0) > 0.05 or math.abs(comps.Fy or 0) > 0.05 or (comps.YawMag and math.abs(comps.YawMag) > 0.05)) then
                        anyForces = true
                    end
                end
                hasActiveForces = anyForces
                if groundedMask ~= lastGroundedMask then
                    lastGroundedMask = groundedMask
                    telemetryDirty = true
                end
                
                wire.ports.Loads = loads_t
                wire.ports.GripPcts = grip_pcts_t
                wire.ports.AbsoluteForces = forces_t
                wire.ports.LongitudinalForces = long_forces_t
                wire.ports.LateralForces = lat_forces_t
                wire.ports.Slips = slips_t
                wire.ports.SlipAngles = slip_angles_t
                wire.ports.AngularDamping = damping_t
                wire.ports.CPULoadPct = cpuLoadPct
                wire.ports.TickSkip = skipTicks
                
                -- Telemetry Sync to Client HUD (Includes Separate Lat & Long Grip & Peak Capacities)
                if Config.ShowTelemetry3D or Config.ShowPacejkaGraph then
                    local now = timer.systime()
                    if telemetryDirty or (now - lastTelemetrySend >= 0.03) then
                        if net.getBytesLeft and net.getBytesLeft() > 384 then
                            net.start("telemetryUpdateV2")
                            net.writeUInt(NumWheels, 8)
                            for _, name in ipairs(OrderedWheelNames) do
                                local wData = Wheels[name]
                                if wData and wData.Entity and wData.Entity:isValid() and wData.LastComponents then
                                    local isSlow = wData.slow
                                    net.writeFloat(isSlow and 0 or (wData.LastComponents.Mag or 0))
                                    net.writeFloat(wData.PeakForce or 0)
                                    net.writeFloat(isSlow and 0 or math.abs(wData.LastComponents.Fy or 0))
                                    net.writeFloat(wData.PeakLatForce or 0)
                                    net.writeFloat(isSlow and 0 or math.abs(wData.LastComponents.Fx or 0))
                                    net.writeFloat(wData.PeakLongForce or 0)
                                    net.writeFloat(isSlow and 0 or (wData.LastComponents.SlipLat or 0))
                                    net.writeFloat(isSlow and 0 or (wData.LastComponents.SlipLong or 0))
                                else
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                    net.writeFloat(0)
                                end
                            end
                            net.send(owner())
                            lastTelemetrySend = now
                            telemetryDirty = false
                        end
                    end
                end
            end 
            
            -- ========================================================================
            -- FORCE APPLICATION (Per-Wheel PropTire Impulses + Steep-Curb Horizontal Hold)
            -- ========================================================================
            if hasActiveForces and isBaseMoveable then
                local maxThrottleNow = math.max(Config.ThrottleMaxValue or 1, math.abs(throttleVal))
                local activeThrottle = (maxThrottleNow > 0) and ((math.abs(throttleVal) / maxThrottleNow) > Config.InputDeadzone) or false
                local maxWheelImpulse = effectiveVehicleMass * Config.Gravity * 2.0

                for i, name in ipairs(OrderedWheelNames) do
                    local wheelData = Wheels[name]
                    if wheelData and wheelData.Entity and wheelData.Entity:isValid() and wheelData.PhysObj and wheelData.PhysObj:isValid() then
                        local comps = wheelData.LastComponents
                        local wheelPhys = wheelData.PhysObj
                        local compoundPreset = (Config.CompoundPresets and wheelData.Preset and Config.CompoundPresets[wheelData.Preset]) or {}
                        local wheelLatMult = math.max(1.0, compoundPreset.LateralGripMultiplier or Config.LateralGripMultiplier or 1.0)
                        local holdDeadzone = math.max(1.0, compoundPreset.LowSpeedLateralHoldDeadzone or Config.LowSpeedLateralHoldDeadzone or 2.0)
                        local holdScale = math.max(0.0, compoundPreset.LowSpeedHoldScale or Config.LowSpeedHoldScale or 1.0)
                        local wPrev, dwOurs = wheelData.wPrev, wheelData.dwOurs or 0
                        wheelData.wPrev, wheelData.dwOurs = nil, 0

                        if comps then
                            local centerForceVec = Vector(0, 0, 0)
                            local wheelMassCenter = wheelPhys:localToWorld(wheelPhys:getMassCenter())
                            local m = math.max(1, wheelPhys:getMass())
                            
                            -- 1. PropTire Longitudinal & Lateral Ground Impulses (susp_lib-invariant velocity)
                            if wheelData.OnGround then
                                local alocal = wheelData.alocal or Vector(0, 1, 0)
                                local rs = wheelData.Radius or Config.TireRadius or 16
                                local hitNormal = wheelData.HitNormal or vehicleUp
                                
                                -- Recompute live wheel basis every tick so force vectors & lever arm never lag or induce steering torque
                                local aRaw = wheelPhys:localToWorldVector(alocal)
                                local a = (aRaw:getLengthSqr() > 0.0001) and aRaw:getNormalized() or vehicleRight
                                local wheelRight = a
                                if wheelRight:dot(vehicleRight) < 0 then wheelRight = -wheelRight end
                                local f = a:cross(hitNormal)
                                if f:getLengthSqr() > 0.0001 then
                                    f = f:getNormalized()
                                else
                                    f = vehicleForward
                                end
                                local rawLat = hitNormal:cross(f)
                                local wheelLat = (rawLat:getLengthSqr() > 0.0001) and rawLat:getNormalized() or vehicleRight
                                if wheelLat:dot(wheelRight) < 0 then wheelLat = -wheelLat end
                                
                                -- Contact point strictly in wheel radial plane from mass center: (p - wheelMassCenter) x f = -rs * a (zero steering torque!)
                                local rawRadial = a:cross(f)
                                local radialDown = (rawRadial:getLengthSqr() > 0.0001) and rawRadial:getNormalized() or -hitNormal
                                local p = wheelMassCenter + radialDown * rs
                                
                                local linvel = math.rad(wheelPhys:getAngleVelocity():dot(alocal)) * rs
                                local rawVel = wheelPhys:getVelocity()
                                local vP = rawVel + vehicleUp * (sysUpVel - rawVel:dot(vehicleUp))
                                local hitEnt = wheelData.HitEntity
                                local gp = (hitEnt and hitEnt:isValid()) and hitEnt:getPhysicsObject() or nil
                                if gp and gp:isValid() then
                                    local vGround = gp:getVelocityAtPoint(p)
                                    vP = vP - vGround
                                end
                                local fv = vP:dot(f)
                                local lvWheel = vP:dot(wheelLat)
                                
                                local maxLatCap = math.max(0, (wheelData.PeakLatForce or 0) * massRatio)
                                
                                local I = math.abs(wheelPhys:getInertia():dot(alocal)) / (U * U)
                                local w = linvel / rs
                                if wPrev and math.abs(dwOurs) > 0.01 then
                                    local ace = (w - wPrev) - dwOurs
                                    local h = (ace * (wPrev + dwOurs) < 0) and math.clamp(-ace / dwOurs, 0, 1) or 0
                                    wheelData.held = (wheelData.held or 0) + (h - (wheelData.held or 0)) * HELD_SMOOTH
                                end
                                wheelData.wPrev = w
                                local denom = (1 / m) + math.max(0, 1 - (wheelData.held or 0)) * (rs * rs) / math.max(0.001, I)
                                local mx = (denom > 0) and (1 / denom) or m
                                local spin = (I > 0.001) and (-rs / I) or 0
                                
                                if wheelData.slow then
                                    -- Zero linear forces on flat/gentle ground; pure horizontal damping only on steep curbs
                                    if hitNormal:dot(vehicleUp) < 0.95 and not activeThrottle and chassisSpeed < 15 then
                                        local vHoriz = vP - vehicleUp * vP:dot(vehicleUp)
                                        if vHoriz:getLengthSqr() > 0.01 and maxLatCap > 0 then
                                            local holdCap = maxLatCap * math.max(0.25, holdScale)
                                            local holdImpulse = -vHoriz * (RELAX * m * holdScale)
                                            local holdLenSqr = holdImpulse:getLengthSqr()
                                            if holdLenSqr > (holdCap * holdCap) then
                                                holdImpulse = holdImpulse:getNormalized() * holdCap
                                            end
                                            centerForceVec = centerForceVec + holdImpulse
                                        end
                                    end
                                else
                                    -- Longitudinal Active Impulse (Clamped by jxMax)
                                    local jxMax = RELAX * math.abs(linvel - fv) * mx
                                    local rawFx = comps.RawFx or 0
                                    if rawFx * (linvel - fv) < 0 then rawFx = 0 end
                                    local jx = math.clamp(rawFx * massRatio, -jxMax, jxMax)
                                    if isSafeNumber(jx) and math.abs(jx) > 0.01 then
                                        safeApplyForceOffset(wheelPhys, f * jx, p, maxWheelImpulse)
                                        wheelData.dwOurs = jx * spin
                                    end
                                    
                                    -- Full-Strength Lateral Grip (Strictly Opposing Lateral Slip Velocity)
                                    local rawFy = comps.RawFy or comps.Fy or 0
                                    if rawFy * lvWheel < 0 then rawFy = 0 end
                                    local microRamp = (wheelLatMult > 1.0) and math.clamp(math.abs(lvWheel) / holdDeadzone, 1.0 / wheelLatMult, 1.0) or 1.0
                                    local pacejkaLat = (math.abs(rawFy) > 0.05) and -(rawFy * massRatio * microRamp) or 0
                                    local jyMax = RELAX * math.abs(lvWheel) * m
                                    local latCap = math.min(jyMax, maxLatCap)
                                    local latImpulse = math.clamp(pacejkaLat, -latCap, latCap)
                                    if isSafeNumber(latImpulse) and math.abs(latImpulse) > 0.01 then
                                        centerForceVec = centerForceVec + (wheelLat * latImpulse)
                                    end
                                end
                            end
                            
                            -- 2. Yaw Stability Assist
                            if not wheelData.slow and comps.YawMag and math.abs(comps.YawMag) > 0.05 then
                                local yawImpulse = comps.YawMag * massRatio
                                if isSafeNumber(yawImpulse) then
                                    centerForceVec = centerForceVec + (vehicleRight * yawImpulse)
                                end
                            end
                            
                            -- Apply linear tire forces strictly to wheelPhys (no basePhys linear/offset force)
                            safeApplyForceCenter(wheelPhys, centerForceVec, maxWheelImpulse)
                        end
                    end
                end
            end
        end)
    end -- end initServer

    _G.TireSimCoreV2 = {
        initServer = initServer,
        Version = Version
    }
end
