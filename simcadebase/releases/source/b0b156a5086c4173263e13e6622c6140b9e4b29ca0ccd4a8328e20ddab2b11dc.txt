local function makeShiftLogic()
    local M = {}
    local function trunc(x)
        return x < 0 and math.ceil(x) or math.floor(x)
    end
    function M.shiftPoints(c, top)
        local up, down = {}, {}
        for i = 0, 11 do up[i] = 0; down[i] = 0 end
        local idle, redline, limiter = c[124], c[125], c[126]
        up[1] = (redline - idle) * c[108] + idle
        local last = top + 1
        local gear = 2
        while gear < last do
            local r0, r1 = c[108 + 9 + gear - 1], c[108 + 9 + gear]
            up[gear] = redline - 100
            down[gear + 1] = math.abs(r0) <= 1e-5 and 0 or r1 / r0 * up[gear]
            gear = gear + 1
        end
        up[gear] = limiter + 500
        return up, down
    end
    return M
end
local function makeVehicleSolver()
local M = {}
local abs, sqrt, floor, ceil, sin, cos, min, max = math.abs, math.sqrt, math.floor, math.ceil, math.sin, math.cos, math.min, math.max

local function clamp(v, a, b)
    if v < a then return a elseif v > b then return b end
    return v
end
local function trunc(x)
    if x < 0 then return ceil(x) end
    return floor(x)
end
local frexp, ldexp = math.frexp, math.ldexp
local function f32(x)
    if x == 0 or x ~= x or x == math.huge or x == -math.huge then return x end
    local m, e = frexp(x)
    local bits = 24
    if e < -125 then bits = 24 - (-125 - e) end     -- a denormal keeps fewer
    if bits < 1 then return 0 * x end
    local s = ldexp(m, bits)
    local r = floor(s)
    local f = s - r
    if f > 0.5 or (f == 0.5 and r % 2 == 1) then r = r + 1 end
    return ldexp(r, e - bits)
end
M.f32 = f32
local function curve(t, x)
    if t.points then
        local points = t.points
        local first = points[1]
        if x <= first.rpm then return first.value end
        if x >= points[t.n].rpm then return points[t.n].value end
        local lower, upper = 1, t.n
        while upper - lower > 1 do
            local middle = floor((lower + upper) * 0.5)
            if x < points[middle].rpm then upper = middle else lower = middle end
        end
        local previous, current = points[lower], points[upper]
        local fraction = (x - previous.rpm) / (current.rpm - previous.rpm)
        return previous.value + (current.value - previous.value) * fraction
    end
    local v = (x - t.lo) * t.k
    local i = v < 0 and ceil(v) or floor(v)
    if i < 0 or v < 0 then return t.v[0] end
    if i >= t.n - 1 then return t.v[t.n - 1] end
    local f = v - i
    return (1 - f) * t.v[i] + f * t.v[i + 1]
end
local function points(p, x)
    local n = floor(p.len / 2)
    if x <= p[0] then return p[1] end
    if x >= p[2 * n - 2] then return p[2 * n - 1] end
    for i = 0, n - 2 do
        if x >= p[2 * i] and x < p[2 * i + 2] then
            return (p[2 * i + 3] - p[2 * i + 1]) * ((x - p[2 * i]) / (p[2 * i + 2] - p[2 * i])) + p[2 * i + 1]
        end
    end
    return p[1]
end
local function ticksOf(dt) return trunc(f32(dt) * 4000 + 0.5) end
local function vec(a, b, c) return { [0] = a, b, c } end
local function dot(a, b) return a[0] * b[0] + a[1] * b[1] + a[2] * b[2] end
local function len(a) return sqrt(a[0] * a[0] + a[1] * a[1] + a[2] * a[2]) end
local function toWorld(R, v) return vec(v[0] * R[0] + v[1] * R[3] + v[2] * R[6], v[0] * R[1] + v[1] * R[4] + v[2] * R[7], v[0] * R[2] + v[1] * R[5] + v[2] * R[8]) end
local function toLocal(R, v) return vec(v[0] * R[0] + v[1] * R[1] + v[2] * R[2], v[0] * R[3] + v[1] * R[4] + v[2] * R[5], v[0] * R[6] + v[1] * R[7] + v[2] * R[8]) end
local function slice(c, from, to)
    local out, n = {}, 0
    for i = from, to - 1 do out[n] = c[i]; n = n + 1 end
    return out
end

-- A negative hold means this pedal has already braked the car to a near-stop.
-- A positive hold keeps the applied brake during the gap before a second press.
local function updateDirectionBrake(latched, hold, launched, pedal, otherPedal, brakeWhileMoving, speed, stopped, dt)
    hold = hold or 0
    if otherPedal >= 0.10 or abs(speed) > 0.75 then hold = 0 end
    if hold > 0 then hold = max(0, hold - dt) end
    if not launched and brakeWhileMoving and pedal > 0.25 then latched = true end
    if pedal < 0.10 then
        launched = false
        if hold < 0 then hold = 0.45 end
        if hold == 0 then latched = false end
    elseif latched and stopped and otherPedal < 0.10 and hold == 0 then
        hold = -1
    end
    return latched, hold, launched
end

function Chassis:driver(steer, gas, brake, handbrake, dt, hold, u16)
    self.ticks = self.ticks + ticksOf(dt)
    -- unoccupied keeps live tire & chassis grip
    local D = self.drv
    D.gas = gas; D.brake = brake; D.handbrake = handbrake
    local v16 = self.forward
    local v18 = D.gas
    D.hbk = D.handbrake
    -- Treat creeping downhill as motion outside this stop deadband (m/s).
    local directionChangeSpeed = 0.25
    local rollingForward = v16 > directionChangeSpeed
    local rollingBackward = v16 < -directionChangeSpeed
    local forwardBrake = D.forwardBrake or v18
    if self._vphysMode and self.classicAuto and not self.manualBox then
        local reverse = self.autoRange == 1
        local swap = reverse and self.reverseWSonReverse
        D.reversing = reverse
        D.thr = swap and (D.reverseGas or brake) or gas
        D.brk = SimcadeBraking.updatePedal(self, D, swap and forwardBrake or brake, dt)
        if self.autoRange == 0 then D.thr, D.brk = 0, 1 end
        if self.ignitionRunning == false then D.thr = 0 end
    elseif self._vphysMode and self.ignitionRunning == false then
        D.reversing = false
        D.reverseLaunch = false
        D.reverseHold = 0
        D.forwardLaunch, D.forwardHold = false, 0
        D.thr = 0
        D.brk = clamp(D.brake, 0, 1)
    elseif self._vphysMode and self.manualBox then
        D.reversing = (D.gear <= 0)
        D.thr = clamp(v18, 0, 1)
        D.brk = SimcadeBraking.updatePedal(self, D, clamp(D.brake, 0, 1), dt)
    elseif self._vphysMode and not self.manualBox then
        -- Symmetric 4-way forward / brake / reverse / brake state machine:
        --   - Pressing W while moving backward (or S while moving forward) always applies full
        --     service brakes to a stop and holds them (latch) regardless of current gear.
        --   - Releasing the latched pedal opens a 0.5s launch window so a second press launches
        --     cleanly even if gravity nudges the car on a steep hill during the release.
        --   - Once launched (forwardLaunch / reverseLaunch), holding the pedal stays in throttle
        --     without oscillating back into brake.
        local stopSpeed = 0.35
        local movingForward = v16 > stopSpeed
        local movingBackward = v16 < -stopSpeed
        local stopped = not movingForward and not movingBackward
        local wPressed = v18 > 0.10
        local sPressed = D.brake > 0.10

        D.forwardHold = max(0, (D.forwardHold or 0) - dt)
        D.reverseHold = max(0, (D.reverseHold or 0) - dt)

        if not sPressed then
            if D.brakeLatch and (stopped or not movingForward or v16 < 1.5) then
                D.reverseHold = 0.50
            end
            D.brakeLatch = false
            D.reverseLaunch = false
        elseif wPressed then
            D.brakeLatch = false
            D.reverseLaunch = false
            D.reverseHold = 0
        end

        if not wPressed then
            if D.revBrakeLatch and (stopped or not movingBackward or v16 > -1.5) then
                D.forwardHold = 0.50
            end
            D.revBrakeLatch = false
            D.forwardLaunch = false
        elseif sPressed then
            D.revBrakeLatch = false
            D.forwardLaunch = false
            D.forwardHold = 0
        end

        local rawBrk = 0
        if wPressed and sPressed then
            -- Both pedals: allow burnout at a stop, otherwise full brake.
            D.thr = stopped and clamp(v18, 0, 1) or 0
            rawBrk = clamp(max(forwardBrake, D.brake), 0, 1)
        elseif wPressed then
            if D.forwardLaunch or (D.forwardHold > 0 and not D.revBrakeLatch) then
                -- Committed to forward acceleration: stay in throttle without bouncing into brake.
                D.reversing = false
                D.forwardLaunch = true
                D.forwardHold = 0
                D.thr = clamp(v18, 0, 1)
                rawBrk = 0
                if D.gear <= 0 then self:gearDrive() end
            elseif D.revBrakeLatch or movingBackward then
                -- Moving backward (in any gear) or holding W after braking to a stop: full brake & hold.
                D.thr = 0
                rawBrk = clamp(forwardBrake, 0, 1)
                D.revBrakeLatch = true
            else
                -- Stopped or moving forward: drive forward.
                D.reversing = false
                D.forwardLaunch = true
                D.thr = clamp(v18, 0, 1)
                rawBrk = 0
                if D.gear <= 0 then self:gearDrive() end
            end
        elseif sPressed then
            if D.reverseLaunch or (D.reverseHold > 0 and not D.brakeLatch) then
                -- Committed to reverse acceleration: stay in throttle without bouncing into brake.
                D.reversing = true
                D.reverseLaunch = true
                D.reverseHold = 0
                D.thr = clamp(D.reverseGas ~= nil and D.reverseGas or D.brake, 0, 1)
                rawBrk = 0
                D.gear = 0
                if self.dt.gear > 0 then self.dt.gear, self.dt.req, self.dt.shifting = 0, 0, 0 end
            elseif D.brakeLatch or movingForward then
                -- Moving forward (in any gear) or holding S after braking to a stop: full brake & hold.
                D.thr = 0
                rawBrk = clamp(D.brake, 0, 1)
                D.brakeLatch = true
            else
                -- Stopped or moving backward: drive reverse.
                D.reversing = true
                D.reverseLaunch = true
                D.thr = clamp(D.reverseGas ~= nil and D.reverseGas or D.brake, 0, 1)
                rawBrk = 0
                D.gear = 0
                if self.dt.gear > 0 then self.dt.gear, self.dt.req, self.dt.shifting = 0, 0, 0 end
            end
        else
            D.thr = 0
            rawBrk = 0
        end

        D.brk = SimcadeBraking.updatePedal(self, D, rawBrk, dt)
        D.reverseGas = nil
    else
        if D.brake <= 0.6000000238418579 or D.gas >= 0.5 then
            if v16 > -2 then D.reversing = false end
        else
            D.reversing = true
        end
        local v7, v17, v21
        if v16 <= 0.5 and D.reversing then
            if v16 < -30 or v16 < -2 then
                v7 = D.brake; v17 = 0; v21 = v18
            else
                v7 = D.brake; v17 = v18; v21 = 0
            end
        else
            v7 = 0; v17 = v18; v21 = D.brake
        end
        local v19 = v7 + v7
        if v19 > 1 then v19 = 1 end
        D.brk = v21
        if v17 + 0.20000000298023224 < v19 then
            D.thr = v19
            D.gear = 0
            local d = self.dt
            if d.gear ~= 0 and (self.restState == 2 or self.speed < 0.1) then
                d.gear, d.req, d.shifting = 0, 0, 0
            end
        else
            if self.isDrag and self.gearShown == 1 then
                local v11 = D.thr
                local v23 = 1 * dt + v11
                local v12 = v11 - 1.5 * dt
                if v17 > v12 then v12 = v17 end
                if v23 < v12 then v12 = v23 end
                v17 = v12
                if D.gas > 0.6000000238418579 and self.rpm < self.c[124] + 1000 and 0.6000000238418579 > v12 then v17 = 0.6000000238418579 end
            end
            D.thr = clamp(v17, 0, 1)
            if D.gear <= 0 then self:gearDrive() end
        end
    end
    D.lamp = hold and brake or D.brk
    if hold then
        if not self.held then self.heldGear = D.gear end
        D.brk, D.gear, self.held = 1, 1, true
    elseif self.held then
        self.held = false
        if self.race then self:gearDrive() else self.drv.gear = self.manualBox and self.heldGear or -1 end
    end
    if self.clutchPedal < 1 then
        self:clutchSet(self.clutchPedal)
    elseif self.dt.mode == 3 then
        self.dt.mode = 0
    end
    if u16 ~= nil then
        local u = trunc(u16) % 65536
        self.steer.u16 = u
        local deg = u * (360 / 65536)
        if deg > 180 then deg = deg - 360 end
        self.drv.u16Deg = deg
    else
        self:steerKeyboard(steer, dt)
    end
end
function Chassis:gearDrive()
    self.drv.gear = self.manualBox and 2 or -1
end
local GRIND = 0.44999998807907104   -- flt_7A18C8
-- Selector positions follow the Red Fox order: 0 Park, 1 Reverse, 2 Neutral, 3 Drive.
function Chassis:setAutoRange(range)
    local d = self.dt
    local gear, request = 1, 1
    if range == 1 then
        gear, request = min(d.gear, 0), 0
    elseif range == 3 then
        gear, request = max(d.gear, 2), -1
    end
    self.autoRange = range
    self.drv.gear, self.drv.reversing = request, range == 1
    d.gear, d.req, d.shifting, d.grind = gear, gear, 0, nil
    if gear == 1 then d.state, d.clutch = 1, 0 end
end

function Chassis:setManual(on)
    if self.isDrag then self.userManual = on; on = true end
    self.manualBox = on
    local d = self.dt
    local D = self.drv
    D.brakeLatch = false
    D.revBrakeLatch = false
    D.reverseLaunch = false
    D.reverseHold = 0
    D.forwardLaunch, D.forwardHold = false, 0
    self.shiftHoldEnd = 0
    self.nextShiftTime = 0
    self.nextDownshiftTime = 0
    self.uphillHoldActive = false
    self._gradeHoldActive = false
    self._gradeHuntHoldUntil = 0
    if d then
        d.shifting = 0
        d.grind = nil
        d.mode = 0
        local isStopped = abs(self.forward or 0) * 3.6 <= (self.stoppedPhaseBrakeSpeed or 5)
        if on then
            if D.reversing or d.gear <= 0 then
                local rg = (d.gear <= 0) and d.gear or 0
                d.gear = rg
                d.req = rg
                D.gear = rg
                D.reversing = true
            else
                if d.gear < 2 and not self.classicAuto then d.gear = 2 end
                d.req = d.gear
                D.gear = d.gear
                D.reversing = false
            end
            if d.gear == 1 then
                d.state = 1
                d.clutch = 0
            elseif isStopped then
                d.state = 2
                d.clutch = 0
            else
                d.state = 3
                d.clutch = 1
            end
        elseif self.classicAuto then
            self:setAutoRange(d.gear <= 0 and 1 or (d.gear == 1 and 2 or 3))
        else
            D.gear = -1
            D.reversing = false
            if d.gear <= 1 then
                d.gear = 2
            end
            d.req = d.gear
            if isStopped then
                d.state = 2
                d.clutch = 0
            else
                d.state = 3
                d.clutch = 1
            end
        end
    else
        D.gear = on and 2 or -1
        D.reversing = false
    end
    if self.held then self.heldGear = D.gear end
end
function Chassis:shiftKey(up)
    if self.parking or self.servoState then return end
    if not self.manualBox then
        if self.classicAuto then
            local range = clamp(self.autoRange + (up and 1 or -1), 0, 3)
            if range == self.autoRange then return end
            if range <= 1 and self.speed * 3.6 > self.speedToReverse then return end
            if range == 3 and -self.forward * 3.6 > self.speedToReverse then return end
            self:setAutoRange(range)
        end
        return
    end
    local D = self.drv
    if up then
        if D.gear < self.top + 1 then D.gear = D.gear + 1 end
    elseif D.gear > (self.minGear or 0) then
        D.gear = D.gear - 1
    end
end
function Chassis:resetGear()
    if not self.manualBox or self.parking or self.servoState then return end
    self.drv.gear = 1
    if self.dt then
        self.dt.req = 1
        self.dt.gear = 1
        self.dt.shifting = 0
        self.dt.grind = nil
    end
end
function Chassis:clutchSet(a)
    local d = self.dt
    d.mode = 3
    if a > 0.949999988079071 then
        d.state = 3; d.clutch = 1
    elseif a < 0.05000000074505806 then
        d.state = 1; d.clutch = 0
    elseif a == 1 then
        d.state = 3; d.clutch = a
    elseif a == 0 then
        d.state = 1; d.clutch = a
    elseif a <= d.clutch then
        d.clutch = a; d.state = 0
    else
        d.state = 2; d.clutch = a
    end
end

function Chassis:slipAngle()
    local a = (self:atan16(self.R[0], self.R[1]) - self:atan16(self.v[0], self.v[1])) % 65536
    a = a * U16_RAD
    if a > PI then return a - 2 * PI end
    return a
end
function Chassis:atan16(x, y)
    local A = self.T.atan
    local a, b = abs(x), abs(y)
    local v
    if a <= b then
        if b <= a then
            v = b == 0 and 0 or 0x2000
        else
            v = 0x4000 - 32 * A[floor(a / b * 256)]     -- (trunc of what's never below 0)
        end
    else
        v = 32 * A[floor(b / a * 256)]
    end
    if x >= 0 and y >= 0 then return v end
    if x < 0 and y >= 0 then return 0x8000 - v end
    if x < 0 and y < 0 then return 0x8000 + v end
    return 0x10000 - v
end

function Chassis:updateWheelMounts()
    local c, WS = self.c, self.wheels
    for i = 0, 3 do
        local o = 4 + 12 * i
        local S = WS[i].su
        S.mount[0], S.mount[1], S.mount[2] = c[o] - c[0], c[o + 1] - c[1], c[o + 2] - c[2]

    end
end

function Chassis:updateDriverState(dt)
    local D, E, Dt = self.drv, self.eng, self.dt
    if self.gearShown ~= 1 then
        if D.thr <= D.brk or D.hbk >= D.thr then
            if D.thr < 0.5 then D.thr = 0 end
        else
            if D.brk < 0.5 then D.brk = 0 end
            if D.hbk < 0.5 then D.hbk = 0 end
        end
    end
    local v12 = D.thr
    if (self.throttleRiseTime or 0) > 0 then
        if v12 > E.T then
            v12 = E.T + (v12 - E.T) * (1 - math.exp(-dt / self.throttleRiseTime))
        end
    elseif v12 > 0.15000000596046448 then
        if v12 > 0 and Dt.gear > 1 and E.w * RPM < 3000 and 0.5 > v12 then v12 = 0.5 end
    else
        v12 = 0
    end
    self:steerWheels(self.steer.u16)
    if self.dt.shifting ~= 0 then
        v12 = 0
    end
    E.T = v12
    if D.want > 1 and D.want <= self.top + 1 then
        if Dt.want == 1 then Dt.timer = 0 end
        Dt.want = D.want
    end
    D.gear = self:gearRequest(D.gear, self.rpm, dt)
    D.want = Dt.want
    self.aero.T = v12
    self.aero.grounded = self.grounded * 0.25
end

function Chassis:gearRequest(a2, a3in, dt)
    if self._vphysMode and not self.manualBox and SimcadeGearbox and SimcadeGearbox.evaluateAutoShift then
        local d = self.dt
        if a2 == -1 and d.gear > 1 and d.shifting == 0 then
            local targetG = SimcadeGearbox.evaluateAutoShift(self, dt)
            if targetG then
                d.req = targetG
            end
            return a2
        elseif a2 == 0 and self.drv and self.drv.reversing and d.gear <= 0 then
            if d.shifting == 0 then
                local targetG = SimcadeGearbox.evaluateAutoShift(self, dt)
                if targetG then
                    d.req = targetG
                end
            end
            return a2
        end
    end
    local d, c = self.dt, self.c
    local a3 = a3in
    if a3 <= -1 then a3 = self.eng.w * RPM end
    local v6 = d.gear
    local manualReverseGear = self.manualBox and a2 <= 0 and a2 >= (self.minGear or 0)
    if a2 == -1 and not manualReverseGear then
        d.want = 1
        if v6 <= 1 then
            d.req = 2
            return a2
        end
        if d.state == 3 then
            local v24 = self:shiftUp(v6, false)
            local v27 = d.rDown + self.down[v6]
            local v7 = 0
            if v6 == 2 then
                v27 = self.eng.T >= 0.20000000298023224 and 0 or c[124] + 100
            else
                local up = self:shiftUp(v6 - 1, true)
                v7 = up * (self:gearRatio(v6) / max(0.05, self:gearRatio(v6 - 1))) - 200
            end
            if self._vphysMode then
                local wRad = max(0.20, self.wheels[2].radius or 0.34)
                local roadMS = max(0, self.forward)
                local curRatio = abs(self:gearRatio(v6))
                if curRatio < 0.01 then curRatio = 1 end
                local roadSyncRPM = (roadMS / wRad) * c[111] * curRatio * RPM
                local wheelSpinMS = max(
                    abs(self.wheels[2].w or 0) * wRad,
                    abs(self.wheels[0].w or 0) * (self.wheels[0].radius or wRad)
                )
                local isWheelSpinningWildly = (wheelSpinMS - roadMS) > 7.78 -- > 28 km/h spin excess
                if a3 < v24 or v6 >= self.top + 1 then
                    if v6 > 2 and (self.now or 0) >= (self.downshiftCooldownEnd or 0) then
                        local lowerRatio = abs(self:gearRatio(v6 - 1))
                        local projectedLowerRPM = (curRatio > 0.05) and (roadSyncRPM * (lowerRatio / curRatio)) or 99999
                        local safeDownCap = (c[125] or 8000) * 0.85
                        if roadSyncRPM <= v27 and roadSyncRPM <= v7 and projectedLowerRPM < safeDownCap then
                            d.req = v6 - 1
                            self.downshiftCooldownEnd = (self.now or 0) + 0.35
                        end
                    end
                elseif not isWheelSpinningWildly then
                    local ok = (self.grounded >= 2) and (roadSyncRPM >= v24 * 0.78)
                    if ok then d.req = v6 + 1 end
                end
            else
                if a3 < v24 or v6 >= self.top + 1 then
                    if a3 <= v27 and a3 <= v7 then d.req = v6 - 1 end
                else
                    local ok = true
                    for i = 0, 3 do
                        local W = self.wheels[i]
                        ok = ok and W.contact ~= 0 and W.slip < 4
                    end
                    if ok then d.req = v6 + 1 end
                end
            end
        end
        return a2
    end
    if a2 ~= -2 or manualReverseGear then
        if d.mode == 3 then
            local g = d.gear
            if a2 ~= g then
                if self.manualBox then
                    if d.state == 1 then
                        d.gear = a2
                        d.req = a2
                        d.shifting = 0
                    else
                        d.req = a2
                        d.shifting = 0
                    end
                    d.grind = nil
                else
                    if d.state ~= 1 then d.shifting = 0; d.grind = self.now; return g end
                    if d.shifting == 0 then d.shifting = 1; d.req = a2 end
                end
            end
            if self.manualBox and d.state == 1 then return a2 end
            return g
        end
        if a2 ~= d.gear and a2 ~= d.req then
            d.req = a2
        end
        return a2
    end
    return a2
end
function Chassis:shiftUp(g, low)
    local d, c = self.dt, self.c
    local v3 = g == 3 and d.r3 or d.rUp
    if low then v3 = self.ranges.up[0] end
    return min(self.isDrag and c[126] or c[125], v3 + self.up[g])
end
function Chassis:integrate(h, skipForces)
    local WS, E, d, c = self.wheels, self.eng, self.dt, self.c
    if self.lim516 ~= 0 then
        local v3 = self.perf * self.lim516 + 1.899999976158142
        local v28, v8 = self.R[2], nil
        if v28 < 0 then
            v28 = -v28
            if v28 > 0.3 then v28 = 0.3 end
            v8 = (1.1 - 1) / 0.3 * v28 + 1
        else
            if v28 > 0.3 then v28 = 0.3 end
            v8 = 1 - (1 - 0.9) / 0.3 * v28
        end
        local fwd = vec(self.R[0], self.R[1], self.R[2])
        if sqrt(self.v[0] * self.v[0] + self.v[1] * self.v[1]) * 2.2360248565673828 > v3 * v8 and dot(fwd, self.F) > 0 then
            local lo = toLocal(self.R, self.F)
            local v10 = -(5 * v28)
            if v10 > -0.5 then v10 = -0.5 end
            lo[0] = v10
            self.F = toWorld(self.R, lo)
        end
    end
    local v0, w, R = self.v, self.w, self.R
    local w0, w1, w2 = w[0], w[1], w[2]
    local r0, r1, r2 = R[0], R[1], R[2]
    local r3, r4, r5 = R[3], R[4], R[5]
    local r6, r7, r8 = R[6], R[7], R[8]
    local dP0, dP1, dP2 = self.F[0], self.F[1], self.F[2]
    local Tq = self.Tq
    local dL0, dL1, dL2 = Tq[0], Tq[1], Tq[2]
    local wl = sqrt(w0 * w0 + w1 * w1 + w2 * w2)
    local L = self.L
    if self.damp340 ~= 0 then
        local v19 = -curve(self.T.ang_damp, wl)
        dL0, dL1, dL2 = dL0 + v19 * L[0], dL1 + v19 * L[1], dL2 + v19 * L[2]
    end
    local gv = self.groundVel
    local gvx, gvy, gvz = gv and gv[0] or 0, gv and gv[1] or 0, gv and gv[2] or 0
    local rvx, rvy, rvz = v0[0] - gvx, v0[1] - gvy, v0[2] - gvz
    local v29 = sqrt(rvx * rvx + rvy * rvy + rvz * rvz) * 0.25
    local v21 = wl * 0.5
    if v21 > v29 then v29 = v21 end
    local dE = curve(self.T.rest, v29) <= self.E and -10 or 0.10000000149011612
    local P = self.P
    if self.damp336 ~= 0 then
        local a, b = -(2 * self.E), -(8 * self.E)
        dL0, dL1, dL2 = dL0 + a * L[0], dL1 + a * L[1], dL2 + a * L[2]
        local gm = self.m
        dP0 = dP0 + b * (P[0] - gm * gvx)
        dP1 = dP1 + b * (P[1] - gm * gvy)
        dP2 = dP2 + b * (P[2] - gm * gvz)
    end
    if self.limit512 ~= 0 then
        local lim = self.limit512
        local tq2 = Tq[2]
        if tq2 > lim then dL2 = lim elseif tq2 < -lim then dL2 = -lim end
    end
    local vphys = self._vphysMode
    for i = 0, 3 do
        local W = WS[i]
        local ww = W.w
        local a
        if W.contact == 0 then
            -- No contact patch to enforce rolling speed or react drive torque.
            W.avgT = 0
            if W.locked ~= 0 then
                ww, a = 0, 0
            else
                local driveImpulse = h * W.T
                if vphys and self.hardRPMLimit and d.gear ~= 1 then
                    local final = i < 2 and c[110] or c[111]
                    local ratio = self:gearRatio(d.gear) * final
                    if abs(ratio) > 0.001 and driveImpulse * ratio > 0 then
                        local mate = WS[i % 2 == 0 and i + 1 or i - 1]
                        local sync = 2 * E.w / ratio - mate.w
                        local remaining = (sync - ww) * W.I
                        driveImpulse = clamp(remaining, min(0, driveImpulse), max(0, driveImpulse))
                    end
                end
                local momentum = W.I * ww + driveImpulse - h * 0.002 * ww
                local brakeImpulse = h * (W.airBrakeTorque or 0)
                if momentum > brakeImpulse then momentum = momentum - brakeImpulse
                elseif momentum < -brakeImpulse then momentum = momentum + brakeImpulse
                else momentum = 0 end
                a = (momentum / W.I - ww) / h
            end
        else
            local wRad = W.radius
            local torque = (W.T - wRad * W.Fx + W.avgT) * 0.5
            W.avgT = torque
            a = W.locked ~= 0 and 0 or ((torque - 0.0020000000949949026 * ww) / W.I)
            if W.sliding == 0 then
                if h > 0.0010000000474974513 then a = a - W.slip / (wRad * h) end
            elseif W.locked == 0 and W.slip * (W.slip + h * a * wRad) < 0 then
                a = -W.slip / (wRad * h)
                W.avgT = 0
            end
        end
        W.ang = W.ang + h * ww
        W.w = ww + h * a
    end
    local dEng = ((self._solvedTq or self:torque()) + E.tload) / c[127]
    local dCl = 0
    if d.mode ~= 3 then
        local rCl = c[115] * self.clutchK
        if d.q > 0 then rCl = rCl * (d.q + 1) elseif d.q < 0 and d.req >= 3 then rCl = rCl * 0.3499999940395355 end
        dCl = rCl * self.T.clutch_dir[d.state]
    end
    local pos = self.pos
    pos[0] = pos[0] + h * v0[0]; P[0] = P[0] + h * dP0; L[0] = L[0] + h * dL0
    pos[1] = pos[1] + h * v0[1]; P[1] = P[1] + h * dP1; L[1] = L[1] + h * dL1
    pos[2] = pos[2] + h * v0[2]; P[2] = P[2] + h * dP2; L[2] = L[2] + h * dL2
    R[0] = r0 + h * (r2 * w1 - w2 * r1); R[1] = r1 + h * (w2 * r0 - r2 * w0); R[2] = r2 + h * (w0 * r1 - r0 * w1)
    R[3] = r3 + h * (r5 * w1 - w2 * r4); R[4] = r4 + h * (w2 * r3 - r5 * w0); R[5] = r5 + h * (w0 * r4 - r3 * w1)
    R[6] = r6 + h * (r8 * w1 - w2 * r7); R[7] = r7 + h * (w2 * r6 - r8 * w0); R[8] = r8 + h * (w0 * r7 - r6 * w1)
    self.E = self.E + h * dE
    E.w = E.w + h * dEng
    d.clutch = d.clutch + h * dCl
    self:post(h, skipForces)
end
function Chassis:postAllWheelsAndSusp(finalSub)
    local WS, c, R, pos, v, w = self.wheels, self.c, self.R, self.pos, self.v, self.w
    local r00, r01, r02 = R[0], R[1], R[2]
    local r10, r11, r12 = R[3], R[4], R[5]
    local r20, r21, r22 = R[6], R[7], R[8]
    local p0, p1, p2 = pos[0], pos[1], pos[2]
    local vx, vy, vz = v[0], v[1], v[2]
    local wx, wy, wz = w[0], w[1], w[2]
    local hasOmega = (wx ~= 0 or wy ~= 0 or wz ~= 0)
    local isDrag = self.isDrag
    local vphys = self._vphysMode
    local a0 = self.steer.a0
    if self._ca0A ~= a0 then
        self._ca0A = a0; local ca, sa = cos(a0), sin(a0); self._ca0, self._sa0 = ca, sa
        local M0 = WS[0].M; M0[0], M0[1], M0[2], M0[3] = ca, sa, -sa, ca
    end
    local a1 = self.steer.a1
    if self._ca1A ~= a1 then
        self._ca1A = a1; local ca, sa = cos(a1), sin(a1); self._ca1, self._sa1 = ca, sa
        local M1 = WS[1].M; M1[0], M1[1], M1[2], M1[3] = ca, sa, -sa, ca
    end
    local ca0, sa0, ca1, sa1 = self._ca0, self._sa0, self._ca1, self._sa1
    for i = 0, 3 do
        local W = WS[i]
        W.Fx = 0; W.Fy = 0; W.T = 0; W.slip = 0
        local ww = W.w
        if W.dir == 0 then
            if ww < 0 then ww = 0; W.w = 0 end
        elseif W.dir == 2 and ww > 0 then
            ww = 0; W.w = 0
        end
        W.dir = ww > 1e-5 and 0 or (ww >= -1e-5 and 1 or 2)
        local ang = W.ang
        if ang > 6.283185 then W.ang = ang - 6.283185 elseif ang < -6.283185 then W.ang = ang + 6.283185 end

        local S = W.su
        local row = S.row
        local mt = S.mount
        local mt0, mt1, mt2 = mt[0], mt[1], mt[2]
        local m0 = mt0 * r00 + mt1 * r10 + mt2 * r20
        local m1 = mt0 * r01 + mt1 * r11 + mt2 * r21
        local m2 = mt0 * r02 + mt1 * r12 + mt2 * r22
        local G = W.ground
        local n = G.n
        local nx, ny, nz = n[0], n[1], n[2]
        local flatN = (nx == 0 and ny == 0 and nz == 1)
        local surf = isDrag and 1 or G.s
        W.bumpZ = 0
        local upn = flatN and r22 or (r20 * nx + r21 * ny + r22 * nz)
        local g = -100
        if upn >= 0.0010000000474974513 then
            local gp = G.p
            if flatN then
                g = (gp[2] - m2 - p2) / upn
            else
                g = ((gp[0] - m0 - p0) * nx + (gp[1] - m1 - p1) * ny + (gp[2] - m2 - p2) * nz) / upn
            end
        else
            surf = 0
        end
        W.surface = surf
        W.n = n
        local v7 = g + W.radius
        local maxx, minx = c[row + 5], c[row + 6]
        local sx = v7 > maxx and maxx or (v7 < minx and minx or v7)
        S.x = sx
        local rx, ry, rz
        local wv = W.vel
        if hasOmega or not finalSub then
            rx = m0 + sx * r20
            ry = m1 + sx * r21
            rz = m2 + sx * r22
        end
        if hasOmega then
            if W._hasGVel then
                local gv = G.v
                wv[0] = (wy * rz - wz * ry) + vx - gv[0]
                wv[1] = (wz * rx - wx * rz) + vy - gv[1]
                wv[2] = (wx * ry - wy * rx) + vz - gv[2]
            else
                wv[0] = (wy * rz - wz * ry) + vx
                wv[1] = (wz * rx - wx * rz) + vy
                wv[2] = (wx * ry - wy * rx) + vz
            end
        elseif W._hasGVel then
            local gv = G.v
            wv[0], wv[1], wv[2] = vx - gv[0], vy - gv[1], vz - gv[2]
        else
            wv[0], wv[1], wv[2] = vx, vy, vz
        end
        local B = W.basis
        local ca, sa
        if i == 0 then ca, sa = ca0, sa0
        elseif i == 1 then ca, sa = ca1, sa1 end
        local b0, b1, b2
        if i < 2 then
            b0, b1, b2 = ca * r00 + sa * r10, ca * r01 + sa * r11, ca * r02 + sa * r12
        else
            b0, b1, b2 = r00, r01, r02
        end
        if vphys and surf ~= 0 then
            if flatN then
                local lx, ly = -b1, b0
                local lenSq = lx * lx + ly * ly
                if lenSq > 1e-6 then
                    local invLen = 1 / sqrt(lenSq)
                    lx, ly = lx * invLen, ly * invLen
                    B[0], B[1], B[2] = ly, -lx, 0
                    B[3], B[4], B[5] = lx, ly, 0
                    B[6], B[7], B[8] = 0, 0, 1
                else
                    B[0], B[1], B[2] = b0, b1, b2
                    if i < 2 then
                        B[3], B[4], B[5] = -sa * r00 + ca * r10, -sa * r01 + ca * r11, -sa * r02 + ca * r12
                    else
                        B[3], B[4], B[5] = r10, r11, r12
                    end
                    B[6], B[7], B[8] = r20, r21, r22
                end
            else
                local lx, ly, lz = ny * b2 - nz * b1, nz * b0 - nx * b2, nx * b1 - ny * b0
                local lenSq = lx * lx + ly * ly + lz * lz
                if lenSq > 1e-6 then
                    local invLen = 1 / sqrt(lenSq)
                    lx, ly, lz = lx * invLen, ly * invLen, lz * invLen
                    B[0], B[1], B[2] = ly * nz - lz * ny, lz * nx - lx * nz, lx * ny - ly * nx
                    B[3], B[4], B[5] = lx, ly, lz
                    B[6], B[7], B[8] = nx, ny, nz
                else
                    B[0], B[1], B[2] = b0, b1, b2
                    if i < 2 then
                        B[3], B[4], B[5] = -sa * r00 + ca * r10, -sa * r01 + ca * r11, -sa * r02 + ca * r12
                    else
                        B[3], B[4], B[5] = r10, r11, r12
                    end
                    B[6], B[7], B[8] = r20, r21, r22
                end
            end
        else
            B[0], B[1], B[2] = b0, b1, b2
            if i < 2 then
                B[3], B[4], B[5] = -sa * r00 + ca * r10, -sa * r01 + ca * r11, -sa * r02 + ca * r12
            else
                B[3], B[4], B[5] = r10, r11, r12
            end
            B[6], B[7], B[8] = r20, r21, r22
        end
        if not finalSub then
            local at = W.at
            at[0], at[1], at[2] = rx + p0, ry + p1, rz + p2
        end
    end
end

function Chassis:post(h, skipForces)
    self:postBody()
    self:postAllWheelsAndSusp(skipForces)
    self:postEngine(h)
    self:postDrive()
    if self._vphysMode and self.updateSimClutch then self:updateSimClutch(h) end
    self.stepH = h
    self:forces(skipForces)
end
function Chassis:postBody()
    local im = self.im
    local P, v = self.P, self.v
    local v0, v1, v2 = P[0] * im, P[1] * im, P[2] * im
    local sSq = v0 * v0 + v1 * v1 + v2 * v2
    if sSq > 40000 then
        local k = 200 / sqrt(sSq)
        P[0], P[1], P[2] = P[0] * k, P[1] * k, P[2] * k
        v0, v1, v2 = v0 * k, v1 * k, v2 * k
    end
    v[0], v[1], v[2] = v0, v1, v2
    local k = im * 0.10204081237316132
    local F, a = self.F, self.a
    a[0], a[1], a[2] = F[0] * k, F[1] * k, F[2] * k
    local I = self.mode344 ~= 0 and self.Iinv2 or self.Iinv
    local R, L = self.R, self.L
    local r0, r1, r2 = R[0], R[1], R[2]
    local r3, r4, r5 = R[3], R[4], R[5]
    local r6, r7, r8 = R[6], R[7], R[8]
    local L0, L1, L2 = L[0], L[1], L[2]
    local l0 = (L0 * r0 + L1 * r1 + L2 * r2) * I[0]
    local l1 = (L0 * r3 + L1 * r4 + L2 * r5) * I[1]
    local l2 = (L0 * r6 + L1 * r7 + L2 * r8) * I[2]
    local w0 = l0 * r0 + l1 * r3 + l2 * r6
    local w1 = l0 * r1 + l1 * r4 + l2 * r7
    local w2 = l0 * r2 + l1 * r5 + l2 * r8
    local wlSq = w0 * w0 + w1 * w1 + w2 * w2
    if wlSq > 62500 then
        local q = 250 / sqrt(wlSq)
        w0, w1, w2 = w0 * q, w1 * q, w2 * q
    end
    local w = self.w
    w[0], w[1], w[2] = w0, w1, w2
    if self.E < 0 then self.E = 0 end
    F[0], F[1], F[2] = 0, 0, 0
    local Tq = self.Tq
    Tq[0], Tq[1], Tq[2] = 0, 0, 0
    if not self._vphysMode and self.grav > 0 then self:addForce3(0, 0, -(self.m * self.grav), 0, 0, 0) end
end
function Chassis:addForce3(f0, f1, f2, r0, r1, r2)
    local l = sqrt(f0 * f0 + f1 * f1 + f2 * f2)
    if l > 1200 then
        local k = 1200 / l
        f0, f1, f2 = f0 * k, f1 * k, f2 * k
    end
    local F = self.F
    F[0] = F[0] + f0; F[1] = F[1] + f1; F[2] = F[2] + f2
    local t0, t1, t2 = r1 * f2 - r2 * f1, r2 * f0 - r0 * f2, r0 * f1 - r1 * f0
    local lt = sqrt(t0 * t0 + t1 * t1 + t2 * t2)
    if lt > 100 then
        local k = 100 / lt
        t0, t1, t2 = t0 * k, t1 * k, t2 * k
    end
    local Tq = self.Tq
    Tq[0] = Tq[0] + t0; Tq[1] = Tq[1] + t1; Tq[2] = Tq[2] + t2
end
function Chassis:postWheel(W)
    W.Fx = 0; W.Fy = 0; W.T = 0; W.slip = 0
    if W.dir == 0 then
        if W.w < 0 then W.w = 0 end
    elseif W.dir == 2 and W.w > 0 then
        W.w = 0
    end
    W.dir = W.w > 1e-5 and 0 or (W.w >= -1e-5 and 1 or 2)
    if W.ang > 6.283185 then W.ang = W.ang - 6.283185 elseif W.ang < -6.283185 then W.ang = W.ang + 6.283185 end
end
function Chassis:postSusp(W)
    local S, c, R = W.su, self.c, self.R
    local row = S.row
    local mt, pos = S.mount, self.pos
    local m0, m1, m2 = mt[0] * R[0] + mt[1] * R[3] + mt[2] * R[6], mt[0] * R[1] + mt[1] * R[4] + mt[2] * R[7], mt[0] * R[2] + mt[1] * R[5] + mt[2] * R[8]
    local G = W.ground
    local n = G.n
    W.surface = self.isDrag and 1 or G.s
    W.bumpZ = 0
    local upn = R[6] * n[0] + R[7] * n[1] + R[8] * n[2]
    local g = -100
    if upn >= 0.0010000000474974513 then
        g = ((G.p[0] - m0 - pos[0]) * n[0] + (G.p[1] - m1 - pos[1]) * n[1] + (G.p[2] - m2 - pos[2]) * n[2] + W.bumpZ * n[2]) / upn
    else
        W.surface = 0
    end
    W.n = n
    local v7 = g + W.radius
    local maxx, minx = c[row + 5], c[row + 6]
    if v7 <= maxx then
        if v7 < minx then
            S.x = minx
        else
            S.x = v7
        end
    else
        S.x = maxx
    end
    local sl = S["local"]
    sl[0], sl[1], sl[2] = mt[0], mt[1], mt[2] + S.x
    local ca, sa = 1, 0      -- cos(0), sin(0): a back wheel's
    if W.i == 0 then
        local a = self.steer.a0
        if self._ca0A ~= a then self._ca0A = a; self._ca0 = cos(a); self._sa0 = sin(a) end
        ca, sa = self._ca0, self._sa0
    elseif W.i == 1 then
        local a = self.steer.a1
        if self._ca1A ~= a then self._ca1A = a; self._ca1 = cos(a); self._sa1 = sin(a) end
        ca, sa = self._ca1, self._sa1
    end
    local M2 = W.M
    M2[0], M2[1], M2[2], M2[3] = ca, sa, -sa, ca
    local r0, r1, r2 = sl[0] * R[0] + sl[1] * R[3] + sl[2] * R[6], sl[0] * R[1] + sl[1] * R[4] + sl[2] * R[7], sl[0] * R[2] + sl[1] * R[5] + sl[2] * R[8]
    local w, v = self.w, self.v
    local gv = G.v
    local wv = W.vel
    if gv then
        wv[0] = (w[1] * r2 - w[2] * r1) + v[0] - gv[0]
        wv[1] = (w[2] * r0 - w[0] * r2) + v[1] - gv[1]
        wv[2] = (w[0] * r1 - w[1] * r0) + v[2] - gv[2]
    else
        wv[0], wv[1], wv[2] = (w[1] * r2 - w[2] * r1) + v[0], (w[2] * r0 - w[0] * r2) + v[1], (w[0] * r1 - w[1] * r0) + v[2]
    end
    local nsa = -sa
    local B = W.basis
    local b0, b1, b2 = ca * R[0] + sa * R[3], ca * R[1] + sa * R[4], ca * R[2] + sa * R[5]
    if self._vphysMode and W.surface ~= 0 then
        -- Resolve tire slip and friction in the road plane even when cargo
        -- rolls the chassis or the suspension has substantial vertical speed.
        local nx, ny, nz = n[0], n[1], n[2]
        local lx, ly, lz = ny * b2 - nz * b1, nz * b0 - nx * b2, nx * b1 - ny * b0
        local lenSq = lx * lx + ly * ly + lz * lz
        if lenSq > 1e-6 then
            local invLen = 1 / sqrt(lenSq)
            lx, ly, lz = lx * invLen, ly * invLen, lz * invLen
            B[0], B[1], B[2] = ly * nz - lz * ny, lz * nx - lx * nz, lx * ny - ly * nx
            B[3], B[4], B[5] = lx, ly, lz
            B[6], B[7], B[8] = nx, ny, nz
        else
            B[0], B[1], B[2] = b0, b1, b2
            B[3], B[4], B[5] = nsa * R[0] + ca * R[3], nsa * R[1] + ca * R[4], nsa * R[2] + ca * R[5]
            B[6], B[7], B[8] = R[6], R[7], R[8]
        end
    else
        B[0], B[1], B[2] = b0, b1, b2
        B[3], B[4], B[5] = nsa * R[0] + ca * R[3], nsa * R[1] + ca * R[4], nsa * R[2] + ca * R[5]
        B[6], B[7], B[8] = R[6], R[7], R[8]
    end
    local at = W.at
    at[0], at[1], at[2] = r0 + pos[0], r1 + pos[1], r2 + pos[2]
end
function Chassis:postEngine(h)
    self._solvedTq = nil
    local E, c = self.eng, self.c
    local lo = (self.ignitionRunning == false or self.physicalIdle) and 0 or (c[124] * RAD_RPM)
    local hi = c[126] * RAD_RPM
    if not self.hardRPMLimit and self.manualBox and self.dt.gear >= 2 and (self.clutchPedal or 1) > 0.5 then
        hi = max(hi, abs(self.dt.wt or 0))
    end
    E.w = clamp(E.w, lo, hi)
    local rpm = E.w * RPM
    self.lastTload = E.tload
    E.tload = 0
    local stepH = h or 0
    if (self.catchFlareTimer or 0) > 0 and stepH > 0 then
        self.catchFlareTimer = max(0, self.catchFlareTimer - stepH)
    end
    if self.physicalIdle then
        if self.ignitionRunning == false then
            self.stallTimer = 0
            self.stallGraceTimer = 0
            self.engineStalled = false
        elseif (self.stallGraceTimer or 0) > 0 then
            self.stallGraceTimer = max(0, self.stallGraceTimer - stepH)
            self.stallTimer = 0
        elseif stepH > 0 then
            local stallRPM = self.stallRPM or 450
            if rpm <= stallRPM then
                self.stallTimer = (self.stallTimer or 0) + stepH
                if self.stallTimer >= (self.stallDelay or 0.35) then
                    self.engineStalled = true
                end
            elseif rpm >= stallRPM + 120 then
                self.stallTimer = 0
            else
                self.stallTimer = max(0, (self.stallTimer or 0) - stepH * 0.5)
            end
        end
    else
        self.stallTimer = 0
        self.engineStalled = false
    end
    if self.isDiesel then
        -- Diesels use a smooth fuel-tapering governor instead of a spark-cut bounce limiter.
        E.lim = false
    else
        local dropRPM = self.revLimitDrop
        if dropRPM == nil then dropRPM = 0.012500000186264515 * c[126] end
        local delaySec = self.revLimitDelay or 0
        local now = self.now or 0
        if E.lim then
            local releaseRPM = max(c[124], c[126] - dropRPM)
            if now >= (E.limUntil or 0) and (rpm <= releaseRPM or ((E.T or 0) < 0.05 and rpm < c[126] - 50)) then
                E.lim = false
            end
        elseif rpm >= c[126] - 25 then
            E.lim = true
            E.limUntil = now + delaySec
        end
    end
end
function Chassis:gearRatio(g)
    return (self.gearRatioMap and self.gearRatioMap[g]) or self.c[116 + (g or 1)] or 0
end
function Chassis:shaftW()
    local c, W = self.c, self.wheels
    local v2 = 0
    local sp = c[112]
    if sp >= 0.0010000000474974513 then v2 = (W[3].w + W[2].w) * 0.5 * c[111] end
    if sp <= 0.9990000128746033 then
        v2 = v2 + (W[1].w + W[0].w) * 0.5 * c[110]
        if sp >= 0.0010000000474974513 then v2 = v2 * 0.5 end
    end
    return v2 * self:gearRatio(self.dt.gear)
end
function Chassis:stalled()
    if self.physicalIdle and self.manualBox and not self.useClutchLerpInManual then return false end
    return self.eng.T < 0.30000001192092896 and self.c[124] * RAD_RPM + 0.0001 >= self.dt.wt
end
function Chassis:matched() return abs(self.eng.w - self.dt.wt) <= 50 end
function Chassis:engineOffCoupled()
    local d = self.dt
    if self.ignitionRunning ~= false then return false end
    if self.clutchType ~= "friction" then return false end
    if d.gear == 1 or d.shifting ~= 0 then return false end
    if (self.clutchPedal or 1) <= 0.5 or (self._stepHandbrake or 0) > 0.5 then return false end
    if self.classicAuto and not self.manualBox and self.autoRange == 0 then return false end
    return self.manualBox or abs(d.wt or 0) > 5
end
function Chassis:postDrive()
    local d, E = self.dt, self.eng
    d.wt = self:shaftW()
    local offCoupled = self:engineOffCoupled()
    local st = d.state
    if d.mode == 3 then
        E.mult = 1
        if st == 1 and d.shifting ~= 0 then d.shifting = 0; d.gear = d.req end
    elseif st == 0 then
        E.mult = 0
        if d.clutch <= 0 then
            d.clutch = 0
            d.state = 1
            if d.shifting ~= 0 then
                self.shiftHoldEnd = (self.now or 0) + (self.shiftDelay or 0)
            else
                self.shiftHoldEnd = 0
            end
        end
    elseif st == 1 then
        E.mult = 0
        d.gear = d.req
        if d.shifting ~= 0 and d.gear ~= 1 and (self.clutchPedal or 1) > 0.5 and d.wt and d.wt > 0 then
            local minSync = self.ignitionRunning == false and 0 or (self.c[124] * RAD_RPM)
            local targetW = clamp(d.wt, minSync, self.c[126] * RAD_RPM)
            local syncRate = clamp((self.substepDt or 0.005) * 8, 0.015, 0.12)
            E.w = E.w + (targetW - E.w) * syncRate
        end
        if (self.now or 0) >= (self.shiftHoldEnd or 0) and (d.shifting ~= 0 or offCoupled or not self:stalled()) then
            d.state = 2
            d.shifting = 0
        end
    elseif st == 2 then
        E.mult = self._vphysMode and 1 or 0.699999988079071
        if d.shifting ~= 0 and d.gear ~= 1 and (self.clutchPedal or 1) > 0.5 and d.wt and d.wt > 0 then
            local minSync = self.ignitionRunning == false and 0 or (self.c[124] * RAD_RPM)
            local targetW = clamp(d.wt, minSync, self.c[126] * RAD_RPM)
            local syncRate = clamp((self.substepDt or 0.005) * 12, 0.02, 0.15)
            E.w = E.w + (targetW - E.w) * syncRate
        end
        if d.clutch >= 1 then
            d.clutch = 1
            if self:matched() or (self._vphysMode and d.gear ~= 1) then d.state = 3 end
        end
        if d.shifting == 0 and not offCoupled and self:stalled() and d.gear ~= 1 then d.state = 0 end
        if d.gear ~= d.req and d.shifting == 0 then
            d.shifting = 1; d.last = E.w * RPM; d.state = 0
        end
    elseif st == 3 then
        if d.q > 0 and d.qAt == 0 then d.qAt = self.now end
        E.mult = 1
        local g = d.gear
        local clutchSlipping = self._vphysMode and (self.simClutch or 0) > 0.01
        -- A manually engaged over-rev stays coupled so the wheels can force engine speed above the limiter.
        local manualOverRev = not self.hardRPMLimit and self.manualBox and g >= 2 and (self.clutchPedal or 1) > 0.5 and d.wt > self.c[126] * RAD_RPM
        if g ~= 1 and not offCoupled and self:stalled() then
            d.state = 0
        elseif g ~= 1 and abs(E.w - (d.wt or 0)) > 150 and (self.clutchPedal or 1) > 0.5 and not manualOverRev and not clutchSlipping and not offCoupled then
            d.state = 2
        end
        local tcUnlocked = (self.clutchType == "torque_converter") and not self.tcLocked
        if g ~= 1 and (self.clutchPedal or 1) > 0.5 and d.wt and d.wt > 0 and not tcUnlocked and not self.clutchSlippingOverload and (not clutchSlipping or d.wt > E.w) then
            local lockRate = clamp((self.substepDt or 0.005) * 16, 0.04, 0.18)
            local minW = (self.physicalIdle or self.ignitionRunning == false) and 0 or (self.c[124] * RAD_RPM)
            local maxW = not self.hardRPMLimit and self.manualBox and g >= 2 and max(self.c[126] * RAD_RPM, d.wt) or self.c[126] * RAD_RPM
            E.w = clamp(E.w + (d.wt - E.w) * lockRate, minW, maxW)
        end
        if g == d.req then
            d.shifting = 0
        else
            d.shifting = 1; d.last = E.w * RPM; d.state = 0
        end
    end
    if d.grind then
        E.mult = 0
        if self.now - d.grind >= GRIND then d.grind = nil end
    elseif d.state == 3 and d.qAt > 0 then
        local e = self.now - d.qAt
        local v = (0.5 - e) / 0.5 * d.q * 2
        if v > 0 then E.mult = v + 1 end
        if e >= 0.5 then d.qAt = 0; d.q = 0 end
    end
end
function Chassis:torque()
    local E = self.eng
    local rpm = E.w * RPM
    if self.ignitionRunning == false then
        -- Unpowered engine still resists rotation via compression and pumping drag.
        local drag = 0
        if rpm > 1 then
            local dragScale = self.physicalStarter and self.cranking and 0.65 or (self.dt.gear == 1 and 0.65 or 0.90)
            drag = -(curve(E.br, rpm) * clamp(rpm / 80, 0, 1) * dragScale)
        end
        local starterTq = 0
        if self.physicalStarter and self.cranking then
            local noLoadRPM = max(250, self.starterNoLoadRPM or 460)
            local speedFrac = clamp(1.0 - rpm / noLoadRPM, 0, 1)
            starterTq = (self.starterTorqueKNm or 0.055) * speedFrac
        end
        return starterTq + drag
    end
    local thrRaw = clamp(E.T or 0, 0, 1)
    local baseCurve = curve(E.tq, rpm) * E.perf
    local brakeCurve = curve(E.br, rpm)
    local manualOverRev = self.manualBox and self.dt.gear >= 2 and (self.clutchPedal or 1) > 0.5 and rpm > self.c[126]
    local useManifold = self.manifoldModel and E.manifoldCharge ~= nil
    local v18
    local brakeFrac
    if useManifold then
        local charge = E.lim and 0 or (E.mult * E.manifoldCharge)
        v18 = baseCurve * charge
        if self.isDiesel then
            local brakeFade = self.engineBrakeThrottleFade or 0.15
            brakeFrac = (E.lim or manualOverRev) and 1.0 or clamp(1.0 - thrRaw / brakeFade, 0, 1)
        else
            brakeFrac = (E.lim or manualOverRev) and 1.0 or clamp(E.manifoldVacuum or 0, 0, 1)
        end
    else
        local thrExp = self.throttleResponseExp or 1.0
        local a = (thrRaw > 0) and (thrExp == 1.0 and thrRaw or (thrRaw ^ thrExp)) or 0
        local thrLoad = a
        local rpmScale = self.isDiesel and 0 or (self.throttleRPMScale or 0)
        if a > 0 and rpmScale > 0 then
            local r = clamp(rpm / max(1000, self.c[125] or 7000), 0.05, 1.0)
            local s = rpmScale * r
            thrLoad = (a * (1.0 + s)) / (a + s)
        end
        local T = E.lim and 0 or E.mult * thrLoad
        v18 = baseCurve * T * (E.inductionTorqueRatio or 1.0)
        local brakeFade = self.engineBrakeThrottleFade or 0.15
        brakeFrac = (E.lim or manualOverRev) and 1.0 or clamp(1.0 - thrRaw / brakeFade, 0, 1)
    end
    if self.isDiesel and self.turboEnabled and (self.dieselSmokeLimit or 0) > 0 and v18 > 0 then
        -- Off-boost AFC / smoke-limit fuel cap: torque builds progressively as compressor boost arrives.
        local boostFrac = clamp((E.boostPSI or 0) / max(2.0, self.dieselSmokeBoostRefPSI or 8.0), 0, 1)
        local fuelCap = (1.0 - self.dieselSmokeLimit) + self.dieselSmokeLimit * boostFrac
        v18 = v18 * fuelCap
    end
    if self.isDiesel and v18 > 0 then
        local govWin = self.dieselGovernorWindow or 280
        local govStart = self.c[126] - govWin
        if rpm > govStart then
            local govFrac = clamp((self.c[126] - rpm) / max(50, govWin), 0.0, 1.0)
            v18 = v18 * govFrac
        end
    elseif self.uphillHoldActive and self.uphillRedlineHold == "soft" and not self.manualBox and self.dt.gear > 1 and self.dt.shifting == 0 and not E.lim and v18 > 0 then
        local softWin = self.revLimiterSoftWindow or 250
        local targetCap = self.c[125] or self.c[126]
        local softStart = targetCap - softWin
        if rpm > softStart then
            local softFrac = clamp((targetCap - rpm) / max(50, softWin), 0.0, 1.0)
            v18 = v18 * softFrac
        end
    elseif self.revLimiterSoft and not E.lim and v18 > 0 then
        local softWin = self.revLimiterSoftWindow or 250
        local softStart = self.c[126] - softWin
        if rpm > softStart then
            local softFrac = clamp((self.c[126] - rpm) / softWin, self.revLimiterSoftMinTorque or 0.05, 1.0)
            v18 = v18 * softFrac
        end
    end
    local jakeStage = self.jakeBrakeStage or 0
    local jakeActive = self.jakeBrakeEnabled and jakeStage > 0 and not manualOverRev
        and thrRaw < 0.05 and rpm >= (self.jakeBrakeMinRPM or ((self.c[124] or 700) + 300))
        and self.dt.gear ~= 1 and (self.clutchPedal or 1) > 0.5 and (self.simClutch or 0) < 0.5
    self.jakeBrakeActive = jakeActive
    local coastScale = self.isDiesel and (self.dieselCoastBrakeScale or 0.55) or 1.0
    local jakeMult = 1.0
    if jakeActive then
        local stageFrac = clamp(jakeStage / max(1, self.jakeBrakeStages or 3), 0, 1)
        coastScale = 1.0
        jakeMult = 1.0 + max(0, (self.jakeBrakeMult or 3.2) - 1.0) * stageFrac
    end
    local v14 = -(brakeCurve * brakeFrac * coastScale * jakeMult)
    if manualOverRev then
        local overRevRatio = rpm / max(1, self.c[126])
        v14 = v14 * clamp(overRevRatio * overRevRatio, 1, 16)
    end
    if E.lim then v14 = (v14 < -0.01) and (v14 * 2) or -0.02 end
    local parasitic = (not E.lim and (E.parasiticDragKNm or 0) > 0) and E.parasiticDragKNm or 0
    local idleTq = 0
    if self.physicalIdle and not E.lim and not manualOverRev then
        local idleRPM = self.c[124] or 800
        if rpm < idleRPM + 150 then
            local stallRPM = self.stallRPM or 450
            local idleBand = clamp((idleRPM + 150 - rpm) / 150, 0, 1)
            local combustEff = clamp((rpm - stallRPM * 0.65) / max(100, idleRPM - stallRPM * 0.65), 0, 1)
            if (self.catchFlareTimer or 0) > 0 then
                combustEff = max(combustEff, 0.85)
            end
            local baseHold = (brakeCurve * brakeFrac + parasitic) * idleBand * combustEff
            local targetIdle = ((self.catchFlareTimer or 0) > 0) and (idleRPM * 1.08) or idleRPM
            local errRPM = max(0, targetIdle - rpm)
            local maxPropTq = baseCurve * (self.idleMaxTorqueFrac or 0.28)
            local safeSlope = (self.c[127] * 18.0) * RAD_RPM
            local gainSlope = baseCurve * (self.idleControllerGain or 0.0018)
            local propTq = min(maxPropTq, errRPM * min(gainSlope, safeSlope)) * combustEff
            idleTq = baseHold + propTq
        end
    end
    local r = v14 + v18 + idleTq - parasitic
    if r < 0 and not E.lim then
        return r * (self.dt.gear == 1 and 0.65 or 0.90)
    end
    return r
end
-- An open differential delivers equal shaft torque. The LSD starts from that
-- same split and couples wheel speeds separately in applyDiffCoupling.
local function driveAxle(a, b, torque, locked)
    local share = 0.5
    if locked then
        if a.contact == 0 and b.contact ~= 0 then share = 0
        elseif b.contact == 0 and a.contact ~= 0 then share = 1 end
    end
    if a.locked == 0 then a.T = a.T + share * torque end
    if b.locked == 0 then b.T = b.T + (1 - share) * torque end
end

function Chassis:evalTorqueCurve(rpm)
    return curve(self.eng.tq, rpm)
end
function Chassis:drive(a2, outMult)
    if self.ignitionRunning == false and not (self.physicalStarter and self.cranking) and not self:engineOffCoupled() then return end
    local d, c, WS, E = self.dt, self.c, self.wheels, self.eng
    if d.gear == 1 then return end
    local allowBrakeClutchLoad = (self.clutchType == "torque_converter") or (self.physicalIdle and self.manualBox and not self.useClutchLerpInManual)
    if self._vphysMode and not allowBrakeClutchLoad and (self.drv.brk or 0) > 0.05 and (self.drv.thr or 0) < 0.05 and a2 > 0 then return end
    local wheelMult = outMult or 1.0
    local prevTq = d.torque
    local nextTq = prevTq + a2 * wheelMult
    d.torque = nextTq
    local eff = self.drivetrainEff or 1.0
    local prevOut = (prevTq >= 0) and (prevTq * eff) or (prevTq * (2.0 - eff))
    local nextOut = (nextTq >= 0) and (nextTq * eff) or (nextTq * (2.0 - eff))
    local v3 = self.grounded * 0.25
    if 0.25 > v3 then v3 = 0.25 end
    E.tload = E.tload - v3 * a2
    local v5 = (nextOut - prevOut) * self:gearRatio(d.gear)
    local v18 = (1 - d.split) * c[110] * v5
    local v7 = v5 * (c[111] * d.split)
    local frontActive = (not self._vphysMode) or ((1 - d.split) >= 0.001)
    local rearActive  = (not self._vphysMode) or (d.split >= 0.001)
    if frontActive then
        self.frontAxleTorque = (self.frontAxleTorque or 0) + v18
        local locked = (self.effFrontDiffType or self.frontDiffType) == 2 or self.frontLockActive or self.allLockActive
        driveAxle(WS[0], WS[1], v18, locked)
    end
    if rearActive then
        self.rearAxleTorque = (self.rearAxleTorque or 0) + v7
        local locked = (self.effRearDiffType or self.rearDiffType) == 2 or self.rearLockActive or self.allLockActive
        driveAxle(WS[2], WS[3], v7, locked)
    end
end
local function biasAxle(h, thrIn, a, b, axleTq, diffType, stiffness, lockRatio, preloadNm, accelLock, decelLock, forceLock)
    if a.locked ~= 0 and b.locked ~= 0 then return false end
    if not forceLock and diffType == 0 then return false end
    local dw = a.w - b.w
    local absDw = abs(dw)
    if absDw <= 0.05 then return false end
    local iw = max(0.0005, min(a.I, b.I))
    local syncTq = 0.45 * iw * absDw / h
    local absAxle = abs(axleTq or 0)
    local lockCap
    if forceLock or diffType == 2 then
        lockCap = max(absAxle, 2.5)
    else
        local avgW = (a.w + b.w) * 0.5
        local isAccel = ((axleTq or 0) * avgW >= 0) and (thrIn > 0.05)
        local maxR = clamp(lockRatio or 0.75, 0.05, 1.0)
        local ramp = clamp(isAccel and (accelLock ~= nil and accelLock or (maxR * 0.65)) or (decelLock ~= nil and decelLock or (maxR * 0.25)), 0, 1)
        local preloadKNm = max(0, preloadNm or 15) * 0.001
        local packTq = preloadKNm + 0.5 * ramp * absAxle
        local viscCap = maxR * max(absAxle * 0.5, 0.35)
        local viscTq = min(viscCap, max(0, stiffness or 3.0) * 0.006 * absDw)
        lockCap = packTq + viscTq
    end
    local biasMag = min(lockCap, syncTq)
    if biasMag <= 0.0001 then return false end
    local bias = (dw > 0) and biasMag or -biasMag
    if a.locked == 0 then a.T = a.T - bias end
    if b.locked == 0 then b.T = b.T + bias end
    return true
end

function Chassis:applyDiffTorque()
    if not self._vphysMode then return end
    local c, W = self.c, self.wheels
    local h = max(0.002, self.substepDt or 0.0044)
    local thrIn = (self.drv and self.drv.thr) or 0
    self.frontDiffTorqueLocked = false
    self.rearDiffTorqueLocked = false

    local rType = self.effRearDiffType ~= nil and self.effRearDiffType or self.rearDiffType
    local fType = self.effFrontDiffType ~= nil and self.effFrontDiffType or self.frontDiffType
    if c[112] >= 0.001 then
        self.rearDiffTorqueLocked = biasAxle(
            h, thrIn, W[2], W[3], self.rearAxleTorque, rType,
            self.effRearStiffness or self.rearLSDStiffness,
            self.effRearLockRatio or self.rearLSDLockRatio,
            self.rearLSDPreload, self.rearLSDAccelLock, self.rearLSDDecelLock,
            self.rearLockActive or self.allLockActive
        )
    end
    if c[112] <= 0.999 then
        self.frontDiffTorqueLocked = biasAxle(
            h, thrIn, W[0], W[1], self.frontAxleTorque, fType,
            self.effFrontStiffness or self.frontLSDStiffness,
            self.effFrontLockRatio or self.frontLSDLockRatio,
            self.frontLSDPreload, self.frontLSDAccelLock, self.frontLSDDecelLock,
            self.frontLockActive or self.allLockActive
        )
    end
    local cType = self.allLockActive and 2 or (self.centerDiffType or 0)
    if cType > 0 and c[112] >= 0.001 and c[112] <= 0.999 then
        local wF = (W[0].w + W[1].w) * 0.5
        local wR = (W[2].w + W[3].w) * 0.5
        local dwC = wF - wR
        local absDwC = abs(dwC)
        if absDwC > 0.05 then
            local totalAxle = abs(self.frontAxleTorque or 0) + abs(self.rearAxleTorque or 0)
            local iwC = max(0.0005, min(W[0].I, W[2].I))
            local syncWheel = 0.45 * iwC * absDwC / h
            local capAxle = (cType == 2) and max(totalAxle, 3.0) or (0.02 + 0.35 * totalAxle + 0.015 * absDwC)
            local biasWheel = ((dwC > 0) and 1 or -1) * min(capAxle * 0.5, syncWheel)
            if W[0].locked == 0 then W[0].T = W[0].T - biasWheel end
            if W[1].locked == 0 then W[1].T = W[1].T - biasWheel end
            if W[2].locked == 0 then W[2].T = W[2].T + biasWheel end
            if W[3].locked == 0 then W[3].T = W[3].T + biasWheel end
        end
    end
end
function Chassis:clutchDrive(cachedTq)
    local d, E = self.dt, self.eng
    d.torque = 0
    self.frontAxleTorque = 0
    self.rearAxleTorque = 0
    self.tcLocked = false
    self.clutchSlippingOverload = false
    if self.clutchType == "torque_converter" then
        local shiftCut = (d.gear == 1 or d.state == 0 or d.state == 1 or d.shifting ~= 0 or (self.clutchPedal == 0) or ((self._stepHandbrake or 0) > 0.5))
        if shiftCut then return end
        local rpmNow = E.w * RPM
        local wt = d.wt or 0
        local wE = max(0, E.w)
        local sr = (wE > 1.0) and clamp(wt / wE, 0, 1.25) or 0
        local thrIn = (self.drv and self.drv.thr) or 0
        local brkIn = (self.drv and self.drv.brk) or 0
        local speedKmh = abs(self.forward or 0) * 3.6
        local stopSpd = self.stoppedPhaseBrakeSpeed or 5
        local lockupGearOk = (d.gear >= 3) or ((self.top or 1) <= 1 and d.gear >= 2)
        if (self.tcLockupRPM or 0) > 0 and lockupGearOk and rpmNow >= self.tcLockupRPM and sr >= 0.88 and thrIn > 0.05 and brkIn < 0.05 then
            self.tcLocked = true
            self:drive(cachedTq or self:torque(), 1.0)
            return
        end
        local kFactor = self.tcKFactor or 0.000004
        local h = max(0.002, self.substepDt or 0.0044)
        local Ie = max(0.00003, self.c[127] or 0.00012)
        local tq = cachedTq or self:torque()
        if wE >= wt then
            local fluidFrac = max(0, 1.0 - sr * sr)
            local tImp = kFactor * (wE * wE) * fluidFrac
            local creepScale = clamp((self.tcCreepTorque or 0.12) / 0.12, 0.0, 3.0)
            if thrIn <= 0.05 then
                tImp = tImp * creepScale
                if brkIn > 0.05 and speedKmh <= stopSpd then
                    tImp = tImp * 0.25
                end
            end
            local maxStable = max(0, tq) + 0.85 * (Ie / h) * (wE - wt)
            tImp = min(tImp, max(0, maxStable))
            local coupRatio = max(0.1, self.tcCouplingRatio or 0.85)
            local tcMult = 1.0 + max(0, (self.tcMaxMult or 2.0) - 1.0) * clamp(1.0 - sr / coupRatio, 0, 1)
            self:drive(tImp, tcMult)
        else
            local tCoast = -kFactor * (wt * wt - wE * wE) * 0.55
            local brakeLimit = -max(abs(tq), curve(E.br, max(80, wt * RPM)))
            local minStable = min(0, tq) + 0.85 * (Ie / h) * (wE - wt)
            tCoast = max(tCoast, brakeLimit, minStable)
            self:drive(tCoast, 1.0)
        end
        return
    end
    local engage = ((d.state == 0 or d.state == 2) and d.clutch or (d.state == 3 and 1.0 or 0.0)) * (1.0 - (self.simClutch or 0))
    if engage <= 0.001 then return end
    local capKNm = (self.clutchMaxTorqueKNm or 0.45) * engage
    local tq = cachedTq or self:torque()
    if self.ignitionRunning == false and not self.cranking and (d.wt or 0) < -1 then
        -- Rolling opposite to the selected gear still faces engine compression drag.
        local oppRPM = abs(d.wt) * RPM
        local oppDrag = curve(E.br, oppRPM) * clamp(oppRPM / 80, 0, 1) * 0.90
        self:drive(min(oppDrag, capKNm))
        return
    end
    if d.state == 0 or d.state == 2 or engage < 0.99 then
        local h = max(0.002, self.substepDt or 0.0044)
        local Ie = max(0.00003, self.c[127] or 0.00012)
        local invI = 1.0 / Ie
        if self.grounded == 0 and d.gear ~= 1 then
            local c, WS = self.c, self.wheels
            if c[112] <= 0.9990000128746033 then
                local rF = self:gearRatio(d.gear) * c[110]
                invI = invI + (1 - d.split) / max(0.0005, WS[0].I) * rF * rF * 0.5
            end
            if c[112] >= 0.0010000000474974513 then
                local rR = self:gearRatio(d.gear) * c[111]
                invI = invI + d.split / max(0.0005, WS[2].I) * rR * rR * 0.5
            end
        end
        local Ieff = 1.0 / invI
        local dw = E.w - (d.wt or 0)
        local tMatch = tq + 0.85 * (Ieff / h) * dw
        local clTq = clamp(tMatch, -capKNm, capKNm)
        if abs(tMatch) > capKNm then self.clutchSlippingOverload = true end
        self:drive(clTq)
    elseif d.state == 3 then
        local clTq = clamp(tq, -capKNm, capKNm)
        if abs(tq) > capKNm then self.clutchSlippingOverload = true end
        self:drive(clTq)
    end
end
function Chassis:shaftAcc()
    local c, W, d = self.c, self.wheels, self.dt
    local v1 = 0
    local W0, W1, W2, W3 = W[0], W[1], W[2], W[3]
    if c[112] >= 0.0010000000474974513 then v1 = ((W3.T - W3.radius * W3.Fx) / W3.I + (W2.T - W2.radius * W2.Fx) / W2.I) * 0.5 * c[111] end
    if c[112] <= 0.9990000128746033 then
        v1 = v1 + ((W1.T - W1.radius * W1.Fx) / W1.I + (W0.T - W0.radius * W0.Fx) / W0.I) * 0.5 * c[110]
        if c[112] >= 0.0010000000474974513 then v1 = v1 * 0.5 end
    end
    return v1 * self:gearRatio(d.gear)
end
local function correctDriveClamped(self, d, delta, minDrive, maxDrive, clutchCap)
    local target = d.torque + delta
    if target > maxDrive or target < minDrive then
        if abs(target) > clutchCap then self.clutchSlippingOverload = true end
        delta = clamp(target, minDrive, maxDrive) - d.torque
    end
    self:drive(delta)
end

function Chassis:solve(finalSub)
    local WS, R = self.wheels, self.R
    local tq = self:torque()
    self._solvedTq = tq
    self:clutchDrive(tq)
    self:applyDiffTorque()
    local W0, W1, W2, W3 = WS[0], WS[1], WS[2], WS[3]
    self:tire(W0); self:tire(W1); self:tire(W2); self:tire(W3)
    local d, c, E = self.dt, self.c, self.eng
    -- Gear selection can persist while the automatic clutch is open or slipping at low RPM.
    local tcUnlocked = (self.clutchType == "torque_converter") and not self.tcLocked
    local lock = (not tcUnlocked) and d.state == 3 and ((self._vphysMode and (self.simClutch or 0) <= 0.01) or (not self._vphysMode and (self.simClutch or 0) < 1))
    -- When tireSimSpecs is active and clutch is not locked (or on finalSub whose F/Tq/T are never integrated), skip the 5-iter loop.
    if (not finalSub) and (lock or not self.tireSimSpecs) then
        local F = self.F
        local Fl0, Fl1 = F[0] * R[0] + F[1] * R[1] + F[2] * R[2], F[0] * R[3] + F[1] * R[4] + F[2] * R[5]
        local im = self.im
        local inv = (self.mode344 ~= 0 and self.Iinv2 or self.Iinv)[2]
        local M, m = W0.M, W0.su.mount
        local a0, b0, c0, d0, p0, q0, y0, r0, I0, free0 = M[0], M[1], M[2], M[3], m[0], m[1], W0.Fy, W0.radius, W0.I, W0.contact ~= 0 and W0.sliding == 0 and W0.locked == 0
        M, m = W1.M, W1.su.mount
        local a1, b1, c1, d1, p1, q1, y1, r1, I1, free1 = M[0], M[1], M[2], M[3], m[0], m[1], W1.Fy, W1.radius, W1.I, W1.contact ~= 0 and W1.sliding == 0 and W1.locked == 0
        m = W2.su.mount
        local p2, q2, y2, r2, I2, free2 = m[0], m[1], W2.Fy, W2.radius, W2.I, W2.contact ~= 0 and W2.sliding == 0 and W2.locked == 0
        m = W3.su.mount
        local p3, q3, y3, r3, I3, free3 = m[0], m[1], W3.Fy, W3.radius, W3.I, W3.contact ~= 0 and W3.sliding == 0 and W3.locked == 0
        -- The locked-clutch iterations reconcile wheel reactions within dry-plate clutch torque capacity.
        local clutchCap = (self.clutchType == "torque_converter") and math.huge or (self.clutchMaxTorqueKNm or 0.45)
        local minDrive, maxDrive = -clutchCap, clutchCap
        if (self.clutchEngageTime or 0) > 0 then
            minDrive = max(minDrive, min(0, d.torque))
            maxDrive = min(maxDrive, max(0, d.torque))
        end
        local c127, v7 = c[127], nil
        local gr = 0
        local c110, c111, c112 = c[110], c[111], c[112]
        local hasRear = c112 >= 0.0010000000474974513
        local hasFront = c112 <= 0.9990000128746033
        local eff = self.drivetrainEff or 1.0
        local fastDrive = false
        local v3, fActive, rActive, fShare0, fShare1, rShare2, rShare3
        local fLockedW0, fLockedW1, rLockedW2, rLockedW3
        local canBrakeGate, invV7
        local invI0, invI1, invI2, invI3 = 1 / I0, 1 / I1, 1 / I2, 1 / I3
        local k0, k1, k2, k3 = I0 / r0, I1 / r1, I2 / r2, I3 / r3
        local omr0, omr1, omr2, omr3 = 1 - r0, 1 - r1, 1 - r2, 1 - r3
        if lock then
            v7 = 1 / c127
            gr = self:gearRatio(d.gear)
            if hasFront then
                local v9 = gr * c110
                v7 = v7 + eff * (1 - d.split) * invI0 * v9 * v9 * 0.5
            end
            if hasRear then
                local v10 = gr * c111
                v7 = v7 + eff * d.split * invI2 * v10 * v10 * 0.5
            end
            invV7 = 1 / v7
            if self.drive == Chassis.drive and d.gear ~= 1
                and (self.ignitionRunning ~= false or (self.physicalStarter and self.cranking) or self:engineOffCoupled()) then
                fastDrive = true
                local allowBrakeClutchLoad = (self.clutchType == "torque_converter") or (self.physicalIdle and self.manualBox and not self.useClutchLerpInManual)
                canBrakeGate = self._vphysMode and not allowBrakeClutchLoad and (self.drv.brk or 0) > 0.05 and (self.drv.thr or 0) < 0.05
                v3 = self.grounded * 0.25
                if v3 < 0.25 then v3 = 0.25 end
                fActive = (not self._vphysMode) or ((1 - d.split) >= 0.001)
                rActive = (not self._vphysMode) or (d.split >= 0.001)
                local fLock = (self.effFrontDiffType or self.frontDiffType) == 2 or self.frontLockActive or self.allLockActive
                local rLock = (self.effRearDiffType or self.rearDiffType) == 2 or self.rearLockActive or self.allLockActive
                fShare0 = (fLock and W0.contact == 0 and W1.contact ~= 0) and 0 or ((fLock and W1.contact == 0 and W0.contact ~= 0) and 1 or 0.5)
                fShare1 = 1 - fShare0
                rShare2 = (rLock and W2.contact == 0 and W3.contact ~= 0) and 0 or ((rLock and W3.contact == 0 and W2.contact ~= 0) and 1 or 0.5)
                rShare3 = 1 - rShare2
                fLockedW0, fLockedW1 = W0.locked == 0, W1.locked == 0
                rLockedW2, rLockedW3 = W2.locked == 0, W3.locked == 0
            end
        end
        local invC127 = 1 / c127
        local splitFront = (1 - d.split) * c110
        local splitRear = d.split * c111
        local effRev = 2.0 - eff
        local x0, x1, x2, x3 = W0.Fx, W1.Fx, W2.Fx, W3.Fx
        local T0, T1, T2, T3 = W0.T, W1.T, W2.T, W3.T
        local dTorque = d.torque
        local eTload = E.tload
        local fAxleTq = self.frontAxleTorque or 0
        local rAxleTq = self.rearAxleTorque or 0
        local baseSx = Fl0 + c0 * y0 + c1 * y1
        local baseSy = Fl1 + d0 * y0 + d1 * y1 + y2 + y3
        local baseST = (d0 * y0) * p0 - (c0 * y0) * q0 + (d1 * y1) * p1 - (c1 * y1) * q1 + y2 * p2 + y3 * p3
        local t0c = b0 * p0 - a0 * q0
        local t1c = b1 * p1 - a1 * q1
        local ka0, kc0, kt0 = a0 * k0, c0 * k0, (p0 * c0 - q0 * a0) * k0
        local ka1, kc1, kt1 = a1 * k1, c1 * k1, (p1 * c1 - q1 * a1) * k1
        local kt2, kt3 = -q2 * k2, -q3 * k3
        local kEng = lock and (invC127 * invV7) or 0
        local f0c, f1c, f0r, f1r, r2c, r3c, r2r, r3r = 0, 0, 0, 0, 0, 0, 0, 0
        if lock then
            local halfScale = (hasFront and hasRear) and 0.25 or 0.5
            local fg = halfScale * c110 * gr * invV7
            local rg = halfScale * c111 * gr * invV7
            if hasFront then
                f0c = fg * invI0; f1c = fg * invI1
                f0r = f0c * r0; f1r = f1c * r1
            end
            if hasRear then
                r2c = rg * invI2; r3c = rg * invI3
                r2r = r2c * r2; r3r = r3c * r3
            end
        end
        local fGr0 = fActive and (splitFront * gr * fShare0) or 0
        local fGr1 = fActive and (splitFront * gr * fShare1) or 0
        local rGr2 = rActive and (splitRear * gr * rShare2) or 0
        local rGr3 = rActive and (splitRear * gr * rShare3) or 0
        local fSplitGr = fActive and (splitFront * gr) or 0
        local rSplitGr = rActive and (splitRear * gr) or 0
        for _ = 1, 5 do
            local v43 = (baseSx + a0 * x0 + a1 * x1 + x2 + x3) * im
            local v44 = (baseSy + b0 * x0 + b1 * x1) * im
            local v42 = (baseST + t0c * x0 + t1c * x1 - q2 * x2 - q3 * x3) * inv
            if free0 then x0 = T0 + omr0 * x0 - (v43 * ka0 + v44 * kc0 + v42 * kt0) end
            if free1 then x1 = T1 + omr1 * x1 - (v43 * ka1 + v44 * kc1 + v42 * kt1) end
            if free2 then x2 = T2 + omr2 * x2 - (v43 * k2 + v42 * kt2) end
            if free3 then x3 = T3 + omr3 * x3 - (v43 * k3 + v42 * kt3) end
            if lock then
                local delta = (tq + eTload) * kEng
                if hasFront then delta = delta - (T0 * f0c - x0 * f0r + T1 * f1c - x1 * f1r) end
                if hasRear then delta = delta - (T2 * r2c - x2 * r2r + T3 * r3c - x3 * r3r) end
                if fastDrive then
                    local target = dTorque + delta
                    if target > maxDrive or target < minDrive then
                        if target > clutchCap or target < -clutchCap then self.clutchSlippingOverload = true end
                        delta = (target < minDrive and minDrive or (target > maxDrive and maxDrive or target)) - dTorque
                    end
                    if not (canBrakeGate and delta > 0) then
                        local prevTq = dTorque
                        local nextTq = prevTq + delta
                        dTorque = nextTq
                        local dOut = (nextTq >= 0 and (nextTq * eff) or (nextTq * effRev))
                            - (prevTq >= 0 and (prevTq * eff) or (prevTq * effRev))
                        eTload = eTload - v3 * delta
                        if fActive then
                            fAxleTq = fAxleTq + dOut * fSplitGr
                            if fLockedW0 then T0 = T0 + dOut * fGr0 end
                            if fLockedW1 then T1 = T1 + dOut * fGr1 end
                        end
                        if rActive then
                            rAxleTq = rAxleTq + dOut * rSplitGr
                            if rLockedW2 then T2 = T2 + dOut * rGr2 end
                            if rLockedW3 then T3 = T3 + dOut * rGr3 end
                        end
                    end
                else
                    W0.Fx, W1.Fx, W2.Fx, W3.Fx = x0, x1, x2, x3
                    W0.T, W1.T, W2.T, W3.T = T0, T1, T2, T3
                    d.torque, E.tload = dTorque, eTload
                    correctDriveClamped(self, d, delta, minDrive, maxDrive, clutchCap)
                    T0, T1, T2, T3 = W0.T, W1.T, W2.T, W3.T
                    dTorque, eTload = d.torque, E.tload
                end
            end
        end
        W0.Fx, W1.Fx, W2.Fx, W3.Fx = x0, x1, x2, x3
        if fastDrive then
            W0.T, W1.T, W2.T, W3.T = T0, T1, T2, T3
            d.torque, E.tload = dTorque, eTload
            self.frontAxleTorque, self.rearAxleTorque = fAxleTq, rAxleTq
        end
    end
    if self.tireSimSpecs then
        local coasting = (self.drv.thr or 0) < 0.05 and (self.drv.brk or 0) < 0.05 and (self.drv.hbk or 0) < 0.05
        for i = 0, 3 do
            local W = WS[i]
            if W.contact ~= 0 then
                local safeRad = W.radius > 0.15 and W.radius or 0.15
                local pLon = W.peakLongKN or W.load or 0
                if pLon < 0 then pLon = 0 end
                if coasting and W.T < 0 then
                    local maxCoastT = pLon * 0.85 * safeRad
                    if W.T < -maxCoastT then W.T = -maxCoastT end
                end
                if W.sliding == 0 and W.locked == 0 then
                    local fx = W.T / safeRad
                    W.Fx = fx < -pLon and -pLon or (fx > pLon and pLon or fx)
                end
            end
        end
    end
    self:ellipse(W0, finalSub); self:ellipse(W1, finalSub); self:ellipse(W2, finalSub); self:ellipse(W3, finalSub)
    self:afterSolve(tq)
end
local function coupleAxle(a, b, diffType, stiffness, lockRatio, forceLock)
    if a.locked ~= 0 and b.locked ~= 0 then return false end
    local wa, wb = a.w, b.w
    local isLocked = false
    if forceLock or diffType == 2 then
        local avg = (wa + wb) * 0.5
        a.w = avg
        b.w = avg
        isLocked = true
    elseif (diffType == nil or diffType == 1) then
        local dw = abs(wa - wb)
        if dw > 0.2 then
            local stiffScale = clamp((stiffness or 3.0) * 0.08, 0.05, 1.0)
            local maxRatio = clamp(lockRatio or 0.90, 0.05, 1.0)
            local lockFrac = clamp((dw - 0.2) * stiffScale, 0, maxRatio)
            local avg = (wa + wb) * 0.5
            -- Torque transfer in applyDiffTorque does the main LSD work; keep light residual damping for low-tick stability.
            local dampFrac = lockFrac * 0.20
            a.w = wa + (avg - wa) * dampFrac
            b.w = wb + (avg - wb) * dampFrac
            isLocked = lockFrac > 0.05
        end
    end
    return isLocked
end

function Chassis:applyDiffCoupling()
    local c, W = self.c, self.wheels
    self.frontDiffLocked = false
    self.rearDiffLocked = false
    local rType = self.effRearDiffType ~= nil and self.effRearDiffType or self.rearDiffType
    local fType = self.effFrontDiffType ~= nil and self.effFrontDiffType or self.frontDiffType
    if c[112] >= 0.001 then
        self.rearDiffLocked = coupleAxle(W[2], W[3], rType, self.effRearStiffness or self.rearLSDStiffness, self.effRearLockRatio or self.rearLSDLockRatio, self.rearLockActive or self.allLockActive) or (self.rearDiffTorqueLocked == true)
    end
    if c[112] <= 0.999 then
        self.frontDiffLocked = coupleAxle(W[0], W[1], fType, self.effFrontStiffness or self.frontLSDStiffness, self.effFrontLockRatio or self.frontLSDLockRatio, self.frontLockActive or self.allLockActive) or (self.frontDiffTorqueLocked == true)
    end
    local cType = self.allLockActive and 2 or (self.centerDiffType or 0)
    if cType > 0 and c[112] >= 0.001 and c[112] <= 0.999 then
        local wF = (W[0].w + W[1].w) * 0.5
        local wR = (W[2].w + W[3].w) * 0.5
        local cLock = (cType == 2) and 1.0 or (clamp(abs(wF - wR) * 0.15, 0, 0.85) * 0.25)
        if cLock > 0.01 then
            local wAvg = (wF + wR) * 0.5
            local dF = (wAvg - wF) * cLock
            local dR = (wAvg - wR) * cLock
            W[0].w = W[0].w + dF; W[1].w = W[1].w + dF
            W[2].w = W[2].w + dR; W[3].w = W[3].w + dR
        end
    end
end
function Chassis:afterSolve(cachedTq)
    if self._vphysMode then
        self:applyDiffCoupling()
    end
    if self.ignitionRunning == false then
        if self:engineOffCoupled() then
            local d, E = self.dt, self.eng
            if self._vphysMode and d.gear ~= 1 and self.grounded > 0 and d.state == 3 and (self.simClutch or 0) <= 0.01 then
                local wt = self:shaftW()
                if wt > 0 then
                    local dw = wt - E.w
                    local lim = max(abs(cachedTq or self:torque()), curve(E.br, max(80, wt * RPM)))
                    E.tload = E.tload + clamp(dw * self.c[113] * 0.65, -lim, lim)
                end
            end
        elseif not (self.physicalStarter and self.cranking) then
            self.eng.tload = 0
        end
        return
    end
    local d, E = self.dt, self.eng
    local tq = cachedTq
    if self._vphysMode and d.gear ~= 1 and self.grounded > 0 then
        local dw = self:shaftW() - E.w
        tq = tq or self:torque()
        local lim = abs(tq)
        local tcUnlocked = (self.clutchType == "torque_converter") and not self.tcLocked
        if d.state == 3 and (self.simClutch or 0) <= 0.01 and not tcUnlocked and not self.clutchSlippingOverload then
            E.tload = E.tload + clamp(dw * self.c[113] * 0.65, -lim, lim)
            if E.T > 0.1 then
                E.tload = clamp(E.tload, -lim * 0.95, lim)
            end
        end
    elseif d.gear ~= 1 and d.state == 3 and self.grounded > 0 then
        local dw = self:shaftW() - E.w
        local avgSlip = self:avgUsage()
        if E.T <= 0.20000000298023224 or avgSlip <= 0.10000000149011612 then
            tq = tq or self:torque()
            local lim = abs(tq)
            E.tload = E.tload + clamp(dw * self.c[113], -lim, lim)
        else
            d.wt = E.w
            self:syncWheels(E.w)
        end
    end
    if d.gear == 2 then
        local v6, v7 = tq or self:torque(), E.tload
        local v12 = v7 + v6
        if v7 < v6 and v12 < 0 then E.tload = E.tload + points(self.T.first_gear, v12) * self.T.firstGearLoadScale * v12 end
    end
end
function Chassis:avgUsage()
    local c, W = self.c, self.wheels
    local v2, v4 = 0, 0
    if c[112] >= 0.0010000000474974513 then v2 = W[3].slip364 + W[2].slip364; v4 = 2 end
    if c[112] <= 0.9990000128746033 then v4 = v4 + 2; v2 = v2 + W[1].slip364 + W[0].slip364 end
    return v2 / v4
end
function Chassis:syncWheels(a2)
    local c, W, d = self.c, self.wheels, self.dt
    local v3 = a2 - self:shaftW()
    local fast = (not self._vphysMode) and (self.speed * MPH < 40)
    self.frontDiffLocked = false
    self.rearDiffLocked = false
    local function one(a, b, final, diffType, stiffness, lockRatio, forceLock)
        local k = v3 / (self:gearRatio(d.gear) * final)
        local l = fast or (a.contact == 0 and b.contact == 0)
        local wa, wb = a.w, b.w
        local isLocked = false
        if l or forceLock or diffType == 2 then
            wb = (wb + wa) * 0.5; wa = wb
            isLocked = true
        elseif (diffType == nil or diffType == 1) then
            local dw = abs(wa - wb)
            if dw > 0.4 then
                local stiffScale = clamp((stiffness or 3.0) * 0.06, 0.05, 1.0)
                local maxRatio = clamp(lockRatio or 0.90, 0.05, 1.0)
                local lockFrac = clamp((dw - 0.4) * stiffScale, 0, maxRatio)
                local avg = (wa + wb) * 0.5
                wa = wa + (avg - wa) * lockFrac
                wb = wb + (avg - wb) * lockFrac
                isLocked = lockFrac > 0.05
            end
        end
        a.w = wa + k; b.w = wb + k
        return isLocked
    end
    local rType = self.effRearDiffType ~= nil and self.effRearDiffType or self.rearDiffType
    local fType = self.effFrontDiffType ~= nil and self.effFrontDiffType or self.frontDiffType
    if c[112] >= 0.0010000000474974513 then
        self.rearDiffLocked = one(W[2], W[3], c[111], rType, self.effRearStiffness or self.rearLSDStiffness, self.effRearLockRatio or self.rearLSDLockRatio, self.rearLockActive or self.allLockActive)
    end
    if c[112] <= 0.9990000128746033 then
        self.frontDiffLocked = one(W[0], W[1], c[110], fType, self.effFrontStiffness or self.frontLSDStiffness, self.effFrontLockRatio or self.frontLSDLockRatio, self.frontLockActive or self.allLockActive)
    end
    local cType = self.allLockActive and 2 or (self.centerDiffType or 0)
    if cType > 0 and c[112] >= 0.001 and c[112] <= 0.999 then
        local wF = (W[0].w + W[1].w) * 0.5
        local wR = (W[2].w + W[3].w) * 0.5
        local cLock = (cType == 2) and 1.0 or clamp(abs(wF - wR) * 0.15, 0, 0.85)
        if cLock > 0.01 then
            local wAvg = (wF + wR) * 0.5
            local dF = (wAvg - wF) * cLock
            local dR = (wAvg - wR) * cLock
            W[0].w = W[0].w + dF; W[1].w = W[1].w + dF
            W[2].w = W[2].w + dR; W[3].w = W[3].w + dR
        end
    end
end
function Chassis:aeroForces()
    Aero.applyForces(self)
end
function Chassis:forces(finalSub)
    if not finalSub then self:aeroForces() end
    local WS, loads = self.wheels, self.vphysLoadsKN
    for i = 0, 3 do
        local W = WS[i]
        local extLoad = loads and loads[i] or 0
        local gndS = (W.ground and W.ground.s) or 0
        if gndS ~= 0 and extLoad > 0.001 then
            W.contact = 1
            W.load = extLoad
            if not finalSub then
                local pt, at, n = W.pt, W.at, W.n
                local nr = -W.radius
                pt[0], pt[1], pt[2] = at[0] + n[0] * nr, at[1] + n[1] * nr, at[2] + n[2] * nr
            end
        else
            W.contact = 0
            W.load = 0
        end
    end
    self:solve(finalSub)
    if finalSub then return end
    local pos, F, Tq = self.pos, self.F, self.Tq
    local shiftX = self.comShiftX_m or 0
    local ox = pos[0] + self.R[0] * shiftX
    local oy = pos[1] + self.R[1] * shiftX
    local oz = pos[2] + self.R[2] * shiftX
    for i = 0, 3 do
        local W = WS[i]
        if W.contact ~= 0 then
            local wf, pt = W.world, W.pt
            local f0, f1, f2 = wf[0], wf[1], wf[2]
            local lsq = f0 * f0 + f1 * f1 + f2 * f2
            if lsq > 1440000 then
                local k = 1200 / sqrt(lsq)
                f0, f1, f2 = f0 * k, f1 * k, f2 * k
            end
            F[0], F[1], F[2] = F[0] + f0, F[1] + f1, F[2] + f2
            local rx, ry, rz = pt[0] - ox, pt[1] - oy, pt[2] - oz
            local t0, t1, t2 = ry * f2 - rz * f1, rz * f0 - rx * f2, rx * f1 - ry * f0
            local ltsq = t0 * t0 + t1 * t1 + t2 * t2
            if ltsq > 10000 then
                local k = 100 / sqrt(ltsq)
                t0, t1, t2 = t0 * k, t1 * k, t2 * k
            end
            Tq[0], Tq[1], Tq[2] = Tq[0] + t0, Tq[1] + t1, Tq[2] + t2
        end
    end
    if self.grounded < 2 then
        local wv = self.w
        local wl = sqrt(wv[0] * wv[0] + wv[1] * wv[1] + wv[2] * wv[2])
        if wl < 0.5 then
            local k = -(points(self.T.yaw_damp, wl) * self.m)
            Tq[0] = Tq[0] + k * wv[0]; Tq[1] = Tq[1] + k * wv[1]; Tq[2] = Tq[2] + k * wv[2]
        end
    end
end
local SAME_N, SAME_D = 0.995, 0.02     -- the same plane again: its normal within about 6 degrees, a point's depth within 2 cm. The chip's
local NEAR, IN = 0.0010000000474974513, 9.999999747378752e-06    -- 0x5B3D20: touching within 0.001, into it within 0.00001
local px, py, pz, ps = {}, {}, {}, {}
function Chassis:orthonormalize()
    local R = self.R
    local a0, a1, a2 = R[0], R[1], R[2]
    local l = sqrt(a0 * a0 + a1 * a1 + a2 * a2)
    if l == 0 then l = 1 end
    a0, a1, a2 = a0 / l, a1 / l, a2 / l
    local r3, r4, r5 = R[3], R[4], R[5]
    local c0, c1, c2 = a1 * r5 - a2 * r4, a2 * r3 - a0 * r5, a0 * r4 - a1 * r3
    l = sqrt(c0 * c0 + c1 * c1 + c2 * c2)
    if l == 0 then l = 1 end
    c0, c1, c2 = c0 / l, c1 / l, c2 / l
    local b0, b1, b2 = c1 * a2 - c2 * a1, c2 * a0 - c0 * a2, c0 * a1 - c1 * a0
    R[0], R[1], R[2], R[3], R[4], R[5], R[6], R[7], R[8] = a0, a1, a2, b0, b1, b2, c0, c1, c2
end
function Chassis:state(dt)
    local R, c = self.R, self.c
    local v = self.cg
    local g0, g1, g2 = v[0] * R[0] + v[1] * R[3] + v[2] * R[6], v[0] * R[1] + v[1] * R[4] + v[2] * R[7], v[0] * R[2] + v[1] * R[5] + v[2] * R[8]
    local Ps, pos = self.Pos, self.pos
    Ps[0], Ps[1], Ps[2] = pos[0] - g0, pos[1] - g1, pos[2] - g2
    local gv = self.groundVel
    local rvx, rvy, rvz = self.v[0] - (gv and gv[0] or 0), self.v[1] - (gv and gv[1] or 0), self.v[2] - (gv and gv[2] or 0)
    self.speed = sqrt(rvx * rvx + rvy * rvy + rvz * rvz)
    self.forward = rvz * R[2] + rvy * R[1] + rvx * R[0]
    local g = 0
    do local i = 0
        if self.wheels[i].contact ~= 0 then g = g + 1 end
    end
    do local i = 1
        if self.wheels[i].contact ~= 0 then g = g + 1 end
    end
    do local i = 2
        if self.wheels[i].contact ~= 0 then g = g + 1 end
    end
    do local i = 3
        if self.wheels[i].contact ~= 0 then g = g + 1 end
    end
    self.grounded = g
    local d, E = self.dt, self.eng
    self.throttleOut = E.lim and 0 or E.mult * E.T
    self.gearShown = d.gear
    self.engaged = d.state == 3 and 1 or 0
    self.clutchTq = d.torque
    local v50 = E.w * RPM
    local shifting = d.shifting ~= 0 and d.req > 2
    local v65 = -(shifting and 15 or 2.5) / c[127]
    if dt > 0 and (v50 - self.rpm) / dt < v65 then
        local v53 = self.rpm + v65 * dt
        if v53 >= v50 then v50 = v53 end
    end
    self.rpm = self.rpm * 0.44999998807907104 + v50 * 0.550000011920929
    self:keepState()
end
function Chassis:keepState()
    local R, v, w = self.R, self.v, self.w
    local S = self.ps
    if not S then S = { up = vec(0, 0, 0) }; self.ps = S end
    S.h, S.v0, S.v1, S.w2 = self:atan16(R[0], R[1]), v[0], v[1], w[2]
    S.up[0], S.up[1], S.up[2] = R[6], R[7], R[8]
end
function Chassis:carryFrom(old)
    do local i = 0
        local W, O = self.wheels[i], old.wheels[i]
        W.ground, W.n, W.surface, W.contact = O.ground, O.n, O.surface, O.contact
        W.w, W.ang = O.w, O.ang
        W.su.x = O.su.x
    end
    do local i = 1
        local W, O = self.wheels[i], old.wheels[i]
        W.ground, W.n, W.surface, W.contact = O.ground, O.n, O.surface, O.contact
        W.w, W.ang = O.w, O.ang
        W.su.x = O.su.x
    end
    do local i = 2
        local W, O = self.wheels[i], old.wheels[i]
        W.ground, W.n, W.surface, W.contact = O.ground, O.n, O.surface, O.contact
        W.w, W.ang = O.w, O.ang
        W.su.x = O.su.x
    end
    do local i = 3
        local W, O = self.wheels[i], old.wheels[i]
        W.ground, W.n, W.surface, W.contact = O.ground, O.n, O.surface, O.contact
        W.w, W.ang = O.w, O.ang
        W.su.x = O.su.x
    end
    self.eng.w = old.eng.w
    self.eng.lim = old.eng.lim
    self.eng.limUntil = old.eng.limUntil and (old.eng.limUntil - old.now + self.now) or 0
    self.eng.parasiticDragKNm = old.eng.parasiticDragKNm or 0
    self.eng.manifoldCharge = old.eng.manifoldCharge
    self.eng.manifoldVacuum = old.eng.manifoldVacuum
    self.eng.manifoldPSI = old.eng.manifoldPSI or 0
    self.stallTimer = old.stallTimer or 0
    self.stallGraceTimer = old.stallGraceTimer or 0
    self.catchFlareTimer = old.catchFlareTimer or 0
    self.engineStalled = old.engineStalled == true
    self.jakeBrakeStage = old.jakeBrakeStage or 0
    local d, od = self.dt, old.dt
    d.gear, d.req, d.state, d.clutch, d.shifting, d.timer, d.want, d.last = od.gear, od.req, od.state, od.clutch, od.shifting, od.timer, od.want, od.last
    self.drv.gear, self.drv.want, self.drv.u16Deg = old.drv.gear, old.drv.want, old.drv.u16Deg
    self.steer.a0, self.steer.a1, self.steer.u16 = old.steer.a0, old.steer.a1, old.steer.u16
    self.held, self.parked = old.held, old.parked
    self.stepH = old.stepH
    self.yawAcc, self.yawPrev = old.yawAcc, old.yawPrev
    self.manualBox, self.clutchPedal, self.heldGear = old.manualBox, old.clutchPedal, old.heldGear
    self.autoRange = old.autoRange
    d.mode, d.grind = od.mode, od.grind and od.grind - old.now + self.now
end
M.Chassis = Chassis
return M

    local Chassis = Phys.Chassis

    function Chassis:cycleJakeBrake()
        if not self.jakeBrakeEnabled then return nil end
        local stages = max(1, self.jakeBrakeStages or 3)
        self.jakeBrakeStage = ((self.jakeBrakeStage or 0) + 1) % (stages + 1)
        return self.jakeBrakeStage
    end

    function Chassis:toggleDiffLock(which)
        if self.diffLockCfg and self.diffLockCfg.toggleable == false then return nil end
        if which == "front" then
            self.frontLockActive = not self.frontLockActive
            return self.frontLockActive
        elseif which == "rear" then
            self.rearLockActive = not self.rearLockActive
            return self.rearLockActive
        elseif which == "all" then
            self.allLockActive = not self.allLockActive
            return self.allLockActive
        end
        return nil
    end

    local function evalActiveDiffAxle(self, cfg, dlc, thr, spdKmh, steerAmt, isFront, wA, wB)
        local manualOn = self.allLockActive or (isFront and self.frontLockActive) or ((not isFront) and self.rearLockActive)
        if manualOn then
            local t = isFront and (dlc.frontType or 2) or (dlc.rearType or 2)
            local s = isFront and (dlc.frontStiffness or 30.0) or (dlc.rearStiffness or 30.0)
            local r = isFront and (dlc.frontRatio or 0.95) or (dlc.rearRatio or 0.95)
            return t, s, r, r * 100
        end
        local targeted = cfg and cfg.enabled and ((isFront and cfg.TargetFrontAxle) or ((not isFront) and cfg.TargetRearAxle))
        if not targeted then
            local baseT = isFront and self.frontDiffType or self.rearDiffType
            local baseS = isFront and self.frontLSDStiffness or self.rearLSDStiffness
            local baseR = isFront and self.frontLSDLockRatio or self.rearLSDLockRatio
            return baseT, baseS, baseR, 0
        end
        local minR = isFront and (cfg.FrontMinLockRatio or 10) or (cfg.RearMinLockRatio or 15)
        local maxR = isFront and (cfg.FrontMaxLockRatio or 95) or (cfg.RearMaxLockRatio or 40)
        local minS = isFront and (cfg.FrontMinStiffness or 1.0) or (cfg.RearMinStiffness or 5.0)
        local maxS = isFront and (cfg.FrontMaxStiffness or 30.0) or (cfg.RearMaxStiffness or 20.0)
        local slipRPM = abs((wA and wA.w or 0) - (wB and wB.w or 0)) * RAD_TO_RPM
        local declSpd = max(5, cfg.SteerDeclineSpeed or 40)
        local spdFactor = clamp((declSpd - spdKmh) / declSpd, 0, 1)
        local lockReduction = 1.0 - (steerAmt * spdFactor * 0.85)
        local lockRange = max(1, maxR - minR)
        local lockPct = (minR + thr * lockRange + slipRPM * (lockRange * 0.1) * thr * (cfg.SlipScale or 2.0) * 0.5) * lockReduction
        local dynThresh = (cfg.SlipThreshold or 12.0) * (1 + steerAmt * 2.0)
        if not (cfg.BypassCapsOnSlip and slipRPM >= dynThresh) then
            local cap = 0
            if thr > 0 then
                if thr < 0.2 then cap = (thr / 0.2) * 25
                elseif thr < 0.6 then cap = 25 + ((thr - 0.2) / 0.4) * 15
                else cap = 40 + ((thr - 0.6) / 0.4) * max(0, maxR - 40) end
            end
            lockPct = min(lockPct, cap)
        end
        lockPct = clamp(lockPct, minR, maxR)
        local stiffness = minS + ((lockPct - minR) / lockRange) * (maxS - minS)
        local activeT = isFront and (cfg.FrontActiveType or 1) or (cfg.RearActiveType or 1)
        return activeT, stiffness, lockPct / 100, lockPct
    end

    local EMPTY_DIFF_LOCK_CFG = {}
    function Chassis:computeActiveDiff(controls)
        local cfg = self.activeDiffCfg
        local dlc = self.diffLockCfg or EMPTY_DIFF_LOCK_CFG
        local thr = clamp((self.drv and self.drv.thr) or (controls and controls.gas) or 0, 0, 1)
        local spdKmh = abs(self.forward or 0) * 3.6
        local steerAmt = clamp(abs((controls and controls.steer) or 0), 0, 1)

        local W = self.wheels
        self.effFrontDiffType, self.effFrontStiffness, self.effFrontLockRatio, self.frontActiveLockPct = evalActiveDiffAxle(self, cfg, dlc, thr, spdKmh, steerAmt, true,  W[0], W[1])
        self.effRearDiffType,  self.effRearStiffness,  self.effRearLockRatio,  self.rearActiveLockPct  = evalActiveDiffAxle(self, cfg, dlc, thr, spdKmh, steerAmt, false, W[2], W[3])
    end

    local sin, atan, atan2, asin = math.sin, math.atan, math.atan2, math.asin
    local function getWheelPeakGrip(self, W, spec)
        local wsMult = W._wsMult or 1.0
        local tireLoadKN = (W.load or 0) * wsMult
        if tireLoadKN < 0 then tireLoadKN = 0 end
        local aeroCfg = self.aeroCfg
        local aeroAdd = 0
        if aeroCfg then
            local factor = (W.i < 2) and (aeroCfg.FrontAeroFactor or 0) or (aeroCfg.RearAeroFactor or 0)
            if factor > 0 then
                local mphDiv100 = abs(self.forward or 0) * 0.0223693629
                aeroAdd = factor * ((self.m or 1450) * 0.004905) * (mphDiv100 * mphDiv100)
            end
        end
        local Fz = tireLoadKN + aeroAdd
        W.effectiveLoadKN = Fz
        if Fz <= 0.001 or (W.contact or 0) == 0 then
            return 0, 0
        end
        local surfId = W.surface or 1
        local matType = W.ground and W.ground.matType
        local surfLat, surfLon
        if W._surfSpec == spec and W._surfId == surfId and W._surfMat == matType then
            surfLat, surfLon = W._surfLat, W._surfLon
        else
            surfLat, surfLon = Tires.computeSurfaceMult(spec, surfId, matType)
            W._surfSpec, W._surfId, W._surfMat = spec, surfId, matType
            W._surfLat, W._surfLon = surfLat, surfLon
            W._surfRR = Tires.computeRollingResistanceMult(spec, surfId, matType)
        end
        local nom = spec.nominalLoadKN or 3.68
        if nom < 0.5 then nom = 0.5 end
        local loadRatio = Fz / nom
        if loadRatio < 0.15 then loadRatio = 0.15 elseif loadRatio > 2.8 then loadRatio = 2.8 end
        local loadSens = 1.0 - (spec.LoadSens or 0.12) * (loadRatio - 1.0)
        if loadSens < 0.65 then loadSens = 0.65 elseif loadSens > 1.20 then loadSens = 1.20 end
        local baseGrip = Fz * spec.CoF * loadSens * (W.grip or 1) * (W.perf or 1)
        return baseGrip * surfLat * spec.latScale, baseGrip * surfLon * spec.longScale
    end

    -- The axle points left on both sides. Rotate static camber about forward, then
    -- rotate the axle about the rearward-tilted steering pivot (Rodrigues' formula).
    function Chassis:updateAlignment(W, steer)
        local wheel = W.alignment or {}
        local right = W.i == 1 or W.i == 2
        local camber = ((wheel.camberDeg or 0) + (W.travelCamberDeg or 0)) * (right and 1 or -1)
        if (wheel.legacySteeringCamber or 0) ~= 0 then
            local limit = max(0.01, self.steerCfg.Lock or self.steerCfg.maxAngle or 38)
            camber = camber - wheel.legacySteeringCamber * clamp(math.deg(steer) / limit, -1, 1)
        end
        W.hubCamberDeg = camber
        local c, k = math.rad(camber), math.rad(wheel.casterDeg or 0)
        local cy, cz = math.cos(c), math.sin(c)
        local px, pz = -math.sin(k), math.cos(k)
        local cs, sn = math.cos(steer), math.sin(steer)
        local d = pz * cz * (1 - cs)
        W.axleX = -pz * cy * sn + px * d
        W.axleY = cy * cs - px * cz * sn
        W.axleZ = cz * cs + px * cy * sn + pz * d
        if (W.beamRollDeg or 0) ~= 0 then
            local roll = math.rad(W.beamRollDeg)
            local cr, sr = math.cos(roll), math.sin(roll)
            W.axleY, W.axleZ = cr * W.axleY - sr * W.axleZ, sr * W.axleY + cr * W.axleZ
        end
    end

    function Chassis:tire(W)
        W.roadCamberRad, W.camberFy = 0, 0
        if not self.tireSimSpecs then
            W.Fx, W.Fy, W.slip, W.slipX, W.slipY = 0, 0, 0, 0, 0
            return
        end
        if W.contact == 0 then
            W.Fx, W.Fy, W.peakLatKN, W.peakLongKN = 0, 0, 0, 0
            W.slip, W.slip364, W.slipX, W.slipY = 0, 0, 0, 0
            W.slipAngleRad, W.slipRatio, W.sliding, W.absActive = 0, 0, 0, false
            local brakes = self.braking
            local service = W.i < 2 and brakes.frontBrakeKNm or brakes.rearBrakeKNm
            W.airBrakeTorque = service * self.drv.brk + brakes.handbrake.torqueKNm * W.hbk
            W.locked = (W.hbk > 0.1 and brakes.handbrake.lockWheels) and 1 or 0
            if W.locked ~= 0 then W.w = 0 end
            return
        end
        local b, vel = W.basis, W.vel
        local vFwd = vel[0] * b[0] + vel[1] * b[1] + vel[2] * b[2]
        local vLat = vel[0] * b[3] + vel[1] * b[4] + vel[2] * b[5]
        W._vFwd, W._vLat = vFwd, vLat
        local absFwd = abs(vFwd)

        local spec = (W.i < 2) and self.tireSimSpecs.front or self.tireSimSpecs.rear
        local peakLatKN, peakLongKN, effLoadKN, rrCoeff
        local loadVal = W.load or 0
        local surfId = W.surface or 1
        local tickId = self._vphysTickId
        if tickId and W._pkTick == tickId and W._pkLoad == loadVal and W._pkSurf == surfId and W._pkSpec == spec then
            peakLatKN, peakLongKN, effLoadKN, rrCoeff = W._pkLat, W._pkLon, W._pkFz, W._pkRRC
            W.effectiveLoadKN = effLoadKN
        else
            local tireLoadKN = loadVal * (W._wsMult or 1.0)
            if tireLoadKN < 0 then tireLoadKN = 0 end
            local aeroCfg = self.aeroCfg
            local aeroAdd = 0
            if aeroCfg then
                local factor = (W.i < 2) and (aeroCfg.FrontAeroFactor or 0) or (aeroCfg.RearAeroFactor or 0)
                if factor > 0 then
                    local mphDiv100 = abs(self.forward or 0) * 0.0223693629
                    aeroAdd = factor * ((self.m or 1450) * 0.004905) * (mphDiv100 * mphDiv100)
                end
            end
            effLoadKN = tireLoadKN + aeroAdd
            W.effectiveLoadKN = effLoadKN
            local matType = W.ground and W.ground.matType
            local surfLat, surfLon, surfRR
            if W._surfSpec == spec and W._surfId == surfId and W._surfMat == matType then
                surfLat, surfLon, surfRR = W._surfLat, W._surfLon, W._surfRR
            else
                surfLat, surfLon = Tires.computeSurfaceMult(spec, surfId, matType)
                surfRR = Tires.computeRollingResistanceMult(spec, surfId, matType)
                W._surfSpec, W._surfId, W._surfMat = spec, surfId, matType
                W._surfLat, W._surfLon, W._surfRR = surfLat, surfLon, surfRR
            end
            if effLoadKN <= 0.001 then
                peakLatKN, peakLongKN, rrCoeff = 0, 0, 0
            else
                local nom = spec.nominalLoadKN or 3.68
                if nom < 0.5 then nom = 0.5 end
                local loadRatio = effLoadKN / nom
                if loadRatio < 0.15 then loadRatio = 0.15 elseif loadRatio > 2.8 then loadRatio = 2.8 end
                local loadSens = 1.0 - (spec.LoadSens or 0.12) * (loadRatio - 1.0)
                if loadSens < 0.65 then loadSens = 0.65 elseif loadSens > 1.20 then loadSens = 1.20 end
                local baseGrip = effLoadKN * spec.CoF * loadSens * (W.grip or 1) * (W.perf or 1)
                peakLatKN, peakLongKN = baseGrip * surfLat * spec.latScale, baseGrip * surfLon * spec.longScale
                rrCoeff = effLoadKN * (spec.rollingRes or 0.015) * (surfRR or 1)
            end
            if tickId then
                W._pkTick, W._pkLoad, W._pkSurf, W._pkSpec = tickId, loadVal, surfId, spec
                W._pkLat, W._pkLon, W._pkFz, W._pkRRC = peakLatKN, peakLongKN, effLoadKN, rrCoeff
            end
        end

        local drv = self.drv
        local hasBrakeInput = (drv and (drv.brk or 0) > 0) or ((W.brk or 0) > 0) or ((W.hbk or 0) > 0)
        local pedalBrk, vBrk, vHbk, absReleasing
        if hasBrakeInput then
            pedalBrk, vBrk, vHbk, absReleasing = Braking.applyWheelTorque(self, W, peakLongKN, vFwd, absFwd)
        else
            pedalBrk, vBrk, vHbk, absReleasing = 0, 0, 0, false
            W.absActive = false
            W.serviceBrakeKNm = 0
            W.locked = 0
        end

        local slipSpeedScale = absFwd + 0.5
        if slipSpeedScale < 2.2 then slipSpeedScale = 2.2 end
        local spinThreshold = 0.16 * slipSpeedScale
        local wRad = W.radius
        local safeRad = wRad > 0.15 and wRad or 0.15
        if W.locked == 0 then
            local unpowered = (W.i < 2 and self.dt.split > 0.999) or (W.i >= 2 and self.dt.split < 0.001)
            if unpowered and pedalBrk < 0.01 and W.hbk < 0.01 then
                W.w, W.avgT = vFwd / wRad, 0
            else
                local wRim = W.w * wRad
                local wT = W.T
                if pedalBrk < 0.01 and W.hbk < 0.01 and abs(wT) < 0.001 and abs(wRim) <= absFwd + spinThreshold
                    and not ((W.i < 2 and ((self.effFrontDiffType == 2) or (self.frontDiffType == 2) or self.frontLockActive or self.allLockActive))
                        or (W.i >= 2 and ((self.effRearDiffType == 2) or (self.rearDiffType == 2) or self.rearLockActive or self.allLockActive))) then
                    W.w, W.avgT = vFwd / safeRad, 0
                elseif pedalBrk < 0.05 and W.hbk < 0.05 and ((wT > 0 and vFwd >= 0 and wRim < vFwd) or (wT < 0 and vFwd < 0 and wRim > vFwd)) then
                    W.w, W.avgT = vFwd / safeRad, 0
                elseif (drv and drv.thr or 0) < 0.05 and pedalBrk < 0.05 and W.hbk < 0.05
                    and ((wT <= 0 and vFwd >= 0 and wRim < vFwd) or (wT >= 0 and vFwd < 0 and wRim > vFwd)) then
                    local thrLim = peakLongKN * 0.55; if thrLim < 0.25 then thrLim = 0.25 end
                    if abs(wT / safeRad) < thrLim then W.w, W.avgT = vFwd / safeRad, 0 end
                end
            end
        end
        if hasBrakeInput then
            peakLatKN = Braking.applyWheelLock(self, W, peakLongKN, peakLatKN, vFwd, absFwd, pedalBrk, vBrk, vHbk, absReleasing)
        end
        local slipVel = wRad * W.w - vFwd
        W.slip = slipVel
        W.slip364 = slipVel

        local alphaRad
        if vLat == 0 then
            alphaRad = 0
        else
            local V0 = spec.V0 or 1.016
            alphaRad = atan2(vLat, absFwd > V0 and absFwd or V0)
            if absFwd < 3.5 and abs(alphaRad) < (spec.lowSpeedDeadzone or 0.08) then
                alphaRad = alphaRad * (spec.lowSpeedHoldScale or 0.35)
            end
        end
        local kappa = slipVel / slipSpeedScale
        if kappa < -1.0 then kappa = -1.0 elseif kappa > 1.0 then kappa = 1.0 end
        W.slipAngleRad = alphaRad
        W.slipRatio = kappa

        W.peakLatKN  = peakLatKN
        W.peakLongKN = peakLongKN

        if peakLatKN <= 0.001 and peakLongKN <= 0.001 then
            W.Fx, W.Fy = 0, 0
            W.sliding = 0
            W.slipX, W.slipY = 0, vLat
            return
        end

        local totalSpd = sqrt(vFwd * vFwd + vLat * vLat)
        W._totalSpd = totalSpd
        local isLocked = W.locked ~= 0
        local latShape = 0
        if not isLocked then
            if vLat == 0 then
                W.Fy = 0
            else
                local latVelRamp = totalSpd * 5.0
                if latVelRamp > 1.0 then latVelRamp = 1.0 end
                local lBx = spec.Lat_B * abs(alphaRad)
                latShape = sin(spec.Lat_C * atan(lBx - spec.Lat_E * (lBx - atan(lBx))))
                W.Fy = peakLatKN * latShape * (vLat > 0 and -latVelRamp or latVelRamp)
            end
        end
        if W.axleX then
            local R, n = self.R, W.n
            local n0, n1, n2 = n[0], n[1], n[2]
            local nx, ny, nz
            if n0 == 0 and n1 == 0 and n2 == 1 then
                nx, ny, nz = R[2], R[5], R[8]
            else
                nx = R[0]*n0 + R[1]*n1 + R[2]*n2
                ny = R[3]*n0 + R[4]*n1 + R[5]*n2
                nz = R[6]*n0 + R[7]*n1 + R[8]*n2
            end
            local dotAx = W.axleX*nx + W.axleY*ny + W.axleZ*nz
            if dotAx < -1 then dotAx = -1 elseif dotAx > 1 then dotAx = 1 end
            local inclination
            if W._lastDotAx == dotAx then
                inclination = W._lastInc
            else
                inclination = asin(dotAx)
                W._lastDotAx, W._lastInc = dotAx, inclination
            end
            W.roadCamberRad = (W.i == 1 or W.i == 2) and inclination or -inclination
            if not isLocked then
                local thrust = -inclination * spec.camberStiffness
                if thrust < -1 then thrust = -1 elseif thrust > 1 then thrust = 1 end
                local fwdRamp = absFwd / 0.55; if fwdRamp > 1 then fwdRamp = 1 end
                local cFy = peakLatKN * thrust * (1 - abs(latShape)) * fwdRamp
                if cFy * vLat > 0 then
                    local afy = abs(W.Fy)
                    if cFy < -afy then cFy = -afy elseif cFy > afy then cFy = afy end
                end
                W.camberFy = cFy
                W.Fy = W.Fy + cFy
            end
        end

        -- The lateral curve already accounts for sideslip. Reduce its available
        -- force by longitudinal spin without restoring grip as sideslip grows.
        local excessSpin = abs(wRad * W.w) - absFwd - spinThreshold
        if excessSpin > 0 and not isLocked then
            local normalizedSpin = excessSpin / spinThreshold
            local retention = 1 / sqrt(1 + normalizedSpin * normalizedSpin)
            W.Fy, W.camberFy = W.Fy * retention, W.camberFy * retention
        end

        local reqFx = W.T / safeRad
        local overGripSpin = ((drv and drv.thr or 0) > 0.1) and (abs(reqFx) > peakLongKN * 1.02)
        if not isLocked and abs(kappa) <= 0.16 and not overGripSpin then
            W.sliding = 0
            W.Fx = reqFx < -peakLongKN and -peakLongKN or (reqFx > peakLongKN and peakLongKN or reqFx)
            W.slipX = 0
            W.slipY = vLat
        else
            W.sliding = 1
            if isLocked then
                local slide = spec._lockedSlide
                if not slide then
                    local sBx = spec.Long_B
                    slide = sin(spec.Long_C * atan(sBx - spec.Long_E * (sBx - atan(sBx))))
                    spec._lockedSlide = slide
                end
                local scale = slide / (totalSpd > 0.12 and totalSpd or 0.12)
                W.Fx = -peakLongKN * vFwd * scale
                W.Fy = -peakLatKN * vLat * scale
            else
                local ak = abs(kappa)
                local slipSign = (ak > 0.01) and ((kappa >= 0) and 1 or -1) or ((reqFx >= 0) and 1 or -1)
                local evalKappa = (overGripSpin and ak < 0.16) and 0.16 or ak
                local kBx = spec.Long_B * evalKappa
                local lonShape = sin(spec.Long_C * atan(kBx - spec.Long_E * (kBx - atan(kBx))))
                if overGripSpin and ak < 0.16 and lonShape < 0.92 then
                    lonShape = 0.92
                end
                W.Fx = peakLongKN * lonShape * slipSign
                if excessSpin > 0 then
                    local scrubSpeed = sqrt(slipVel * slipVel + vLat * vLat)
                    local spinBlend = excessSpin / spinThreshold; if spinBlend > 1 then spinBlend = 1 end
                    W.Fx = W.Fx * (1 - spinBlend * (1 - abs(slipVel) / scrubSpeed))
                end
            end
            W.slipX = slipVel
            W.slipY = vLat
        end

        if absFwd > 0.15 and effLoadKN > 0.001 then
            local rrRamp = absFwd / 1.5; if rrRamp > 1.0 then rrRamp = 1.0 end
            W.rrForce = rrCoeff * rrRamp
        else
            W.rrForce = 0
        end
    end

    function Chassis:ellipse(W, finalSub)
        if not self.tireSimSpecs then
            W.Fx, W.Fy = 0, 0
            return
        end
        if (W.contact or 0) == 0 or (W.load or 0) <= 0.001 then
            W.Fx, W.Fy = 0, 0
            W.usage = 0
            W.skidIntensity = 0
            local wf = W.world
            wf[0], wf[1], wf[2] = 0, 0, 0
            return
        end

        local pLon = W.peakLongKN or W.load
        if pLon < 0.05 then pLon = 0.05 end
        local pLat = W.peakLatKN or W.load
        if pLat < 0.05 then pLat = 0.05 end
        local fx, fy = W.Fx, W.Fy
        local rx, ry = fx / pLon, fy / pLat
        local r = sqrt(rx * rx + ry * ry)

        if r > 1.0 then
            local invR = 1 / r
            fx = fx * invR
            fy = fy * invR
            W.camberFy = W.camberFy * invR
        end

        local usage = r > 1.35 and 1.35 or r
        W.usage = usage
        local b = W.basis
        local vFwd, vLat = W._vFwd, W._vLat
        if vFwd == nil then
            local vel = W.vel
            vFwd = vel[0] * b[0] + vel[1] * b[1] + vel[2] * b[2]
            vLat = vel[0] * b[3] + vel[1] * b[4] + vel[2] * b[5]
        end
        local hdt = self.substepDt or 0.004
        if hdt < 0.002 then hdt = 0.002 end
        local cornerMass = self.m * 0.25
        if cornerMass < 0.08 then cornerMass = 0.08 end
        local jyMax = (0.85 * abs(vLat) * cornerMass) / (self.coastContactDt or hdt) + abs(W.camberFy)
        local afy = abs(fy)
        if afy > jyMax then
            W.camberFy = W.camberFy * (jyMax / afy)
            fy = (fy >= 0) and jyMax or -jyMax
        end
        local isLocked = W.locked ~= 0
        if isLocked then
            local jxMax = (0.90 * abs(vFwd) * cornerMass) / hdt
            if abs(fx) > jxMax then
                fx = (fx >= 0) and jxMax or -jxMax
            end
        end

        local vphys = self._vphysMode
        local wRad = W.radius
        local slipVelLong = wRad * W.w - vFwd
        if finalSub or not vphys then
            local rimSpd = wRad * W.w
            local spd2 = W._totalSpd or sqrt(vFwd * vFwd + vLat * vLat)
            local absRim = abs(rimSpd)
            local activeSpd = spd2 > absRim and spd2 or absRim
            local totalSlipSpeedIn = sqrt(slipVelLong * slipVelLong + vLat * vLat) * 39.3700787
            local sr = (W.slipRatio or 0) * 8.333333333333334
            local sa = (W.slipAngleRad or 0) * 6.25
            local overRatio = sqrt(sr * sr + sa * sa)
            local speedGate = (totalSlipSpeedIn - 24) * 0.010526315789473684
            if speedGate < 0 then speedGate = 0 elseif speedGate > 1 then speedGate = 1 end
            local skid = (overRatio - 0.58) * 0.8
            if skid < 0 then skid = 0 elseif skid > 1 then skid = 1 end
            skid = skid * speedGate
            if (isLocked or W.sliding ~= 0 or (usage > 0.92 and not W.absActive)) and activeSpd > 0.8 then
                local uSkid = usage * 0.85
                if uSkid < 0.35 then uSkid = 0.35 elseif uSkid > 1.0 then uSkid = 1.0 end
                local sGate = speedGate > 0.35 and speedGate or 0.35
                local altSkid = uSkid * sGate
                if altSkid > skid then skid = altSkid end
            end
            if W.absActive and not isLocked and W.sliding == 0 and (self.braking and self.braking.abs and self.braking.abs.threshold or 0.32) <= 0.14 then
                skid = 0
            end
            W.skidIntensity = skid
        end

        local rrF = W.rrForce or 0
        if rrF > 0.0001 and abs(vFwd) > 0.15 then
            fx = fx - ((vFwd >= 0) and rrF or -rrF)
        end

        if vphys and W.sliding ~= 0 and not isLocked and fx * slipVelLong > 0 then
            local syncForce = W.T / wRad + W.I * slipVelLong / (hdt * wRad * wRad)
            if fx > 0 then
                local cap = syncForce > 0 and syncForce or 0
                if fx > cap then fx = cap end
            else
                local cap = syncForce < 0 and syncForce or 0
                if fx < cap then fx = cap end
            end
        end
        W.Fx, W.Fy = fx, fy
        if not (finalSub and vphys) then
            local wf = W.world
            if vphys then
                wf[0] = fx * b[0] + fy * b[3]
                wf[1] = fx * b[1] + fy * b[4]
                wf[2] = fx * b[2] + fy * b[5]
            else
                local normalForce = W.load
                wf[0] = fx * b[0] + fy * b[3] + normalForce * b[6]
                wf[1] = fx * b[1] + fy * b[4] + normalForce * b[7]
                wf[2] = fx * b[2] + fy * b[5] + normalForce * b[8]
            end
        end
    end

    function Chassis:updateMass(newMassKg)
        local newMassT = max(0.35, (newMassKg or 1500) / 1000)
        local oldMassT = self.baseParams[68] or self.m or 1.22
        local ratio = newMassT / max(0.1, oldMassT)
        if abs(ratio - 1) < 0.005 then return end

        local function scaleCarArray(arr)
            arr[68] = newMassT
            arr[72] = arr[72] * ratio
            arr[77] = arr[77] * ratio
            arr[82] = arr[82] * ratio
            arr[153] = arr[153] * ratio
            arr[154] = arr[154] * ratio
        end

        scaleCarArray(self.baseParams)
        self.c = self.baseParams
        if self.rearDriveShare ~= nil then self.c[112] = self.rearDriveShare end
        if self.rearDriveShare ~= nil and self.dt then self.dt.split = self.rearDriveShare end

        self.m = self.c[68]
        self.im = 1 / self.m
        self.Iinv = { [0] = 1 / self.c[72], 1 / self.c[77], 1 / self.c[82] }
        self.Iinv2 = { [0] = self.Iinv[0], self.Iinv[1], self.Iinv[2] / self.c[88] }
        if self.tireSimSpecs then
            local cornerKN = (self.m * 0.25) * 9.800000190734863
            self.tireSimSpecs.front.nominalLoadKN = cornerKN
            self.tireSimSpecs.rear.nominalLoadKN  = cornerKN
        end
        self:updateWheelMounts()
    end

    function Chassis:updateSimClutch(elapsed)
        local d = self.dt
        local speedKmh = abs(self.forward or 0) * 3.6
        local stopSpd = self.stoppedPhaseBrakeSpeed or 5
        local thrIn = (self.drv and self.drv.thr) or 0
        local brkIn = (self.drv and self.drv.brk) or 0
        local hasThrInput = thrIn > 0.05
        local simClutch = 0
        if d.gear == 1 or d.state == 0 or d.state == 1 or d.shifting ~= 0 or (self.clutchPedal == 0) or ((self._stepHandbrake or 0) > 0.5) then
            simClutch = 1
        elseif self.clutchType == "torque_converter" then
            if self.tcLocked then
                simClutch = 0
            else
                local wE = max(1, self.eng.w or 0)
                local wt = max(0, d.wt or 0)
                local sr = clamp(wt / wE, 0, 1)
                simClutch = clamp(1.0 - sr, 0.05, 0.95)
            end
        elseif self.clutchType == "centrifugal" then
            local rpmNow = self.eng.w * RAD_TO_RPM
            local engRPM = self.centrifugalEngageRPM or 1800
            local fullRPM = max(engRPM + 100, self.centrifugalFullRPM or 3200)
            local engage = clamp((rpmNow - engRPM) / (fullRPM - engRPM), 0, 1)
            simClutch = 1.0 - engage
        else
            if self:engineOffCoupled() then
                simClutch = 0
            else
                local autoAntiStall = not (self.physicalIdle and self.manualBox and not self.useClutchLerpInManual)
                if autoAntiStall then
                    if brkIn > 0.05 and not hasThrInput and speedKmh <= stopSpd then
                        simClutch = 1
                    elseif brkIn > 0.05 and not hasThrInput and speedKmh <= (stopSpd + 15) then
                        simClutch = 1.0 - clamp((speedKmh - stopSpd) / 15.0, 0, 1)
                    elseif not hasThrInput and speedKmh <= stopSpd then
                        simClutch = 1
                    end
                end
                if simClutch < 1 and (not self.manualBox or self.useClutchLerpInManual) then
                    local rpmNow = self.eng.w * RAD_TO_RPM
                    local idleRPM = self.c[124] or 800
                    local redline = max(1000, self.c[126] or 8000)
                    local lerpLim = (d.gear <= 0 or (self.drv and self.drv.reversing)) and (self.clutchLerpReverse or 0.5) or (self.clutchLerp or 0.4)
                    local fullRPM = max(idleRPM + 300, lerpLim * redline)
                    local wtRPM = abs(d.wt or 0) * RAD_TO_RPM
                    local autoEngage = clamp((rpmNow - idleRPM) / (fullRPM - idleRPM), 0, 1)
                    local syncFactor
                    if d.gear > 2 then
                        syncFactor = clamp((wtRPM - idleRPM * 0.90) / max(150, idleRPM * 0.25), 0, 1)
                    else
                        syncFactor = clamp((wtRPM - idleRPM * 1.05) / max(300, fullRPM - idleRPM * 1.05), 0, 1)
                    end
                    local rpmClutch = 1.0 - max(autoEngage * autoEngage, syncFactor)
                    simClutch = max(simClutch, rpmClutch)
                end
            end
        end
        if (self.clutchEngageTime or 0) > 0 and self.clutchType == "friction" then
            simClutch = max(simClutch, (self.simClutch or 1) - elapsed / self.clutchEngageTime)
        end
        self.simClutch = clamp(simClutch, 0, 1)
    end

    -- Synchronize the solver state from GMod VPhysics and advance across 0.0044s substeps
    function Chassis:stepVPhysics(dt, pos_m, R, vel_m, angVel_rad, wheelGrounds, controls)
        dt = clamp(dt or 0.015, 0.001, 0.067)
        self._vphysMode = true
        self._vphysTickId = (self._vphysTickId or 0) + 1

        -- 1. Update wheel ground planes from Base raycasts
        local gvx, gvy, gvz, gvc = 0, 0, 0, 0
        local nx, ny, nz = 0, 0, 0
        local WS = self.wheels
        if wheelGrounds then
            for i = 0, 3 do
                local g = wheelGrounds[i]
                if g then
                    local wg = WS[i].ground
                    local gp, gn, gv = g.p, g.n, g.v
                    local wgp, wgn, wgv = wg.p, wg.n, wg.v
                    wgp[0], wgp[1], wgp[2] = gp[0], gp[1], gp[2]
                    wgn[0], wgn[1], wgn[2] = gn[0], gn[1], gn[2]
                    if gv then
                        local gx, gy, gz = gv[0], gv[1], gv[2]
                        wgv[0], wgv[1], wgv[2] = gx, gy, gz
                        WS[i]._hasGVel = (gx ~= 0 or gy ~= 0 or gz ~= 0)
                    else
                        wgv[0], wgv[1], wgv[2] = 0, 0, 0
                        WS[i]._hasGVel = false
                    end
                    local s = g.s or 1
                    wg.s = s
                    wg.matType = g.matType
                    if s ~= 0 then
                        gvx, gvy, gvz = gvx + wgv[0], gvy + wgv[1], gvz + wgv[2]
                        gvc = gvc + 1
                        nx, ny, nz = nx + wgn[0], ny + wgn[1], nz + wgn[2]
                    end
                end
            end
        end
        local r0, r1, r2 = R[0], R[1], R[2]
        local horiz = sqrt(r0 * r0 + r1 * r1)
        local along = horiz > 0.01 and -(nx * r0 + ny * r1) / horiz or 0
        self.uphillSin = nz > 0.01 and max(0, along) / sqrt(along * along + nz * nz) or 0
        local sGV = self.groundVel
        if not sGV then sGV = { [0] = 0, 0, 0 }; self.groundVel = sGV end
        if gvc > 0 then
            local invGvc = 1 / gvc
            sGV[0], sGV[1], sGV[2] = gvx * invGvc, gvy * invGvc, gvz * invGvc
        else
            sGV[0], sGV[1], sGV[2] = 0, 0, 0
        end

        -- 2. Copy GMod VPhysics rigid-body pose and velocity into the solver body
        local sPos, sR, sV, sW = self.pos, self.R, self.v, self.w
        sPos[0], sPos[1], sPos[2] = pos_m[0], pos_m[1], pos_m[2]
        sR[0], sR[1], sR[2] = r0, r1, r2
        local r3, r4, r5 = R[3], R[4], R[5]
        local r6, r7, r8 = R[6], R[7], R[8]
        sR[3], sR[4], sR[5] = r3, r4, r5
        sR[6], sR[7], sR[8] = r6, r7, r8
        local vx, vy, vz = vel_m[0], vel_m[1], vel_m[2]
        sV[0], sV[1], sV[2] = vx, vy, vz
        local avx, avy, avz = angVel_rad[0], angVel_rad[1], angVel_rad[2]
        sW[0], sW[1], sW[2] = avx, avy, avz
        local rvx, rvy, rvz = vx - sGV[0], vy - sGV[1], vz - sGV[2]
        self.speed = sqrt(rvx * rvx + rvy * rvy + rvz * rvz)
        self.forward = rvx * r0 + rvy * r1 + rvz * r2

        self.mode344 = (gvc > 0) and 1 or 0

        local sm = self.m
        self.P[0], self.P[1], self.P[2] = vx * sm, vy * sm, vz * sm
        local I = (self.mode344 ~= 0) and self.Iinv2 or self.Iinv
        local wx = avx * r0 + avy * r1 + avz * r2
        local wy = avx * r3 + avy * r4 + avz * r5
        local wz = avx * r6 + avy * r7 + avz * r8
        local lx, ly, lz = wx / I[0], wy / I[1], wz / I[2]
        self.L[0] = lx * r0 + ly * r3 + lz * r6
        self.L[1] = lx * r1 + ly * r4 + lz * r7
        self.L[2] = lx * r2 + ly * r5 + lz * r8

        -- Initialize wheel travel from the current ground height.
        if not self._vphysInit then
            self._vphysInit = true
            self:updateWheelMounts()
            for i = 0, 3 do
                self:postWheel(self.wheels[i])
                self:postSusp(self.wheels[i])
            end
            self:keepState()
        end

        -- 3. Apply driver controls & Clutch Pedal
        controls = controls or {}
        if controls.aeroCfg ~= nil then self.aeroCfg = controls.aeroCfg end
        if controls.weightShiftCfg ~= nil then self.weightShiftCfg = controls.weightShiftCfg end

        -- Simulated Dynamic Weight Shift (adjusts per-wheel normal load Fz from accel Gs)
        local wsCfg = self.weightShiftCfg
        local rawFwdG = controls.fwdAccelG
        if rawFwdG == nil then
            local prevFwd = self._prevFwdSpeed or (self.forward or 0)
            rawFwdG = ((self.forward or 0) - prevFwd) / (max(0.002, dt) * 9.81)
        end
        self._prevFwdSpeed = self.forward or 0
        local rawLatG = controls.latAccelG or 0
        local rawVertG = controls.vertAccelG or 0
        local wsSmooth = clamp(((wsCfg and wsCfg.Smoothing) or 10.0) * dt, 0.05, 1.0)
        self._wsFwdG  = (self._wsFwdG  or 0) + (rawFwdG  - (self._wsFwdG  or 0)) * wsSmooth
        self._wsLatG  = (self._wsLatG  or 0) + (rawLatG  - (self._wsLatG  or 0)) * wsSmooth
        self._wsVertG = (self._wsVertG or 0) + (rawVertG - (self._wsVertG or 0)) * wsSmooth

        local wsEnabled = (wsCfg == nil) or (wsCfg.Enabled ~= false)
        local dLong, dLat, vertMult = 0, 0, 1.0
        if wsEnabled then
            local cgH = (wsCfg and wsCfg.CGHeight) or 0.55
            local wb  = max(0.5, (wsCfg and wsCfg.Wheelbase) or 2.55)
            local tw  = max(0.5, (wsCfg and wsCfg.TrackWidth) or 1.55)
            local maxFrac = (wsCfg and wsCfg.MaxTransferFrac) or 0.45
            local slowingDown = self._wsFwdG * (self.forward or 0) < 0
            local longMul = 1.0
            if wsCfg then
                local phaseMul
                if slowingDown then
                    phaseMul = wsCfg.LongitudinalBrakeTransfer
                else
                    phaseMul = wsCfg.LongitudinalAccelTransfer
                end
                if phaseMul ~= nil then
                    longMul = phaseMul
                elseif wsCfg.LongitudinalTransfer ~= nil then
                    longMul = wsCfg.LongitudinalTransfer
                end
            end
            local latMul  = (wsCfg and wsCfg.LateralTransfer) or 1.0
            local vertMul = (wsCfg and wsCfg.VerticalTransfer) or 1.0
            local minVert = (wsCfg and wsCfg.MinVertLoadMult) or 0.20
            local maxVert = (wsCfg and wsCfg.MaxVertLoadMult) or 2.00
            dLong    = clamp(-self._wsFwdG * (cgH / wb) * longMul, -maxFrac, maxFrac)
            dLat     = clamp(-self._wsLatG * (cgH / tw) * latMul,  -maxFrac, maxFrac)
            vertMult = clamp(1.0 + self._wsVertG * vertMul, minVert, maxVert)
        end
        local minLoadFrac = (wsCfg and wsCfg.MinWheelLoadFrac) or 0.15
        local gripEffect  = (wsCfg and wsCfg.GripEffectMult ~= nil) and wsCfg.GripEffectMult or 0.65
        local gripLong    = (wsCfg and wsCfg.GripEffectLongMult ~= nil) and wsCfg.GripEffectLongMult or 1.0
        local gripLat     = (wsCfg and wsCfg.GripEffectLatMult  ~= nil) and wsCfg.GripEffectLatMult  or 1.0
        local fLongM      = (wsCfg and wsCfg.FrontLongMult    ~= nil) and wsCfg.FrontLongMult    or 1.0
        local rLongM      = (wsCfg and wsCfg.RearLongMult     ~= nil) and wsCfg.RearLongMult     or 1.0
        local fLatM       = (wsCfg and wsCfg.FrontLateralMult ~= nil) and wsCfg.FrontLateralMult or 1.0
        local rLatM       = (wsCfg and wsCfg.RearLateralMult  ~= nil) and wsCfg.RearLateralMult  or 1.0
        local frontSum, rearSum, leftSum, rightSum, totalW = 0, 0, 0, 0, 0
        local rawMultSum, activeCount = 0, 0
        for i = 0, 3 do
            local wObj = self.wheels[i]
            if wObj then
                local isFront   = (i < 2)
                local frontSign = isFront and 1 or -1
                local rightSign = (i == 1 or i == 2) and 1 or -1 -- 0=FL, 1=FR, 2=RR, 3=RL
                local axLongM   = isFront and fLongM or rLongM
                local axLatM    = isFront and fLatM  or rLatM
                local dispMult  = wsEnabled and max(minLoadFrac, 1.0 + frontSign * dLong + rightSign * dLat) or 1.0
                local physDelta = (frontSign * dLong * gripLong * axLongM) + (rightSign * dLat * gripLat * axLatM)
                local physMult  = wsEnabled and max(minLoadFrac, 1.0 + physDelta * gripEffect) or 1.0
                wObj._wsMult = physMult
                if (wObj.load or 0) > 0.001 then
                    rawMultSum = rawMultSum + physMult
                    activeCount = activeCount + 1
                end
                local wLoad = max(0, (wObj.load or 0) * dispMult)
                totalW = totalW + wLoad
                if i < 2 then frontSum = frontSum + wLoad else rearSum = rearSum + wLoad end
                if i == 0 or i == 3 then leftSum = leftSum + wLoad else rightSum = rightSum + wLoad end
            end
        end
        if activeCount > 0 and rawMultSum > 0.001 then
            local normScale = (activeCount / rawMultSum) * vertMult
            for i = 0, 3 do
                local wObj = self.wheels[i]
                if wObj and (wObj.load or 0) > 0.001 then
                    wObj._wsMult = wObj._wsMult * normScale
                end
            end
        end
        local fb = self.frontWeightBias or 0.50
        local fPct = (totalW > 0.001) and (frontSum / totalW * 100) or clamp((fb + 0.5 * dLong) * 100, 5, 95)
        local lPct = (totalW > 0.001) and (leftSum / totalW * 100) or clamp((0.5 - 0.5 * dLat) * 100, 5, 95)
        local wsd = self.weightShiftDebug
        if not wsd then
            wsd = {}
            self.weightShiftDebug = wsd
        end
        wsd.enabled  = wsEnabled
        wsd.fwdG     = self._wsFwdG
        wsd.latG     = self._wsLatG
        wsd.vertG    = self._wsVertG
        wsd.vertMult = vertMult
        wsd.dLong    = dLong
        wsd.dLat     = dLat
        wsd.frontPct = fPct
        wsd.rearPct  = 100 - fPct
        wsd.leftPct  = lPct
        wsd.rightPct = 100 - lPct
        self.ignitionRunning = (controls.ignitionRunning ~= false)
        self.cranking = (controls.cranking == true)
        if not self.ignitionRunning and not self.physicalStarter and self.cranking and self.eng then
            self.eng.w = (controls.crankRPM or 0) * 0.10471975512
            self.rpm = controls.crankRPM or 0
        end
        self.vphysLoadsKN = controls.vphysLoadsKN
        if controls.clutchPedal ~= nil then self.clutchPedal = controls.clutchPedal else self.clutchPedal = 1 end
        if self.drv then self.drv.reverseGas = controls.reverseGas; self.drv.forwardBrake = controls.forwardBrake end
        self:driver(
            controls.steer or 0,
            controls.gas or 0,
            controls.brake or 0,
            controls.handbrake or 0,
            dt,
            controls.hold and true or false,
            controls.u16
        )

        -- 4. Frame pre-substep setup & velocity-based suspension sync
        local hbkIn = (self.drv and self.drv.hbk) or 0
        for i = 0, 3 do
            self.wheels[i].hbk = (i >= 2) and hbkIn or 0
        end
        self:prepareFrame()
        self:updateDriverState(dt)
        -- VPhysics receives one accumulated impulse before reporting new
        -- contact velocities. Bound coasting slip correction over that interval.
        self.coastContactDt = controls.useContactYaw and self.drv.thr < 0.05 and dt or nil

        for i = 0, 3 do
            self:updateAlignment(self.wheels[i], i == 0 and self.steer.a0 or (i == 1 and self.steer.a1 or 0))
        end
        self:postBody()
        self:postAllWheelsAndSusp()
        self:postEngine(0)
        self:postDrive()
        local maxSub = (controls and controls.maxSubsteps) or 4
        local n = min(maxSub, max(1, math.ceil(dt / 0.010)))
        local h = dt / n
        self.substepDt = h

        -- 5. Record start-of-substeps rigid-body velocities & run bounded substeps (2 max by default)
        local vx0, vy0, vz0 = self.v[0], self.v[1], self.v[2]
        local wx0, wy0, wz0 = self.w[0], self.w[1], self.w[2]
        local ux, uy, uz    = R[6], R[7], R[8]
        local fx, fy, fz    = R[0], R[1], R[2]
        local lfx, lfy, lfz = R[3], R[4], R[5]



        for i = 0, 3 do
            local W = self.wheels[i]
            W.absActiveTick = false
            W.tickFx, W.tickFy = 0, 0
        end

        -- Red Fox Engine Master Clutch Control (0 = engaged/driving, 1 = disengaged)
        self._stepHandbrake = (controls and controls.handbrake or 0)
        self:updateSimClutch(0)

        self:computeActiveDiff(controls)
        self:forces()
        local tireYawImpulse = 0
        for stepIdx = 1, n do
            -- integrate consumes these forces, then solves new ones for the next substep.
            for i = 0, 3 do
                local W = self.wheels[i]
                W.tickFx = W.tickFx + W.Fx / n
                W.tickFy = W.tickFy + W.Fy / n
                if controls.useContactYaw and W.contact ~= 0 then
                    -- Integrate the contact moment about the suspension origin, in kN m s.
                    -- Use the same road-tangent forces that accelerate the chassis.
                    local f, pt = W.world, W.pt
                    local x, y, z = pt[0] - self.pos[0], pt[1] - self.pos[1], pt[2] - self.pos[2]
                    tireYawImpulse = tireYawImpulse + h * (
                        ux * (y * f[2] - z * f[1]) + uy * (z * f[0] - x * f[2]) + uz * (x * f[1] - y * f[0]))
                end
            end
            self:integrate(h, stepIdx == n)
            self.now = self.now + h
            self:orthonormalize()
            for i = 0, 3 do
                if self.wheels[i].absActive then self.wheels[i].absActiveTick = true end
            end
        end

        self:state(dt)
        if dt > 0 then
            self.yawAcc  = (self.w[2] - self.yawPrev) / dt
            self.yawPrev = self.w[2]
        end

        -- 6. Convert the tire and aero velocity delta to car-frame axes
        local dvx = self.v[0] - vx0
        local dvy = self.v[1] - vy0
        local dvz = self.v[2] - vz0

        local dv_fwd_m  = dvx * fx  + dvy * fy  + dvz * fz
        local dv_left_m = dvx * lfx + dvy * lfy + dvz * lfz
        local dv_up_m   = dvx * ux  + dvy * uy  + dvz * uz

        local dwx = self.w[0] - wx0
        local dwy = self.w[1] - wy0
        local dwz = self.w[2] - wz0
        local dw_roll_rad  = dwx * fx  + dwy * fy  + dwz * fz
        local dw_pitch_rad = dwx * lfx + dwy * lfy + dwz * lfz
        local dw_yaw_rad   = dwx * ux  + dwy * uy  + dwz * uz

        local boostPSI = self.eng and self.eng.boostPSI or 0

        local g = self.gearShown or 2
        local gName
        if self.classicAuto and not self.manualBox and self.autoRange == 0 then
            gName = "P"
        elseif g == 1 then
            gName = "N"
        elseif g >= 2 then
            gName = (self.classicAuto and not self.manualBox and "D" or "") .. tostring(g - 1)
        else
            gName = ((self.minGear or 0) == 0) and "R" or ("R" .. tostring(1 - g))
        end
        local peakTqKNm = self._peakTqKNm
        if not peakTqKNm then
            peakTqKNm = 0
            for i = 128, 136 do
                local v = self.c[i] or 0
                if v > peakTqKNm then peakTqKNm = v end
            end
            self._peakTqKNm = peakTqKNm
        end
        local engineLoad = clamp(math.abs(self.lastTload or (self.eng and self.eng.tload) or 0) / math.max(0.05, peakTqKNm * (self.eng and self.eng.perf or 1)), 0, 1.5)
        return {
            dv_fwd_m      = dv_fwd_m,
            dv_left_m     = dv_left_m,
            dv_up_m       = dv_up_m,
            dw_roll_rad   = dw_roll_rad,
            dw_pitch_rad  = dw_pitch_rad,
            dw_yaw_rad    = dw_yaw_rad,
            tireYawImpulse = tireYawImpulse,
            grounded      = self.grounded,
            restState     = self.restState,
            speed_ms      = self.speed,
            forward_ms    = self.forward,
            rpm           = self.rpm,
            flywheel_rpm  = self.eng.w * RAD_TO_RPM,
            gearIndex     = self.dt.gear,
            gearShown     = g,
            gearName      = gName,
            clutch        = self.simClutch ~= nil and self.simClutch or (1.0 - (self.dt.clutch or 0)),
            shifting      = self.dt.shifting,
            throttle      = (self.dt.shifting ~= 0) and 0 or (self.eng and self.eng.T or (self.drv and self.drv.thr or self.throttleOut)),
            brake         = self.drv and self.drv.brk or 0,
            frontDiffLocked = (self.frontDiffType == 2 and self.c[112] <= 0.999) or (self.frontDiffLocked == true) or (self.frontLockActive == true) or (self.allLockActive == true),
            rearDiffLocked  = (self.rearDiffType == 2 and self.c[112] >= 0.001) or (self.rearDiffLocked == true) or (self.rearLockActive == true) or (self.allLockActive == true),
            allDiffLocked   = (self.allLockActive == true),
            frontActiveLockPct = self.frontActiveLockPct or 0,
            rearActiveLockPct  = self.rearActiveLockPct or 0,
            torqueNm      = (self._solvedTq or self:torque()) * 1000,
            engineLoad    = engineLoad,
            boostPSI      = boostPSI,
            manifoldPSI   = self.eng and self.eng.manifoldPSI or 0,
            engineStalled = self.engineStalled == true,
            jakeBrakeStage = self.jakeBrakeStage or 0,
            jakeBrakeActive = self.jakeBrakeActive == true,
            slipDeg       = self:slipAngle() * 57.2957795,
            manual        = self.manualBox,
            a0            = self.steer.a0,
            a1            = self.steer.a1,
        }
    end
end


initializeVehiclePhysicsBridge(SimcadeVehiclePhysics)

_G.SimcadeVehiclePhysics = SimcadeVehiclePhysics
return SimcadeVehiclePhysics
