--@name Simcade Particle Effects Module (Smoke, Dirt, Grass, Snow - Skid & Roll)
--@shared

SimcadeEffects = SimcadeEffects or {}

if SERVER then return end

local clamp = math.clamp or function(v, a, b) return v < a and a or (v > b and b or v) end

function SimcadeEffects.initClient(cfg)
    local FX = cfg.EFFECT_CFG or {}
    local TIRE = cfg.TIRE_CFG or {}
    if TIRE.EnableParticles == false or FX.enabled == false then return end

    local emitter = nil
    local emitterOrigin = nil
    local permWarned = false
    local nextFxTick = 0
    local latestWheels = {}
    local lastCarFwd = Vector(1, 0, 0)
    local lastCarUp  = Vector(0, 0, 1)
    local wheelEmitStamp = {}
    local surfacesCfg = FX.Surfaces or {}

    local defaultSmokePaths = FX.SmokeMaterials or { FX.SmokeMaterial or "particle/particle_smokegrenade", "particle/particle_smokegrenade1" }
    local defaultDirtPaths  = FX.DirtMaterials  or { FX.DirtMaterial  or "particle/particle_smokegrenade", "particle/particle_smokegrenade1" }
    local defaultGrassPaths = FX.GrassMaterials or { "particle/debris2/debris_2", "particle/particle_smokegrenade" }
    local matCache = {}

    local function loadFxMat(path)
        if not path or path == "" then return nil end
        if matCache[path] ~= nil then return matCache[path] end
        local ok, m = pcall(function() return material.load(path) end)
        matCache[path] = (ok and m) or false
        return matCache[path]
    end

    local function pickRandomMat(paths, fallbackPaths)
        local list = (type(paths) == "table" and #paths > 0) and paths or fallbackPaths
        if type(list) == "string" then list = { list } end
        if type(list) ~= "table" or #list == 0 then return nil end
        local tries = #list
        local startIdx = math.random(1, tries)
        for k = 0, tries - 1 do
            local idx = ((startIdx + k - 1) % tries) + 1
            local m = loadFxMat(list[idx])
            if m then return m end
        end
        return nil
    end

    local DEFAULT_SKID_SMOKE = {
        enabled = true, spawnOnSkid = true,
        path = defaultSmokePaths, color = Color(255, 255, 255), affectByLight = false,
        startAt = 0.14, maxAt = 1.0,
        sizeStart = 30, sizeEnd = 120, alphaStart = 255, alphaEnd = 0,
        multMin = 0.9, multMax = 1.1,
        velocityMult = Vector(0.1, 1.0, 0.01), gravity = Vector(0, 0, 20),
        airResistance = 50, ttl = 5.0, interval = 0.10,
    }

    local DEFAULT_SURFACES = {
        [68] = { -- MAT_DIRT (wheels_fx roaddust)
            roll = {
                enabled = true, spawnOnSkid = false, path = defaultDirtPaths,
                color = Color(189, 157, 121), affectByLight = false,
                startAt = 20, maxAt = 800, sizeStart = 20, sizeEnd = 300,
                alphaStart = 255, alphaEnd = 0, multMin = 0.9, multMax = 1.1,
                velocityMult = Vector(0.1, 1.0, 0.01), gravity = Vector(0, 0, 20),
                airResistance = 50, ttl = 2.0, interval = 0.07,
            },
        },
        [71] = { enabled = false }, -- MAT_GRATE
        [74] = { -- MAT_SNOW (wheels_fx snow)
            roll = {
                enabled = true, spawnOnSkid = false, path = defaultSmokePaths,
                color = Color(255, 255, 255), affectByLight = false,
                startAt = 60, maxAt = 1800, sizeStart = 20, sizeEnd = 200,
                alphaStart = 255, alphaEnd = 0, multMin = 0.9, multMax = 1.1,
                velocityMult = Vector(0.1, 1.0, 0.1), gravity = Vector(0, 0, 60),
                airResistance = 80, ttl = 2.0, interval = 0.07,
            },
        },
        [78] = { -- MAT_SAND (wheels_fx roaddust)
            roll = {
                enabled = true, spawnOnSkid = false, path = defaultDirtPaths,
                color = Color(189, 157, 121), affectByLight = false,
                startAt = 20, maxAt = 800, sizeStart = 20, sizeEnd = 300,
                alphaStart = 255, alphaEnd = 0, multMin = 0.9, multMax = 1.1,
                velocityMult = Vector(0.1, 1.0, 0.01), gravity = Vector(0, 0, 20),
                airResistance = 50, ttl = 2.0, interval = 0.07,
            },
        },
        [85] = { -- MAT_GRASS (wheels_fx grass)
            roll = {
                enabled = true, spawnOnSkid = false, path = { "particle/debris2/debris_2" },
                color = Color(100, 120, 70), affectByLight = true,
                startAt = 200, maxAt = 800, sizeStart = 10, sizeEnd = 10,
                alphaStart = 255, alphaEnd = 255, multMin = 0.5, multMax = 1.1,
                velocityMult = Vector(0.1, 1.0, 0.1), gravity = Vector(0, 0, -600),
                airResistance = 5, ttl = 2.0, interval = 0.05,
            },
        },
    }

    local function resolveSurfaceCfg(matType)
        local sCfg = surfacesCfg[matType] or DEFAULT_SURFACES[matType] or surfacesCfg[0] or surfacesCfg[88]
        if sCfg then return sCfg end
        return { enabled = true, skid = DEFAULT_SKID_SMOKE }
    end

    local function isCpuHot(limit)
        local qMax = quotaMax and quotaMax() or 0
        if qMax <= 0 then return false end
        if quotaUsed and (quotaUsed() / qMax) > limit then return true end
        if quotaAverage and (quotaAverage() / qMax) > limit then return true end
        return false
    end

    local skidEffectExp = math.max(0.05, FX.skidEffectExp or 1)
    local fxThresh = FX.smokeThreshold or 0.14
    local emitInterval = math.max(0.015, FX.emitInterval or 0.025)
    local maxPerBatch = FX.maxParticlesPerBatch or 12

    function SimcadeEffects.sampleWheel(i, pos, skid, matType, rimSpeed, grounded, now, fwd, up)
        lastCarFwd, lastCarUp = fwd, up
        latestWheels[i] = {pos = pos, skid = skid, matType = matType, rimSpeed = rimSpeed, grounded = grounded, time = now}
    end

    local function ensureEmitter(anchorPos)
        if hasPermission and not hasPermission("particle.create") then
            if not permWarned then
                permWarned = true
                print("[SimcadeEffects] Particle creation permission not granted (particle.create).")
            end
            return nil
        end
        if not (particle and particle.create) then return nil end
        if emitter and emitterOrigin and emitterOrigin:getDistance(anchorPos) > 2048 then
            if emitter.destroy then pcall(function() emitter:destroy() end) end
            emitter = nil
        end
        if not emitter then
            local ok, em = pcall(function() return particle.create(anchorPos, false) end)
            if ok and em then
                emitter = em
                emitterOrigin = anchorPos
            elseif not permWarned then
                permWarned = true
                print("[SimcadeEffects] particle.create failed: " .. tostring(em))
            end
        elseif emitter.setPos then
            pcall(function() emitter:setPos(anchorPos) end)
        end
        return emitter
    end

    local function spawnLayer(em, item, layerCfg, isRollLayer, now, key, carFwd, carUp)
        if not layerCfg or layerCfg.enabled == false then return false end
        local interval = layerCfg.interval or emitInterval
        local lastT = wheelEmitStamp[key] or 0
        if (now - lastT) < interval then return false end

        local strength = 0
        if isRollLayer then
            local startAt = layerCfg.startAt or 60
            local maxAt   = math.max(startAt + 1, layerCfg.maxAt or 700)
            if item.rimSpeed < startAt then return false end
            strength = clamp((item.rimSpeed - startAt) / (maxAt - startAt), 0.1, 1.0)
        else
            local startAt = layerCfg.startAt or fxThresh
            local maxAt   = math.max(startAt + 0.05, layerCfg.maxAt or 1.0)
            if item.skid < startAt then return false end
            strength = clamp((item.skid - startAt) / (maxAt - startAt), 0, 1) ^ skidEffectExp
        end

        if em.getParticlesLeft and em:getParticlesLeft() == 0 then return false end

        local fallback = isRollLayer and defaultDirtPaths or defaultSmokePaths
        local pMat = pickRandomMat(layerCfg.path, fallback)
        if not pMat then return false end

        local rMin = layerCfg.multMin or 0.85
        local rMax = layerCfg.multMax or 1.15
        local rMult = (rMin == rMax) and rMin or (rMin + math.random() * (rMax - rMin))

        local sizeStart  = math.max(2, (layerCfg.sizeStart or 12) * (0.5 + strength * 0.7) * rMult)
        local sizeEnd    = math.max(sizeStart, (layerCfg.sizeEnd or 48) * (0.5 + strength * 0.7) * rMult)
        local startA     = math.floor(clamp((layerCfg.alphaStart or 160) * (0.4 + strength * 0.6), 20, 255))
        local endA       = layerCfg.alphaEnd or 0
        local ttl        = math.max(0.15, (layerCfg.ttl or 1.0) * (0.7 + strength * 0.4) * rMult)
        local col        = layerCfg.color or Color(235, 235, 235)
        local grav       = layerCfg.gravity or Vector(0, 0, 20)
        local airRes     = layerCfg.airResistance or 50
        local vm         = layerCfg.velocityMult or Vector(-0.35, 0.15, 0.06)

        local latSide = carFwd:cross(carUp):getNormalized()
        local baseSpd = math.max(60, item.rimSpeed)
        local backMult = (vm.x < 0) and (-vm.x) or math.max(0.28, vm.x)
        local latMult  = math.min(0.18, math.abs(vm.y) * 0.15)
        local vel = (-carFwd * (backMult * baseSpd) + latSide * (latMult * baseSpd * (math.random() * 2 - 1)) + carUp * (vm.z * baseSpd)) * rMult

        local okSpawn = pcall(function()
            local p = em:add(pMat, item.pos + carUp * 3, sizeStart, sizeEnd, 0, 0, startA, endA, ttl)
            if p then
                if p.setStartSize then p:setStartSize(sizeStart) end
                if p.setEndSize then p:setEndSize(sizeEnd) end
                if p.setStartAlpha then p:setStartAlpha(startA) end
                if p.setEndAlpha then p:setEndAlpha(endA) end
                if p.setCollide then p:setCollide(true) end
                if p.setRoll then p:setRoll(math.random(0, 360)) end
                if p.setRollDelta then p:setRollDelta((math.random() * 2 - 1) * 1.0) end
                if p.setLighting then p:setLighting(layerCfg.affectByLight == true) end
                if p.setColor then p:setColor(col) end
                if p.setVelocity then p:setVelocity(vel) end
                if p.setGravity then p:setGravity(grav) end
                if p.setAirResistance then p:setAirResistance(airRes) end
            end
        end)

        if okSpawn then
            wheelEmitStamp[key] = now
            if SimcadeAddons and SimcadeAddons._clientHost then
                SimcadeAddons._clientHost.runHook("OnEffectEmit", item, layerCfg, isRollLayer)
            end
            return true
        end
        return false
    end

    hook.add("think", "SimcadeEffects_Update", function()
        local now = timer.curtime()
        if now < nextFxTick then return end
        nextFxTick = now + emitInterval
        if isCpuHot(0.78) then return end

        local sample = SimcadePrediction and SimcadePrediction.getTireEffects and SimcadePrediction.getTireEffects()
        local wheels = sample and sample.wheels or latestWheels
        local carFwd, carUp = sample and sample.fwd or lastCarFwd, sample and sample.up or lastCarUp
        local anchorPos = nil
        for i = 1, #wheels do
            local w = wheels[i]
            if w and w.grounded and (now - w.time) <= 0.40 and (w.skid >= fxThresh or w.rimSpeed >= 40) then
                anchorPos = w.pos
                break
            end
        end
        if not anchorPos then return end

        local em = ensureEmitter(anchorPos)
        if not em then return end

        local emitted = 0
        for i = 1, #wheels do
            if emitted >= maxPerBatch then break end
            local item = wheels[i]
            if item and item.grounded and (now - item.time) <= 0.40 then
                local sCfg = resolveSurfaceCfg(item.matType)
                if sCfg.enabled ~= false then
                    local skidLayer = sCfg.skid or (sCfg.path and sCfg) or ((not sCfg.roll) and DEFAULT_SKID_SMOKE) or nil
                    local rollLayer = sCfg.roll
                    if skidLayer and spawnLayer(em, item, skidLayer, false, now, "s" .. i, carFwd, carUp) then
                        emitted = emitted + 1
                    end
                    if rollLayer and emitted < maxPerBatch and spawnLayer(em, item, rollLayer, true, now, "r" .. i, carFwd, carUp) then
                        emitted = emitted + 1
                    end
                end
            end
        end
    end)

    hook.add("removed", "SimcadeEffects_Cleanup", function()
        if emitter and emitter.destroy then pcall(function() emitter:destroy() end) end
        emitter = nil
    end)
end
