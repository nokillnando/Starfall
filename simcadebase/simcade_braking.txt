--@name Simcade Braking & ABS Module
--@shared

SimcadeBraking = SimcadeBraking or {}

local abs = math.abs
local clamp = math.clamp or function(v, a, b) return v < a and a or (v > b and b or v) end
local max = math.max
local min = math.min

function SimcadeBraking.configure(brakeCfg, massT, massScale, engineCfg)
    brakeCfg = brakeCfg or {}
    engineCfg = engineCfg or {}
    local function pick(k)
        if brakeCfg[k] ~= nil then return brakeCfg[k] end
        return engineCfg[k]
    end

    local baseBrakeNm = max(50, pick("brakeTorque") or 2800)
    local brakeMultiplier = baseBrakeNm / 2800
    local frontBias = 0.535
    local cfgFrontBias = pick("frontBrakeBias")
    if cfgFrontBias then
        frontBias = clamp(cfgFrontBias, 0.15, 0.85)
    end
    local baseBrakeKNm = baseBrakeNm * 0.001
    local fallbackRearBrake = max(0.5 * massScale * brakeMultiplier, 2.2 * massT)
    local brakeHighSpeedStartKMH = max(0, tonumber(pick("BrakeHighSpeedStartKMH")) or 80)
    local frontBrakeTorque = pick("FrontBrakeTorque")
    local rearBrakeTorque = pick("RearBrakeTorque")
    local handbrakeTorque = pick("HandbrakeTorque")

    return {
        frontBrakeKNm = frontBrakeTorque and max(0.05, frontBrakeTorque * 0.001) or (baseBrakeKNm * frontBias * 2.0),
        rearBrakeKNm = rearBrakeTorque and max(0.05, rearBrakeTorque * 0.001) or (baseBrakeKNm * (1.0 - frontBias) * 2.0),
        rampTime = clamp(tonumber(pick("BrakeRampTime")) or 0.18, 0.02, 1.0),
        highSpeedScale = clamp(tonumber(pick("BrakeHighSpeedScale")) or 1.0, 0.1, 2.0),
        highSpeedStartKMH = brakeHighSpeedStartKMH,
        highSpeedRefKMH = max(brakeHighSpeedStartKMH + 10, tonumber(pick("BrakeHighSpeedRefKMH")) or 200),
        burnoutAssist = (pick("BurnoutAssist") ~= false),
        abs = {
            enabled = (pick("ABSEnabled") ~= false),
            threshold = clamp(tonumber(pick("ABSLockThreshold")) or 0.22, 0.02, 0.85),
            pulseRate = clamp(tonumber(pick("ABSPulseRate")) or 14, 4, 120),
            release = clamp(tonumber(pick("ABSReleaseRatio")) or 0.25, 0.05, 0.90),
        },
        handbrake = {
            torqueKNm = handbrakeTorque and max(0.05, handbrakeTorque * 0.001) or (fallbackRearBrake * 10),
            lockWheels = (pick("HandbrakeLockWheels") ~= false),
        },
    }
end

function SimcadeBraking.updatePedal(car, driver, rawBrake, dt)
    local brakeCfg = car.braking
    if rawBrake > 0.05 then
        local rampTime = max(0.02, brakeCfg.rampTime or 0.18)
        driver.brakeRamp = min(1.0, (driver.brakeRamp or 0.35) + (dt or 0.015) / rampTime * 0.65)
    else
        driver.brakeRamp = 0.35
    end
    return rawBrake * (driver.brakeRamp or 1.0)
end

function SimcadeBraking.applyWheelTorque(car, wheel, peakLongKN, vFwd, absFwd)
    local brakeCfg = car.braking
    local driver = car.drv
    local isBurnout = (brakeCfg.burnoutAssist ~= false) and (driver and (driver.thr or 0) > 0.5 and (driver.brk or 0) > 0.3 and absFwd < 8.0)
    local pedalBrake = (isBurnout and wheel.i >= 2) and (absFwd * 0.05) or clamp((driver and driver.brk) or wheel.brk or 0, 0, 1)
    local brakeTorque = (wheel.i < 2) and (brakeCfg.frontBrakeKNm or 2.00) or (brakeCfg.rearBrakeKNm or 1.22)
    if wheel.brakeTempMult ~= nil then
        brakeTorque = brakeTorque * clamp(wheel.brakeTempMult, 0.15, 1.25)
    end
    local highSpeedScale = brakeCfg.highSpeedScale or 1.0
    if abs(highSpeedScale - 1.0) > 0.001 then
        local startMS = (brakeCfg.highSpeedStartKMH or 80) / 3.6
        if absFwd > startMS then
            local refMS = max(startMS + 2.0, (brakeCfg.highSpeedRefKMH or 200) / 3.6)
            local speedFraction = clamp((absFwd - startMS) / (refMS - startMS), 0, 1)
            brakeTorque = brakeTorque * (1.0 + (highSpeedScale - 1.0) * speedFraction)
        end
    end
    local brakeForce = brakeTorque * pedalBrake
    local absCfg = brakeCfg.abs
    local absReleasing = false
    wheel.absActive = false

    if absCfg and absCfg.enabled and wheel.hbk < 0.1 and pedalBrake > 0.05 and absFwd > 2.2 and peakLongKN > 0.001 then
        local kappaEstimate = (wheel.radius * wheel.w - vFwd) / max(2.2, absFwd + 0.5)
        local absSlip = (vFwd >= 0) and (kappaEstimate < -absCfg.threshold) or (kappaEstimate > absCfg.threshold)
        local overTorque = brakeForce > (wheel.radius * peakLongKN * 0.98)
        local allowLock = (absCfg.threshold >= 0.45) or (absCfg.release >= 0.65)
        if (wheel.locked ~= 0) or absSlip or (not allowLock and overTorque) then
            wheel.absActive = true
            if not allowLock then wheel.locked = 0 end
            local period = 1.0 / max(4, absCfg.pulseRate)
            local phase = (car.now or 0) % period
            if phase < period * 0.55 then
                brakeForce = brakeForce * absCfg.release
                absReleasing = true
                local recovery = clamp(1.0 - absCfg.release, 0.15, 0.85)
                local targetWheelSpeed = (vFwd * (1.0 - absCfg.threshold * 0.45)) / max(0.15, wheel.radius)
                wheel.w = wheel.w + (targetWheelSpeed - wheel.w) * recovery
            else
                local applyCap = allowLock and (0.95 + absCfg.threshold * 0.5) or (0.88 + absCfg.threshold * 0.35)
                brakeForce = min(brakeForce, wheel.radius * peakLongKN * applyCap)
                local floorWheelSpeed = (vFwd * (1.0 - absCfg.threshold * 0.90)) / max(0.15, wheel.radius)
                if (vFwd >= 0 and wheel.w < floorWheelSpeed) or (vFwd < 0 and wheel.w > floorWheelSpeed) then
                    wheel.w = floorWheelSpeed
                end
            end
        end
    end

    local handbrakeCfg = brakeCfg.handbrake
    local handbrakeTorque = (handbrakeCfg and handbrakeCfg.torqueKNm or 0) * wheel.hbk
    if pedalBrake > 0.05 and (driver and driver.thr or 0) < 0.05 and ((vFwd >= 0 and wheel.T > 0) or (vFwd < 0 and wheel.T < 0)) then
        wheel.T = 0
    end
    local capacity = brakeForce + handbrakeTorque
    wheel.serviceBrakeKNm = brakeForce
    if wheel.locked == 0 and capacity > 0 then
        local h = car.substepDt
        local rimSpeed = wheel.w * wheel.radius
        local stopTorque = wheel.I * wheel.w / h
        if abs(rimSpeed - vFwd) <= 0.16 * max(2.2, absFwd + 0.5) then
            -- Rolling contact couples the brake to this corner's chassis mass.
            stopTorque = car.m * 0.25 * vFwd * wheel.radius / h
        end
        -- A stationary brake can balance drive torque up to its capacity.
        -- At rest with no drive, its requested torque is exactly zero.
        local appliedTorque = clamp(stopTorque + wheel.T, -capacity, capacity)
        wheel.T = wheel.T - appliedTorque
    end

    return pedalBrake, brakeForce, handbrakeTorque, absReleasing
end

function SimcadeBraking.applyWheelLock(car, wheel, peakLongKN, peakLatKN, vFwd, absFwd, pedalBrake, brakeForce, handbrakeTorque, absReleasing)
    local brakeCfg = car.braking
    local handbrakeCfg = brakeCfg.handbrake
    if wheel.i >= 2 and (wheel.hbk or 0) > 0.1 then
        local lockWheels = (not handbrakeCfg) or handbrakeCfg.lockWheels
        if lockWheels or handbrakeTorque > wheel.radius * peakLongKN * 1.02 then
            wheel.locked = 1
            wheel.w = 0
        else
            wheel.locked = 0
        end
    elseif peakLongKN > 0.001 then
        local absCfg = brakeCfg.abs
        local allowAbsLock = absCfg and absCfg.enabled and ((absCfg.threshold >= 0.45) or (absCfg.release >= 0.65))
        local totalBrakeTorque = handbrakeTorque + abs(brakeForce)
        if absReleasing or (wheel.absActive and not allowAbsLock) or (absCfg and absCfg.enabled and not allowAbsLock and absFwd > 2.2) or totalBrakeTorque <= wheel.radius * peakLongKN * 1.02 then
            wheel.locked = 0
        elseif totalBrakeTorque > wheel.radius * peakLongKN * 1.02 then
            wheel.locked = 1
            wheel.w = 0
        else
            wheel.locked = 0
        end
    else
        wheel.locked = 0
    end
    return peakLatKN
end

return SimcadeBraking
