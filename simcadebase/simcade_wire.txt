--@name Simcade Wiremod & Net Sync Module
--@shared

SimcadeWire = SimcadeWire or {}
if CLIENT then return end

local solverSlotsInWireOrder = { 0, 1, 3, 2 } -- FL, FR, RL, RR
local detailedWheelOutputs = false

function SimcadeWire.initPorts(VEHICLE_CFG, WHEELS, extraInputs, extraOutputs)
    local inNames = { "Pod", "Base", "EngineSound", "ExtraFilterStatic", "ExtraFilterSemiDynamic" }
    local inTypes = { "Entity", "Entity", "Entity", "Array", "Array" }

    if VEHICLE_CFG and VEHICLE_CFG.enableExternalWireOverrides then
        local extNames = { "Active", "FrontWheels", "RearWheels", "EngineThrottle", "EngineBrake", "Handbrake", "Clutch", "GearUp", "GearDown", "Ignition", "TrailerMode" }
        local extTypes = { "Number", "Array", "Array", "Number", "Number", "Number", "Number", "Number", "Number", "Number", "Number" }
        for i = 1, #extNames do
            table.insert(inNames, extNames[i])
            table.insert(inTypes, extTypes[i])
        end
    end
    if type(extraInputs) == "table" then
        for _, inp in ipairs(extraInputs) do
            if inp.name then
                table.insert(inNames, inp.name)
                table.insert(inTypes, inp.type or "Number")
            end
        end
    end
    wire.adjustInputs(inNames, inTypes)

    local outNames = {
        "SpeedKMH", "SpeedMPH",
        "Throttle", "Brake", "Handbrake", "Ignition", "Active",
        "RPM", "FlywheelRPM", "Gear", "GearName", "Clutch",
        "Torque", "MaxTorque", "EngineLoad", "BoostPSI", "TurboPSI",
        "SlipAngle", "LongSlipRatio", "IsDrifting", "IsSlipping",
        "ActiveCompound", "AppliedForce",
        "FrontDiffLocked", "RearDiffLocked", "AllDiffLocked", "FrontActiveLockPct", "RearActiveLockPct", "ABSActive",
        "FL_Load", "FR_Load", "RL_Load", "RR_Load",
        "GroundedCount", "TotalMass",
        "SteeringAngleDeg", "FrontAxle", "RearAxle"
    }
    local outTypes = {
        "Number", "Number",
        "Number", "Number", "Number", "Number", "Number",
        "Number", "Number", "Number", "String", "Number",
        "Number", "Number", "Number", "Number", "Number",
        "Number", "Number", "Number", "Number",
        "String", "Number",
        "Number", "Number", "Number", "Number", "Number", "Number",
        "Number", "Number", "Number", "Number",
        "Number", "Number",
        "Number", "Entity", "Entity"
    }
    detailedWheelOutputs = VEHICLE_CFG and VEHICLE_CFG.detailedWheelWireOutputs == true
    if detailedWheelOutputs then
        for _, name in ipairs({ "SteeringInput", "FL_SteerDeg", "FR_SteerDeg" }) do
            table.insert(outNames, name)
            table.insert(outTypes, "Number")
        end
        for _, name in ipairs({ "WheelCamberDeg", "WheelCamberThrustKN", "WheelLateralForceKN", "WheelSlipAngleDeg", "WheelGrounded" }) do
            table.insert(outNames, name)
            table.insert(outTypes, "Array")
        end
    end
    local wheelPortNames = {}
    if type(WHEELS) == "table" and #WHEELS > 0 then
        for i, w in ipairs(WHEELS) do
            local pName = (type(w) == "table" and w.name and w.name ~= "")
                and ("Wheel_" .. tostring(w.name):gsub("%W+", "_"))
                or ("Wheel" .. i)
            wheelPortNames[i] = pName
            table.insert(outNames, pName)
            table.insert(outTypes, "Entity")
        end
    else
        for i = 1, 4 do
            local pName = "Wheel" .. i
            wheelPortNames[i] = pName
            table.insert(outNames, pName)
            table.insert(outTypes, "Entity")
        end
    end
    SimcadeWire._wheelPortNames = wheelPortNames
    if type(extraOutputs) == "table" then
        for _, outp in ipairs(extraOutputs) do
            if outp.name then
                table.insert(outNames, outp.name)
                table.insert(outTypes, outp.type or "Number")
            end
        end
    end
    wire.adjustOutputs(outNames, outTypes)
    return inNames
end

function SimcadeWire.writeOutputs(data)
    wire.ports.SpeedKMH        = data.kmh or 0
    wire.ports.SpeedMPH        = data.mph or 0
    wire.ports.Throttle        = math.round(data.throttle or 0, 3)
    wire.ports.Brake           = math.round(data.brake or 0, 3)
    wire.ports.Handbrake       = math.round(data.handbrake or 0, 3)
    wire.ports.Ignition        = data.ignitionMode or data.ignitionState or 0
    wire.ports.Active          = data.engineRunning and 1 or 0
    wire.ports.RPM             = math.round(data.rpm or 0)
    wire.ports.FlywheelRPM     = math.round(data.flywheelRPM or 0)
    wire.ports.Gear            = tonumber(data.gearNumber) or 0
    wire.ports.GearName        = tostring(data.gearName or "N")
    wire.ports.Clutch          = math.round(data.clutch or 0, 2)
    wire.ports.Torque          = math.round(data.torqueNm or 0)
    wire.ports.MaxTorque       = math.round(data.maxTorque or data.maxTorqueNm or 0)
    wire.ports.EngineLoad      = math.round(data.engineLoad or 0, 3)
    wire.ports.BoostPSI        = data.boostPSI or data.turboPSI or 0
    wire.ports.TurboPSI        = data.turboPSI or data.boostPSI or 0
    wire.ports.SlipAngle       = math.round(data.slipAngleDeg or 0, 1)
    wire.ports.LongSlipRatio   = math.round(data.longSlipPct or 0, 1)
    wire.ports.IsDrifting      = data.isDrifting and 1 or 0
    wire.ports.IsSlipping      = data.isSlipping and 1 or 0
    wire.ports.ActiveCompound  = tostring(data.activeCompound or "Sport")
    wire.ports.AppliedForce    = math.round(data.appliedForce or 0)
    wire.ports.FrontDiffLocked    = data.frontDiffLocked and 1 or 0
    wire.ports.RearDiffLocked     = data.rearDiffLocked and 1 or 0
    wire.ports.AllDiffLocked      = data.allDiffLocked and 1 or 0
    wire.ports.FrontActiveLockPct = math.round(data.frontActiveLockPct or 0, 1)
    wire.ports.RearActiveLockPct  = math.round(data.rearActiveLockPct or 0, 1)
    wire.ports.ABSActive       = data.absActive and 1 or 0
    wire.ports.FL_Load         = math.round(data.flLoad or 0)
    wire.ports.FR_Load         = math.round(data.frLoad or 0)
    wire.ports.RL_Load         = math.round(data.rlLoad or 0)
    wire.ports.RR_Load         = math.round(data.rrLoad or 0)
    wire.ports.GroundedCount   = data.groundedCount or 0
    wire.ports.TotalMass       = math.round(data.totalMass or 0)
    wire.ports.SteeringAngleDeg = math.round(data.steerDeg or 0, 2)
    if detailedWheelOutputs then
        wire.ports.SteeringInput   = data.steerInput or 0
        wire.ports.FL_SteerDeg = math.round(math.deg(data.frontLeftSteerRad or 0), 2)
        wire.ports.FR_SteerDeg = math.round(math.deg(data.frontRightSteerRad or 0), 2)
        -- Wire arrays use vehicle corner order FL, FR, RL, RR. Solver slots are FL, FR, RR, RL.
        local camber, thrust, lateral, slip, contact = {}, {}, {}, {}, {}
        for corner, slot in ipairs(solverSlotsInWireOrder) do
            local w = data.solverWheels and data.solverWheels[slot]
            local grounded = w and w.alignment and w.alignment.isGrounded
            contact[corner] = grounded and 1 or 0
            camber[corner] = grounded and math.deg(w.roadCamberRad or 0) or 0
            thrust[corner] = grounded and w.camberFy or 0
            lateral[corner] = grounded and w.Fy or 0
            slip[corner] = grounded and math.deg(w.slipAngleRad or 0) or 0
        end
        wire.ports.WheelCamberDeg = camber
        wire.ports.WheelCamberThrustKN = thrust
        wire.ports.WheelLateralForceKN = lateral
        wire.ports.WheelSlipAngleDeg = slip
        wire.ports.WheelGrounded = contact
    end
    if type(data.wheelEntities) == "table" then
        local names = SimcadeWire._wheelPortNames or {}
        for i = 1, #data.wheelEntities do
            local pName = names[i] or ("Wheel" .. i)
            wire.ports[pName] = data.wheelEntities[i]
        end
    end
end
