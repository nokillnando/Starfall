--@name Simcade HUD Module (HUD & 3D Wheel Telemetry)
--@shared

SimcadeHUD = SimcadeHUD or {}

-- A reliable wheel page carries absolute field values only where they changed.
-- Baselines are per wheel and are committed by the server after a successful send.
function SimcadeHUD.wheelFields(previous, writing)
    local values, index = {}, 0
    local io = {}
    local function field(value, write, read, bits)
        index = index + 1
        local old
        if previous then old = previous[index] end
        if writing then
            local changed = value ~= old
            net.writeBool(changed)
            if changed then
                if bits then write(value, bits) else write(value) end
            end
        elseif net.readBool() then
            if bits then value = read(bits) else value = read() end
        else
            value = old
        end
        values[index] = value
        return value
    end
    function io.int(v, bits) return field(v, net.writeInt, net.readInt, bits) end
    function io.uint(v, bits) return field(v, net.writeUInt, net.readUInt, bits) end
    function io.bool(v) return field(v, net.writeBool, net.readBool) end
    function io.float(v) return field(v, net.writeFloat, net.readFloat) end
    function io.point(v)
        local function coordinate(x)
            return io.int(x and math.clamp(math.round(x * 8), -32768, 32767), 16) / 8
        end
        local x, y, z = coordinate(v and v.x), coordinate(v and v.y), coordinate(v and v.z)
        if not writing then return Vector(x, y, z) end
    end
    function io.direction(v)
        local function coordinate(x)
            return io.int(x and math.clamp(math.round(x * 127), -127, 127), 8)
        end
        local x, y, z = coordinate(v and v.x), coordinate(v and v.y), coordinate(v and v.z)
        if not writing then return Vector(x, y, z):getNormalized() end
    end
    return io, values
end

if SERVER then return end

local clamp = math.clamp or function(v, a, b) return v < a and a or (v > b and b or v) end

local function readDirection()
    return Vector(net.readInt(8), net.readInt(8), net.readInt(8)):getNormalized()
end

local hudState = nil
function SimcadeHUD.getState()
    return hudState
end

function SimcadeHUD.initClient(cfg)
    local HUD  = cfg.HUD_CFG or {}
    local TIRE = cfg.TIRE_CFG or {}

    local wheelDebug = { base = nil, fwdLocal = Vector(1, 0, 0), rightLocal = Vector(0, -1, 0), wheels = {} }
    local wheelBaselines = {}
    local smoothTelemetry = {}
    local cachedPacejkaLat = nil
    local cachedPacejkaLong = nil
    local lastPacejkaKey = nil

    local specCompound, frontSpec, rearSpec

    local function evalPacejka(B, C, D, E, x)
        local bx = B * x
        return D * math.sin(C * math.atan(bx - E * (bx - math.atan(bx))))
    end

    local mathFont, smallFont, speedFont, gearFont, hudSubFont, unitFont
    if render and render.createFont then
        pcall(function()
            mathFont   = render.createFont("Roboto Mono", 15, 500, true)
            smallFont  = render.createFont("Roboto Mono", 13, 600, true)
            speedFont  = render.createFont("Roboto Mono", 60, 700, true)
            gearFont   = render.createFont("Roboto Mono", 70, 800, true)
            hudSubFont = render.createFont("Roboto Mono", 16, 600, true)
            unitFont   = render.createFont("Roboto Mono", 13, 700, true)
        end)
    end
    local gtArcCache, gtArcKey = nil, nil
    local smoothThr, smoothBrk, smoothClu = 0, 0, 0
    local function setFontSafe(f)
        if f and render.setFont then render.setFont(f) end
    end
    local function drawRoundedSafe(r, x, y, w, h)
        if render.drawRoundedBox then
            render.drawRoundedBox(r, x, y, w, h)
        else
            render.drawRect(x, y, w, h)
        end
    end
    local function drawCircleOutlineSafe(cx, cy, r)
        if render.drawCircle then
            render.drawCircle(cx, cy, r)
        else
            drawOutlineSafe(cx - r, cy - r, r * 2, r * 2)
        end
    end
    local function drawTextAligned(x, y, txt, ax, ay)
        if render.drawSimpleText then
            render.drawSimpleText(x, y, txt, ax or 0, ay or 0)
        else
            render.drawText(x, y, txt)
        end
    end
    local function drawOutlineSafe(x, y, w, h)
        if render.drawRectOutline then
            render.drawRectOutline(x, y, w, h)
        end
    end
    local function forceReading(kn)
        if math.abs(kn) < 0.005 then return "0.00 kN" end
        return string.format("%.2f kN %s", math.abs(kn), kn > 0 and "left" or "right")
    end

    local function placeWheelCard(points, i, sw, sh, width, height, driving)
        local p = points[i]
        local gap = 5
        if driving and #cfg.WHEELS <= 4 then
            -- Keep each corner in its own slot through suspension travel and contact changes.
            local inset = math.min(sw * 0.085, math.max(0, sw / 2 - width - 4))
            local x = sw / 2 + (p.wheel.isRightSide and inset or -inset - width)
            local top = clamp(sh * 0.69 - 155, 4, sh - 159)
            return x, top + (p.wheel.isFront and 0 or 80)
        end
        local x = clamp(p.x - width / 2, 4, sw - width - 4)
        local idealY = p.y + 18
        local y, bestScore
        local function tryY(candidate)
            candidate = clamp(candidate, 4, sh - height - 4)
            local overlap = 0
            for j = 1, i - 1 do
                local other = points[j]
                local dx = width + gap - math.abs(x - other.cardX)
                local dy = math.min(candidate + height, other.cardBottom) - math.max(candidate, other.cardY) + gap
                if dx > 0 and dy > 0 then overlap = overlap + dx * dy end
            end
            local score = overlap * sh + math.abs(candidate - idealY)
            if not bestScore or score < bestScore then y, bestScore = candidate, score end
        end
        tryY(idealY)
        for j = 1, i - 1 do
            local other = points[j]
            tryY(other.cardY - height - gap)
            tryY(other.cardBottom + gap)
        end
        return x, y
    end

    local function drawWheelCallouts(points, sw, sh, driving)
        local width = 216
        for i, p in ipairs(points) do
            local w = p.wheel
            local height = w.isGrounded and 75 or 25
            local x, y = placeWheelCard(points, i, sw, sh, width, height, driving)
            p.cardX, p.cardY, p.cardBottom = x, y, y + height
            local accent = w.isFront and Color(95, 220, 245) or Color(255, 186, 96)
            render.setColor(accent)
            render.drawLine(p.x, p.y, clamp(p.x, x, x + width), clamp(p.y, y, y + height))
            render.setColor(Color(16, 22, 29, 206))
            render.drawRect(x, y, width, height)
            render.setColor(accent)
            render.drawRect(x, y + 5, 2, 15)
            setFontSafe(mathFont)
            drawTextAligned(x + 8, y + 4, w.name, 0, 0)
            if not w.isGrounded then
                drawTextAligned(x + width - 8, y + 4, "No ground contact", 2, 0)
            else
                drawTextAligned(x + width - 8, y + 4, string.format("%.1f kN load", w.loadKN), 2, 0)
                setFontSafe(smallFont)
                render.setColor(Color(242, 245, 250))
                local lean = math.abs(w.camberDeg) < 0.05 and "upright" or string.format("%.1f° %s", math.abs(w.camberDeg), w.camberDeg < 0 and "in" or "out")
                drawTextAligned(x + 8, y + 23, "Camber " .. lean, 0, 0)
                drawTextAligned(x + width - 8, y + 23, string.format("Slip %.1f°", w.slipAngle), 2, 0)
                drawTextAligned(x + 8, y + 40, "Camber thrust", 0, 0)
                drawTextAligned(x + width - 8, y + 40, forceReading(w.camberFy), 2, 0)
                drawTextAligned(x + 8, y + 56, "Total lateral", 0, 0)
                drawTextAligned(x + width - 8, y + 56, forceReading(w.lateralFy), 2, 0)
            end
        end
    end
    local function drawCircleSafe(x, y, r)
        if render.drawFilledCircle then
            render.drawFilledCircle(x, y, r)
        else
            render.drawRect(x - r, y - r, r * 2, r * 2)
        end
    end

    net.receive("Simcade_StateSync", function()
        hudState = {
            base         = net.readEntity(),
            manual       = net.readBool(),
            gear         = net.readString(),
            rpm          = net.readFloat(),
            redline      = net.readFloat(),
            kmh          = net.readFloat(),
            compound     = net.readString(),
            halfThrottle = net.readBool(),
            clutchDown   = net.readBool(),
            ignition     = net.readUInt(2),
            running      = net.readBool(),
            steer        = net.readFloat(),
            pod          = net.readBool() and net.readEntity() or nil,
            wsEnabled    = net.readBool(),
            wsFrontPct   = net.readFloat(),
            wsLeftPct    = net.readFloat(),
            wsFwdG       = net.readFloat(),
            wsLatG       = net.readFloat(),
            wsVertMult   = net.readFloat(),
            clutch       = net.readUInt(8) / 255,
        }
    end)

    net.receive("Simcade_WheelDebug", function()
        wheelDebug.base       = net.readEntity()
        wheelDebug.fwdLocal   = readDirection()
        wheelDebug.rightLocal = readDirection()
        wheelDebug.comLocal = net.readVector()
        local hasRays = net.readBool()
        if net.readBool() then
            wheelBaselines, wheelDebug.wheels, smoothTelemetry = {}, {}, {}
        end
        local full = net.readBool()
        local first, count = net.readUInt(32), net.readUInt(4)
        if not full and not wheelBaselines[first] then return end
        local list = wheelDebug.wheels
        for i = first, first + count - 1 do
            local io, values = SimcadeHUD.wheelFields(not full and wheelBaselines[i], false)
            wheelBaselines[i] = values
            local hx = io.int(nil, 16) * 0.1
            local hy = io.int(nil, 16) * 0.1
            local hz = io.int(nil, 16) * 0.1
            list[i] = {
                hubLocal   = Vector(hx, hy, hz),
                latRatio   = io.int(nil, 16) * 0.001,
                longRatio  = io.int(nil, 16) * 0.001,
                slipAngle  = io.int(nil, 16) * 0.01,
                slipRatio  = io.int(nil, 16) * 0.001,
                isFront    = io.bool(),
                name       = (cfg.WHEELS[i] and cfg.WHEELS[i].name) or ("W" .. i),
                isGrounded = io.bool(),
                loadKN     = io.uint(nil, 16) * 0.01,
                camberDeg  = io.int(nil, 16) * 0.01,
                camberFy   = io.float(),
                lateralFy  = io.float(),
                isRightSide = io.bool(),
                axleLocal   = io.direction(),
                normalLocal = io.direction(),
            }
            if io.bool() then
                list[i].springTop = io.point()
                list[i].seatNear = io.uint(nil, 8) / 255
            end
            if hasRays then
                local debugRays = {}
                list[i].rays = debugRays
                local mount, axle = io.point(), io.direction()
                local up = wheelDebug.rightLocal:cross(wheelDebug.fwdLocal):getNormalized()
                local function readRay(start)
                    local state = io.uint(nil, 2)
                    if state == 3 then return end
                    local depth = io.uint(nil, 16) / 8
                    debugRays[#debugRays + 1] = { startLocal = start, hitLocal = start - up * depth, state = state }
                end
                if io.bool() then
                    local w = cfg.WHEELS[i]
                    SimcadeSuspension.eachTreadPoint(mount, up, axle, w.radius or 15, w.width or 9, readRay)
                else
                    readRay(mount)
                end
            end
        end
    end)

    hook.add("drawhud", "SimcadeHUD_Draw", function()
        local hudEnabled = (HUD.enabled ~= false)
        local debugEnabled = (HUD.debugEnabled ~= nil) and (HUD.debugEnabled == true) or (HUD.debugEnabled == nil and hudEnabled)
        if not hudEnabled and not debugEnabled then return end
        local qMax = quotaMax and quotaMax() or 0
        if qMax > 0 then
            if quotaUsed and (quotaUsed() / qMax) > 0.80 then return end
            if quotaAverage and (quotaAverage() / qMax) > 0.80 then return end
        end
        local lp = player()
        if not (lp and lp:isValid()) then return end
        local st = hudState
        local inThisSeat = lp:inVehicle() and (not st or not st.pod or not st.pod:isValid() or not lp.getVehicle or lp:getVehicle() == st.pod)
        if lp ~= owner() and not inThisSeat then return end
        local sw, sh = render.getResolution()

        local scale = (sh and sh > 0) and (sh / 1080) or 1

        if hudEnabled and HUD.showSpeedo ~= false and inThisSeat and st then
            local pred = SimcadePrediction and SimcadePrediction.get and SimcadePrediction.get(st.base)
            local kmh = math.max(0, math.floor((pred and math.abs(pred.result.forward_ms) * 3.6 or st.kmh or 0) + 0.5))
            local rpm = math.max(0, math.floor((pred and pred.result.flywheel_rpm or st.rpm or 0) + 0.5))
            local redline = math.max(1000, st.redline or 8000)
            local rpmFrac = clamp(rpm / redline, 0, 1)
            local gearStr = tostring(pred and pred.result.gearName or st.gear or "1")
            local shiftFlash = st.running and rpmFrac >= 0.94 and (math.sin(timer.curtime() * 42) > 0)

            -- Live smoothed pedal bars (clutch / brake / throttle / handbrake)
            local keyFwd = lp.keyDown and lp:keyDown(IN_KEY.FORWARD) or false
            local keyBack = lp.keyDown and lp:keyDown(IN_KEY.BACK) or false
            local keyHbk = lp.keyDown and lp:keyDown(IN_KEY.JUMP) or false
            local inRev = (gearStr == "R" or gearStr:sub(1, 1) == "R")
            local rawThr = (inRev and keyBack or keyFwd) and (st.halfThrottle and 0.5 or 1.0) or 0
            local rawBrk = (inRev and keyFwd or keyBack) and 1.0 or 0
            local rawClu = st.clutch
            if not st.running then rawThr = 0 end
            smoothThr = smoothThr + (rawThr - smoothThr) * 0.35
            smoothBrk = smoothBrk + (rawBrk - smoothBrk) * 0.35
            smoothClu = smoothClu + (rawClu - smoothClu) * 0.35

            local s = scale * 1.28
            local cx = sw * 0.5
            local baseY = sh - math.floor(114 * s)
            local bgW = math.floor(510 * s)
            local bgH = math.floor(98 * s)

            -- Subtle GT-style translucent floating backdrop (bottom-center)
            render.setColor(Color(8, 10, 14, 155))
            drawRoundedSafe(math.max(4, math.floor(10 * s)), cx - bgW * 0.5, baseY - math.floor(4 * s), bgW, bgH)

            -- Curved fine-tick RPM bar across top center (pre-computed per resolution)
            local tickCount = 68
            local arcKey = string.format("%d_%d_%.2f", sw, sh, s)
            if not gtArcCache or gtArcKey ~= arcKey then
                gtArcCache = {}
                gtArcKey = arcKey
                local arcR = 540 * s
                local halfAng = math.rad(16.5)
                local arcCenterY = baseY + math.floor(10 * s) + arcR
                for i = 1, tickCount do
                    local t = (i - 1) / (tickCount - 1)
                    local ang = -math.pi * 0.5 - halfAng + t * (halfAng * 2)
                    local cosA, sinA = math.cos(ang), math.sin(ang)
                    local tickLen = (t >= 0.84) and (14 * s) or (11 * s)
                    gtArcCache[i] = {
                        x1 = cx + cosA * arcR,
                        y1 = arcCenterY + sinA * arcR,
                        x2 = cx + cosA * (arcR - tickLen),
                        y2 = arcCenterY + sinA * (arcR - tickLen),
                        isWarn = (t >= 0.74 and t < 0.84),
                        isRed = (t >= 0.84),
                    }
                end
            end

            local litTicks = math.floor(rpmFrac * tickCount + 0.5)
            for i = 1, tickCount do
                local tk = gtArcCache[i]
                if i <= litTicks then
                    if tk.isRed then
                        render.setColor(shiftFlash and Color(255, 240, 240) or Color(245, 42, 42))
                    elseif tk.isWarn then
                        render.setColor(Color(250, 195, 50))
                    else
                        render.setColor(Color(246, 250, 255))
                    end
                else
                    render.setColor(tk.isRed and Color(175, 46, 46, 155) or Color(135, 146, 165, 115))
                end
                render.drawLine(tk.x1, tk.y1, tk.x2, tk.y2)
                render.drawLine(tk.x1 + 1, tk.y1, tk.x2 + 1, tk.y2)
            end

            -- Center SPEED | GEAR split
            local divTop = baseY + math.floor(25 * s)
            local divH = math.floor(56 * s)
            render.setColor(Color(220, 226, 236, 175))
            render.drawRect(math.floor(cx - 1), divTop, math.max(1, math.floor(1.5 * s)), divH)

            -- Speed (left of divider) + km/h underneath
            setFontSafe(speedFont)
            render.setColor(Color(246, 249, 255))
            drawTextAligned(cx - math.floor(13 * s), divTop + math.floor(21 * s), tostring(kmh), 2, 1)
            setFontSafe(hudSubFont)
            render.setColor(Color(195, 204, 218, 225))
            drawTextAligned(cx - math.floor(13 * s), divTop + math.floor(49 * s), "km/h", 2, 1)

            -- Gear (right of divider) + RPM subtly beneath
            setFontSafe(gearFont)
            local gearCol = shiftFlash and Color(255, 70, 70)
                or (inRev and Color(255, 195, 75) or Color(248, 250, 255))
            render.setColor(gearCol)
            drawTextAligned(cx + math.floor(14 * s), divTop + math.floor(22 * s), gearStr, 0, 1)
            setFontSafe(unitFont)
            render.setColor(rpmFrac > 0.88 and Color(255, 90, 90) or Color(165, 176, 195, 210))
            drawTextAligned(cx + math.floor(14 * s), divTop + math.floor(52 * s), string.format("%d RPM", rpm), 0, 1)

            -- Left (Clutch + Brake) & Right (Throttle) GT7 Vertical Pedal Brackets
            local barTop = baseY + math.floor(12 * s)
            local barH = math.floor(68 * s)
            local barW = math.max(6, math.floor(8 * s))
            local barGap = math.max(2, math.floor(3 * s))
            local tickW = math.floor(10 * s)

            local brkBarX = cx - math.floor(172 * s)
            local cluBarX = brkBarX - barW - barGap
            local thrBarX = cx + math.floor(164 * s)

            -- Left bracket (Clutch blue + Brake red)
            render.setColor(Color(210, 218, 230, 190))
            render.drawLine(cluBarX - tickW, barTop, brkBarX + barW + 2, barTop)
            render.drawLine(cluBarX - tickW, barTop + barH, brkBarX + barW + 2, barTop + barH)
            render.drawLine(brkBarX + barW + 2, barTop, brkBarX + barW + 2, barTop + barH)

            render.setColor(Color(28, 33, 44, 185))
            render.drawRect(cluBarX, barTop + 2, barW, barH - 4)
            render.drawRect(brkBarX, barTop + 2, barW, barH - 4)
            if smoothClu > 0.01 then
                local ch = math.floor((barH - 4) * clamp(smoothClu, 0, 1))
                render.setColor(Color(65, 185, 255))
                render.drawRect(cluBarX, barTop + barH - 2 - ch, barW, ch)
            end
            if smoothBrk > 0.01 then
                local bh = math.floor((barH - 4) * clamp(smoothBrk, 0, 1))
                render.setColor(Color(240, 48, 48))
                render.drawRect(brkBarX, barTop + barH - 2 - bh, barW, bh)
            end

            -- Right bracket (Throttle white)
            render.setColor(Color(210, 218, 230, 190))
            render.drawLine(thrBarX - 2, barTop, thrBarX + barW + tickW, barTop)
            render.drawLine(thrBarX - 2, barTop + barH, thrBarX + barW + tickW, barTop + barH)
            render.drawLine(thrBarX - 2, barTop, thrBarX - 2, barTop + barH)
            render.setColor(Color(28, 33, 44, 185))
            render.drawRect(thrBarX, barTop + 2, barW, barH - 4)
            if smoothThr > 0.01 then
                local th = math.floor((barH - 4) * clamp(smoothThr, 0, 1))
                render.setColor(Color(244, 248, 255))
                render.drawRect(thrBarX, barTop + barH - 2 - th, barW, th)
            end

            -- Flanking circular GT-style indicator rings (symmetrical 2 left, 2 right)
            local ringR = math.floor(15 * s)
            local function drawBadge(bx, by, active, col, label)
                render.setColor(active and Color(col.r, col.g, col.b, 40) or Color(22, 26, 34, 130))
                drawCircleSafe(bx, by, ringR)
                render.setColor(active and col or Color(130, 140, 158, 120))
                drawCircleOutlineSafe(bx, by, ringR)
                setFontSafe(unitFont)
                render.setColor(active and col or Color(130, 140, 158, 145))
                drawTextAligned(bx, by, label, 1, 1)
            end

            local row1Y = barTop + math.floor(18 * s)
            local row2Y = barTop + math.floor(52 * s)
            local leftCol = cluBarX - math.floor(28 * s)
            local rightCol = thrBarX + barW + math.floor(32 * s)

            -- Left badges: IGN (top), HBK (bottom)
            local ignCol = st.running and Color(95, 235, 140)
                or (st.ignition == 2 and Color(255, 195, 65) or Color(240, 75, 75))
            drawBadge(leftCol, row1Y, true, ignCol, st.ignition == 2 and "CRK" or "IGN")
            drawBadge(leftCol, row2Y, keyHbk, Color(245, 70, 70), "HBK")

            -- Right badges: AT/MT (top), 1/2 throttle (bottom)
            drawBadge(rightCol, row1Y, true, st.manual and Color(255, 150, 75) or Color(105, 225, 145), st.manual and "MT" or "AT")
            drawBadge(rightCol, row2Y, st.halfThrottle == true, Color(100, 230, 185), "1/2")
        end

        local canSeeDebug = debugEnabled and ((lp == owner()) or inThisSeat)

        if canSeeDebug and #wheelDebug.wheels > 0 then
            for i, w in ipairs(wheelDebug.wheels) do
                smoothTelemetry[i] = smoothTelemetry[i] or { latRatio = 0, longRatio = 0, slipLat = 0, slipLong = 0 }
                local sm = smoothTelemetry[i]
                if w.isGrounded then
                    sm.latRatio  = sm.latRatio  + (math.abs(w.latRatio or 0) - sm.latRatio) * 0.4
                    sm.longRatio = sm.longRatio + (math.abs(w.longRatio or 0) - sm.longRatio) * 0.4
                    sm.slipLat   = sm.slipLat   + (math.abs(math.rad(w.slipAngle or 0)) - sm.slipLat) * 0.4
                    sm.slipLong  = sm.slipLong  + (math.abs(w.slipRatio or 0) - sm.slipLong) * 0.4
                else
                    sm.latRatio, sm.longRatio, sm.slipLat, sm.slipLong = 0, 0, 0, 0
                end
            end
        end

        local show3D = canSeeDebug and (TIRE.ShowTelemetryHUD ~= false) and ((TIRE.ShowTelemetry3D == true) or (TIRE.ShowTelemetry3D == nil and TIRE.ShowTireDebug == true))
        local baseEnt = wheelDebug.base
        if canSeeDebug and baseEnt and baseEnt:isValid() and cfg.SUSPENSION_CFG and cfg.SUSPENSION_CFG.showAxleDebug then
            local function line(a, b)
                local p, q = baseEnt:localToWorld(a):toScreen(), baseEnt:localToWorld(b):toScreen()
                if p.visible and q.visible then render.drawLine(p.x, p.y, q.x, q.y) end
            end
            local up = wheelDebug.rightLocal:cross(wheelDebug.fwdLocal):getNormalized()
            for _, left in ipairs(wheelDebug.wheels) do
                if left.springTop and not left.isRightSide then
                    for _, right in ipairs(wheelDebug.wheels) do
                        if right.springTop and right.isRightSide and right.isFront == left.isFront then
                            local l, r = left.hubLocal, right.hubLocal
                            local near = left.seatNear
                            render.setColor(Color(255, 182, 58))
                            line(l, r)
                            local mid = (l + r) * 0.5
                            line(mid - up * 2, mid + up * 2)
                            render.setColor(Color(95, 235, 155))
                            line(left.springTop, l * near + r * (1 - near))
                            line(right.springTop, r * near + l * (1 - near))
                            local label = baseEnt:localToWorld(mid + up * 3):toScreen()
                            if label.visible then
                                local roll = math.deg(math.atan2((l - r):dot(up), (r - l):dot(wheelDebug.rightLocal)))
                                setFontSafe(smallFont)
                                drawTextAligned(label.x, label.y, string.format("%s beam  Z %+.1f su  roll %+.1f°",
                                    left.isFront and "Front" or "Rear", (mid - wheelDebug.comLocal):dot(up), roll), 1, 1)
                            end
                        end
                    end
                end
            end
        end
        if show3D and baseEnt and baseEnt:isValid() and #wheelDebug.wheels > 0 then
            local callouts = TIRE.ShowCamberDebug and {} or nil
            local cPos = baseEnt:getPos()
            local liveFwd = baseEnt:localToWorldVector(wheelDebug.fwdLocal)
            local liveRight = baseEnt:localToWorldVector(wheelDebug.rightLocal)
            local liveUp = liveRight:cross(liveFwd):getNormalized()
            local sCenter = cPos:toScreen()
            local sFront  = (cPos + liveFwd * 65):toScreen()
            local sRight  = (cPos + liveRight * 45):toScreen()
            local sUp     = (cPos + liveUp * 45):toScreen()
            if sCenter.visible and sFront.visible then
                render.setColor(Color(60, 220, 255, 220))
                render.drawLine(sCenter.x, sCenter.y, sFront.x, sFront.y)
                setFontSafe(smallFont)
                drawTextAligned(sFront.x, sFront.y - 8 * scale, ">>> FRONT (FWD) >>>", 1, 1)
            end
            if sCenter.visible and sRight.visible then
                render.setColor(Color(255, 120, 80, 190))
                render.drawLine(sCenter.x, sCenter.y, sRight.x, sRight.y)
                setFontSafe(smallFont)
                drawTextAligned(sRight.x, sRight.y - 8 * scale, "RIGHT", 1, 1)
            end
            if sCenter.visible and sUp.visible then
                render.setColor(Color(90, 150, 255, 220))
                render.drawLine(sCenter.x, sCenter.y, sUp.x, sUp.y)
                setFontSafe(smallFont)
                drawTextAligned(sUp.x, sUp.y - 8 * scale, "UP", 1, 1)
            end

            for i, w in ipairs(wheelDebug.wheels) do
                local hub = w.hubLocal and baseEnt:localToWorld(w.hubLocal)
                local sPos = hub and hub:toScreen()
                if sPos and sPos.visible and callouts then
                    callouts[#callouts + 1] = { wheel = w, x = sPos.x, y = sPos.y }
                end
                if sPos and sPos.visible and not w.isGrounded then
                    setFontSafe(smallFont)
                    render.setColor(Color(225, 225, 225))
                    drawTextAligned(sPos.x, sPos.y, w.name .. " · air", 1, 1)
                elseif sPos and sPos.visible then
                    if callouts then
                        local axle = baseEnt:localToWorldVector(w.axleLocal)
                        local normal = baseEnt:localToWorldVector(w.normalLocal)
                        local wheelUp = normal - axle * normal:dot(axle)
                        if wheelUp:getLengthSqr() > 0.001 then
                            local roadTip = (hub + normal * 30):toScreen()
                            local tireTip = (hub + wheelUp:getNormalized() * 30):toScreen()
                            if roadTip.visible and tireTip.visible then
                                render.setColor(Color(245, 245, 245))
                                render.drawLine(sPos.x, sPos.y, roadTip.x, roadTip.y)
                                render.setColor(w.isFront and Color(95, 220, 245) or Color(255, 186, 96))
                                render.drawLine(sPos.x, sPos.y, tireTip.x, tireTip.y)
                                render.drawRect(tireTip.x - 2, tireTip.y - 2, 4, 4)
                            end
                        end
                    end
                    local sm = smoothTelemetry[i] or {}
                    local barW = 10 * scale
                    local gap = 12 * scale
                    local barH = 44 * scale
                    local latX = sPos.x - barW - gap * 0.5 - 6 * scale
                    local longX = sPos.x + 6 * scale
                    local latR = clamp(sm.latRatio or w.latRatio or 0, 0, 1.5)
                    local longR = clamp(sm.longRatio or w.longRatio or 0, 0, 1.5)
                    local cLat = (latR >= 1.0 and Color(255, 40, 40)) or (latR > 0.85 and Color(255, 255, 0)) or Color(0, 255, 80)
                    local cLng = (longR >= 1.0 and Color(255, 40, 40)) or (longR > 0.85 and Color(255, 180, 0)) or Color(0, 200, 255)

                    render.setColor(Color(0, 0, 0, 200))
                    drawOutlineSafe(latX - 1, sPos.y - barH - 1, barW + 2, barH + 2)
                    drawOutlineSafe(longX - 1, sPos.y - barH - 1, barW + 2, barH + 2)
                    render.setColor(Color(0, 0, 0, 150))
                    render.drawRect(latX, sPos.y - barH, barW, barH)
                    render.drawRect(longX, sPos.y - barH, barW, barH)

                    local latFill = barH * math.min(1, latR)
                    local longFill = barH * math.min(1, longR)
                    render.setColor(cLat)
                    render.drawRect(latX, sPos.y - latFill, barW, latFill)
                    render.setColor(cLng)
                    render.drawRect(longX, sPos.y - longFill, barW, longFill)

                    setFontSafe(smallFont)
                    render.setColor(cLat)
                    drawTextAligned(latX + barW * 0.5, sPos.y - barH - 10 * scale, string.format("L:%d%%", math.floor(latR * 100)), 1, 1)
                    render.setColor(cLng)
                    drawTextAligned(longX + barW * 0.5, sPos.y - barH - 10 * scale, string.format("X:%d%%", math.floor(longR * 100)), 1, 1)

                    local axleTag = w.isFront and "[FRONT]" or "[REAR]"
                    local wName = (w.name and w.name ~= "") and w.name or ("W" .. i)
                    render.setColor(w.isFront and Color(80, 230, 255) or Color(255, 180, 80))
                    local label = TIRE.ShowCamberDebug and wName or string.format("%s %s %.0f° | %.1fkN", axleTag, wName, w.slipAngle or 0, w.loadKN or 0)
                    drawTextAligned(sPos.x, sPos.y + 10 * scale, label, 1, 1)
                end
            end
            if callouts then drawWheelCallouts(callouts, sw, sh, inThisSeat) end
        end

        local wsCfg = cfg.WEIGHT_SHIFT_CFG
        local showWS = canSeeDebug and (wsCfg == nil or wsCfg.ShowDebug ~= false) and (TIRE.ShowTelemetryHUD ~= false or (wsCfg and wsCfg.ShowDebug))
        if showWS and (st or #wheelDebug.wheels > 0) then
            local wx = 40 * scale
            local wy = sh - 454 * scale
            local ww = 268 * scale
            local wh = 94 * scale
            render.setColor(Color(12, 16, 24, 195))
            render.drawRect(wx, wy, ww, wh)
            render.setColor(Color(70, 90, 120, 220))
            drawOutlineSafe(wx, wy, ww, wh)

            setFontSafe(smallFont)
            local wsOn  = (st and st.wsEnabled ~= nil) and st.wsEnabled or (wheelDebug.wsEnabled ~= false)
            local fPct  = clamp((st and st.wsFrontPct) or wheelDebug.wsFrontPct or 50, 0, 100)
            local rPct  = 100 - fPct
            local lPct  = clamp((st and st.wsLeftPct) or wheelDebug.wsLeftPct or 50, 0, 100)
            local riPct = 100 - lPct
            local fwdG  = (st and st.wsFwdG) or wheelDebug.wsFwdG or 0
            local latG  = (st and st.wsLatG) or wheelDebug.wsLatG or 0
            local vertM = (st and st.wsVertMult) or 1.0
            render.setColor(wsOn and Color(110, 230, 160) or Color(180, 180, 180))
            drawTextAligned(wx + 8 * scale, wy + 13 * scale, wsOn and "Weight Shift [ON]" or "Weight Shift [OFF]", 0, 1)
            render.setColor(Color(230, 235, 245))
            drawTextAligned(wx + 8 * scale, wy + 33 * scale, string.format("F: %d%%   R: %d%%", math.floor(fPct + 0.5), math.floor(rPct + 0.5)), 0, 1)
            drawTextAligned(wx + 8 * scale, wy + 51 * scale, string.format("L: %d%%   R: %d%%", math.floor(lPct + 0.5), math.floor(riPct + 0.5)), 0, 1)
            render.setColor(Color(160, 195, 235))
            drawTextAligned(wx + 8 * scale, wy + 71 * scale, string.format("G: %.2fF / %.2fL", fwdG, latG), 0, 1)

            -- 2D Mass Center / Horizontal G-ball
            local bx = wx + ww - 76 * scale
            local by = wy + wh * 0.5
            local br = 26 * scale
            render.setColor(Color(35, 42, 55, 220))
            render.drawRect(bx - br, by - br, br * 2, br * 2)
            render.setColor(Color(90, 110, 140, 220))
            render.drawLine(bx - br, by, bx + br, by)
            render.drawLine(bx, by - br, bx, by + br)
            local dotX = bx + clamp((riPct - 50) / 50, -1, 1) * (br - 3 * scale)
            local dotY = by - clamp((fPct - 50) / 50, -1, 1) * (br - 3 * scale)
            render.setColor(Color(255, 190, 70))
            render.drawRect(dotX - 3 * scale, dotY - 3 * scale, 6 * scale, 6 * scale)

            -- Separate Vertical G / Slope Load Gauge (1.0g center = normal weight)
            local vx = wx + ww - 34 * scale
            local vy = by - br
            local vw = 12 * scale
            local vh = br * 2
            render.setColor(Color(35, 42, 55, 220))
            render.drawRect(vx, vy, vw, vh)
            render.setColor(Color(120, 140, 175, 220))
            render.drawLine(vx - 2 * scale, by, vx + vw + 2 * scale, by)
            local vFrac = clamp(vertM - 1.0, -1.0, 1.0)
            local vCol = (vertM >= 1.05 and Color(80, 225, 255)) or (vertM <= 0.92 and Color(255, 110, 80)) or Color(120, 230, 150)
            render.setColor(vCol)
            local barH = math.abs(vFrac) * (br - 1 * scale)
            if vFrac >= 0 then
                render.drawRect(vx + 2 * scale, by - barH, vw - 4 * scale, math.max(2 * scale, barH))
            else
                render.drawRect(vx + 2 * scale, by, vw - 4 * scale, math.max(2 * scale, barH))
            end
            render.setColor(vCol)
            drawTextAligned(vx + vw * 0.5, vy + vh + 10 * scale, string.format("%.2fg", vertM), 1, 1)
        end

        local showGraph = canSeeDebug and (TIRE.ShowTelemetryHUD ~= false) and ((TIRE.ShowPacejkaGraph == true) or (TIRE.ShowPacejkaGraph == nil and TIRE.ShowTireDebug == true))
        if showGraph and #wheelDebug.wheels > 0 then
            local activeCompound = (st and st.compound) or TIRE.DefaultCompound or "Sport"
            if specCompound ~= activeCompound then
                specCompound = activeCompound
                frontSpec = SimcadeTires.buildAxleSpec(TIRE.Front, activeCompound, nil, TIRE, "Front")
                rearSpec = SimcadeTires.buildAxleSpec(TIRE.Rear, activeCompound, nil, TIRE, "Rear")
            end
            local latScaleMult = (frontSpec.latScale + rearSpec.latScale) * 0.5
            local longScaleMult = (frontSpec.longScale + rearSpec.longScale) * 0.5

            local gx = 40 * scale
            local gy = sh - 320 * scale
            local gw = 380 * scale
            local gh = 220 * scale

            render.setColor(Color(0, 0, 0, 165))
            render.drawRect(gx, gy, gw, gh)
            render.setColor(Color(255, 255, 255, 45))
            drawOutlineSafe(gx, gy, gw, gh)

            local maxMult = math.max(latScaleMult, longScaleMult, 0.01)
            local D_Lat  = (gh * 0.75) * (latScaleMult / maxMult)
            local D_Long = (gh * 0.75) * (longScaleMult / maxMult)

            local cfgKey = string.format("%s_%d_%d_%.3f", activeCompound, sw, sh, scale)
            if not cachedPacejkaLat or not cachedPacejkaLong or lastPacejkaKey ~= cfgKey then
                cachedPacejkaLat  = {}
                cachedPacejkaLong = {}
                lastPacejkaKey    = cfgKey
                for x_slip = 0, 1.0001, 0.04 do
                    local sx = gx + (x_slip * gw)
                    local fy = (evalPacejka(frontSpec.Lat_B, frontSpec.Lat_C, D_Lat, frontSpec.Lat_E, x_slip * 0.5) + evalPacejka(rearSpec.Lat_B, rearSpec.Lat_C, D_Lat, rearSpec.Lat_E, x_slip * 0.5)) * 0.5
                    local fx = (evalPacejka(frontSpec.Long_B, frontSpec.Long_C, D_Long, frontSpec.Long_E, x_slip) + evalPacejka(rearSpec.Long_B, rearSpec.Long_C, D_Long, rearSpec.Long_E, x_slip)) * 0.5
                    table.insert(cachedPacejkaLat,  { x = sx, y = gy + gh - fy })
                    table.insert(cachedPacejkaLong, { x = sx, y = gy + gh - fx })
                end
            end

            render.setColor(Color(120, 255, 160, 145))
            for j = 1, #cachedPacejkaLat - 1 do
                local p1, p2 = cachedPacejkaLat[j], cachedPacejkaLat[j + 1]
                render.drawLine(p1.x, p1.y, p2.x, p2.y)
            end

            render.setColor(Color(80, 200, 255, 145))
            for j = 1, #cachedPacejkaLong - 1 do
                local p1, p2 = cachedPacejkaLong[j], cachedPacejkaLong[j + 1]
                render.drawLine(p1.x, p1.y, p2.x, p2.y)
            end

            setFontSafe(mathFont)
            render.setColor(Color(255, 255, 255, 225))
            drawTextAligned(gx + 8 * scale, gy - 24 * scale, string.format("Tire Simulation V2 (%s) [B=Rays]", tostring(activeCompound)), 0, 0)
            render.setColor(Color(120, 255, 160, 220))
            drawTextAligned(gx + 8 * scale, gy + 6 * scale, string.format("● Lateral (Eff: x%.2f)", latScaleMult), 0, 0)
            render.setColor(Color(80, 200, 255, 220))
            drawTextAligned(gx + 178 * scale, gy + 6 * scale, string.format("■ Longitudinal (Eff: x%.2f)", longScaleMult), 0, 0)

            for i, w in ipairs(wheelDebug.wheels) do
                local sm = smoothTelemetry[i] or {}
                local latRatio  = math.min(1.0, sm.latRatio or w.latRatio or 0)
                local longRatio = math.min(1.0, sm.longRatio or w.longRatio or 0)
                local wName = (w.name and w.name ~= "") and w.name or ("W" .. i)

                local spec = w.isFront and frontSpec or rearSpec
                local simSlipLat = sm.slipLat or math.abs(math.rad(w.slipAngle or 0))
                local x_lat = clamp(simSlipLat / 0.5, 0, 1)
                local dotLatX = gx + x_lat * gw
                local fY = (simSlipLat > 0.0001) and evalPacejka(spec.Lat_B, spec.Lat_C, D_Lat, spec.Lat_E, simSlipLat) or 0
                local dotLatY = gy + gh - fY

                local cLat = (latRatio >= 1.0 and Color(255, 40, 40)) or (latRatio > 0.85 and Color(255, 255, 0)) or Color(0, 255, 80)
                render.setColor(cLat)
                drawCircleSafe(dotLatX, dotLatY, 4 * scale)

                local simSlipLong = sm.slipLong or math.abs(w.slipRatio or 0)
                local x_long = clamp(simSlipLong / 1.0, 0, 1)
                local dotLongX = gx + x_long * gw
                local fX = (simSlipLong > 0.0001) and evalPacejka(spec.Long_B, spec.Long_C, D_Long, spec.Long_E, simSlipLong) or 0
                local dotLongY = gy + gh - fX

                local cLong = (longRatio >= 1.0 and Color(255, 40, 40)) or (longRatio > 0.85 and Color(255, 180, 0)) or Color(0, 200, 255)
                render.setColor(cLong)
                local sqSz = 7 * scale
                render.drawRect(dotLongX - sqSz * 0.5, dotLongY - sqSz * 0.5, sqSz, sqSz)

                setFontSafe(smallFont)
                render.setColor(Color(255, 255, 255, 220))
                drawTextAligned(dotLatX + 6 * scale, dotLatY, string.format("%s L:%d%%", wName, math.floor(latRatio * 100)), 0, 1)
                if simSlipLong > 0.02 then
                    render.setColor(cLong)
                    drawTextAligned(dotLongX + 6 * scale, dotLongY, string.format("%s X:%d%% (s:%.2f)", wName, math.floor(longRatio * 100), simSlipLong), 0, 1)
                end
            end
        end

        if hudEnabled and SimcadeAddons and SimcadeAddons._clientHost then
            SimcadeAddons._clientHost.runHook("OnHudDraw", st, inThisSeat)
        end
    end)

    hook.add("postdrawtranslucentrenderables", "SimcadeHUD_DebugRays", function()
        local debugEnabled = (HUD.debugEnabled ~= nil) and (HUD.debugEnabled == true) or (HUD.debugEnabled == nil and HUD.enabled ~= false)
        if not debugEnabled or not TIRE.ShowSuspensionRays or not (render and render.draw3DLine) then return end
        local qMax = quotaMax and quotaMax() or 0
        if qMax > 0 then
            if quotaUsed and (quotaUsed() / qMax) > 0.80 then return end
            if quotaAverage and (quotaAverage() / qMax) > 0.80 then return end
        end
        local baseEnt = wheelDebug.base
        if not baseEnt or not baseEnt:isValid() then return end
        for _, w in ipairs(wheelDebug.wheels) do
            for _, r in ipairs(w.rays or {}) do
                local col = r.state == 2 and Color(80, 255, 110) or r.state == 1 and Color(80, 190, 255) or Color(255, 70, 70)
                local hit = baseEnt:localToWorld(r.hitLocal)
                render.setColor(col)
                render.draw3DLine(baseEnt:localToWorld(r.startLocal), hit, false)
                if r.state == 2 and render.draw3DWireframeBox then
                    render.draw3DWireframeBox(hit, Angle(0, 0, 0), Vector(-1, -1, -1), Vector(1, 1, 1), false)
                end
            end
        end
        if baseEnt and baseEnt:isValid() then
            local p = baseEnt:getPos()
            local fwd = baseEnt:localToWorldVector(wheelDebug.fwdLocal):getNormalized()
            local right = baseEnt:localToWorldVector(wheelDebug.rightLocal):getNormalized()
            local up = right:cross(fwd):getNormalized()
            render.setColor(Color(255, 60, 60))
            render.draw3DLine(p, p + fwd * 45, true)
            render.setColor(Color(60, 255, 60))
            render.draw3DLine(p, p + right * 45, true)
            render.setColor(Color(80, 140, 255))
            render.draw3DLine(p, p + up * 45, true)
        end
    end)
