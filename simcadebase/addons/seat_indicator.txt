--@name Simcade Seat Indicator Addon (LVS-Style Billboard + 3D Bezier)
--@shared

--[[
    LVS-style visual seat indicator for any number of seats (1, 2, 5, 10+).
    Works without HUD permission or player.enterVehicle permission:
      - Auto-discovers Driver seat (ctx.pod) + any constrained/parented passenger
        seats on Base, or uses an optional PassengerSeats (Array) wire input.
      - Client-side aim detection (no server think loop).
      - Constant O(1) resources regardless of seat count:
          * 1 RenderTarget + 1 billboard quad holo for crisp "[E] Driver" text
            (uses render.offscreen, which defaults to allowed for everyone)
          * 1 fixed pool of thin dash holos for the animated LVS cubic Bezier curve
          * 1 ghost seat holo highlighting the targeted seat
      - Optional silent enterVehicle fallback when AllowEnterVehicle = true AND
        the server/player has granted player.enterVehicle permission.
]]

return function(options)
    options = options or {}

    local AutoDetectSeats   = options.AutoDetectSeats ~= false
    local MaxDistance       = options.MaxDistance or 130
    local AimRadius         = options.AimRadius or 24
    local AimDotMin         = options.AimDotMin or 0.86
    local ThroughWalls      = options.ThroughWalls ~= false
    local ShowLabel         = options.ShowLabel ~= false
    local ShowCurve         = options.ShowCurve ~= false
    local ShowSeatGhost     = options.ShowSeatGhost ~= false
    local ShowOccupied      = options.ShowOccupied ~= false
    local CurveSegments     = math.clamp(math.floor(options.CurveSegments or 24), 6, 36)
    local CurveThickness    = options.CurveThickness or 0.24
    local CurveWaveSpeed    = options.CurveWaveSpeed or 5.0
    local LabelWidth        = options.LabelWidth or 22
    local LabelHeight       = options.LabelHeight or 11
    local LabelOffsetZ      = options.LabelOffsetZ or 16
    local DriverLabel       = options.DriverLabel or "Driver"
    local PassengerPrefix   = options.PassengerPrefix or "Passenger"
    local SeatLabels        = type(options.SeatLabels) == "table" and options.SeatLabels or nil
    local AllowEnterVehicle = options.AllowEnterVehicle == true
    local RequestEnterPerm  = options.RequestEnterPermission ~= false

    local DriverColor       = options.DriverColor    or Color(110, 235, 175, 255)
    local PassengerColor    = options.PassengerColor or Color(100, 185, 255, 255)
    local OccupiedColor     = options.OccupiedColor  or Color(255, 95, 95, 220)
    local CurveColor        = options.CurveColor     or Color(255, 255, 255, 235)

    local NET_LIST  = "Simcade_SeatInd_List"
    local NET_REQ   = "Simcade_SeatInd_Req"
    local NET_ENTER = "Simcade_SeatInd_Enter"

    if SERVER then
        local seats = {}
        local wiredPassengers = nil
        local lastSignature = ""
        local ctxRef = nil

        local function isSeatEntity(ent)
            if not isValid(ent) then return false end
            local okVeh, isVeh = pcall(function() return ent:isVehicle() end)
            if okVeh and isVeh then return true end
            local okCls, cls = pcall(function() return ent:getClass() end)
            return okCls and cls == "prop_vehicle_prisoner_pod"
        end

        local function collectSeats()
            local list = {}
            local seen = {}

            local pod = (ctxRef and isValid(ctxRef.pod)) and ctxRef.pod or (wire and wire.ports and wire.ports.Pod)
            local base = (ctxRef and isValid(ctxRef.base)) and ctxRef.base or (wire and wire.ports and wire.ports.Base)

            if isValid(pod) then
                seen[pod] = true
                list[#list + 1] = {
                    ent = pod,
                    isDriver = true,
                    label = (SeatLabels and SeatLabels[1]) or DriverLabel,
                }
            end

            local passList = {}
            if type(wiredPassengers) == "table" and #wiredPassengers > 0 then
                for _, ent in ipairs(wiredPassengers) do
                    if isValid(ent) and not seen[ent] and isSeatEntity(ent) then
                        seen[ent] = true
                        passList[#passList + 1] = ent
                    end
                end
            elseif AutoDetectSeats and isValid(base) then
                local candidates = {}
                local function consider(ent)
                    if not isValid(ent) or seen[ent] then return end
                    if isSeatEntity(ent) then
                        seen[ent] = true
                        candidates[#candidates + 1] = ent
                    end
                end

                local okCon, conList = pcall(function() return base:getAllConstrained() end)
                if okCon and type(conList) == "table" then
                    for _, ent in pairs(conList) do
                        consider(ent)
                        local okCh, ch = pcall(function() return ent:getChildren() end)
                        if okCh and type(ch) == "table" then
                            for _, child in pairs(ch) do consider(child) end
                        end
                    end
                end

                local okCh, baseCh = pcall(function() return base:getChildren() end)
                if okCh and type(baseCh) == "table" then
                    for _, child in pairs(baseCh) do
                        consider(child)
                        local okSub, subCh = pcall(function() return child:getChildren() end)
                        if okSub and type(subCh) == "table" then
                            for _, sub in pairs(subCh) do consider(sub) end
                        end
                    end
                end

                table.sort(candidates, function(a, b)
                    local okA, la = pcall(function() return base:worldToLocal(a:getPos()) end)
                    local okB, lb = pcall(function() return base:worldToLocal(b:getPos()) end)
                    if okA and okB then
                        if math.abs(la.x - lb.x) > 8 then return la.x > lb.x end
                        if math.abs(la.y - lb.y) > 8 then return la.y > lb.y end
                    end
                    return a:entIndex() < b:entIndex()
                end)
                passList = candidates
            end

            for i, ent in ipairs(passList) do
                local custom = SeatLabels and SeatLabels[i + 1]
                local lbl = custom or (#passList == 1 and PassengerPrefix or (PassengerPrefix .. " " .. tostring(i)))
                list[#list + 1] = {
                    ent = ent,
                    isDriver = false,
                    label = lbl,
                }
            end

            return list, base
        end

        local function sendSeatList(targetPly)
            local base = (ctxRef and isValid(ctxRef.base)) and ctxRef.base or (wire and wire.ports and wire.ports.Base)
            pcall(function()
                net.start(NET_LIST)
                net.writeEntity(isValid(base) and base or entity(0))
                net.writeUInt(math.min(#seats, 63), 6)
                for i = 1, math.min(#seats, 63) do
                    local s = seats[i]
                    net.writeEntity(s.ent)
                    net.writeBool(s.isDriver)
                    net.writeString(s.label)
                end
                if isValid(targetPly) then
                    net.send(targetPly)
                else
                    net.send()
                end
            end)
        end

        local function refreshSeats(forceSend)
            local newSeats, base = collectSeats()
            local sigParts = { isValid(base) and tostring(base:entIndex()) or "0" }
            for i = 1, #newSeats do
                local s = newSeats[i]
                sigParts[#sigParts + 1] = tostring(s.ent:entIndex()) .. ":" .. (s.isDriver and "D" or "P") .. ":" .. s.label
            end
            local sig = table.concat(sigParts, "|")
            if forceSend or sig ~= lastSignature then
                lastSignature = sig
                seats = newSeats
                sendSeatList(nil)
            end
        end

        return {
            name = "SeatIndicator",
            OnInit = function(ctx)
                ctxRef = ctx
                ctx.addWireInput("PassengerSeats", "Array", function(_, arr)
                    wiredPassengers = type(arr) == "table" and arr or nil
                    refreshSeats(true)
                end)

                net.receive(NET_REQ, function(_, ply)
                    if isValid(ply) then
                        refreshSeats(false)
                        sendSeatList(ply)
                    end
                end)

                if AllowEnterVehicle then
                    net.receive(NET_ENTER, function(_, ply)
                        if not isValid(ply) or not ply:isAlive() or ply:inVehicle() then return end
                        local seat = net.readEntity()
                        if not isValid(seat) then return end

                        local found = false
                        for i = 1, #seats do
                            if seats[i].ent == seat then found = true break end
                        end
                        if not found then return end

                        local okDrv, drv = pcall(function() return seat:getDriver() end)
                        if okDrv and isValid(drv) then return end

                        if ply:getShootPos():getDistance(seat:getPos()) > MaxDistance + 25 then return end

                        if hasPermission then
                            local okPerm, allowed = pcall(hasPermission, "player.enterVehicle", ply)
                            if okPerm and not allowed then return end
                        end

                        pcall(function() ply:enterVehicle(seat) end)
                    end)
                end

                timer.simple(1.0, function() refreshSeats(true) end)
                timer.create("SimcadeSeatInd_Scan", 2.0, 0, function() refreshSeats(false) end)
            end,
            OnCleanup = function()
                timer.remove("SimcadeSeatInd_Scan")
            end,
        }
    end

    -- ========================================================================
    -- CLIENT
    -- ========================================================================
    local seats = {}
    local baseEnt = nil

    local rtName = "SimcadeSeatInd_RT_" .. tostring(chip():entIndex())
    local rtReady = false
    local rtDirty = false
    local rtTitle = ""
    local rtSub = ""
    local rtAccent = DriverColor
    local lastRenderKey = ""

    local labelHolo = nil
    local labelMesh = nil
    local labelMat  = nil
    local labelFont = nil
    local subFont   = nil

    local ghostHolo = nil
    local ghostMat  = nil
    local ghostModel = ""

    local curveHolos = {}
    local curveMat   = nil

    local targetedSeat = nil
    local lastUseDown = false
    local enterCooldown = 0
    local permPrompted = false
    local isVisible = false

    local function canUseOffscreen()
        if not (render and render.createRenderTarget) then return false end
        if hasPermission then
            local ok, allowed = pcall(hasPermission, "render.offscreen")
            if ok then return allowed end
        end
        return true
    end

    local function makeUnlitMat(ignoreZ)
        local ok, m = pcall(function() return material.create("UnlitGeneric") end)
        if not ok or not m then return nil end
        pcall(function()
            m:setTexture("$basetexture", "vgui/white")
            m:setInt("$model", 1)
            m:setInt("$translucent", 1)
            m:setInt("$vertexcolor", 1)
            m:setInt("$vertexalpha", 1)
            m:setInt("$ignorez", ignoreZ and 1 or 0)
        end)
        return m
    end

    local function ensureLabelAssets()
        if not ShowLabel or labelHolo then return end
        if not (hologram and hologram.create and mesh and mesh.createFromTable) then return end
        if not canUseOffscreen() then return end

        local okRT = pcall(function() render.createRenderTarget(rtName) end)
        if not okRT then return end
        rtReady = true

        pcall(function()
            labelFont = render.createFont("Roboto", 116, 700, true)
            subFont   = render.createFont("Roboto", 84, 600, true)
        end)

        local okMat, m = pcall(function() return material.create("UnlitGeneric") end)
        if not okMat or not m then return end
        labelMat = m
        pcall(function()
            -- Opaque alpha-test WITH Z-write ($ignorez = 0): since labelHolo is projected
            -- 10-18 units in front of the camera, writing depth causes translucent car glass
            -- (which renders in the later translucent pass) to fail the depth test behind it.
            -- $flags: ALPHATEST (256) | MODEL (2048) | NOCULL (8192) — explicitly NO IGNOREZ (32768) or TRANSLUCENT (2097152).
            labelMat:setInt("$model", 1)
            labelMat:setInt("$nocull", 1)
            labelMat:setInt("$translucent", 0)
            labelMat:setInt("$alphatest", 1)
            labelMat:setFloat("$alphatestreference", 0.35)
            labelMat:setInt("$ignorez", 0)
            labelMat:setInt("$flags", 256 + 2048 + 8192)
            labelMat:setTextureRenderTarget("$basetexture", rtName)
        end)

        local hw = LabelWidth * 0.5
        local hh = LabelHeight * 0.5
        local vSpan = math.clamp(LabelHeight / math.max(1, LabelWidth), 0.2, 1.0)
        local vMin = 0.5 - vSpan * 0.5
        local vMax = 0.5 + vSpan * 0.5
        local nF = Vector(1, 0, 0)
        local nB = Vector(-1, 0, 0)
        local vCol = Color(255, 255, 255, 255)
        local tl = Vector(0, -hw,  hh)
        local tr = Vector(0,  hw,  hh)
        local br = Vector(0,  hw, -hh)
        local bl = Vector(0, -hw, -hh)

        local okM, qm = pcall(function()
            return mesh.createFromTable({
                { pos = tl, normal = nF, u = 0, v = vMin, color = vCol },
                { pos = tr, normal = nF, u = 1, v = vMin, color = vCol },
                { pos = br, normal = nF, u = 1, v = vMax, color = vCol },
                { pos = br, normal = nF, u = 1, v = vMax, color = vCol },
                { pos = bl, normal = nF, u = 0, v = vMax, color = vCol },
                { pos = tl, normal = nF, u = 0, v = vMin, color = vCol },
                { pos = tl, normal = nB, u = 0, v = vMin, color = vCol },
                { pos = bl, normal = nB, u = 0, v = vMax, color = vCol },
                { pos = br, normal = nB, u = 1, v = vMax, color = vCol },
                { pos = br, normal = nB, u = 1, v = vMax, color = vCol },
                { pos = tr, normal = nB, u = 1, v = vMin, color = vCol },
                { pos = tl, normal = nB, u = 0, v = vMin, color = vCol },
            })
        end)
        if not okM or not qm then return end
        labelMesh = qm

        local okH, h = pcall(function()
            return hologram.create(Vector(), Angle(), "models/Combine_Helicopter/helicopter_bomb01.mdl")
        end)
        if not okH or not isValid(h) then
            pcall(function() labelMesh:destroy() end)
            labelMesh = nil
            return
        end

        h:setMesh(labelMesh)
        h:setMeshMaterial(labelMat)
        h:setRenderBounds(Vector(-2, -hw - 2, -hh - 2), Vector(2, hw + 2, hh + 2))
        h:suppressEngineLighting(true)
        h:setNoDraw(true)
        labelHolo = h
    end

    local function ensureCurveAssets()
        if not ShowCurve or #curveHolos > 0 then return end
        if not (hologram and hologram.create) then return end

        for i = 1, CurveSegments do
            if hologram.canSpawn and not hologram.canSpawn() then break end
            local ok, h = pcall(function()
                return hologram.create(Vector(), Angle(), "models/holograms/cube.mdl")
            end)
            if ok and isValid(h) then
                pcall(function()
                    h:setMaterial("models/debug/debugwhite")
                    h:suppressEngineLighting(true)
                    h:setRenderMode(4)
                    h:setNoDraw(true)
                end)
                curveHolos[#curveHolos + 1] = h
            end
        end
    end

    local function ensureGhostHolo(seat)
        if not ShowSeatGhost or not isValid(seat) then return nil end
        local mdl = "models/nova/airboat_seat.mdl"
        pcall(function()
            local m = seat:getModel()
            if m and m ~= "" then mdl = m end
        end)

        if isValid(ghostHolo) then
            if ghostModel ~= mdl then
                pcall(function() ghostHolo:setModel(mdl) end)
                ghostModel = mdl
            end
            return ghostHolo
        end

        if hologram.canSpawn and not hologram.canSpawn() then return nil end
        local ok, h = pcall(function()
            return hologram.create(seat:getPos(), seat:getAngles(), mdl)
        end)
        if not ok or not isValid(h) then return nil end

        pcall(function()
            h:setMaterial("models/debug/debugwhite")
            h:suppressEngineLighting(true)
            h:setRenderMode(4)
            h:setNoDraw(true)
        end)
        ghostHolo = h
        ghostModel = mdl
        return ghostHolo
    end

    local function hideAll()
        if not isVisible then return end
        isVisible = false
        if isValid(labelHolo) then pcall(function() labelHolo:setNoDraw(true) end) end
        if isValid(ghostHolo) then pcall(function() ghostHolo:setNoDraw(true) end) end
        for i = 1, #curveHolos do
            local h = curveHolos[i]
            if isValid(h) then pcall(function() h:setNoDraw(true) end) end
        end
    end

    local function evalBezier(p0, p1, p2, p3, t)
        local u = 1 - t
        local uu = u * u
        local tt = t * t
        return p0 * (uu * u) + p1 * (3 * uu * t) + p2 * (3 * u * tt) + p3 * (tt * t)
    end

    local function getSeatCenter(seat)
        local ok, pos = pcall(function() return seat:localToWorld(seat:obbCenter()) end)
        if ok and pos then return pos end
        return seat:getPos()
    end

    local function getUseKeyName()
        if input and input.lookupBinding then
            local ok, key = pcall(input.lookupBinding, "+use")
            if ok and key then
                if type(key) == "number" and input.getKeyName then
                    local okN, name = pcall(input.getKeyName, key)
                    if okN and name and name ~= "" then return string.upper(name) end
                elseif type(key) == "string" and key ~= "" then
                    return string.upper(key)
                end
            end
        end
        return "E"
    end

    local function drawLabelRT()
        if not (rtReady and rtDirty) then return end
        rtDirty = false
        pcall(function()
            render.selectRenderTarget(rtName)
            render.clear(Color(0, 0, 0, 0), true)

            local w, h = 1024, 1024
            local cx, cy = w * 0.5, h * 0.5

            -- Clean LVS-style shadowed floating text (opaque alpha for crisp $alphatest)
            if labelFont and render.setFont then render.setFont(labelFont) end
            render.setColor(Color(0, 0, 0, 255))
            render.drawSimpleText(cx + 6, cy - 68 + 6, rtTitle, 1, 1)
            render.setColor(Color(255, 255, 255, 255))
            render.drawSimpleText(cx, cy - 68, rtTitle, 1, 1)

            if subFont and render.setFont then render.setFont(subFont) end
            render.setColor(Color(0, 0, 0, 255))
            render.drawSimpleText(cx + 5, cy + 54 + 5, rtSub, 1, 1)
            render.setColor(Color(rtAccent.r, rtAccent.g, rtAccent.b, 255))
            render.drawSimpleText(cx, cy + 54, rtSub, 1, 1)

            render.selectRenderTarget()
        end)
    end

    return {
        name = "SeatIndicator",
        OnClientInit = function()
            if AllowEnterVehicle and RequestEnterPerm and setupPermissionRequest then
                pcall(setupPermissionRequest, { "player.enterVehicle" }, "Allow entering vehicle seats when pressing [E] while aiming at them.", true)
            end

            net.receive(NET_LIST, function()
                local b = net.readEntity()
                baseEnt = isValid(b) and b or nil
                local count = net.readUInt(6)
                seats = {}
                for i = 1, count do
                    local ent = net.readEntity()
                    local isDrv = net.readBool()
                    local lbl = net.readString()
                    if isValid(ent) then
                        seats[#seats + 1] = { ent = ent, isDriver = isDrv, label = lbl }
                    end
                end
            end)

            local function requestList()
                pcall(function()
                    net.start(NET_REQ)
                    net.send()
                end)
            end
            requestList()
            timer.simple(1.5, function()
                if #seats == 0 then requestList() end
            end)

            hook.add("renderoffscreen", "SimcadeSeatInd_RT", drawLabelRT)

            hook.add("think", "SimcadeSeatInd_Think", function()
                local lp = player()
                if #seats == 0 or not isValid(lp) or not lp:isAlive() or lp:inVehicle() then
                    targetedSeat = nil
                    lastUseDown = true
                    enterCooldown = timer.curtime() + 1.0
                    hideAll()
                    return
                end

                local eyePos = lp:getShootPos()
                if isValid(baseEnt) and eyePos:getDistanceSqr(baseEnt:getPos()) > (MaxDistance + 260) ^ 2 then
                    targetedSeat = nil
                    hideAll()
                    return
                end

                local eyeDir = lp:getAimVector()
                local best = nil
                local bestScore = -1e9
                local bestCenter = nil

                for i = 1, #seats do
                    local s = seats[i]
                    local seat = s.ent
                    if not isValid(seat) then continue end

                    local okDrv, drv = pcall(function() return seat:getDriver() end)
                    local occupied = okDrv and isValid(drv)
                    if occupied and not ShowOccupied then continue end

                    local center = getSeatCenter(seat)
                    local toSeat = center - eyePos
                    local dist = toSeat:getLength()
                    if dist > MaxDistance or dist < 1 then continue end

                    local dir = toSeat / dist
                    local dot = eyeDir:dot(dir)
                    if dot <= 0 then continue end

                    -- Closest distance from aim ray to seat center
                    local perpDist = (toSeat - eyeDir * (toSeat:dot(eyeDir))):getLength()
                    if perpDist <= AimRadius or dot >= AimDotMin then
                        local score = dot * 100 - perpDist - dist * 0.08
                        if score > bestScore then
                            bestScore = score
                            best = s
                            bestCenter = center
                            best._occupied = occupied
                            local drvName = "Player"
                            if occupied then
                                local okN, n = pcall(function() return drv:getName() end)
                                if okN and n and n ~= "" then drvName = n end
                            end
                            best._driverName = drvName
                        end
                    end
                end

                if not best then
                    targetedSeat = nil
                    hideAll()
                    return
                end

                targetedSeat = best
                isVisible = true
                local now = timer.curtime()
                local accent = best._occupied and OccupiedColor or (best.isDriver and DriverColor or PassengerColor)

                -- Pull anchor toward the camera past bodywork/holo glass so it doesn't sit inside the cabin
                local toEyeVec = eyePos - bestCenter
                local eyeDist = toEyeVec:getLength()
                local toEyeDir = eyeDist > 1 and (toEyeVec / eyeDist) or -eyeDir
                local anchorPos = bestCenter + toEyeDir * math.min(26, eyeDist * 0.32)
                if trace and trace.line then
                    local okTr, tr = pcall(trace.line, eyePos, bestCenter, lp)
                    if okTr and tr and tr.Hit and tr.Entity ~= best.ent then
                        local trAnchor = tr.HitPos + toEyeDir * 8
                        if eyePos:getDistanceSqr(trAnchor) < eyePos:getDistanceSqr(anchorPos) then
                            anchorPos = trAnchor
                        end
                    end
                end

                -- 1. Billboard text label anchored stably in world space (no eyePos lag/jitter)
                if ShowLabel then
                    ensureLabelAssets()
                    if isValid(labelHolo) then
                        local title = best._occupied and "OCCUPIED" or ("[" .. getUseKeyName() .. "] Enter")
                        local sub = best._occupied and (best.label .. " - " .. best._driverName) or best.label
                        local key = title .. "|" .. sub .. "|" .. tostring(accent.r) .. "," .. tostring(accent.g) .. "," .. tostring(accent.b)
                        if key ~= lastRenderKey then
                            lastRenderKey = key
                            rtTitle = title
                            rtSub = sub
                            rtAccent = accent
                            rtDirty = true
                        end

                        local lblPos = anchorPos + Vector(0, 0, LabelOffsetZ * 0.5)
                        labelHolo:setPos(lblPos)
                        labelHolo:setAngles((eyePos - lblPos):getAngle())
                        labelHolo:setScale(Vector(1, 1, 1))
                        labelHolo:setNoDraw(false)
                    end
                end

                -- 2. Animated LVS-style thin 3D cubic Bezier curve
                if ShowCurve then
                    ensureCurveAssets()
                    local n = #curveHolos
                    if n > 0 then
                        local p0 = lp:getPos() + Vector(0, 0, 10)
                        local toTarget = anchorPos - p0
                        local dist = toTarget:getLength()
                        local fwd = dist > 1 and (toTarget / dist) or eyeDir
                        local right = fwd:cross(Vector(0, 0, 1))
                        if right:getLengthSqr() > 0.001 then
                            right = right:getNormalized()
                        else
                            right = Vector(0, 1, 0)
                        end
                        -- Subtle lateral + vertical S-curve so it reads as a true curve from the player's view
                        local p1 = p0 + fwd * (dist * 0.42) + right * (dist * 0.14) + Vector(0, 0, 26)
                        local p2 = anchorPos - fwd * (dist * 0.25) - right * (dist * 0.10) - Vector(0, 0, 14)
                        local p3 = anchorPos

                        local prevPt = p0
                        for i = 1, n do
                            local t = i / n
                            local pt = evalBezier(p0, p1, p2, p3, t)
                            local seg = pt - prevPt
                            local segLen = seg:getLength()
                            local h = curveHolos[i]
                            if isValid(h) and segLen > 0.05 then
                                -- Match LVS cosine wave + taper thickness so near-camera segments stay razor-thin
                                local wave = math.abs(math.cos(-now * CurveWaveSpeed + i * 0.28))
                                local edgeFade = math.min(t * 4, (1 - t) * 5, 1)
                                local alpha = math.floor(wave * edgeFade * CurveColor.a)
                                local camDist = eyePos:getDistance((prevPt + pt) * 0.5)
                                local thick = math.clamp(CurveThickness * (camDist / 65), 0.10, CurveThickness * 1.4)
                                h:setPos((prevPt + pt) * 0.5)
                                h:setAngles(seg:getAngle())
                                if h.setSize then
                                    h:setSize(Vector(segLen + 0.15, thick, thick))
                                else
                                    h:setScale(Vector((segLen + 0.15) / 12, thick / 12, thick / 12))
                                end
                                h:setColor(Color(CurveColor.r, CurveColor.g, CurveColor.b, alpha))
                                h:setNoDraw(alpha < 4)
                            end
                            prevPt = pt
                        end
                    end
                end

                -- 3. Pulsing ghost highlight on the targeted seat
                if ShowSeatGhost then
                    local g = ensureGhostHolo(best.ent)
                    if isValid(g) then
                        local pulse = math.sin(now * 3.5) * 0.5 + 0.5
                        local alpha = math.floor(30 + pulse * 85)
                        g:setPos(best.ent:getPos())
                        g:setAngles(best.ent:getAngles())
                        g:setColor(Color(accent.r, accent.g, accent.b, alpha))
                        g:setNoDraw(false)
                    end
                end

                -- 4. Optional enterVehicle + one-time permission prompt when pressing +use
                if AllowEnterVehicle and not best._occupied and now >= enterCooldown then
                    local blockedUI = lp:isTyping()
                    if not blockedUI and vgui and vgui.cursorVisible then
                        local okV, vis = pcall(vgui.cursorVisible)
                        if okV and vis then blockedUI = true end
                    end
                    if not blockedUI and gui and gui.isConsoleVisible then
                        local okC, vis = pcall(gui.isConsoleVisible)
                        if okC and vis then blockedUI = true end
                    end
                    if not blockedUI and lp.getEntityInUse then
                        local okU, inUse = pcall(function() return lp:getEntityInUse() end)
                        if okU and isValid(inUse) then blockedUI = true end
                    end
                    -- Ignore E while holding LMB/RMB (e.g. rotating props with physgun or using tools)
                    if not blockedUI and lp.keyDown and IN_KEY then
                        local okA, atk = pcall(function()
                            return (IN_KEY.ATTACK and lp:keyDown(IN_KEY.ATTACK))
                                or (IN_KEY.ATTACK2 and lp:keyDown(IN_KEY.ATTACK2))
                        end)
                        if okA and atk then blockedUI = true end
                    end

                    -- Use Source's +use command state (not raw KEY.E) so chat/console/editor never trigger it
                    local useDown = false
                    if lp.keyDown and IN_KEY and IN_KEY.USE then
                        local okK, down = pcall(function() return lp:keyDown(IN_KEY.USE) end)
                        useDown = okK and down or false
                    end

                    if not blockedUI and useDown and not lastUseDown then
                        local hasPerm = true
                        if permissionRequestSatisfied then
                            local okS, sat = pcall(permissionRequestSatisfied)
                            if okS then hasPerm = sat end
                        end
                        if not hasPerm and RequestEnterPerm and not permPrompted and sendPermissionRequest then
                            permPrompted = true
                            pcall(sendPermissionRequest)
                        end
                        pcall(function()
                            net.start(NET_ENTER)
                            net.writeEntity(best.ent)
                            net.send()
                        end)
                        enterCooldown = now + 1.2
                    end
                    lastUseDown = useDown
                end
            end)
        end,
        OnCleanup = function()
            hook.remove("think", "SimcadeSeatInd_Think")
            hook.remove("renderoffscreen", "SimcadeSeatInd_RT")
            if isValid(labelHolo) then pcall(function() labelHolo:remove() end) end
            if isValid(ghostHolo) then pcall(function() ghostHolo:remove() end) end
            if labelMesh and labelMesh.destroy then pcall(function() labelMesh:destroy() end) end
            for i = 1, #curveHolos do
                if isValid(curveHolos[i]) then pcall(function() curveHolos[i]:remove() end) end
            end
            curveHolos = {}
        end,
    }
end
