--@name Simcade Brake Temperature Addon
--@shared

return function(options)
    local ambient = options.AmbientTemp or options.ambientC or 20
    local ceiling = math.max(ambient, options.MaxTemp or options.maxC or 800)
    local heatRate = options.HeatRate or (0.35 * 4 / math.max(0.01, options.heatCapacityKJPerC or 4))
    local heatScale = math.max(0, heatRate * (options.SpeedHeatScale or 1))
    local coolRate = math.max(0, options.CoolRate or 0.08)
    local airflow = math.max(0, options.SpeedCoolScale or 0.15)
    local stoppedMult = math.max(0, options.StoppedCoolMult or 0.5)
    -- Solver slots run FL, FR, RR, RL.
    local fields = { [0] = "TempFL", "TempFR", "TempRR", "TempRL" }
    local temperatures = {}
    local extraWheels = {}
    local extraTemperatures = {}
    local rpmToKmh = math.max(0, options.WheelRPMToSpeed or 0.15)
    local values = { TempMax = ambient, WheelTemps = {}, ExtraWheelTemps = {} }
    for slot = 0, 3 do
        values[fields[slot]] = ambient
    end

    return {
        name = "BrakeTemperature",
        OnClientInit = function(ctx)
            ctx.syncedState.BrakeTemperature = values
        end,
        OnInit = function(ctx)
            ctx.syncedState.BrakeTemperature = values
            for slot = 0, 3 do
                ctx.addWireOutput("Brake" .. fields[slot], "Number")
            end
            ctx.addWireOutput("BrakeTempMax", "Number")
            ctx.addWireOutput("BrakeWheelTemps", "Array")
            ctx.addWireOutput("ExtraBrakeWheelTemps", "Array")
            ctx.addWireOutput("ExtraBrakeWheels", "Array")
            ctx.addWireInput("ExtraBrakeWheels", "Array", function(c, wheels)
                extraWheels = type(wheels) == "table" and wheels or {}
                -- Keep the temperature with the entity when a trailer is rewired.
                local retained = {}
                for _, ent in ipairs(extraWheels) do
                    if isValid(ent) then retained[ent] = extraTemperatures[ent] or ambient end
                end
                extraTemperatures = retained
                c.setWireOutput("ExtraBrakeWheels", extraWheels)
            end)
        end,
        OnPrePhysics = function(ctx)
            -- Wheels can skip their brake solve while sleeping or airborne.
            for slot = 0, 3 do
                ctx.car.wheels[slot].serviceBrakeKNm = 0
            end
        end,
        OnPostPhysics = function(ctx, dt)
            local speed = math.abs(ctx.state.kmh)
            local passive = speed < 2 and coolRate * stoppedMult or coolRate
            -- Convert the engine sim's 66 Hz cooling step to Simcade's tick length.
            local loss = (passive + speed * airflow / 66) / math.max(1, ceiling)
            local retention = math.max(0, 1 - loss) ^ (66 * dt)
            local brakes = ctx.car.braking
            local fullBrake = math.max(0.001, brakes.frontBrakeKNm, brakes.rearBrakeKNm)
            local hottest = ambient
            local function integrate(previous, demand, kmh)
                local heat = demand * kmh * heatScale * dt
                return math.clamp(ambient + (previous + heat - ambient) * retention, ambient, ceiling)
            end
            local corners = {}
            for slot = 0, 3 do corners[slot] = ambient end
            local wheelTemps = {}
            for i, runtime in ipairs(ctx.wheels) do
                local slot = runtime.physicsIndex
                local wheel = ctx.car.wheels[slot]
                local wheelKmh = math.abs(wheel.w) * runtime.radius * 0.0254 * 3.6
                temperatures[i] = integrate(temperatures[i] or ambient, wheel.serviceBrakeKNm / fullBrake, wheelKmh)
                wheelTemps[i] = math.round(temperatures[i], 1)
                corners[slot] = math.max(corners[slot], wheelTemps[i])
                hottest = math.max(hottest, wheelTemps[i])
            end
            local extraTemps = {}
            local updated = {}
            for i, ent in ipairs(extraWheels) do
                local temp = ambient
                if isValid(ent) and ent.getAngleVelocity then
                    -- Spin magnitude is independent of how the wheel prop was mounted.
                    local rpm = ent:getAngleVelocity():getLength() / 6
                    temp = updated[ent] or integrate(extraTemperatures[ent] or ambient, ctx.state.brake, rpm * rpmToKmh)
                    updated[ent] = temp
                    extraTemps[i] = math.round(temp, 1)
                else
                    extraTemps[i] = ambient
                end
                hottest = math.max(hottest, extraTemps[i])
            end
            extraTemperatures = updated
            for slot = 0, 3 do
                local field = fields[slot]
                values[field] = corners[slot]
                ctx.setWireOutput("Brake" .. field, corners[slot])
            end
            values.TempMax = math.round(hottest, 1)
            values.WheelTemps = wheelTemps
            values.ExtraWheelTemps = extraTemps
            ctx.setWireOutput("BrakeTempMax", values.TempMax)
            ctx.setWireOutput("BrakeWheelTemps", wheelTemps)
            ctx.setWireOutput("ExtraBrakeWheelTemps", extraTemps)
        end,
    }
end
