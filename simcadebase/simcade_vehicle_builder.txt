--@name Simcade Vehicle Builder
--@shared

local SimcadeVehicleBuilder = {}
local abs, min, max = math.abs, math.min, math.max
local floor = math.floor
local clamp = math.clamp or function(v, a, b)
    if v < a then return a elseif v > b then return b else return v end
end
local IN_TO_M = 0.0254
local Tires = assert(SimcadeTires, "Load simcade_tires.txt before simcade_vehicle_builder.txt")
local Braking = assert(SimcadeBraking, "Load simcade_braking.txt before simcade_vehicle_builder.txt")
local Aero = assert(SimcadeAero, "Load simcade_aero.txt before simcade_vehicle_builder.txt")
local DEFAULT_VEHICLE_PARAMETERS = {
    referenceMassT = 1.22,
    frontAxleXM = 1.3,
    rearAxleXM = -1.3,
    trackHalfM = 0.75,
    wheelCenterZM = 0.08,
    wheelRadiusM = 0.3135,
    wheelWidthM = 0.205,
    groundYawInertiaMultiplier = 2.2,
    upshiftFraction = 0.8,
    clutchCoupling = 0.04,
    reverseGearRatio = -3.0,
    forwardGearRatios = { 1.0 },
    idleRPM = 800,
    redlineRPM = 6500,
    limiterRPM = 7000,
    flywheelInertia = 0.00025,
    torqueCurveKNm = { 0.14, 0.15, 0.16, 0.18, 0.20, 0.216, 0.203, 0.17, 0.15 },
    engineBrakeCurveKNm = { 0.12, 0.16, 0.12 },
    brakeAssist = 17,
}

local function makeVehicleParameters()
    local defaults = DEFAULT_VEHICLE_PARAMETERS
    local c = {
        [0] = 0, [1] = 0, [2] = 0,
        [68] = defaults.referenceMassT,
        [88] = defaults.groundYawInertiaMultiplier,
        [108] = defaults.upshiftFraction,
        [116] = defaults.reverseGearRatio,
        [117] = 0,
        [113] = defaults.clutchCoupling,
        [124] = defaults.idleRPM,
        [125] = defaults.redlineRPM,
        [126] = defaults.limiterRPM,
        [127] = defaults.flywheelInertia,
        [162] = defaults.brakeAssist,
        [163] = 0,
    }

    local wheelPositions = {
        { defaults.frontAxleXM, defaults.trackHalfM },
        { defaults.frontAxleXM, -defaults.trackHalfM },
        { defaults.rearAxleXM, -defaults.trackHalfM },
        { defaults.rearAxleXM, defaults.trackHalfM },
    }
    for i = 0, 3 do
        local offset = 4 + 12 * i
        local position = wheelPositions[i + 1]
        c[offset] = position[1]
        c[offset + 1] = position[2]
        c[offset + 2] = defaults.wheelCenterZM
        c[offset + 4] = defaults.wheelRadiusM
        c[offset + 5] = defaults.wheelWidthM
    end
    for i = 0, 8 do
        c[128 + i] = defaults.torqueCurveKNm[i + 1]
        c[137 + i] = 0
    end
    for i = 1, #defaults.forwardGearRatios do
        c[117 + i] = defaults.forwardGearRatios[i]
    end
    for i = 0, 2 do
        c[146 + i] = defaults.engineBrakeCurveKNm[i + 1]
    end
    return c
end

local function evalBezierTable(tbl, t)
    if type(tbl) ~= "table" or #tbl == 0 then return 1.0 end
    local n = #tbl - 1
    if n <= 0 then return tbl[1] or 1.0 end
    t = clamp(t, 0, 1)
    -- Piecewise smooth-step interpolation preserves exact peak/control points without Bernstein flattening
    local s = t * n
    local idx = min(n - 1, floor(s))
    local f = clamp(s - idx, 0, 1)
    local sf = f * f * (3 - 2 * f)
    local v0 = tbl[idx + 1] or 1.0
    local v1 = tbl[idx + 2] or v0
    return v0 + (v1 - v0) * sf
end

function SimcadeVehicleBuilder.buildCarData(vehicleCfg, tireCfg, suspCfg, engineCfg, steerCfg, runtimeWheels, totalMassKg, compoundName, aeroCfg, dependencies)
    dependencies = dependencies or {}
    local T = assert(dependencies.T, "Physics curves are required to build vehicle data")
    local Tune = assert(dependencies.Tune, "Shift logic is required to build vehicle data")
    vehicleCfg = vehicleCfg or {}
    tireCfg    = tireCfg    or {}
    suspCfg    = suspCfg    or {}
    engineCfg  = engineCfg  or {}
    steerCfg   = steerCfg   or {}

    local c = makeVehicleParameters()

    -- Coordinates are already CoM-centered in runtimeWheels
    c[0], c[1], c[2] = 0, 0, 0

    local targetMassT = max(0.35, (totalMassKg or 1500) / 1000)
    local massScale = targetMassT / max(0.4, c[68] or 1.1)
    c[68] = targetMassT

    local function getAxleSusp(isFront)
        if isFront and suspCfg.Front then return suspCfg.Front end
        if (not isFront) and suspCfg.Rear then return suspCfg.Rear end
        return suspCfg
    end
    local function getAxleTire(isFront)
        if isFront and tireCfg.Front then return tireCfg.Front end
        if (not isFront) and tireCfg.Rear then return tireCfg.Rear end
        return tireCfg
    end

    -- Wheel slots: front left, front right, rear right, rear left.
    local frontSumX, rearSumX, maxAbsY = 0, 0, 0.75
    local frontN, rearN = 0, 0
    local avgRadiusM, radCount = 0, 0
    if runtimeWheels and #runtimeWheels >= 4 then
        local slotByQuad = {}
        for _, rw in ipairs(runtimeWheels) do
            if rw.isFront and (rw.pos.y >= 0) and not slotByQuad[0] then slotByQuad[0] = rw
            elseif rw.isFront and (rw.pos.y < 0) and not slotByQuad[1] then slotByQuad[1] = rw
            elseif (not rw.isFront) and (rw.pos.y < 0) and not slotByQuad[2] then slotByQuad[2] = rw
            elseif (not rw.isFront) and (rw.pos.y >= 0) and not slotByQuad[3] then slotByQuad[3] = rw
            end
        end
        slotByQuad[0] = slotByQuad[0] or runtimeWheels[1]
        slotByQuad[1] = slotByQuad[1] or runtimeWheels[2]
        slotByQuad[2] = slotByQuad[2] or runtimeWheels[4] or runtimeWheels[3]
        slotByQuad[3] = slotByQuad[3] or runtimeWheels[3] or runtimeWheels[4]

        for ni = 0, 3 do
            local rw = slotByQuad[ni]
            if rw then
                local o = 4 + 12 * ni
                local isFront = (ni == 0 or ni == 1)
                local aSusp = getAxleSusp(isFront)
                local restComp = clamp(aSusp.restCompression or 0.45, 0.25, 0.70)
                local rideH_m  = max(8, aSusp.rideHeight or 24) * IN_TO_M
                local x_m = rw.pos.x * IN_TO_M
                local y_m = rw.pos.y * IN_TO_M
                local z_rest_m = SimcadeSuspension.mountHeight(rw, aSusp) * IN_TO_M - (1.0 - restComp) * rideH_m
                local rad_m = max(0.15, (rw.radius or 15) * IN_TO_M)

                c[o]     = x_m
                c[o + 1] = y_m
                c[o + 2] = z_rest_m
                c[o + 4] = rad_m
                c[o + 5] = max(0.10, (rw.width or 9) * IN_TO_M)
                avgRadiusM = avgRadiusM + rad_m
                radCount = radCount + 1

                if isFront then frontSumX = frontSumX + x_m; frontN = frontN + 1
                else rearSumX = rearSumX + x_m; rearN = rearN + 1 end
                maxAbsY = max(maxAbsY, abs(y_m))
            end
        end
    end
    avgRadiusM = (radCount > 0) and (avgRadiusM / radCount) or (15 * IN_TO_M)

    -- Update box dimensions & inertia diagonal from wheel layout
    local wheelbase = (frontN > 0 and rearN > 0) and max(1.6, (frontSumX / frontN) - (rearSumX / rearN)) or 2.44
    local lenM   = max(3.0, wheelbase * 1.75)
    local widthM = max(1.4, maxAbsY * 2.1)
    local heightM = 1.29
    c[69], c[70], c[71] = lenM, widthM, heightM
    for i = 72, 87 do c[i] = 0 end
    c[72] = (heightM * heightM + widthM * widthM) * targetMassT * 0.0833333358168602
    c[77] = (heightM * heightM + lenM * lenM) * targetMassT * 0.0833333358168602
    c[82] = (widthM * widthM + lenM * lenM) * targetMassT * 0.0833333358168602
    c[87] = 1

    -- Wheel travel limits used to position the tire relative to its mount.
    local cornerMassT = targetMassT * 0.25
    for axle = 0, 1 do
        local row = 92 + 8 * axle
        local isFront = (axle == 0)
        local aSusp = getAxleSusp(isFront)
        local rideH_in = max(8, aSusp.rideHeight or 24)
        local airTrav_in = max(-(rideH_in - 2), aSusp.airTravel ~= nil and aSusp.airTravel or 6)
        local rideH_m  = rideH_in * IN_TO_M
        local airTrav_m = airTrav_in * IN_TO_M
        c[row + 5] = clamp(rideH_m * 0.95, 0.18, 1.50)
        c[row + 6] = -airTrav_m
    end

    -- Tires: store TireSimV2 compound & per-axle specs + reversible Stage ratios
    local compName = compoundName or "Sport"
    local fTire = getAxleTire(true)
    local rTire = getAxleTire(false)

    local staticCornerKN = cornerMassT * 9.800000190734863
    local tireSimSpecs = {
        tireCfg  = tireCfg,
        frontCfg = fTire,
        rearCfg  = rTire,
        front    = Tires.buildAxleSpec(fTire, compName, staticCornerKN, tireCfg, "Front"),
        rear     = Tires.buildAxleSpec(rTire, compName, staticCornerKN, tireCfg, "Rear"),
    }

    -- Brake configuration is owned by the braking module.
    local braking = Braking.configure(engineCfg, targetMassT, massScale)
    c[152], c[153], c[154] = Aero.buildSolverCoefficients(aeroCfg, massScale)

    -- Transmission, Ratios & Automatic gearSpeeds Calibration (unlimited forward + reverse gears)
    local top = #DEFAULT_VEHICLE_PARAMETERS.forwardGearRatios
    local gearRatioMap = { [1] = 0 }
    local revList = {}
    local fwdList = {}

    local revSrc = engineCfg.ReverseGearRatios
    local fwdSrc = engineCfg.ForwardGearRatios or engineCfg.Gearbox_Ratios or engineCfg.gearRatios or engineCfg.GearRatios
    if type(fwdSrc) == "table" and #fwdSrc >= 1 then
        local zeroIdx = nil
        for idx = 1, #fwdSrc do
            if fwdSrc[idx] == 0 then zeroIdx = idx; break end
        end
        if zeroIdx and not revSrc then
            for idx = 1, zeroIdx - 1 do revList[#revList + 1] = -math.abs(fwdSrc[idx]) end
            for idx = zeroIdx + 1, #fwdSrc do fwdList[#fwdList + 1] = math.abs(fwdSrc[idx]) end
        else
            for idx = 1, #fwdSrc do
                if fwdSrc[idx] and fwdSrc[idx] ~= 0 then fwdList[#fwdList + 1] = math.abs(fwdSrc[idx]) end
            end
        end
    end
    if type(revSrc) == "table" and #revSrc >= 1 then
        revList = {}
        for idx = 1, #revSrc do
            if revSrc[idx] and revSrc[idx] ~= 0 then revList[#revList + 1] = -math.abs(revSrc[idx]) end
        end
    elseif #revList == 0 then
        revList[1] = -math.abs(engineCfg.ReverseRatio or c[116] or 3.45)
    end
    if #fwdList == 0 then
        for i = 1, top do fwdList[i] = math.abs(c[117 + i] or 1.0) end
    end

    top = #fwdList
    local minGear = 1 - #revList
    for i = 1, #revList do gearRatioMap[1 - i] = revList[i] end
    for i = 1, #fwdList do gearRatioMap[1 + i] = fwdList[i] end
    c[116] = gearRatioMap[0] or -3.45
    c[117] = 0
    for i = 1, 6 do c[117 + i] = gearRatioMap[1 + i] or 0 end

    -- Engine RPM Grid
    if engineCfg.idleRPM then c[124] = max(500, engineCfg.idleRPM) end
    if engineCfg.redlineRPM then c[125] = max(c[124] + 1500, engineCfg.redlineRPM) end
    c[126] = max(c[125] + 250, engineCfg.limiterRPM or (c[125] + 500))

    local rawFinal = engineCfg.finalDrive or engineCfg.FinalDrive or 4.10
    local fwdSpeeds = engineCfg.ForwardGearSpeeds or engineCfg.GearSpeeds or engineCfg.gearSpeeds
    local revSpeeds = engineCfg.ReverseGearSpeeds

    if engineCfg.AutoCalibrateRatiosFromSpeeds == true and rawFinal > 0.05 then
        local kNumerator = (c[125] * 2.0 * math.pi * avgRadiusM * 3.6) / (60.0 * rawFinal)
        if type(fwdSpeeds) == "table" and #fwdSpeeds >= 1 then
            fwdList = {}
            for i = 1, #fwdSpeeds do
                local spd = tonumber(fwdSpeeds[i]) or 0
                if spd > 1 then
                    fwdList[#fwdList + 1] = clamp(kNumerator / spd, 0.2, 15.0)
                end
            end
            if #fwdList >= 1 then
                top = #fwdList
                for i = 1, #fwdList do gearRatioMap[1 + i] = fwdList[i] end
                for i = 1, 6 do c[117 + i] = gearRatioMap[1 + i] or 0 end
            end
        end
        if type(revSpeeds) == "table" and #revSpeeds >= 1 then
            revList = {}
            for i = 1, #revSpeeds do
                local spd = tonumber(revSpeeds[i]) or 0
                if spd > 1 then
                    revList[#revList + 1] = -clamp(kNumerator / spd, 0.2, 15.0)
                end
            end
            if #revList >= 1 then
                minGear = 1 - #revList
                for i = 1, #revList do gearRatioMap[1 - i] = revList[i] end
                c[116] = gearRatioMap[0] or -3.45
            end
        end
    elseif engineCfg.AutoCalibrateFinalDrive == true then
        local topGearIdx = 1 + top
        local topRatio = abs(gearRatioMap[topGearIdx] or 0.763)
        local targetTopKmh = (type(fwdSpeeds) == "table") and fwdSpeeds[#fwdSpeeds] or nil
        if targetTopKmh and targetTopKmh > 60 and topRatio > 0.1 then
            local calFinal = (c[125] * 2.0 * math.pi * avgRadiusM * 3.6) / (60.0 * topRatio * targetTopKmh)
            if calFinal > rawFinal * 1.12 or calFinal < rawFinal * 0.88 then
                rawFinal = clamp(calFinal, rawFinal * 0.85, rawFinal * 1.56)
            end
        end
    end
    c[110] = rawFinal
    c[111] = rawFinal
    local shiftDelay = clamp(tonumber(engineCfg.shiftDelay) or 0.14, 0.01, 2.0)
    c[115] = 1.0 / max(0.02, shiftDelay * 0.45)
    local frontDriveShare = clamp((engineCfg.PowerDistribution or 0) / 100, 0, 1)
    local rearDriveShare  = 1.0 - frontDriveShare
    c[112] = rearDriveShare

    -- Engine Torque Curve (VirtualEngine torqueTable + Engine Specs + Turbo Boost -> kN.m)
    local baseNm = (engineCfg.baseTorque or 520) * (engineCfg.TorqueMultiplier or 1.25) * (engineCfg.ForceMult or 1.0)
    if type(engineCfg.torqueTable) == "table" and #engineCfg.torqueTable >= 2 then
        local loRpm, redRpm, limRpm = c[124], c[125], c[126]
        local spanAll = max(500, limRpm - loRpm)
        local spanRed = max(500, redRpm - loRpm)
        local cyl = engineCfg.Engine_Cylinders or 4
        local bore = engineCfg.Engine_Bore or 87
        local stroke = engineCfg.Engine_Stroke or 84
        local dispL = (math.pi * 0.25 * (bore * 0.1)^2 * (stroke * 0.1) * cyl) / 1000
        local dispBonus = clamp(0.95 + (dispL - 1.8) * 0.04, 0.90, 1.18)
        local airflow = engineCfg.Engine_Airflow or 165
        local airBonus = clamp(0.94 + (airflow - 140) / 400, 0.90, 1.15)
        local breathingMult = dispBonus * airBonus * 1.12
        for i = 0, 8 do
            local rpm = loRpm + (i / 8) * spanAll
            local u = clamp((rpm - loRpm) / spanRed, 0, 1)
            local bz = evalBezierTable(engineCfg.torqueTable, u)
            if rpm > redRpm then
                local overU = clamp((rpm - redRpm) / max(100, limRpm - redRpm), 0, 1)
                bz = bz * (1.0 - 0.06 * overU)
            end
            c[128 + i] = max(0.05, (baseNm * bz * breathingMult) / 1000)
        end
    else
        local curPeak = 0.001
        for i = 0, 8 do
            if c[128 + i] > curPeak then curPeak = c[128 + i] end
        end
        local desiredPeakKNm = baseNm / 1000
        if desiredPeakKNm > 0.05 then
            local tqScale = desiredPeakKNm / curPeak
            for i = 0, 8 do c[128 + i] = c[128 + i] * tqScale end
        end
    end

    local engineTorqueCurve, enginePeakTorqueNm
    if SimcadeEngine and engineCfg and (engineCfg.PeakTorqueNm or engineCfg.PeakHP or engineCfg.DisplacementLiters or engineCfg.TorqueCurve) then
        local spec = SimcadeEngine.synthesizeSpec(engineCfg)
        c[124] = spec.idleRPM
        c[125] = spec.redlineRPM
        c[126] = spec.limiterRPM
        c[127] = clamp(spec.flywheelInertia * 0.001, 0.00003, 0.0015)
        engineTorqueCurve = spec.torqueCurve
        enginePeakTorqueNm = spec.peakTorqueNm
        for i = 0, 8 do
            c[128 + i] = spec.torqueCurveKNm[i]
        end
        for i = 0, 2 do
            c[146 + i] = spec.brakeCurveKNm[i]
        end
    end

    -- Engine Braking
    if engineCfg.engineBrakeTorque ~= nil then
        local ebMult = max(0, tonumber(engineCfg.EngineBrakeMult) or 1.0)
        local brkKNm = clamp((engineCfg.engineBrakeTorque / 1000) * ebMult, 0.0, 5.0)
        c[146], c[147], c[148] = brkKNm * 0.45, brkKNm * 0.78, brkKNm
    else
        for i = 0, 2 do
            c[146 + i] = c[146 + i] * max(0.6, massScale)
        end
    end

    -- Zero disables the speed limiter.
    c[164] = engineCfg.SpeedLimitMPH or 0

    local up, down = Tune.shiftPoints(c, top)
    -- Ensure auto shifts happen crisply near peak power / redline without bouncing on limiter
    local redRpm = c[125]
    for g = 2, top + 1 do
        if up[g] then
            up[g] = min(up[g], redRpm * 0.935)
        end
    end
    local frontWeightBias = clamp(tonumber(vehicleCfg.frontWeightBias) or 0.50, 0.30, 0.70)
    local wheelbase_m = max(1.0, abs((c[4] or 1.32) - (c[28] or -1.32)))
    local comShiftX_m = (frontWeightBias - 0.50) * wheelbase_m

    return {
        c                 = c,
        engineTorqueCurve = engineTorqueCurve,
        enginePeakTorqueNm = enginePeakTorqueNm,
        top               = top,
        minGear           = minGear,
        gearRatioMap      = gearRatioMap,
        fwdSpeeds         = fwdSpeeds,
        reverseGearSpeeds = (engineCfg and engineCfg.ReverseGearSpeeds) or nil,
        up              = up,
        down            = down,
        T               = T,
        hull            = {},
        tireSimSpecs    = tireSimSpecs,
        steerCfg        = steerCfg,
        frontDriveShare = frontDriveShare,
        rearDriveShare  = rearDriveShare,
        turboEnabled    = (engineCfg.Turbo == true),
        shiftDelay      = shiftDelay,
        frontWeightBias = frontWeightBias,
        comShiftX_m     = comShiftX_m,
        braking         = braking,
    }
end
_G.SimcadeVehicleBuilder = SimcadeVehicleBuilder
return SimcadeVehicleBuilder
