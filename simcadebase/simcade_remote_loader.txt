--@name Simcade Remote Thin Loader
--@shared

--[[
    Lightweight remote loader for SimcadeBase (~5 KB in AdvDupe2).
    Zero core/addon include directives:
      - Server fetches the latest matching core release (core.bin) + any enabled
        addons in CONFIG.ADDON_MODULES that are NOT locally included.
      - Any addon that IS locally included in your vehicle config takes
        priority (great for custom per-car addons or testing local edits).
      - Streams the compiled bundle once to each client via net.writeStream
        (clients do not need http.get permission to load core or addons).
]]

local modules = {
    "simcadebase/simcade_engine.txt",
    "simcadebase/simcade_aero.txt",
    "simcadebase/simcade_tires.txt",
    "simcadebase/simcade_gearbox.txt",
    "simcadebase/simcade_suspension.txt",
    "simcadebase/simcade_controls.txt",
    "simcadebase/simcade_camera.txt",
    "simcadebase/simcade_sounds.txt",
    "simcadebase/simcade_skidmarks.txt",
    "simcadebase/simcade_effects.txt",
    "simcadebase/simcade_hud.txt",
    "simcadebase/simcade_wire.txt",
    "simcadebase/simcade_addons.txt",
    "simcadebase/simcade_braking.txt",
    "simcadebase/simcade_steering.txt",
    "simcadebase/simcade_vehicle_builder.txt",
    "simcadebase/simcade_vehicle_physics.txt",
    "simcadebase/simcade_core.txt",
    "simcadebase/simcade_prediction.txt",
}

local MAX_BYTES = 4 * 1024 * 1024

local function parsePositiveInteger(value)
    if not value or #value > 9 or value:find("[^0-9]") then return end
    local number = tonumber(value)
    if number and number > 0 then return number end
end

local function validDigest(value)
    return value and #value == 64 and not value:find("[^0-9a-f]")
end

-- Wire format streamed to clients:
--   <coreByteCount>\n<coreBundleBytes><addonCount>\n
--   For each remote addon:
--     <pathLen>\n<path><srcLen>\n<source>
local function compilePayload(payload, ready, fail)
    if type(payload) ~= "string" or #payload > MAX_BYTES then return fail("Invalid payload size") end

    local nl = payload:find("\n", 1, true)
    if not nl or nl > 10 then return fail("Invalid core header") end
    local coreLen = tonumber(payload:sub(1, nl - 1):match("^%d+$") or "")
    if not coreLen or coreLen < 1 or nl + coreLen > #payload then return fail("Incomplete core payload") end

    local coreData = payload:sub(nl + 1, nl + coreLen)
    local addonPos = nl + coreLen + 1

    local compiledCore = {}
    local compiledAddons = {}

    -- Parse addon table upfront so compileNext can step through core + addons smoothly
    local addonEntries = {}
    if addonPos <= #payload then
        local cnl = payload:find("\n", addonPos, true)
        if not cnl or cnl - addonPos > 6 then return fail("Invalid addon count header") end
        local addonCount = tonumber(payload:sub(addonPos, cnl - 1):match("^%d+$") or "")
        if not addonCount then return fail("Invalid addon count") end
        local p = cnl + 1
        for i = 1, addonCount do
            local pnl = payload:find("\n", p, true)
            if not pnl or pnl - p > 6 then return fail("Invalid addon path header") end
            local pLen = tonumber(payload:sub(p, pnl - 1):match("^%d+$") or "")
            if not pLen or pnl + pLen > #payload then return fail("Incomplete addon path") end
            local aPath = payload:sub(pnl + 1, pnl + pLen)
            p = pnl + pLen + 1

            local snl = payload:find("\n", p, true)
            if not snl or snl - p > 8 then return fail("Invalid addon source header") end
            local sLen = tonumber(payload:sub(p, snl - 1):match("^%d+$") or "")
            if not sLen or snl + sLen > #payload then return fail("Incomplete addon source: " .. aPath) end
            local aSrc = payload:sub(snl + 1, snl + sLen)
            p = snl + sLen + 1

            addonEntries[i] = { path = aPath, source = aSrc }
        end
    end

    local corePos, coreIndex, addonIndex = 1, 1, 1
    local function compileNext()
        local path = modules[coreIndex]
        if path then
            local stop = coreData:find("\n", corePos, true)
            if not stop or stop - corePos > 7 then return fail("Invalid module header for " .. path) end
            local size = tonumber(coreData:sub(corePos, stop - 1):match("^%d+$") or "")
            if not size or size < 1 or stop + size > #coreData then return fail("Incomplete " .. path) end
            local source = coreData:sub(stop + 1, stop + size)
            if path == "simcadebase/simcade_addons.txt" then
                source = "local require = ...\n" .. source
            end
            local fn, err = loadstring(source, path)
            if not fn then return fail(tostring(err)) end
            compiledCore[path] = fn
            corePos = stop + size + 1
            coreIndex = coreIndex + 1
            timer.simple(0, compileNext)
            return
        end

        local aEntry = addonEntries[addonIndex]
        if aEntry then
            local fn, err = loadstring(aEntry.source, aEntry.path)
            if not fn then return fail("Addon " .. aEntry.path .. ": " .. tostring(err)) end
            compiledAddons[aEntry.path] = fn
            addonIndex = addonIndex + 1
            timer.simple(0, compileNext)
            return
        end

        ready(compiledCore, compiledAddons)
    end
    compileNext()
end

local function boot(config)
    local schema = config.SCHEMA_VERSION
    if schema == nil then schema = 1 end
    if type(schema) ~= "number" or schema < 1 or schema > 999999999 or schema ~= math.floor(schema) then
        if SERVER then print("[Simcade] Vehicle stopped. CONFIG.SCHEMA_VERSION must be a positive integer.") end
        return
    end

    local loadedAddons = {}
    local function start(compiledCore, compiledAddons)
        local function requireAddon(path)
            -- Local include always takes priority over remote copy
            if getScript and getScript(path) then
                return require(path)
            end
            local fn = compiledAddons and compiledAddons[path]
            if not fn then
                return require(path)
            end
            if loadedAddons[path] == nil then
                loadedAddons[path] = fn()
            end
            return loadedAddons[path]
        end

        for _, path in ipairs(modules) do
            compiledCore[path](requireAddon)
        end
        SimcadeCore.init(config)
    end

    local REQ_NET = "simcade_remote_req"
    local REP_NET = "simcade_remote_rep"

    -- ========================================================================
    -- CLIENT: receive combined core + remote addons bundle from server
    -- ========================================================================
    if CLIENT then
        local busy, started = false, false
        net.receive(REP_NET, function()
            if started or busy then return end
            busy = true
            local ok = pcall(net.readStream, function(data)
                if not data then busy = false return end
                compilePayload(data, function(compiledCore, compiledAddons)
                    started = true
                    timer.remove(REQ_NET)
                    start(compiledCore, compiledAddons)
                end, function(reason)
                    busy = false
                    print("[Simcade Remote] Client compile error: " .. tostring(reason))
                end)
            end)
            if not ok then busy = false end
        end)

        timer.create(REQ_NET, 2, 0, function()
            if busy or started then return end
            net.start(REQ_NET)
            if net.getBytesLeft() < 0 then return net.abort() end
            net.send()
        end)
        return
    end

    -- ========================================================================
    -- SERVER: fetch core.bin + any un-included enabled addons from GitHub
    -- ========================================================================
    local opt = config.CORE_UPDATE or {}
    local releaseBase = (opt.baseURL or "https://raw.githubusercontent.com/nokillnando/Starfall/main/simcadebase/releases"):gsub("/+$", "")
    local repoBase = (opt.repoURL or releaseBase:gsub("/simcadebase/releases$", "")):gsub("/+$", "")

    if not releaseBase:match("^https://[^/]+") then
        print("[Simcade Remote] Vehicle stopped. Set CONFIG.CORE_UPDATE.baseURL to a valid https:// URL.")
        return
    end

    local readyPayload = nil
    local lastSent = setmetatable({}, { __mode = "k" })
    net.receive(REQ_NET, function(_, ply)
        if not readyPayload or (lastSent[ply] or -10) + 3 > timer.curtime() then return end
        if not net.canWriteStream() then return end
        net.start(REP_NET)
        net.writeStream(readyPayload, false)
        if net.getBytesLeft() < 0 then return net.abort() end
        net.send(ply)
        lastSent[ply] = timer.curtime()
    end)

    local timeoutName = "simcade_remote_timeout"
    local failed = false
    local function failBoot(reason)
        if failed then return end
        failed = true
        timer.remove(timeoutName)
        print("[Simcade Remote] Vehicle stopped: " .. tostring(reason))
    end

    local function fetchURL(url, limit, onOk, onMissing)
        if failed then return end
        timer.create(timeoutName, 20, 1, function() failBoot("Timed out fetching " .. url) end)
        local function send()
            if failed then return end
            if not http.canRequest() then return timer.simple(0.25, send) end
            local ok, err = pcall(http.get, url, function(body, _, _, code)
                if failed then return end
                if code == 404 and onMissing then
                    timer.remove(timeoutName)
                    return onMissing()
                end
                if code ~= 200 or not body or #body > limit then
                    return failBoot("HTTP " .. tostring(code) .. " for " .. url)
                end
                timer.remove(timeoutName)
                onOk(body)
            end, function(reason)
                failBoot("Could not fetch " .. url .. (reason and (": " .. tostring(reason)) or ""))
            end)
            if not ok then failBoot(tostring(err)) end
        end
        send()
    end

    -- Determine which enabled addons are NOT locally included and need remote fetching
    local remoteAddonsToFetch = {}
    local seenAddon = {}
    local function queueAddon(path, customURL)
        if not path or seenAddon[path] then return end
        seenAddon[path] = true
        if getScript and getScript(path) then return end -- Locally included!
        remoteAddonsToFetch[#remoteAddonsToFetch + 1] = {
            path = path,
            url  = customURL or (repoBase .. "/" .. path:gsub("^/+", "")),
        }
    end

    if config.ADDON_MODULES then
        for _, spec in ipairs(config.ADDON_MODULES) do
            if spec.enabled ~= false and spec.file then
                queueAddon(spec.file, spec.remoteURL)
            end
        end
    else
        queueAddon("simcadebase/addons/lights.txt")
    end

    local function finishWithCore(coreData, revisionLabel)
        local addonParts = {}
        local function fetchAddonAt(i)
            local entry = remoteAddonsToFetch[i]
            if not entry then
                local header = tostring(#coreData) .. "\n" .. coreData .. tostring(#addonParts) .. "\n"
                local payload = header .. table.concat(addonParts)
                compilePayload(payload, function(compiledCore, compiledAddons)
                    readyPayload = payload
                    start(compiledCore, compiledAddons)
                    print("[Simcade Remote] Loaded " .. revisionLabel .. " + " .. tostring(#addonParts) .. " remote addon(s).")
                end, failBoot)
                return
            end
            fetchURL(entry.url, 512 * 1024, function(src)
                src = src:gsub("\r\n", "\n")
                addonParts[#addonParts + 1] = tostring(#entry.path) .. "\n" .. entry.path .. tostring(#src) .. "\n" .. src
                timer.simple(0, function() fetchAddonAt(i + 1) end)
            end)
        end
        fetchAddonAt(1)
    end

    local function handleManifest(body)
        local fields = body:gmatch("%S+")
        local revision, digest, required = fields(), fields(), fields()
        local revisionNumber = parsePositiveInteger(revision)
        if fields() or not revisionNumber or not validDigest(digest) then
            return failBoot("Invalid release manifest")
        end
        local requiredSchema = parsePositiveInteger(required)
        if not requiredSchema then
            return failBoot("Release r" .. revision .. " manifest has no valid config version.")
        end
        if requiredSchema ~= schema then
            return failBoot("Release r" .. revision .. " requires config v" .. required .. ", vehicle uses v" .. schema .. ".")
        end

        local function activateCore(coreData)
            if bit.sha256(coreData) ~= digest then
                return failBoot("Release checksum mismatch")
            end
            finishWithCore(coreData, "core r" .. revision)
        end

        local function fetchBundle(onMissing)
            fetchURL(releaseBase .. "/" .. revision .. "/core.bin", MAX_BYTES, activateCore, onMissing)
        end

        local function fetchFromSources(onMissing)
            fetchURL(releaseBase .. "/" .. revision .. "/sources.txt", #modules * 120, function(index)
                local tokens = index:gmatch("%S+")
                local hashes = {}
                for i, path in ipairs(modules) do
                    local indexedPath, hash = tokens(), tokens()
                    if indexedPath ~= path or not validDigest(hash) then
                        return failBoot("Invalid source index for r" .. revision)
                    end
                    hashes[i] = hash
                end
                if tokens() then return failBoot("Unexpected files in source index") end

                local records, total = {}, 0
                local function fetchMod(i)
                    local path = modules[i]
                    if not path then
                        return activateCore(table.concat(records))
                    end
                    local hash = hashes[i]
                    fetchURL(releaseBase .. "/source/" .. hash .. ".txt", MAX_BYTES - total, function(src)
                        if bit.sha256(src) ~= hash then
                            return failBoot("Checksum mismatch for " .. path)
                        end
                        local record = tostring(#src) .. "\n" .. src
                        total = total + #record
                        if total > MAX_BYTES then return failBoot("Release exceeds size limit") end
                        records[i] = record
                        timer.simple(0, function() fetchMod(i + 1) end)
                    end)
                end
                fetchMod(1)
            end, onMissing)
        end

        if opt.format == "source" then
            fetchFromSources(function() fetchBundle() end)
        else
            fetchBundle(function() fetchFromSources() end)
        end
    end

    print("[Simcade Remote] Downloading Simcade core (config v" .. schema .. ")...")
    -- Try schema-pinned manifest first (latest_v<schema>.txt) so older dupes keep
    -- working forever even after latest.txt moves to a newer SCHEMA_VERSION.
    fetchURL(releaseBase .. "/latest_v" .. tostring(schema) .. ".txt", 128, handleManifest, function()
        fetchURL(releaseBase .. "/latest.txt", 128, handleManifest)
    end)
end

return function(config)
    boot(config)
end
