--@name Simcade Camera Module (Chase, Freecam & FPV, ported from Advanced Car Camera V1.8)
--@shared

SimcadeCamera = SimcadeCamera or {}

local MODE_NAMES = {
    [0] = "Off",
    [1] = "Chase",
    [2] = "FPV",
    [3] = "Freecam",
}

function SimcadeCamera.modeName(mode)
    return MODE_NAMES[mode] or "Off"
end

local function modeEnabled(cam, mode, isPassenger)
    if mode == 0 then return true end
    if cam.enabled == false then return false end
    if isPassenger then
        if cam.passengerCamEnabled == false then return false end
        if mode == 1 then return cam.passengerChaseEnabled ~= false end
        if mode == 2 then return cam.passengerFpvEnabled ~= false end
        if mode == 3 then return cam.passengerFreecamEnabled ~= false end
        return false
    end
    if mode == 1 then return cam.chaseEnabled ~= false end
    if mode == 2 then return cam.fpvEnabled ~= false end
    if mode == 3 then return cam.freecamEnabled == true end
    return false
end

local function resolveMode(cam, mode, isPassenger)
    if modeEnabled(cam, mode, isPassenger) then return mode end
    if modeEnabled(cam, 1, isPassenger) then return 1 end
    if modeEnabled(cam, 3, isPassenger) then return 3 end
    if modeEnabled(cam, 2, isPassenger) then return 2 end
    return 0
end

local function nextMode(cam, curMode, isPassenger)
    local mode = curMode or 0
    for _ = 1, 4 do
        mode = (mode + 1) % 4
        if modeEnabled(cam, mode, isPassenger) then
            return mode
        end
    end
    return 0
end

if SERVER then
    local _srvState = { camMode = 1, trailerMode = false, cameraConfig = {} }
    function SimcadeCamera.initServer(state, cfg)
        state = state or _srvState
        local cam = cfg and cfg.CAMERA_CFG or {}
        state.cameraConfig = cam
        state.camMode = resolveMode(cam, cam.defaultMode or 1, false)
        state.trailerMode = false
        return state
    end

    function SimcadeCamera.resolvePassengerMode(cfg, mode)
        local cam = cfg and (cfg.CAMERA_CFG or cfg) or {}
        if mode == nil then mode = cam.passengerDefaultMode end
        if mode == nil then mode = 1 end
        return resolveMode(cam, mode, true)
    end

    function SimcadeCamera.togglePassengerMode(cfg, curMode, mode)
        local cam = cfg and (cfg.CAMERA_CFG or cfg) or {}
        if mode == nil then
            return nextMode(cam, curMode, true)
        end
        return resolveMode(cam, mode, true)
    end

    function SimcadeCamera.toggleMode(state, mode)
        state = state or _srvState
        local cam = state.cameraConfig or {}
        if mode == nil then
            mode = nextMode(cam, state.camMode, false)
        end
        state.camMode = resolveMode(cam, mode, false)
        return state.camMode
    end
    return
end

local clamp = math.clamp or function(v, a, b) return v < a and a or (v > b and b or v) end

local camState = {
    camMode       = 1,
    trailerActive = false,
    currentDist   = 185,
    targetDist    = 185,
    freeDist      = 190,
    distVel       = 0,
    currentAng    = nil,
    targetAng     = nil,
    lastMouseTime = 0,
    lastSpeed     = 0,
    currentFOV    = 90,
    trackAng      = nil,
    smoothSeatAng = nil,
    initialized   = false,
    base          = nil,
    seat          = nil,
    isPassenger   = false,
}

function SimcadeCamera.initClient(cfg)
    local CAM = cfg.CAMERA_CFG or {}
    local anyDriverMode = modeEnabled(CAM, 1, false) or modeEnabled(CAM, 2, false) or modeEnabled(CAM, 3, false)
    local anyPassMode   = modeEnabled(CAM, 1, true)  or modeEnabled(CAM, 2, true)  or modeEnabled(CAM, 3, true)
    if not (anyDriverMode or anyPassMode) then return end

    pcall(function() if enableHud then enableHud(player(), true) end end)

    camState.camMode     = resolveMode(CAM, CAM.defaultMode or 1, false)
    camState.targetDist  = CAM.maxDist or CAM.chaseDist or 190
    camState.currentDist = camState.targetDist
    camState.freeDist    = CAM.passengerDist or CAM.maxDist or CAM.chaseDist or 190

    local localFwd = (cfg.VEHICLE_CFG and cfg.VEHICLE_CFG.chassisForward or Vector(1, 0, 0)):getNormalized()
    local localRight = localFwd:cross(Vector(0, 0, 1))
    if localRight:getLengthSqr() < 0.01 then localRight = Vector(0, -1, 0) end
    localRight = localRight:getNormalized()
    local localUp = localRight:cross(localFwd):getNormalized()

    local function getVehicleContext()
        local lp = player()
        if not (lp and lp:isValid() and lp:inVehicle() and lp.getVehicle) then return nil, nil, false end
        local veh = lp:getVehicle()
        if not (veh and veh:isValid()) then return nil, nil, false end

        local hudSt = SimcadeHUD and SimcadeHUD.getState and SimcadeHUD.getState() or nil
        local targetPod = hudSt and hudSt.pod or nil
        if targetPod and targetPod:isValid() and veh == targetPod then
            local base = (hudSt and hudSt.base and hudSt.base:isValid() and hudSt.base)
                or (camState.base and camState.base:isValid() and camState.base) or nil
            return base, targetPod, false
        end

        if camState.seat and camState.seat:isValid() and veh == camState.seat then
            local base = (camState.base and camState.base:isValid() and camState.base)
                or (hudSt and hudSt.base and hudSt.base:isValid() and hudSt.base) or nil
            return base, camState.seat, camState.isPassenger == true
        end

        return nil, nil, false
    end

    local function isInThisVehicle()
        local base, seat = getVehicleContext()
        return base ~= nil and seat ~= nil
    end

    net.receive("Simcade_CameraSeat", function()
        local base = net.readEntity()
        local seat = net.readEntity()
        local isPassenger = net.readBool()
        local mode = net.readUInt(2)
        camState.base = (base and base:isValid()) and base or nil
        camState.seat = (seat and seat:isValid()) and seat or nil
        camState.isPassenger = isPassenger
        camState.camMode = resolveMode(CAM, mode, isPassenger)
        camState.initialized = false
        camState.currentAng = nil
        camState.targetAng = nil
        camState.smoothSeatAng = nil
        camState.trackAng = nil
    end)

    net.receive("Simcade_CameraMode", function()
        local mode = net.readUInt(2)
        local _, _, isPassenger = getVehicleContext()
        camState.camMode = resolveMode(CAM, mode, isPassenger)
        camState.initialized = false
        camState.currentAng = nil
        camState.targetAng = nil
    end)

    local function isMouseBlockedByUI()
        if SimcadeCamera._blockMouseInput then return true end
        if input and input.isCursorVisible then
            local ok, vis = pcall(input.isCursorVisible)
            if ok and vis then return true end
        end
        return false
    end

    hook.add("mouseMoved", "SimcadeCam_MouseLook", function(dx, dy)
        if camState.camMode == 0 then return end
        if isMouseBlockedByUI() then return end
        if not isInThisVehicle() then return end
        local thresh = CAM.mouseThreshold or 2
        if math.abs(dx) < thresh and math.abs(dy) < thresh then return end
        local sens = CAM.mouseSens or 0.05
        local pSign = CAM.invertPitch and -1 or 1
        local minP = (camState.camMode == 3) and -80 or -65
        local maxP = (camState.camMode == 3) and 85 or 75
        if camState.targetAng then
            camState.targetAng.y = camState.targetAng.y - dx * sens
            camState.targetAng.p = clamp(camState.targetAng.p + dy * sens * pSign, minP, maxP)
        end
        camState.lastMouseTime = timer.curtime()
    end)

    hook.add("mouseWheeled", "SimcadeCam_Zoom", function(delta)
        if isMouseBlockedByUI() then return end
        if not isInThisVehicle() then return end
        if camState.camMode == 1 then
            local maxD = clamp((CAM.maxDist or CAM.chaseDist or 190) - delta * 14, 90, 450)
            CAM.maxDist = maxD
            CAM.minDist = math.max(70, maxD - 30)
            camState.targetDist = maxD
        elseif camState.camMode == 3 then
            camState.freeDist = clamp((camState.freeDist or CAM.maxDist or 190) - delta * 14, 65, 450)
        end
    end)

    hook.add("CalcView", "SimcadeCam_CalcView", function(origin, angles, fov)
        local base, seatEnt, isPassenger = getVehicleContext()
        if not (base and base:isValid() and seatEnt and seatEnt:isValid()) then
            camState.initialized = false
            camState.currentAng = nil
            camState.targetAng = nil
            return
        end
        if camState.camMode == 0 then return end

        local hudSt = SimcadeHUD and SimcadeHUD.getState and SimcadeHUD.getState() or nil
        local lp = player()
        local dt = clamp(timer.frametime() or 0.015, 0.001, 0.05)
        local vel = base:getVelocity()
        local fwd   = base:localToWorldVector(localFwd):getNormalized()
        local right = base:localToWorldVector(localRight):getNormalized()
        local up    = base:localToWorldVector(localUp):getNormalized()
        local vehAng = fwd:getAngle()

        if camState.camMode == 2 then
            -- First-Person View (Advanced Car Camera V1.8) - anchored to occupant's seat
            local curTime = timer.curtime()
            local fpv = CAM.fpv or {}
            local fpvOff = isPassenger
                and (CAM.passengerFpvOffset or Vector(0, 0, 28))
                or (fpv.offset or CAM.fpvOffset or Vector(2, 0, 28))
            local rawSeatAng = base.localToWorldAngles and base:localToWorldAngles(localFwd:getAngle()) or vehAng
            local seatBump = fpv.seatBumpSmoothing or 5
            camState.smoothSeatAng = camState.smoothSeatAng and math.lerpAngle(clamp(seatBump * dt, 0, 1), camState.smoothSeatAng, rawSeatAng) or rawSeatAng
            local sAng = camState.smoothSeatAng
            camState.trackAng = camState.trackAng and math.lerpAngle(clamp(1.5 * dt, 0, 1), camState.trackAng, sAng) or sAng

            local anchorPos = seatEnt:getPos()
            local camPos = anchorPos + fwd * fpvOff.x - right * fpvOff.y + up * fpvOff.z
            local isLocked = (curTime - camState.lastMouseTime) > (CAM.idleTimeout or 2.0)
            local mph = vel:getLength() * 0.0568

            local camAng
            if isLocked or not camState.targetAng then
                local hLockP = fpv.horizonLockPitch or 0.5
                local hLockR = fpv.horizonLockRoll or 0.7
                local idealP = math.lerp(hLockP, sAng.p, camState.trackAng.p)
                local idealR = math.lerp(hLockR, sAng.r, 0)
                local idealY = sAng.y

                camState.angHistory = camState.angHistory or {}
                camState.velHistory = camState.velHistory or {}
                table.insert(camState.angHistory, { a = sAng, t = curTime })
                table.insert(camState.velHistory, { v = vel, t = curTime })
                while #camState.angHistory > 0 and (curTime - camState.angHistory[1].t) > 0.1 do table.remove(camState.angHistory, 1) end
                while #camState.velHistory > 0 and (curTime - camState.velHistory[1].t) > 0.1 do table.remove(camState.velHistory, 1) end

                local rawYawRate = 0
                local rawAccel = Vector(0, 0, 0)
                if #camState.angHistory > 1 and #camState.velHistory > 1 then
                    local oa, na = camState.angHistory[1], camState.angHistory[#camState.angHistory]
                    local tda = na.t - oa.t
                    if tda > 0.01 then rawYawRate = math.normalizeAngle(na.a.y - oa.a.y) / tda end
                    local ov, nv = camState.velHistory[1], camState.velHistory[#camState.velHistory]
                    local tdv = nv.t - ov.t
                    if tdv > 0.01 then rawAccel = (nv.v - ov.v) / tdv end
                end

                local yawOffset = 0
                local maxApex = fpv.apexMaxYaw or 65
                if not isPassenger and fpv.apexLookSteer == true and hudSt then
                    local tgtSteer = (hudSt.steer or 0) * (fpv.apexSteerFactor or 12)
                    local recov = math.abs(camState.smoothSteerOff or 0) > math.abs(tgtSteer) + 0.1
                    local damp = clamp((recov and 10 or (fpv.headSpringSpeed or 8)) * dt, 0, 1)
                    camState.smoothSteerOff = math.lerp(damp, camState.smoothSteerOff or 0, tgtSteer)
                    yawOffset = clamp(camState.smoothSteerOff, -maxApex, maxApex)
                elseif fpv.apexLookYaw ~= false then
                    local recov = math.abs(camState.smoothYawRate or 0) > math.abs(rawYawRate) + 0.5
                    local damp = clamp((recov and 6 or (fpv.physicsDamping or 1.8)) * dt, 0, 1)
                    camState.smoothYawRate = math.lerp(damp, camState.smoothYawRate or 0, rawYawRate)
                    yawOffset = clamp(camState.smoothYawRate * (fpv.apexYawFactor or 0.45), -maxApex, maxApex)
                end
                idealY = idealY + yawOffset

                if fpv.enableGForces ~= false then
                    local fAcc = clamp(rawAccel:dot(fwd), -1500, 1500)
                    local rAcc = clamp(rawAccel:dot(right), -1500, 1500)
                    local gDamp = clamp((fpv.physicsDamping or 1.8) * dt, 0, 1)
                    camState.smoothFA = math.lerp(gDamp, camState.smoothFA or 0, fAcc)
                    camState.smoothRA = math.lerp(gDamp, camState.smoothRA or 0, rAcc)
                    idealP = idealP - camState.smoothFA * (fpv.gForcePitch or 0.009)
                    idealR = idealR - camState.smoothRA * (fpv.gForceRoll or 0.007)
                    camPos = camPos + fwd * clamp(-camState.smoothFA * 0.005, -1.5, 2.5) + right * clamp(-camState.smoothRA * 0.0025, -1.0, 1.0)
                end

                local idealAng = Angle(idealP + (fpv.pitchOffset or 0), idealY, idealR)
                local headLerp = clamp((fpv.headSpringSpeed or 8) * dt, 0, 1)
                camState.currentAng = math.lerpAngle(headLerp, camState.currentAng or sAng, idealAng)
                camState.targetAng = Angle(camState.currentAng.p, camState.currentAng.y, camState.currentAng.r)
                camAng = camState.currentAng
            else
                local seatYawDelta = math.normalizeAngle(sAng.y - (camState.lastFrameSeatYaw or sAng.y))
                camState.targetAng.y = camState.targetAng.y + seatYawDelta
                local idealRoll = math.lerp(fpv.horizonLockRoll or 0.7, sAng.r, 0)
                camState.targetAng.r = math.lerp(clamp(5 * dt, 0, 1), camState.targetAng.r, idealRoll)
                camAng = camState.targetAng
                camState.currentAng = camAng
            end
            camState.lastFrameSeatYaw = sAng.y

            local baseFOV = fpv.baseFOV or 90
            local targetFOV = baseFOV
            if fpv.enableFOV ~= false and mph > (fpv.fovSpeedThreshold or 20) then
                targetFOV = math.min(baseFOV + (mph - (fpv.fovSpeedThreshold or 20)) * (fpv.fovScale or 0.18), fpv.maxFOV or 120)
            end
            camState.currentFOV = math.lerp(clamp((fpv.fovSmoothing or 1.5) * dt, 0, 1), camState.currentFOV or baseFOV, targetFOV)

            local viewOut = { origin = camPos, angles = camAng, fov = camState.currentFOV, drawviewer = false }
            if SimcadeAddons and SimcadeAddons._clientHost then
                SimcadeAddons._clientHost.runHook("OnCalcView", viewOut, camState)
            end
            return viewOut
        end

        if camState.camMode == 3 then
            -- 360-degree Orbit Freecam (follows vehicle position without auto-recentering)
            local hOff = (isPassenger and CAM.passengerHeightOffset) or CAM.heightOffset or CAM.chaseHeight or 55
            local pOff = CAM.pitchOffset or 5
            local target = base:getPos() + up * hOff
            local targetDist = clamp(camState.freeDist or CAM.passengerDist or CAM.maxDist or 190, 65, 450)

            if not camState.initialized then
                camState.currentDist = targetDist
                camState.distVel     = 0
                camState.currentAng  = Angle(vehAng.p + pOff, vehAng.y, 0)
                camState.targetAng   = Angle(vehAng.p + pOff, vehAng.y, 0)
                camState.initialized = true
            end

            local finalCamAng = camState.targetAng or Angle(vehAng.p + pOff, vehAng.y, 0)
            finalCamAng.r = 0
            camState.currentAng = finalCamAng
            camState.currentDist = math.lerp(clamp(12 * dt, 0, 1), camState.currentDist or targetDist, targetDist)

            local desiredOrigin = target - (finalCamAng:getForward() * camState.currentDist)
            if CAM.doCollision ~= false then
                local tr = trace.line(target, desiredOrigin, { base, seatEnt, lp }, CAM.colMask or 131083)
                if tr.Hit then
                    desiredOrigin = tr.HitPos + tr.HitNormal * (CAM.colExtra or 8)
                    camState.currentDist = (tr.HitPos - target):getLength()
                end
            end

            return { origin = desiredOrigin, angles = finalCamAng, fov = CAM.chaseFOV or 90, drawviewer = true }
        end

        -- 3rd-Person Chase Camera (Advanced Car Camera V1.8)
        local maxDist = camState.trailerActive and (CAM.trailerDist or 350) or (CAM.maxDist or CAM.chaseDist or 190)
        local minDist = camState.trailerActive and (CAM.trailerMinDist or 300) or (CAM.minDist or math.max(70, maxDist - 30))
        local hOff    = camState.trailerActive and (CAM.trailerHeight or 80) or ((isPassenger and CAM.passengerHeightOffset) or CAM.heightOffset or CAM.chaseHeight or 55)
        local pOff    = camState.trailerActive and (CAM.trailerPitch or 10) or (CAM.pitchOffset or 5)

        local target = base:getPos() + up * hOff
        local isLocked = (timer.curtime() - camState.lastMouseTime) > (CAM.idleTimeout or 2.0)

        if not camState.initialized then
            local initDist = (minDist + maxDist) * 0.5
            camState.currentDist = initDist
            camState.targetDist  = initDist
            camState.distVel     = 0
            camState.lastSpeed   = vel:dot(fwd)
            camState.smoothAccel = 0
            camState.currentAng  = Angle(vehAng.p + pOff, vehAng.y, 0)
            camState.targetAng   = Angle(vehAng.p + pOff, vehAng.y, 0)
            camState.initialized = true
        end

        local finalCamAng
        if isLocked then
            local lookTarget = target
            if CAM.lookAhead ~= false and vel:getLengthSqr() > 2500 then
                lookTarget = target + vel:getNormalized() * (CAM.lookAheadDist or 200)
            end
            local idealAng = (lookTarget - (target - fwd)):getAngle()
            idealAng.p = idealAng.p + pOff
            idealAng.r = 0
            local resp = clamp((CAM.angleResponsiveness or 3.5) * dt, 0, 1)
            camState.currentAng = math.lerpAngle(resp, camState.currentAng or idealAng, idealAng)
            camState.targetAng  = Angle(camState.currentAng.p, camState.currentAng.y, 0)
            finalCamAng = camState.currentAng
        else
            finalCamAng = camState.targetAng or vehAng
            camState.currentAng = finalCamAng
        end

        local fwdSpd = vel:dot(fwd)
        local speedDelta = fwdSpd - (camState.lastSpeed or fwdSpd)
        camState.lastSpeed = fwdSpd
        local instAccel = speedDelta / dt
        camState.smoothAccel = math.lerp(clamp(8 * dt, 0, 1), camState.smoothAccel or 0, instAccel)

        local restDist = (minDist + maxDist) * 0.5
        local halfSpan = math.max(12, (maxDist - minDist) * 0.5)
        local dynamicTarget = restDist + clamp(camState.smoothAccel * (CAM.accelOffsetScale or 0.045), -halfSpan, halfSpan)
        local impulse = clamp(speedDelta * (CAM.accelImpulseScale or 0.18), -12, 12)

        local springForce = (dynamicTarget - camState.currentDist) * (CAM.centerForce or 15)
        local dampingForce = -camState.distVel * (CAM.posDamping or 10)
        local boundaryImpulse = 0
        if camState.currentDist > maxDist then
            boundaryImpulse = (maxDist - camState.currentDist) * (CAM.boundaryForce or 75)
        elseif camState.currentDist < minDist then
            boundaryImpulse = (minDist - camState.currentDist) * (CAM.boundaryForce or 75)
        end
        camState.distVel = camState.distVel + (springForce + dampingForce + boundaryImpulse) * dt + impulse
        camState.currentDist = clamp(camState.currentDist + camState.distVel * dt, 45, 600)

        local desiredOrigin = target - (finalCamAng:getForward() * camState.currentDist)
        if CAM.doCollision ~= false then
            local tr = trace.line(target, desiredOrigin, { base, seatEnt, lp }, CAM.colMask or 131083)
            if tr.Hit then
                desiredOrigin = tr.HitPos + tr.HitNormal * (CAM.colExtra or 8)
                camState.distVel = 0
                camState.currentDist = (tr.HitPos - target):getLength()
            end
        end

        return { origin = desiredOrigin, angles = finalCamAng, fov = CAM.chaseFOV or 90, drawviewer = true }
    end)
end
