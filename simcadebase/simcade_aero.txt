--@name Simcade Aerodynamics Module
--@shared

SimcadeAero = SimcadeAero or {}

local clamp = math.clamp or function(v, a, b)
    if v < a then return a elseif v > b then return b else return v end
end
local max = math.max
local sqrt = math.sqrt

SimcadeAero.frontForceOffset = SimcadeAero.frontForceOffset or { len = 3, [0] = 1, 0, 0 }
SimcadeAero.rearForceOffset = SimcadeAero.rearForceOffset or { len = 3, [0] = -1, 0, 0 }

local function computeWheelLoadKN(aeroCfg, totalMassKG, speedMs, isFront)
    if not aeroCfg then return 0 end
    local factor = isFront and (aeroCfg.FrontAeroFactor or 0) or (aeroCfg.RearAeroFactor or 0)
    if factor <= 0 then return 0 end
    local speedMph = math.abs(speedMs or 0) * 2.23693629
    local mphDiv100 = speedMph / 100.0
    return max(0, factor * (totalMassKG * 0.004905) * (mphDiv100 * mphDiv100))
end

function SimcadeAero.addTireLoadKN(tireLoadKN, aeroCfg, totalMassKG, speedMs, isFront)
    return (tireLoadKN or 0) + computeWheelLoadKN(aeroCfg, totalMassKG, speedMs, isFront)
end

function SimcadeAero.buildSolverCoefficients(aeroCfg, massScale)
    local dragCoeff
    if aeroCfg and (aeroCfg.DragCoeff or aeroCfg.FrontalAreaM2) then
        local cd = max(0.01, aeroCfg.DragCoeff or 0.32)
        local area = max(0.1, aeroCfg.FrontalAreaM2 or 2.15)
        dragCoeff = clamp(0.5 * 1.225 * cd * area * 0.001, 0.0, 0.0025)
    else
        dragCoeff = clamp(0.00034 * (0.85 + 0.15 * massScale), 0.00024, 0.00058)
    end

    return dragCoeff,
        clamp(0.00020 * massScale, 0.00012, 0.00055),
        clamp(0.00024 * massScale, 0.00015, 0.00065)
end

function SimcadeAero.applyForces(chassis)
    local A, c, v, R = chassis.aero, chassis.c, chassis.v, chassis.R
    local speed = sqrt(v[0] * v[0] + v[1] * v[1] + v[2] * v[2])
    local scaledSpeed = speed / chassis.perf
    local drag = -(A.grip * c[152] * scaledSpeed * A.on)
    local armz = 0
    if A.grounded >= 0.5 then armz = (1 - A.T) * -0.10000000149011612 end
    chassis:addForce3(drag * v[0], drag * v[1], drag * v[2], 0, 0, armz)

    local groundFactor = R[8]
    if groundFactor < 0 then groundFactor = 0 end
    if A.grounded >= 0.5 then groundFactor = 1 end
    local n0, n1, n2 = 1, 0, 0
    if scaledSpeed > 0.0001 then
        n0, n1, n2 = v[0] / scaledSpeed, v[1] / scaledSpeed, v[2] / scaledSpeed
    end
    local alignment = n2 * R[2] + n1 * R[1] + n0 * R[0]
    if alignment < 0 then alignment = 0 end
    local alignmentSq = alignment * alignment
    if alignmentSq < 0.4000000059604645 then alignmentSq = 0.4000000059604645 end

    local frontLoad, rearLoad
    if A.grounded == 0 then
        local totalLoad = groundFactor * ((c[154] + c[153]) * alignmentSq) * scaledSpeed
        frontLoad = 0.4000000059604645 * totalLoad
        rearLoad = totalLoad * 0.4099999964237213
    else
        frontLoad = alignmentSq * c[153] * groundFactor * scaledSpeed
        rearLoad = alignmentSq * c[154] * groundFactor * scaledSpeed
    end

    local af = chassis.T.aero_front or SimcadeAero.frontForceOffset
    local ar = chassis.T.aero_rear or SimcadeAero.rearForceOffset
    local force = -(frontLoad * A.on)
    chassis:addForce3(force * R[6], force * R[7], force * R[8],
        af[0] * R[0] + af[1] * R[3] + af[2] * R[6],
        af[0] * R[1] + af[1] * R[4] + af[2] * R[7],
        af[0] * R[2] + af[1] * R[5] + af[2] * R[8])

    force = -(rearLoad * A.on)
    chassis:addForce3(force * R[6], force * R[7], force * R[8],
        ar[0] * R[0] + ar[1] * R[3] + ar[2] * R[6],
        ar[0] * R[1] + ar[1] * R[4] + ar[2] * R[7],
        ar[0] * R[2] + ar[1] * R[5] + ar[2] * R[8])
end

return SimcadeAero
