--@name Simcade Modular Addon, Custom Key & Full-Chip Override System
--@shared
--@include simcadebase/addons/lights.txt

SimcadeAddons = SimcadeAddons or {}

local function safeCall(tag, fn, ...)
    if type(fn) ~= "function" then return false, nil end
    local ok, res = pcall(fn, ...)
    if not ok then
        SimcadeAddons._loggedErrors = SimcadeAddons._loggedErrors or {}
        if not SimcadeAddons._loggedErrors[tag] then
            SimcadeAddons._loggedErrors[tag] = true
            print("[SimcadeAddons] Error in " .. tostring(tag) .. ": " .. tostring(res))
        end
        return false, nil
    end
    return true, res
end

-- Compatibility for cars using the original LIGHTS_CFG and inline ADDONS format.
function SimcadeAddons.createLightsAddon(cfg)
    return require("simcadebase/addons/lights.txt")(cfg.LIGHTS_CFG or {}, cfg)
end

function SimcadeAddons.collectAddons(cfg)
    local list = {}
    cfg = cfg or {}
    if cfg.ADDON_MODULES then
        for _, spec in ipairs(cfg.ADDON_MODULES) do
            if spec.enabled ~= false then
                local ok, addon = safeCall(tostring(spec.file), function()
                    local create = require(spec.file)
                    return create(spec.options or {}, cfg)
                end)
                if ok and type(addon) == "table" then
                    addon.name = addon.name or spec.file
                    list[#list + 1] = addon
                end
            end
        end
    else
        local lights = SimcadeAddons.createLightsAddon(cfg)
        if lights then list[#list + 1] = lights end
    end

    local userAddons = cfg and cfg.ADDONS
    if type(userAddons) == "table" then
        for k, entry in pairs(userAddons) do
            if type(entry) == "table" and entry.enabled ~= false then
                if not entry.name then entry.name = tostring(k) end
                list[#list + 1] = entry
            end
        end
    end
    return list
end

if CLIENT then
    function SimcadeAddons.initClient(cfg)
        local addons = SimcadeAddons.collectAddons(cfg)
        local keyBindings = {} -- id -> keyCode
        local controlsCfg = (cfg and cfg.CONTROLS_CFG) or {}

        local ctx = {
            config = cfg,
            syncedState = {},
            bindKey = function(id, defaultKey)
                keyBindings[id] = controlsCfg[id] or defaultKey
            end,
            addWireInput = function() end,
            addWireOutput = function() end,
            setWireOutput = function() end,
            getWireInput = function() return nil end,
            override = function() end,
            wrap = function() end,
        }

        for _, a in ipairs(addons) do
            if a.OnClientInit then
                safeCall((a.name or "addon") .. ".OnClientInit", a.OnClientInit, ctx, cfg)
            end
        end

        if next(keyBindings) ~= nil then
            local function isInSeat()
                local lp = player()
                if not (lp and lp:isValid() and lp:inVehicle()) then return false end
                local st = SimcadeHUD and SimcadeHUD.getState and SimcadeHUD.getState() or nil
                if st and st.pod and st.pod:isValid() and lp.getVehicle then
                    return lp:getVehicle() == st.pod
                end
                return true
            end
            local function sendAddonKey(id, down)
                pcall(function()
                    net.start("Simcade_AddonKey")
                    net.writeString(id)
                    net.writeBool(down and true or false)
                    net.send()
                end)
            end
            hook.add("inputPressed", "SimcadeAddons_KeyDown", function(btn)
                if not isInSeat() then return end
                for id, kCode in pairs(keyBindings) do
                    if btn == kCode then sendAddonKey(id, true) end
                end
            end)
            hook.add("inputReleased", "SimcadeAddons_KeyUp", function(btn)
                if not isInSeat() then return end
                for id, kCode in pairs(keyBindings) do
                    if btn == kCode then sendAddonKey(id, false) end
                end
            end)
        end

        net.receive("Simcade_AddonState", function()
            ctx.syncedState = net.readTable()
        end)

        local clientHost = {
            addons = addons,
            ctx = ctx,
            runHook = function(hookName, ...)
                for _, a in ipairs(addons) do
                    if a[hookName] then
                        safeCall((a.name or "addon") .. "." .. hookName, a[hookName], ctx, ...)
                    end
                end
            end,
        }
        net.receive("Simcade_AddonEvent", function()
            local name = net.readString()
            local data = net.readTable()
            clientHost.runHook("OnClientEvent", name, data)
        end)
        hook.add("removed", "Simcade_ClientAddonCleanup", function()
            clientHost.runHook("OnCleanup")
        end)
        SimcadeAddons._clientHost = clientHost
        return clientHost
    end
    return
end

-- SERVER SIDE
function SimcadeAddons.initServer(cfg, notifyDriverFn)
    local addons = SimcadeAddons.collectAddons(cfg)
    local extraInputs = {}
    local extraOutputs = {}
    local inputHandlers = {}
    local keyHandlers = {}
    local wireValues = {}
    local overrides = {}
    local wraps = {}
    local carMethodOrigs = {}
    local pendingEvents = {}

    local ctx = {
        config = cfg,
        syncedState = {},
        base = nil,
        pod = nil,
        phys = nil,
        car = nil,
        wheels = nil,
        driver = nil,
        state = {},
        pipeline = {},
        notifyDriver = notifyDriverFn,
        emitEvent = function(name, data)
            pendingEvents[name] = { data = data, time = timer.curtime() }
        end,
        bindKey = function(id, defaultKey, onPress, onRelease)
            keyHandlers[id] = { key = (cfg.CONTROLS_CFG and cfg.CONTROLS_CFG[id]) or defaultKey, onPress = onPress, onRelease = onRelease }
        end,
        addWireInput = function(name, portType, onChange)
            extraInputs[#extraInputs + 1] = { name = name, type = portType or "Number" }
            if onChange then inputHandlers[name] = onChange end
        end,
        addWireOutput = function(name, portType)
            extraOutputs[#extraOutputs + 1] = { name = name, type = portType or "Number" }
        end,
        setWireOutput = function(name, val)
            if wire and wire.ports then
                wire.ports[name] = val
            end
        end,
        getWireInput = function(name)
            return wireValues[name]
        end,
        override = function(target, fn)
            if type(fn) == "function" then overrides[target] = fn end
        end,
        wrap = function(target, fn)
            if type(fn) == "function" then
                wraps[target] = wraps[target] or {}
                wraps[target][#wraps[target] + 1] = fn
            end
        end,
    }

    for _, a in ipairs(addons) do
        if a.OnInit then
            safeCall((a.name or "addon") .. ".OnInit", a.OnInit, ctx, cfg)
        end
    end

    net.receive("Simcade_AddonKey", function(_, ply)
        if not (ctx.car and isValid(ctx.pod)) then return end
        local drv = ctx.pod:getDriver()
        if not isValid(drv) or ply ~= drv then return end
        local id = net.readString()
        local down = net.readBool()
        local h = keyHandlers[id]
        if not h then return end
        if down and h.onPress then
            safeCall("AddonKey.onPress." .. id, h.onPress, ctx, drv)
        elseif (not down) and h.onRelease then
            safeCall("AddonKey.onRelease." .. id, h.onRelease, ctx, drv)
        end
    end)

    local function buildWrapped(target, origFn)
        local fn = overrides[target] or origFn
        local wList = wraps[target]
        if wList then
            for i = 1, #wList do
                local wrapper = wList[i]
                local prev = fn
                fn = function(...)
                    return wrapper(prev, ...)
                end
            end
        end
        return fn
    end

    local CAR_METHODS = {
        "stepVPhysics", "driver", "gearbox", "auto", "clutchDrive", "torque", "drive",
        "tire", "ellipse", "aeroForces", "computeActiveDiff", "afterSolve",
        "postEngine", "postDrive", "shiftKey", "setManual", "toggleDiffLock",
    }

    local MODULE_TARGETS = {
        { tbl = SimcadeSuspension, key = "computeWheelForce" },
        { tbl = SimcadeSuspension, key = "applyCurbBlock" },
        { tbl = SimcadeEngine,     key = "synthesizeSpec" },
        { tbl = SimcadeEngine,     key = "stepIgnition" },
        { tbl = SimcadeControls,   key = "pollInputs" },
        { tbl = SimcadeGearbox,    key = "evaluateAutoShift" },
        { tbl = SimcadeWire,       key = "writeOutputs" },
    }
    local moduleOrigs = {}

    return {
        ctx = ctx,
        addons = addons,
        pendingEvents = pendingEvents,
        extraInputs = extraInputs,
        extraOutputs = extraOutputs,
        onWireInput = function(name, value)
            wireValues[name] = value
            local cb = inputHandlers[name]
            if cb then safeCall("WireInput." .. name, cb, ctx, value) end
        end,
        applyOverrides = function(car, pipeline)
            if pipeline then
                for name, origFn in pairs(pipeline) do
                    if overrides[name] or wraps[name] then
                        pipeline[name] = buildWrapped(name, origFn)
                    end
                end
            end
            if car then
                for _, mName in ipairs(CAR_METHODS) do
                    if (overrides[mName] or wraps[mName]) and type(car[mName]) == "function" then
                        carMethodOrigs[mName] = carMethodOrigs[mName] or car[mName]
                        car[mName] = buildWrapped(mName, carMethodOrigs[mName])
                    end
                end
            end
            for _, mt in ipairs(MODULE_TARGETS) do
                local tbl, k = mt.tbl, mt.key
                if tbl and type(tbl[k]) == "function" and (overrides[k] or wraps[k]) then
                    moduleOrigs[k] = moduleOrigs[k] or tbl[k]
                    tbl[k] = buildWrapped(k, moduleOrigs[k])
                end
            end
        end,
        runHook = function(hookName, ...)
            for _, a in ipairs(addons) do
                if a[hookName] then
                    safeCall((a.name or "addon") .. "." .. hookName, a[hookName], ctx, ...)
                end
            end
        end,
    }
end
