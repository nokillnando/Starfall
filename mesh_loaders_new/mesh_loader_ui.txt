-- Global Font Cache (Prevents "You have reached the max created fonts!" error)
_G.MeshLoaderFontCache = _G.MeshLoaderFontCache or {}
local function getSafeFont(fontName, size, weight, antialias, fallback)
    local key = string.format("%s_%d_%d_%s", fontName, size or 14, weight or 400, antialias and "1" or "0")
    if _G.MeshLoaderFontCache[key] then
        return _G.MeshLoaderFontCache[key]
    end
    
    local ok, res = pcall(function()
        return render.createFont(fontName, size, weight, antialias)
    end)
    
    if ok and res then
        _G.MeshLoaderFontCache[key] = res
        return res
    end
    
    local defFont = fallback or "DermaDefault"
    if size and size >= 48 then
        defFont = "CloseCaption_Bold"
    elseif size and size >= 24 then
        defFont = "Trebuchet24"
    elseif size and size >= 18 then
        defFont = "Trebuchet18"
    elseif size and size >= 14 then
        defFont = "DermaDefaultBold"
    else
        defFont = "DermaDefault"
    end
    _G.MeshLoaderFontCache[key] = defFont
    return defFont
end

local function parseColorString(str)
    if not str or str == "" then return nil end
    local s = str:gsub("^%s+", ""):gsub("%s+$", "")
    
    -- Hex: #RRGGBB or #RRGGBBAA or RRGGBB
    if s:sub(1, 1) == "#" or (#s >= 6 and s:match("^%x+$")) then
        local hex = s:sub(1, 1) == "#" and s:sub(2) or s
        if #hex == 3 then
            local r = tonumber(hex:sub(1,1) .. hex:sub(1,1), 16)
            local g = tonumber(hex:sub(2,2) .. hex:sub(2,2), 16)
            local b = tonumber(hex:sub(3,3) .. hex:sub(3,3), 16)
            if r and g and b then return Color(r, g, b, 255) end
        elseif #hex >= 6 then
            local r = tonumber(hex:sub(1, 2), 16)
            local g = tonumber(hex:sub(3, 4), 16)
            local b = tonumber(hex:sub(5, 6), 16)
            local a = #hex >= 8 and tonumber(hex:sub(7, 8), 16) or 255
            if r and g and b then return Color(r, g, b, a or 255) end
        end
    end
    
    -- RGB / RGBA (comma or space separated)
    local parts = {}
    for part in s:gmatch("[^,%s]+") do
        local n = tonumber(part)
        if n then table.insert(parts, n) end
    end
    if #parts >= 3 then
        local r = parts[1] <= 1.0 and math.floor(parts[1] * 255) or math.floor(parts[1])
        local g = parts[2] <= 1.0 and math.floor(parts[2] * 255) or math.floor(parts[2])
        local b = parts[3] <= 1.0 and math.floor(parts[3] * 255) or math.floor(parts[3])
        local a = parts[4] and (parts[4] <= 1.0 and math.floor(parts[4] * 255) or math.floor(parts[4])) or 255
        return Color(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255), math.clamp(a, 0, 255))
    end
    return nil
end

local function toVector(val, fallback)
    if not val then return fallback or Vector(0, 0, 0) end
    if type(val) == "Vector" then return val end
    if type(val) == "table" then
        return Vector(val[1] or val.x or 0, val[2] or val.y or 0, val[3] or val.z or 0)
    end
    return fallback or Vector(0, 0, 0)
end

        if not isValid(ent) then return nil end
        local curr = ent
        local visited = {}
        while isValid(curr) and not visited[curr] do
            visited[curr] = true
            local p = curr:getParent()
            if isValid(p) then
                curr = p
            else
                break
            end
        end
        return curr
    end

    local function belongsToVehicle(ent, entIdx)
        if not isValid(ent) and not entIdx then return false end
        if isValid(ent) and vehicleEntities and vehicleEntities[ent] then return true end
        local idx = entIdx or (isValid(ent) and ent:entIndex())
        if idx and vehicleEntityIndices and vehicleEntityIndices[idx] then return true end
        if isValid(ent) and isValid(base) and ent == base then return true end
        if isValid(ent) and ent == chip() then return true end
        if isValid(ent) and isValid(base) and isValid(ent:getParent()) and ent:getParent() == base then return true end
        if isValid(ent) and isValid(chip():getParent()) and ent == chip():getParent() then return true end
        local myRoot = (isValid(base) and getRootEntity(base)) or (isValid(chip()) and getRootEntity(chip()))
        local entRoot = isValid(ent) and getRootEntity(ent)
        if isValid(myRoot) and isValid(entRoot) and myRoot == entRoot then return true end
        return false
    end

    local function isModuleOnSameVehicle(pId, ent)
        if not pId or pId == currentChipId then return true end
        
        if isValid(ent) and (vehicleEntities[ent] or (ent.entIndex and vehicleEntityIndices[ent:entIndex()])) then
            return true
        end
        if pId and vehicleEntityIndices[pId] then
            return true
        end
        
        local netMod = discoveredNetworkModules[pId]
        if netMod and currentVehicleId and netMod.vehicleId == currentVehicleId and netMod.vehicleId > 0 then
            return true
        end
        
        local otherReg = _G.MeshLoaderRegistry and _G.MeshLoaderRegistry[pId]
        if otherReg and currentVehicleId and otherReg.vehicleId == currentVehicleId and otherReg.vehicleId > 0 then
            return true
        end

        local otherChip = (otherReg and isValid(otherReg.chip)) and otherReg.chip or (isValid(ent) and ent or nil)
        local otherBase = (otherReg and isValid(otherReg.base)) and otherReg.base or otherChip

        local myBase = isValid(base) and base or chip()
        if isValid(myBase) and isValid(otherChip) then
            if otherChip == myBase or otherBase == myBase then return true end
            if isValid(chip()) and (otherChip == chip() or otherBase == chip()) then return true end
            if isValid(otherChip:getParent()) and otherChip:getParent() == myBase then return true end
            if isValid(otherBase) and isValid(otherBase:getParent()) and otherBase:getParent() == myBase then return true end
            
            local myRoot = getRootEntity(myBase) or myBase
            local otherRoot = getRootEntity(otherBase) or otherBase
            if isValid(myRoot) and isValid(otherRoot) and myRoot == otherRoot then return true end
        end
        
        if isValid(otherChip) and otherChip.getAllConstrained then
            local cons = otherChip:getAllConstrained()
            if type(cons) == "table" then
                for _, c in ipairs(cons) do
                    if c == myBase or c == chip() or vehicleEntities[c] then return true end
                end
            end
        end
        
        return false
    end

    local function cleanupDeadModuleFiles()
        pcall(function()
            local files = (file.find and (file.find("mesh_presets/module_*.txt") or file.find("mesh_presets/*.txt"))) or {}
            for _, fname in ipairs(files) do
                if fname:sub(1, 7) == "module_" and fname:sub(-4) == ".txt" then
                    local idStr = fname:sub(8, -5)
                    local fId = tonumber(idStr)
                    local shouldDelete = false
                    if not fId or fId <= 0 then
                        shouldDelete = true
                    elseif fId ~= currentChipId then
                        local ent = entity and entity(fId)
                        if not isValid(ent) then
                            shouldDelete = true
                        else
                            local okClass, cls = pcall(function() return ent:getClass() end)
                            if okClass and cls ~= "starfall_processor" then
                                shouldDelete = true
                            end
                        end
                    end
                    if shouldDelete then
                        pcall(function() file.delete("mesh_presets/" .. fname) end)
                    end
                end
            end
        end)
    end

    local lastDiskScanTime = 0
    local function getDiscoveredModules()
        local now = timer.curtime()
        if cachedDiscoveredMods and (now - lastModScanTime < 1.0) then
            return cachedDiscoveredMods
        end
        lastModScanTime = now

        local list = {}
        local seenIds = {}
        local seenNames = {}
        
        local function tryAddModule(fId, fName, fEnt)
            if not fId or fId <= 0 or seenIds[fId] then return end
            if not isModuleOnSameVehicle(fId, fEnt) then return end
            
            local nameKey = (fName or ("Module #" .. fId)):lower()
            if seenNames[nameKey] then return end
            
            seenIds[fId] = true
            seenNames[nameKey] = true
            table.insert(list, {
                id = fId,
                name = fName or ("Module #" .. fId),
                isSelf = (fId == currentChipId)
            })
        end

        if isBoothEnabled then
            tryAddModule(currentChipId, myModuleName, chip())
        end
        
        if _G.MeshLoaderRegistry then
            for rId, rData in pairs(_G.MeshLoaderRegistry) do
                if rData and (rData.isBoothEnabled ~= false) then
                    tryAddModule(rData.id or rId, rData.name, rData.chip)
                end
            end
        end

        if discoveredNetworkModules then
            for nId, nData in pairs(discoveredNetworkModules) do
                if nData then
                    tryAddModule(nData.id or nId, nData.name, nData.ent)
                end
            end
        end

        -- Low-frequency disk fallback (maximum once every 3 seconds, never per-frame)
        if (now - lastDiskScanTime >= 3.0) then
            lastDiskScanTime = now
            pcall(function()
                local files = (file.find and (file.find("mesh_presets/module_*.txt") or file.find("mesh_presets/*.txt"))) or {}
                for _, fname in ipairs(files) do
                    if fname:sub(1, 7) == "module_" and fname:sub(-4) == ".txt" then
                        local raw = file.read("mesh_presets/" .. fname)
                        if raw and raw ~= "" then
                            local parts = string.split(raw, ":")
                            local fId = tonumber(parts[1])
                            local fName = parts[2] or ("Module #" .. tostring(fId))
                            local ent = fId and entity and entity(fId)
                            if fId and fId > 0 then
                                if isValid(ent) then
                                    if not seenIds[fId] then
                                        tryAddModule(fId, fName, ent)
                                    end
                                else
                                    pcall(function() file.delete("mesh_presets/" .. fname) end)
                                end
                            else
                                pcall(function() file.delete("mesh_presets/" .. fname) end)
                            end
                        end
                    end
                end
            end)
        end
        
        table.sort(list, function(a, b) return a.id < b.id end)
        cachedDiscoveredMods = list
        return list
    end
    pcall(timer.simple, 0.5, cleanupDeadModuleFiles)
    pcall(timer.simple, 2.0, cleanupDeadModuleFiles)

    local cachedVehicleGeo = nil
    local lastGeoCalcTime = 0
    local function getVehicleVisualGeometry(force)
        local now = timer.curtime()
        if not force and cachedVehicleGeo and (now - lastGeoCalcTime < 0.3) then
            return cachedVehicleGeo
        end
        lastGeoCalcTime = now

        local rootBase = (isValid(base) and getRootEntity(base)) or (isValid(chip()) and getRootEntity(chip())) or (isValid(base) and base) or chip()
        local upDir = Vector(0, 0, 1)
        
        local allHolos = {}
        for _, h in ipairs(spawnedHolograms or {}) do
            if isValid(h) then table.insert(allHolos, h) end
        end
        for _, reg in pairs(_G.MeshLoaderRegistry or {}) do
            if reg.spawnedHolograms and reg.id ~= currentChipId then
                for _, h in ipairs(reg.spawnedHolograms) do
                    if isValid(h) then table.insert(allHolos, h) end
                end
            end
        end
        
        local originRef = (isValid(rootBase) and rootBase:getPos()) or (isValid(chip()) and chip():getPos()) or Vector(0, 0, 0)
        local bFwd = isValid(rootBase) and rootBase:getForward() or Vector(1, 0, 0)
        local bRight = isValid(rootBase) and rootBase:getRight() or Vector(0, 1, 0)
        local hFwd = bFwd - upDir * bFwd:dot(upDir)
        if hFwd:getLength() > 0.01 then hFwd = hFwd:getNormalized() else hFwd = Vector(1, 0, 0) end
        local hRight = bRight - upDir * bRight:dot(upDir)
        if hRight:getLength() > 0.01 then hRight = hRight:getNormalized() else hRight = Vector(0, 1, 0) end
        
        local testMinFwd, testMaxFwd = 999999, -999999
        local testMinRight, testMaxRight = 999999, -999999
        for _, holo in ipairs(allHolos) do
            local wPos = holo:getPos()
            local dF = (wPos - originRef):dot(hFwd)
            local dR = (wPos - originRef):dot(hRight)
            if dF < testMinFwd then testMinFwd = dF end
            if dF > testMaxFwd then testMaxFwd = dF end
            if dR < testMinRight then testMinRight = dR end
            if dR > testMaxRight then testMaxRight = dR end
        end
        local spanFwd = testMaxFwd - testMinFwd
        local spanRight = testMaxRight - testMinRight
        
        local fwdDir = hFwd
        local rightDir = hRight
        if spanRight > spanFwd * 1.15 then
            fwdDir = hRight
            rightDir = -hFwd
        end
        
        local minFwd, maxFwd = 999999, -999999
        local minRight, maxRight = 999999, -999999
        local minUp, maxUp = 999999, -999999
        local totalCount = 0
        
        for _, holo in ipairs(allHolos) do
            local cfg = holo.MeshConfig or {}
            local rawPart = cfg.Part or cfg.part
            local isVis = holo:getScale():getLength() > 0.01
            if isVis or not rawPart then
                local wPos = holo:getPos()
                if cfg.meshBounds and cfg.meshBounds.center then
                    local bc = toVector(cfg.meshBounds.center, Vector(0, 0, 0))
                    local s = holo:getScale()
                    wPos = holo:localToWorld(Vector(bc.x * s.x, bc.y * s.y, bc.z * s.z))
                end
                
                local rel = wPos - originRef
                local dFwd = rel:dot(fwdDir)
                local dRight = rel:dot(rightDir)
                local dUp = rel:dot(upDir)
                
                if dFwd < minFwd then minFwd = dFwd end
                if dFwd > maxFwd then maxFwd = dFwd end
                if dRight < minRight then minRight = dRight end
                if dRight > maxRight then maxRight = dRight end
                if dUp < minUp then minUp = dUp end
                if dUp > maxUp then maxUp = dUp end
                totalCount = totalCount + 1
            end
        end
        
        if totalCount == 0 then
            minFwd, maxFwd = -60, 60
            minRight, maxRight = -30, 30
            minUp, maxUp = 0, 40
        end
        
        local lengthSpan = math.max(20, maxFwd - minFwd)
        local widthSpan = math.max(20, maxRight - minRight)
        local heightSpan = math.max(10, maxUp - minUp)
        
        local centerFwd = (minFwd + maxFwd) * 0.5
        local centerRight = (minRight + maxRight) * 0.5
        local centerUp = (minUp + maxUp) * 0.5
        local worldCenter = originRef + fwdDir * centerFwd + rightDir * centerRight + upDir * centerUp
        local worldRoof = originRef + fwdDir * centerFwd + rightDir * centerRight + upDir * maxUp
        
        local result = {
            center = worldCenter,
            roof = worldRoof,
            rootBase = rootBase,
            up = upDir,
            forward = fwdDir,
            right = rightDir,
            length = lengthSpan,
            width = widthSpan,
            height = heightSpan,
            count = totalCount
        }
        cachedVehicleGeo = result
        return result
    end

    local function getWheelWorldPos(wheelKey)
        local wLower = tostring(wheelKey):lower()

    local function getPartMeshWorldPos(pName)
        if not pName then return nil end
        local pLower = pName:lower()
        local baseEnt = (isValid(base) and base) or (resolveParent and resolveParent("Base")) or chip()
        
        for _, holo in ipairs(spawnedHolograms or {}) do
            if isValid(holo) and holo.MeshConfig then
                local rawPart = holo.MeshConfig.Part or holo.MeshConfig.part
                if rawPart and rawPart:lower() == pLower and holo:getScale():getLength() > 0.01 then
                    local cfg = holo.MeshConfig
                    if cfg.meshBounds and cfg.meshBounds.center then
                        local bc = toVector(cfg.meshBounds.center, Vector(0, 0, 0))
                        local s = holo:getScale()
                        return holo:localToWorld(Vector(bc.x * s.x, bc.y * s.y, bc.z * s.z))
                    else
                        return holo:getPos()
                    end
                end
            end
        end
        
        local bestMesh = nil
        local bestBounds = nil
        local targetParent = baseEnt
        
        for _, v in ipairs(activeConfig.meshes or {}) do
            local rawPart = v.Part or v.part
            if rawPart and rawPart:lower() == pLower then
                if v.meshBounds and v.meshBounds.center then
                    bestMesh = v
                    bestBounds = v.meshBounds
                    targetParent = resolveParent(v.Parent) or baseEnt
                    break
                elseif not bestMesh then
                    bestMesh = v
                    targetParent = resolveParent(v.Parent) or baseEnt
                end
            end
        end
        
        if bestMesh then
            local rawPos = bestMesh.offsetPos or (activeConfig and activeConfig.offsetPos) or Vector(0, 0, 0)
            local rawScale = bestMesh.scale or Vector(1, 1, 1)
            local finalPos = toVector(rawPos, Vector(0, 0, 0))
            local subScale = toVector(rawScale, Vector(1, 1, 1))
            local gScale = toVector(activeConfig and activeConfig.globalScale, Vector(1, 1, 1))
            local combinedScale = Vector(subScale.x * gScale.x, subScale.y * gScale.y, subScale.z * gScale.z)
            
            if bestBounds and bestBounds.center then
                local c = toVector(bestBounds.center, Vector(0, 0, 0))
                local localCenter = Vector(c.x * combinedScale.x, c.y * combinedScale.y, c.z * combinedScale.z) + finalPos
                if isValid(targetParent) then
                    return targetParent:localToWorld(localCenter)
                end
            else
                if isValid(targetParent) then
                    return targetParent:localToWorld(finalPos)
                end
            end
        end
        return nil
    end

    local function focusWorkshopCamera(targetSlot, targetData, instantReset)
        local geo = getVehicleVisualGeometry()
        local rootBase = geo.rootBase
        if not isValid(rootBase) then return end
        local cs = getSharedCamState()
        cs.focusedSlot = targetSlot
        cs.isOrbiting = false
        cs.isPanning = false
        
        local vUp = geo.up
        local vFwd = geo.forward
        local vRight = geo.right
        
        if targetSlot == "overview" or targetSlot == "paint" then
            cs.targetPivot = rootBase:worldToLocal(geo.center)
            local camDist = math.clamp(geo.length * 1.55 + 50, 180, 500)
            local camOffset = (vFwd * 0.68 + vRight * 0.56 + vUp * 0.48):getNormalized() * camDist
            local camWorldPos = geo.center + camOffset
            local lookDir = (geo.center - camWorldPos):getNormalized()
            cs.targetAngle = rootBase:worldToLocalAngles(lookDir:getAngle())
            cs.targetDist = camDist
            if instantReset then
                cs.currentPivot = cs.targetPivot
                cs.currentAngle = cs.targetAngle
                cs.currentDist = camDist
            end
        elseif targetSlot == "part" and targetData then
            local partWorldPos = getPartMeshWorldPos(tostring(targetData)) or (geo.center + vFwd * 40)
            cs.targetPivot = rootBase:worldToLocal(partWorldPos)
            
            local dirFromCenter = partWorldPos - geo.center
            local horizDir = dirFromCenter - vUp * dirFromCenter:dot(vUp)
            local outDir = (horizDir:getLength() > 8) and horizDir:getNormalized() or vFwd
            local camDist = math.clamp(geo.length * 0.75 + 40, 120, 260)
            local camOffset = (outDir * 0.82 + vUp * 0.44):getNormalized() * camDist
            local camWorldPos = partWorldPos + camOffset
            local lookDir = (partWorldPos - camWorldPos):getNormalized()
            cs.targetAngle = rootBase:worldToLocalAngles(lookDir:getAngle())
            cs.targetDist = camDist
        elseif targetSlot == "wheel" then
            local wWorldPos = getWheelWorldPos(targetData) or geo.center
            cs.targetPivot = rootBase:worldToLocal(wWorldPos)
            
            local dirFromCenter = wWorldPos - geo.center
            local sideDot = dirFromCenter:dot(vRight)
            local fwdDot = dirFromCenter:dot(vFwd)
            local outSide = (sideDot >= 0) and vRight or -vRight
            local outFwd = (fwdDot >= 0) and (vFwd * 0.35) or (-vFwd * 0.35)
            local camDist = math.clamp(geo.width * 0.85 + 30, 70, 140)
            local camOffset = (outSide + outFwd):getNormalized() * camDist + vUp * 24
            local camWorldPos = wWorldPos + camOffset
            local lookDir = (wWorldPos - camWorldPos):getNormalized()
            cs.targetAngle = rootBase:worldToLocalAngles(lookDir:getAngle())
            cs.targetDist = camDist
        elseif targetSlot == "swaps" then
            local baseEnt = (isValid(base) and base) or (resolveParent and resolveParent("Base")) or chip()
            local swapWorldPos = (geo and geo.roof and (geo.roof + geo.right * 20 + geo.up * 6)) or (isValid(baseEnt) and baseEnt:localToWorld(Vector(0, 20, 48))) or geo.center
            cs.targetPivot = rootBase:worldToLocal(swapWorldPos)
            
            local camDist = math.clamp(geo.length * 1.55 + 50, 180, 500)
            local camOffset = (vFwd * 0.68 + vRight * 0.56 + vUp * 0.48):getNormalized() * camDist
            local camWorldPos = geo.center + camOffset
            local lookDir = (geo.center - camWorldPos):getNormalized()
            cs.targetAngle = rootBase:worldToLocalAngles(lookDir:getAngle())
            cs.targetDist = camDist
        end
    end

    -- Camera View Hook for Workshop Mode
    hook.add("calcview", "MeshLoader_WorkshopCamera_" .. currentChipId, function(pos, angles, fov)
        if not _G.MeshLoaderBoothOpen or not isBoothActiveOnMe then return end
        if _G.MeshLoaderBoothViewMode ~= "workshop" then return end
        
        local rootBase = (isValid(base) and getRootEntity(base)) or (isValid(chip()) and getRootEntity(chip())) or (isValid(base) and base) or chip()
        if not isValid(rootBase) then return end
        
        local dt = timer.frametime()
        if dt <= 0 or dt > 0.1 then dt = 0.016 end
        
        local cs = getSharedCamState()
        cs.currentPivot = cs.currentPivot or Vector(0, 0, 28)
        cs.targetPivot = cs.targetPivot or Vector(0, 0, 28)
        cs.currentAngle = cs.currentAngle or Angle(16, 45, 0)
        cs.targetAngle = cs.targetAngle or Angle(16, 45, 0)
        
        if cs.turntable and not cs.isOrbiting then
            cs.targetAngle = Angle(cs.targetAngle.p, (cs.targetAngle.y + dt * 18) % 360, cs.targetAngle.r)
        end
        
        local lerpRate = math.clamp(dt * 10, 0, 1)
        cs.currentPivot = math.lerpVector(lerpRate, cs.currentPivot, cs.targetPivot)
        cs.currentDist = math.lerp(lerpRate, cs.currentDist or 320, cs.targetDist or 320)
        
        local curAng = cs.currentAngle
        local tarAng = cs.targetAngle
        local diffP = math.normalizeAngle(tarAng.p - curAng.p)
        local diffY = math.normalizeAngle(tarAng.y - curAng.y)
        local diffR = math.normalizeAngle(tarAng.r - curAng.r)
        cs.currentAngle = Angle(curAng.p + diffP * lerpRate, curAng.y + diffY * lerpRate, curAng.r + diffR * lerpRate)
        
        local worldPivot = rootBase:localToWorld(cs.currentPivot)
        local worldAng = rootBase:localToWorldAngles(cs.currentAngle)
        local camPos = worldPivot - worldAng:getForward() * (cs.currentDist or 320)
        
        local tr = trace.line(worldPivot, camPos, { rootBase, base, chip(), player() })
        if tr and tr.hit and tr.fraction < 0.98 then
            camPos = tr.hitPos + tr.hitNormal * 4
        end
        
        local lookAng = (worldPivot - camPos):getAngle()
        return {
            origin = camPos,
            angles = lookAng,
            fov = cs.fov or 75,
            drawviewer = true
        }
    end)

    local function inRect(rx, ry, rw, rh)
        if not rx or not ry or not rw or not rh then return false end
        local mx, my = input.getCursorPos()
        if not mx or not my then return false end
        return mx >= rx and mx <= rx + rw and my >= ry and my <= ry + rh
    end

    local function isCursorOverBoothUI(mx, my)
        if not mx or not my then return false end
        local scrW, scrH = render.getResolution()
        if not scrH or scrH == 0 then scrH = 1080 end
        if not scrW or scrW == 0 then scrW = 1920 end
        local hudScale = math.clamp(scrH / 1080, 0.75, 1.75)
        
        local isWorkshop = (_G.MeshLoaderBoothViewMode == "workshop")
        if isWorkshop then
            local sideW = math.floor(360 * hudScale)
            local sideX = 24
            local sideY = 68
            local sideH = scrH - 120
            
            -- Sidebar bounds
            if inRect(sideX, sideY, sideW, sideH) then return true end
            -- Top navigation, module switcher, and header pills
            if my <= 68 then return true end
            -- Bottom controls hint bar
            if my >= scrH - 45 then return true end
            return false
        else

    -- UNIFIED BOOTH INPUT / CONTROLS HOOK
    -- =========================================================================
    hook.add("inputPressed", "MeshLoader_UnifiedBoothInputPressed_" .. currentChipId, function(key)
        if not isValid(player()) then return end
        if not _G.MeshLoaderBoothOpen or not isBoothActiveOnMe then return end
        
        local activeReg = _G.MeshLoaderRegistry and _G.MeshLoaderRegistry[currentChipId] or ctx
        
        -- ESCAPE KEY (Key 70 or 27 or KEY.ESCAPE)
        if key == 70 or key == 27 or (KEY and key == KEY.ESCAPE) then
            _G.MeshLoaderBoothOpen = false
            _G.MeshLoaderActiveBoothChip = nil
            input.enableCursor(false)
            setPresetInputActive(false)
            pcall(function()
                file.write("mesh_presets/active_booth.txt", "CLOSED:" .. tostring(timer.realtime()))
            end)
            return
        end

        -- 'R' KEY: RESET CAMERA (Key 28 or KEY.R)
        if (key == 28 or (KEY and key == KEY.R)) and not boothHexInputActive and not boothPresetInputActive and not boothSliderInputId then
            focusWorkshopCamera("overview")
            return
        end

        -- KEYBOARD INPUT FOR SLIDER VALUE
        if boothSliderInputId and (key < 107) then
            if key == 66 or (KEY and key == KEY.BACKSPACE) then
                boothSliderInputVal = boothSliderInputVal:sub(1, -2)
                return
            elseif key == 64 or (KEY and (key == KEY.ENTER or key == KEY.PAD_ENTER)) then
                local num = tonumber(boothSliderInputVal)
                if num then
                    applySliderInputValue(boothSliderInputId, num)
                end
                boothSliderInputId = nil
                setPresetInputActive(false)
                return
            else
                local sourceNumMap = {
                    [1] = "0", [2] = "1", [3] = "2", [4] = "3", [5] = "4",
                    [6] = "5", [7] = "6", [8] = "7", [9] = "8", [10] = "9",
                    [37] = "0", [38] = "1", [39] = "2", [40] = "3", [41] = "4",
                    [42] = "5", [43] = "6", [44] = "7", [45] = "8", [46] = "9",
                    [48] = "-", [53] = ".",
                }
                local numChar = sourceNumMap[key]
                if not numChar and input.getKeyName then
                    local kName = input.getKeyName(key)
                    if kName and #kName == 1 and kName:match("[%d%.%-]") then numChar = kName end
                end
                if numChar and #boothSliderInputVal < 10 then
                    boothSliderInputVal = boothSliderInputVal .. numChar
                    return
                end
            end
        end

        -- KEYBOARD INPUT FOR PRESET NAME
        if boothPresetInputActive and (key < 107) then
            if key == 66 or (KEY and key == KEY.BACKSPACE) then
                boothPresetInput = boothPresetInput:sub(1, -2)
                return
            elseif key == 64 or (KEY and (key == KEY.ENTER or key == KEY.PAD_ENTER)) then
                if activeReg and activeReg.savePreset and boothPresetInput ~= "" then
                    activeReg.savePreset(boothPresetInput)
                    boothPresetInput = ""
                    setPresetInputActive(false)
                end
                return
            elseif key == 67 or (KEY and key == KEY.ESCAPE) then
                setPresetInputActive(false)
                return
            elseif key == 65 or (KEY and key == KEY.SPACE) then
                if #boothPresetInput < 25 then
                    boothPresetInput = boothPresetInput .. " "
                    lastPresetCharTime = timer.realtime()
                end
                return
            else

    -- Unified Chat Trigger (Owner Only, Fast & Regex-Free)
    hook.add("playerChat", "MeshLoader_UnifiedBoothChat_" .. currentChipId, function(ply, text, team, isDead)
        if ply ~= player() then return end
        if not text or text == "" then return end
        
        local activeReg = _G.MeshLoaderRegistry and _G.MeshLoaderRegistry[currentChipId] or ctx

        local firstChar = text:sub(1, 1)
        if firstChar ~= "!" and firstChar ~= "/" then return end
        
        local spacePos = text:find(" ")
        local cmd = spacePos and text:sub(1, spacePos - 1):lower() or text:lower()
        local arg = spacePos and text:sub(spacePos + 1) or ""
        
        -- Simple trim without regex
        while #arg > 0 and arg:sub(1, 1) == " " do arg = arg:sub(2) end
        while #arg > 0 and arg:sub(-1) == " " do arg = arg:sub(1, -2) end
        
        local myNameLower = (activeConfig.meshName or "module"):lower()
        local argLower = arg:lower()
        
        -- Strict raycast vehicle targeting check
        local function isTargetingVehicle()
            local me = player()
            if not isValid(me) then return false end
            
            if me:inVehicle() then
                local seat = me:getVehicle()
                if isValid(seat) then
                    if vehicleEntities and vehicleEntities[seat] then return true end
                    if vehicleEntityIndices and vehicleEntityIndices[seat:entIndex()] then return true end
                    if belongsToVehicle(seat, seat:entIndex()) then return true end
                end
                return false
            end
            
            local tr = me:getEyeTrace()
            local hitEnt = tr.Entity
            if not isValid(hitEnt) then return false end
            
            if hitEnt == chip() then return true end
            if isValid(base) and hitEnt == base then return true end
            local myBase = resolveParent and resolveParent("Base")
            if isValid(myBase) and hitEnt == myBase then return true end
            
            if vehicleEntities and vehicleEntities[hitEnt] then return true end
            if vehicleEntityIndices and vehicleEntityIndices[hitEnt:entIndex()] then return true end
            if belongsToVehicle(hitEnt, hitEnt:entIndex()) then return true end
            
            for _, h in ipairs(spawnedHolograms or {}) do
                if isValid(h) and hitEnt == h then return true end
            end
            
            return false
        end

        if cmd == "!booth" or cmd == "/booth" then
            if activeConfig.EnableBooth == false or activeConfig.enableBooth == false or isBoothEnabled == false then return end
            if not isTargetingVehicle() then return end
            
            local now = timer.realtime()
            
            if _G.MeshLoaderBoothOpen and (isBoothActiveOnMe or _G.MeshLoaderActiveBoothChip == currentChipId) then
                _G.MeshLoaderBoothOpen = false
                _G.MeshLoaderActiveBoothChip = nil
                pcall(function()
                    file.write("mesh_presets/active_booth.txt", "0:CLOSED:" .. tostring(now))
                end)
                setBoothState(false, false)
                pcall(function() input.enableCursor(false) end)
                pcall(function() input.lockControls(false) end)
                return true
            end
            
            local priorOpenChipId = 0
            local lastWriteTime = 0
            pcall(function()
                if file.exists("mesh_presets/active_booth.txt") then
                    local raw = file.read("mesh_presets/active_booth.txt")
                    if raw and raw ~= "" then
                        local colon = raw:find(":")
                        if colon then
                            local tId = tonumber(raw:sub(1, colon - 1))
                            local rest = raw:sub(colon + 1)
                            local colon2 = rest:find(":")
                            local tName = colon2 and rest:sub(1, colon2 - 1) or rest
                            local tTime = colon2 and tonumber(rest:sub(colon2 + 1)) or 0
                            if tName ~= "CLOSED" and tId and tId > 0 then
                                priorOpenChipId = tId
                                lastWriteTime = tTime or 0
                            end
                        end
                    end
                end
            end)
            
            if priorOpenChipId > 0 and (now - lastWriteTime > 0.25) then
                _G.MeshLoaderBoothOpen = false
                _G.MeshLoaderActiveBoothChip = nil
                pcall(function()
                    file.write("mesh_presets/active_booth.txt", "0:CLOSED:" .. tostring(now))
                end)
                setBoothState(false, false)
                pcall(function() input.enableCursor(false) end)
                pcall(function() input.lockControls(false) end)
                return true
            end
            
            if priorOpenChipId > 0 and (now - lastWriteTime <= 0.25) then
                return true
            end
            
            if argLower ~= "" then
                if myNameLower:find(argLower, 1, true) then
                    pcall(function()
                        file.write("mesh_presets/active_booth.txt", currentChipId .. ":" .. myModuleName .. ":" .. tostring(now))
                    end)
                    setBoothState(true)
                    focusWorkshopCamera("overview", nil, true)
                end
            else
                local isPrimary = true
                local allMods = getDiscoveredModules()
                for _, m in ipairs(allMods) do
                    if m.id < currentChipId then
                        isPrimary = false
                        break
                    end
                end
                
                if isPrimary then
                    pcall(function()
                        file.write("mesh_presets/active_booth.txt", currentChipId .. ":" .. myModuleName .. ":" .. tostring(now))
                    end)
                    setBoothState(true)
                    focusWorkshopCamera("overview", nil, true)
                end
            end
            return true
        elseif cmd == "!savepreset" or cmd == "/savepreset" then
            if not isTargetingVehicle() then return end
            if arg ~= "" then
                activeReg.savePreset(arg)
            else
                print(Color(255, 200, 0), "[Mesh Loader] Usage: !savepreset <preset_name>")
            end
            return true
        elseif cmd == "!loadpreset" or cmd == "/loadpreset" then
            if not isTargetingVehicle() then return end
            if arg ~= "" then
                activeReg.loadPreset(arg)
            else
                print(Color(255, 200, 0), "[Mesh Loader] Usage: !loadpreset <preset_name>")
            end
            return true
        elseif cmd == "!listpresets" or cmd == "/listpresets" then
            if not isTargetingVehicle() then return end
            local list = activeReg.refreshPresetList and activeReg.refreshPresetList() or {}
            print(Color(0, 200, 255), "[Mesh Loader] Saved presets for ", Color(255, 255, 255), myModuleName, Color(0, 200, 255), ":")
            if #list == 0 then
                print(Color(180, 180, 180), "  (No presets saved yet)")
            else
                for _, p in ipairs(list) do
                    print(Color(0, 255, 120), "  - ", Color(255, 255, 255), p)
                end
            end
            return true
        elseif cmd == "!delpreset" or cmd == "/delpreset" then
            if not isTargetingVehicle() then return end
            if arg ~= "" then
                activeReg.deletePreset(arg)
            end
            return true
        elseif cmd == "!resetcar" or cmd == "/resetcar" or cmd == "!factoryreset" or cmd == "/factoryreset" or cmd == "!resetall" or cmd == "/resetall" then
            if not isTargetingVehicle() then return end
            activeReg.resetAllSettings()
            return true
        elseif cmd == "!paint" or cmd == "/paint" or cmd == "!col" or cmd == "/col" or cmd == "!color" or cmd == "/color" or cmd == "!body" or cmd == "/body" then
            if not isTargetingVehicle() then return end
            if arg ~= "" or cmd == "!body" or cmd == "/body" then
                local targetCmd = nil
                local colorStr = arg
                local sp = arg:find(" ")
                if sp and cmd ~= "!body" and cmd ~= "/body" then
                    local firstW = arg:sub(1, sp - 1):lower()
                    if registeredColorCommands and registeredColorCommands[firstW] then
                        targetCmd = firstW
                        colorStr = arg:sub(sp + 1)
                    end
                end
                if not targetCmd then
                    if registeredColorCommands and registeredColorCommands["body"] then
                        targetCmd = "body"
                    else
                        local firstCmd = next(registeredColorCommands or {})
                        if firstCmd then targetCmd = firstCmd end
                    end
                end
                if targetCmd then
                    local parsed = parseColorString(colorStr)
                    if parsed then
                        local st = activeReg.currentCommandColors and activeReg.currentCommandColors[targetCmd] or {}
                        st.color = parsed
                        if boothMatchSpecular then
                            st.phongTint = Color(parsed.r, parsed.g, parsed.b, 255)
                            st.envMapTint = Color(parsed.r, parsed.g, parsed.b, 255)
                        end
                        activeReg.currentCommandColors[targetCmd] = st
                        for _, v in ipairs(activeReg.config.meshes or {}) do
                            if v.ColorCommand and v.ColorCommand:lower() == targetCmd then
                                activeReg.applyColorCommandUpdate(v, parsed, st.phongTint, st.envMapTint, st.phongMult, st.envMapMult, st.phongBoost, st.phongExp, st.rimLightBoost, st.rimLightExp)
                            end
                        end
                        if player() == owner() then
                            net.start("clientUpdateColor")
                            net.writeString(targetCmd)
                            net.writeUInt(st.color.r or 255, 8)
                            net.writeUInt(st.color.g or 255, 8)
                            net.writeUInt(st.color.b or 255, 8)
                            net.writeUInt(st.color.a or 255, 8)
                            if st.phongTint then
                                net.writeBool(true)
                                net.writeUInt(st.phongTint.r or 255, 8)
                                net.writeUInt(st.phongTint.g or 255, 8)
                                net.writeUInt(st.phongTint.b or 255, 8)
                            else
                                net.writeBool(false)
                            end
                            if st.envMapTint then
                                net.writeBool(true)
                                net.writeUInt(st.envMapTint.r or 255, 8)
                                net.writeUInt(st.envMapTint.g or 255, 8)
                                net.writeUInt(st.envMapTint.b or 255, 8)
                            else
                                net.writeBool(false)
                            end
                            net.writeFloat(st.phongMult or 1)
                            net.writeFloat(st.envMapMult or 1)
                            net.writeFloat(st.phongBoost or 1)
                            net.writeUInt(math.clamp(math.floor(st.phongExp or 64), 1, 2048), 12)
                            net.writeFloat(st.rimLightBoost or 0.25)
                            net.writeUInt(math.clamp(math.floor(st.rimLightExp or 4), 1, 2048), 12)
                            net.send()
                        end
                        print(Color(0, 255, 120), "[Mesh Loader] Applied paint ", Color(parsed.r, parsed.g, parsed.b), string.format("#%02X%02X%02X", parsed.r, parsed.g, parsed.b), Color(0, 255, 120), " to ", Color(255, 255, 255), targetCmd)
                    end
                else
                    print(Color(255, 200, 0), "[Mesh Loader] Usage: !paint [#hex / R,G,B] or !paint <command> [#hex / R,G,B]")
                end
            end
            return true
        elseif cmd == "!part" or cmd == "/part" then
            if not isTargetingVehicle() then return end
            local pSpace = arg:find(" ")
            local pName = pSpace and arg:sub(1, pSpace - 1) or arg
            local vArg = pSpace and arg:sub(pSpace + 1) or "1"
            if pName ~= "" then
                activeReg.setPartVariant(pName, vArg)
                print(Color(0, 255, 100), "[Mesh Loader] Set part '", pName, "' to variant: ", vArg)
            end
            return true
        elseif cmd == "!swap" or cmd == "/swap" then
            if not isTargetingVehicle() then return end
            local sSpace = arg:find(" ")
            local sName = sSpace and arg:sub(1, sSpace - 1) or arg
            local sVal = sSpace and arg:sub(sSpace + 1) or ""
            if sName ~= "" then
                local foundKey = nil
                for _, m in ipairs(activeReg.config.meshes or {}) do
                    if m.TextureSwaps then
                        for swKey, swDef in pairs(m.TextureSwaps) do
                            local inName = (type(swDef) == "table" and swDef.input) or swKey
                            if inName:lower() == sName:lower() or swKey:lower() == sName:lower() then
                                foundKey = swKey
                                break
                            end
                        end
                    end
                    if foundKey then break end
                end
                if foundKey then
                    activeReg.activeWireSwaps = activeReg.activeWireSwaps or {}
                    local curVal = activeReg.activeWireSwaps[foundKey] or 0
                    local val = tonumber(sVal)
                    if not val then
                        local s = sVal:lower()
                        if s == "on" or s == "1" or s == "true" then val = 1
                        elseif s == "off" or s == "0" or s == "false" then val = 0
                        else val = (curVal > 0) and 0 or 1 end
                    end
                    activeReg.activeWireSwaps[foundKey] = val
                    for _, holo in ipairs(activeReg.spawnedHolograms or {}) do
                        if isValid(holo) and holo.MeshConfig and holo.MeshConfig.TextureSwaps and holo.MeshConfig.TextureSwaps[foundKey] then
                            holo.ActiveSwaps = holo.ActiveSwaps or {}
                            holo.ActiveSwaps[foundKey] = (val > 0) or nil
                            activeReg.applyHoloSwaps(holo)
                        end
                    end
                    print(Color(0, 255, 100), "[Mesh Loader] Texture swap '", foundKey, "' set to: ", val > 0 and "ON" or "OFF")
                end
            end
            return true
        end
    end)

    -- Ensure cursor is restored when chat is opened/closed while booth is open
    hook.add("finishchat", "MeshLoader_RestoreCursorOnChatClose_" .. currentChipId, function()
        if _G.MeshLoaderBoothOpen and isBoothActiveOnMe then
            timer.simple(0.05, function()
                if _G.MeshLoaderBoothOpen and isBoothActiveOnMe then
                    input.enableCursor(true)
                end
            end)
        end
    end)

    -- Block gameplay binds (WASD movement, jumping, ducking, weapon slots) while menu is open
    hook.add("playerbindpress", "MeshLoader_BlockPlayerBinds_" .. currentChipId, function(ply, bind, pressed)
        if not _G.MeshLoaderBoothOpen or not isBoothActiveOnMe then return end
        if not isValid(ply) or ply ~= player() then return end
        
        local bLower = bind:lower()
        -- NEVER block mouse clicks (+attack, +attack2) so mouse clicking always functions seamlessly!
        if bLower:find("+attack") then
            return
        end
        
        -- While booth is open, block movement and inventory switching
        if bLower:find("+forward") or bLower:find("+back") or bLower:find("+moveleft") or bLower:find("+moveright")
           or bLower:find("+jump") or bLower:find("+duck") or bLower:find("+speed") or bLower:find("+walk")
           or bLower:find("+use") or bLower:find("invnext") or bLower:find("invprev") or bLower:find("slot") then
            return true
        end
    end)
end
