    local function parseColorString(str)
        if not str or str == "" then return nil end
        local s = str:gsub("^%s+", ""):gsub("%s+$", "")
        
        -- Hex: #RRGGBB or #RRGGBBAA or RRGGBB
        if s:sub(1, 1) == "#" or (#s >= 6 and s:match("^%x+$")) then
            local hex = s:sub(1, 1) == "#" and s:sub(2) or s
            if #hex == 3 then
                local r = tonumber(hex:sub(1,1) .. hex:sub(1,1), 16)
                local g = tonumber(hex:sub(2,2) .. hex:sub(2,2), 16)
                local b = tonumber(hex:sub(3,3) .. hex:sub(3,3), 16)
                if r and g and b then return Color(r, g, b, 255) end
            elseif #hex >= 6 then
                local r = tonumber(hex:sub(1, 2), 16)
                local g = tonumber(hex:sub(3, 4), 16)
                local b = tonumber(hex:sub(5, 6), 16)
                local a = #hex >= 8 and tonumber(hex:sub(7, 8), 16) or 255
                if r and g and b then return Color(r, g, b, a or 255) end
            end
        end
        
        -- RGB / RGBA (comma or space separated)
        local parts = {}
        for part in s:gmatch("[^,%s]+") do
            local n = tonumber(part)
            if n then table.insert(parts, n) end
        end
        if #parts >= 3 then
            local r = parts[1] <= 1.0 and math.floor(parts[1] * 255) or math.floor(parts[1])
            local g = parts[2] <= 1.0 and math.floor(parts[2] * 255) or math.floor(parts[2])
            local b = parts[3] <= 1.0 and math.floor(parts[3] * 255) or math.floor(parts[3])
            local a = parts[4] and (parts[4] <= 1.0 and math.floor(parts[4] * 255) or math.floor(parts[4])) or 255
            return Color(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255), math.clamp(a, 0, 255))
        end
        return nil
    end

    local function getSafeFont(fontName, size, weight, antialias, fallback)
        local key = string.format("%s_%d_%d_%s", fontName, size or 14, weight or 400, antialias and "1" or "0")
        if _G.MeshLoaderFontCache[key] then
            return _G.MeshLoaderFontCache[key]
        end
        
        local ok, res = pcall(function()
            return render.createFont(fontName, size, weight, antialias)
        end)
        
        if ok and res then
            _G.MeshLoaderFontCache[key] = res
            return res
        end
        
        -- Safe fallback mapping if font quota is exhausted
        local defFont = fallback or "DermaDefault"
        if size and size >= 48 then
            defFont = "CloseCaption_Bold"
        elseif size and size >= 24 then
            defFont = "Trebuchet24"
        elseif size and size >= 18 then
            defFont = "Trebuchet18"
        elseif size and size >= 14 then
            defFont = "DermaDefaultBold"
        else
            defFont = "DermaDefault"
        end
        
        _G.MeshLoaderFontCache[key] = defFont
        return defFont
    end

    local function markTextureLoaded(url)
        if type(url) ~= "string" or url == "" or (url:sub(1,4) ~= "http" and not url:find("%.vtf")) then return end
        if not loadedTexturesSet[url] then
            loadedTexturesSet[url] = true
            local count = 0
            for _ in pairs(loadedTexturesSet) do count = count + 1 end
            if activeConfig and activeConfig.progressData then
                activeConfig.progressData.loadedTextures = count
            end
        end
    end

    local function markDownloadCompleted(url)
        if type(url) ~= "string" or url == "" or (url:sub(1,4) ~= "http" and not url:find("%.vtf")) then return end
        if downloadsNeededSet[url] and not downloadedUrlsSet[url] then
            downloadedUrlsSet[url] = true
            local count = 0
            for _ in pairs(downloadedUrlsSet) do count = count + 1 end
            if activeConfig and activeConfig.progressData then
                activeConfig.progressData.completedDownloads = count
            end
        end
    end

    local function applyColorCommandUpdate(v, newColor, phongTint, envMapTint, phongMult, envMapMult, phongBoost, phongExp, rimLightBoost, rimLightExp)
        if not v then return end
        
        newColor = newColor or v.ColorAndAlpha or v.OriginalColor or Color(255, 255, 255, 255)
        local r = newColor.r or 255
        local g = newColor.g or 255
        local b = newColor.b or 255
        local a = newColor.a or (v.OriginalColor and v.OriginalColor.a) or 255
        local finalColor = Color(r, g, b, a)
        
        v.ColorAndAlpha = finalColor
        v.ActivePhongTint = phongTint
        v.ActiveEnvMapTint = envMapTint
        v.ActivePhongMult = phongMult
        v.ActiveEnvMapMult = envMapMult
        
        if phongBoost ~= nil then
            v.PhongBoost = phongBoost
            if v.mat and v.mat.setFloat then v.mat:setFloat("$phongboost", tonumber(phongBoost) or 1) end
        end
        if phongExp ~= nil then
            v.PhongExp = phongExp
            if v.mat and v.mat.setFloat then v.mat:setFloat("$phongexponent", tonumber(phongExp) or 64) end
        end
        if rimLightBoost ~= nil then
            v.RimLightBoost = rimLightBoost
            if v.mat and v.mat.setFloat then v.mat:setFloat("$rimlightboost", tonumber(rimLightBoost) or 0.25) end
        end
        if rimLightExp ~= nil then
            v.RimLightExp = rimLightExp
            if v.mat and v.mat.setFloat then v.mat:setFloat("$rimlightexponent", tonumber(rimLightExp) or 4) end
        end
        
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) and holo.MeshConfig == v then
                if holo.ActiveSwaps and next(holo.ActiveSwaps) and applyHoloSwaps then
                    applyHoloSwaps(holo)
                else
                    holo:setColor(finalColor)
                end
            end
        end
        
        local function scaleVector(origVec, origColor, targetColor)
            if not origVec then return nil end
            origColor = origColor or Color(255, 255, 255, 255)
            targetColor = targetColor or Color(255, 255, 255, 255)
            local newVec = Vector(0, 0, 0)
            newVec[1] = origColor.r > 0 and (origVec[1] * (targetColor.r / origColor.r)) or (origVec[1] * (targetColor.r / 255))
            newVec[2] = origColor.g > 0 and (origVec[2] * (targetColor.g / origColor.g)) or (origVec[2] * (targetColor.g / 255))
            newVec[3] = origColor.b > 0 and (origVec[3] * (targetColor.b / origColor.b)) or (origVec[3] * (targetColor.b / 255))
            return newVec
        end
        
        local pM = phongMult or 1
        local eM = envMapMult or 1
        
        v.OriginalColor = v.OriginalColor or Color(255, 255, 255, 255)
        v.OriginalPhong = v.OriginalPhong or (v.PhongTint and Vector(v.PhongTint[1], v.PhongTint[2], v.PhongTint[3])) or Vector(1, 1, 1)
        v.OriginalEnvMap = v.OriginalEnvMap or (v.EnvMapShinyness and Vector(v.EnvMapShinyness[1], v.EnvMapShinyness[2], v.EnvMapShinyness[3])) or Vector(0.2, 0.2, 0.2)

        if v.OriginalPhong and v.mat then
            local newPhong
            if phongTint then
                newPhong = Vector((phongTint.r or 255) / 255, (phongTint.g or 255) / 255, (phongTint.b or 255) / 255)
            else
                newPhong = scaleVector(v.OriginalPhong, v.OriginalColor, finalColor)
            end
            
            if newPhong then
                newPhong = newPhong * pM
                v.PhongTint = newPhong
                if v.mat.setVector then
                    v.mat:setVector("$phongtint", newPhong)
                end
            end
        end
        
        if v.OriginalEnvMap and v.mat then
            local newEnvMap
            if envMapTint then
                newEnvMap = Vector((envMapTint.r or 255) / 255, (envMapTint.g or 255) / 255, (envMapTint.b or 255) / 255)
            else
                newEnvMap = scaleVector(v.OriginalEnvMap, v.OriginalColor, finalColor)
            end
            
            if newEnvMap then
                newEnvMap = newEnvMap * eM
                v.EnvMapShinyness = newEnvMap
                if v.mat.setVector then
                    v.mat:setVector("$envmaptint", newEnvMap)
                end
            end
        end
    end

    local function applyMeshMaterialOverrides(v)
        if not v or not v.mat then return end
        if v.mat.setFloat then
            if v.ColorAndAlpha and v.ColorAndAlpha.a then
                v.mat:setFloat("$alpha", (v.ColorAndAlpha.a or 255) / 255)
            elseif v.Alpha ~= nil then
                v.mat:setFloat("$alpha", tonumber(v.Alpha) or 1)
            end
            if v.PhongBoost ~= nil then v.mat:setFloat("$phongboost", tonumber(v.PhongBoost) or 1) end
            if v.PhongExp ~= nil then v.mat:setFloat("$phongexponent", tonumber(v.PhongExp) or 64) end
            if v.RimLightBoost ~= nil then v.mat:setFloat("$rimlightboost", tonumber(v.RimLightBoost) or 0.25) end
            if v.RimLightExp ~= nil then v.mat:setFloat("$rimlightexponent", tonumber(v.RimLightExp) or 4) end
            if v.EnvMapContrast ~= nil then v.mat:setFloat("$envmapcontrast", tonumber(v.EnvMapContrast) or 0) end
            if v.EnvMapSaturation ~= nil then v.mat:setFloat("$envmapsaturation", tonumber(v.EnvMapSaturation) or 1) end
            if v.EnvMapFresnel ~= nil then v.mat:setFloat("$envmapfresnel", tonumber(v.EnvMapFresnel) or 1) end
        end
        if v.mat.setVector then
            if v.ActivePhongTint then
                local pt = v.ActivePhongTint
                local pM = v.ActivePhongMult or 1
                v.mat:setVector("$phongtint", Vector((pt.r / 255) * pM, (pt.g / 255) * pM, (pt.b / 255) * pM))
            elseif v.PhongTint then
                local pM = v.ActivePhongMult or 1
                if type(v.PhongTint) == "Vector" then
                    v.mat:setVector("$phongtint", v.PhongTint * pM)
                end
            end
            if v.ActiveEnvMapTint then
                local et = v.ActiveEnvMapTint
                local eM = v.ActiveEnvMapMult or 1
                v.mat:setVector("$envmaptint", Vector((et.r / 255) * eM, (et.g / 255) * eM, (et.b / 255) * eM))
            elseif v.EnvMapShinyness then
                local eM = v.ActiveEnvMapMult or 1
                if type(v.EnvMapShinyness) == "Vector" then
                    v.mat:setVector("$envmaptint", v.EnvMapShinyness * eM)
                end
            end
        end
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) and holo.MeshConfig == v then
                if holo.setMeshMaterial and v.mat then
                    holo:setMeshMaterial(v.mat)
                end
                if holo.ActiveSwaps and next(holo.ActiveSwaps) and applyHoloSwaps then
                    applyHoloSwaps(holo)
                else
                    if v.ColorAndAlpha then holo:setColor(v.ColorAndAlpha) end
                    if v.SuppressEngineLighting ~= nil then holo:suppressEngineLighting(v.SuppressEngineLighting) end
                end
            end
        end
    end

    local function updatePartVisibility(targetPart)
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) and holo.MeshConfig then
                local rawPart = holo.MeshConfig.Part or holo.MeshConfig.part
                if rawPart and type(rawPart) == "string" and rawPart ~= "" then
                    local pName = rawPart
                    if not targetPart or pName:lower() == targetPart:lower() then
                        local activeVar = activePartVariants[pName:lower()] or activePartVariants[pName] or 1
                        local thisVar = tonumber(holo.MeshConfig.Variant or holo.MeshConfig.variant) or 1
                        if thisVar == activeVar then
                            if updateHoloTransforms then
                                updateHoloTransforms(holo)
                            else
                                local sScale = toVector(holo.MeshConfig.scale, Vector(1, 1, 1))
                                local gScale = toVector(activeConfig and activeConfig.globalScale, Vector(1, 1, 1))
                                holo:setScale(Vector(sScale[1] * gScale[1], sScale[2] * gScale[2], sScale[3] * gScale[3]))
                            end
                            holo:setColor(holo.MeshConfig.ColorAndAlpha or Color(255, 255, 255, 255))
                        else
                            holo:setScale(Vector(0, 0, 0))
                            holo:setColor(Color(0, 0, 0, 0))
                        end
                    end
                end
            end
        end
    end

    local function setPartVariant(partName, variantNum)
        if not partName then return end
        local pLower = partName:lower()
        local vNum = tonumber(variantNum)
        if not vNum then
            local s = tostring(variantNum):lower()
            if s == "none" or s == "off" or s == "remove" then
                vNum = 0
            else
                vNum = 1
            end
        end
        activePartVariants[pLower] = vNum
        activePartVariants[partName] = vNum
        updatePartVisibility(partName)
        
        if player() == owner() then
            net.start("clientUpdatePart")
            net.writeString(partName)
            net.writeInt(vNum, 8)
            net.send()
        end
    end

    local function getCarSafeName()
        local name = activeConfig and activeConfig.meshName
        local isGeneric = (not name or name == "" or name:lower() == "vehicle" or name:lower() == "body" or name:lower() == "car" or name:lower() == "module" or name:lower() == "mesh")
        
        if isGeneric and activeConfig and activeConfig.meshes and activeConfig.meshes[1] and activeConfig.meshes[1].mdl then
            local primaryMdl = activeConfig.meshes[1].mdl:lower():gsub("\\", "/")
            local parts = string.explode("/", primaryMdl)
            if #parts >= 2 then
                local folder = parts[#parts - 1]:gsub("%.mdl$", "")
                local file = parts[#parts]:gsub("%.mdl$", "")
                if folder ~= "" and folder ~= "props" and folder ~= "models" and folder ~= "props_c17" then
                    name = (name and name ~= "" and name:lower() ~= "vehicle") and (folder .. "_" .. name) or (folder .. "_" .. file)
                else
                    name = file
                end
            end
        end
        
        name = name or "Vehicle"
        return name:gsub("[^%w_%-]", "_")
    end

    local function refreshPresetList()
        local safeName = getCarSafeName()
        boothPresetsList = {}
        pcall(function()
            file.createDir("mesh_presets")
            file.createDir("mesh_presets/" .. safeName)
            local seen = {}
            local files = (file.find and file.find("mesh_presets/" .. safeName .. "/*.txt")) or {}
            for _, f in ipairs(files) do
                local pName = f:gsub("%.txt$", "")
                if pName ~= "" and not seen[pName] then
                    seen[pName] = true
                    table.insert(boothPresetsList, pName)
                end
            end
            local files2 = (file.find and file.find("mesh_presets/" .. safeName .. "_*.txt")) or {}
            for _, f in ipairs(files2) do
                local pName = f:sub(#safeName + 2):gsub("%.txt$", "")
                if pName ~= "" and not seen[pName] then
                    seen[pName] = true
                    table.insert(boothPresetsList, pName)
                end
            end
        end)
        table.sort(boothPresetsList)
        boothSelectedPreset = math.clamp(boothSelectedPreset or 1, 1, math.max(1, #boothPresetsList))
        return boothPresetsList
    end

    local function savePreset(presetName)
        if not presetName or presetName == "" then return false end
        presetName = presetName:gsub("[^%w_%- ]", ""):gsub("%s+", " ")
        if presetName == "" then return false end
        local safeName = getCarSafeName()
        if file.createDir then
            file.createDir("mesh_presets")
            file.createDir("mesh_presets/" .. safeName)
        end
        
        local data = {
            car = activeConfig.meshName or "Vehicle",
            version = Version,
            timestamp = os.time(),
            colors = {},
            parts = activePartVariants or {},
            swaps = activeWireSwaps or {},
            submeshes = {},
            submeshOverrides = {}
        }
        
        for cmdLower, state in pairs(currentCommandColors or {}) do
            data.colors[cmdLower] = {
                color = { state.color.r, state.color.g, state.color.b, state.color.a or 255 },
                phongTint = state.phongTint and { state.phongTint.r, state.phongTint.g, state.phongTint.b } or nil,
                envMapTint = state.envMapTint and { state.envMapTint.r, state.envMapTint.g, state.envMapTint.b } or nil,
                phongMult = state.phongMult or 1,
                envMapMult = state.envMapMult or 1,
                phongBoost = state.phongBoost or 1,
                phongExp = state.phongExp or 64,
                rimLightBoost = state.rimLightBoost or 0.25,
                rimLightExp = state.rimLightExp or 4
            }
        end

        for i, v in ipairs(activeConfig and activeConfig.meshes or {}) do
            local pt = v.ActivePhongTint or (v.PhongTint and Color(math.clamp(math.floor(v.PhongTint[1] * 255), 0, 255), math.clamp(math.floor(v.PhongTint[2] * 255), 0, 255), math.clamp(math.floor(v.PhongTint[3] * 255), 0, 255))) or Color(255, 255, 255)
            local et = v.ActiveEnvMapTint or (v.EnvMapShinyness and Color(math.clamp(math.floor(v.EnvMapShinyness[1] * 255), 0, 255), math.clamp(math.floor(v.EnvMapShinyness[2] * 255), 0, 255), math.clamp(math.floor(v.EnvMapShinyness[3] * 255), 0, 255))) or Color(51, 51, 51)
            local col = v.ColorAndAlpha or v.OriginalColor or Color(255, 255, 255, 255)
            
            data.submeshes[i] = {
                mdl = v.mdl or nil,
                texture = v.texture or nil,
                part = v.Part or v.part or nil,
                variant = v.Variant or v.variant or nil,
                isIsolatedMaterial = (v.isIsolatedMaterial == true),
                color = { r = col.r, g = col.g, b = col.b, a = col.a or 255 },
                phongTint = { r = pt.r, g = pt.g, b = pt.b },
                envMapTint = { r = et.r, g = et.g, b = et.b },
                phongMult = v.ActivePhongMult or 1,
                envMapMult = v.ActiveEnvMapMult or 1,
                phongBoost = v.PhongBoost or 1,
                phongExp = v.PhongExp or 64,
                rimLightBoost = v.RimLightBoost or 0.25,
                rimLightExp = v.RimLightExp or 4,
                hidden = (v.hidden == true),
                suppressLighting = (v.SuppressEngineLighting == true),
                offsetPos = v.offsetPos and { v.offsetPos[1], v.offsetPos[2], v.offsetPos[3] } or nil,
                offsetAngle = v.offsetAngle and { v.offsetAngle[1], v.offsetAngle[2], v.offsetAngle[3] } or nil,
                scale = v.scale and { v.scale[1], v.scale[2], v.scale[3] } or nil,
            }
            data.submeshOverrides[i] = {
                color = { col.r, col.g, col.b, col.a or 255 },
                phongBoost = v.PhongBoost,
                phongExp = v.PhongExp,
                rimLightBoost = v.RimLightBoost,
                rimLightExp = v.RimLightExp,
                envMapShinyness = { et.r / 255, et.g / 255, et.b / 255 },
                offsetPos = v.offsetPos and { v.offsetPos[1], v.offsetPos[2], v.offsetPos[3] } or nil,
                scale = v.scale and { v.scale[1], v.scale[2], v.scale[3] } or nil,
                fullbright = v.SuppressEngineLighting
            }
        end
        
        local jsonStr = json.encode(data)
        local path = "mesh_presets/" .. safeName .. "/" .. presetName .. ".txt"
        local ok, err = pcall(file.write, path, jsonStr)
        if ok then
            print(Color(0, 255, 100), "[Mesh Loader] Preset '", Color(255, 255, 255), presetName, Color(0, 255, 100), "' saved successfully for ", Color(255, 255, 255), activeConfig.meshName or "Vehicle")
            refreshPresetList()
            return true
        else
            print(Color(255, 0, 0), "[Mesh Loader] Failed to save preset: ", tostring(err))
            return false
        end
    end

    local function loadPreset(presetName)
        if not presetName or presetName == "" then return false end
        local safeName = getCarSafeName()
        local path = "mesh_presets/" .. safeName .. "/" .. presetName .. ".txt"
        if not file.exists(path) then
            path = "mesh_presets/" .. safeName .. "_" .. presetName .. ".txt"
        end
        if not file.exists(path) then
            path = "mesh_presets/" .. presetName .. ".txt"
        end
        if not file.exists(path) then
            print(Color(255, 0, 0), "[Mesh Loader] Preset '", presetName, "' does not exist for ", activeConfig.meshName or "Vehicle")
            return false
        end
        
        local str = file.read(path)
        if not str or str == "" then return false end
        local data = json.decode(str)
        if not data then return false end
        
        -- 1. Apply Parts
        if data.parts then
            for pName, vNum in pairs(data.parts) do
                setPartVariant(pName, tonumber(vNum) or 1)
            end
        end

        -- 2. Apply Colors & ColorCommands
        if data.colors then
            for cmdLower, cData in pairs(data.colors) do
                local col = Color(cData.color[1], cData.color[2], cData.color[3], cData.color[4] or 255)
                local pTint = cData.phongTint and Color(cData.phongTint[1], cData.phongTint[2], cData.phongTint[3]) or nil
                local eTint = cData.envMapTint and Color(cData.envMapTint[1], cData.envMapTint[2], cData.envMapTint[3]) or nil
                local pBoost = cData.phongBoost or 1
                local pExp = cData.phongExp or 64
                local rBoost = cData.rimLightBoost or 0.25
                local rExp = cData.rimLightExp or 4
                
                currentCommandColors[cmdLower] = {
                    color = col,
                    phongTint = pTint,
                    envMapTint = eTint,
                    phongMult = cData.phongMult or 1,
                    envMapMult = cData.envMapMult or 1,
                    phongBoost = pBoost,
                    phongExp = pExp,
                    rimLightBoost = rBoost,
                    rimLightExp = rExp
                }
                
                for _, v in ipairs(activeConfig and activeConfig.meshes or {}) do
                    if v.ColorCommand and v.ColorCommand:lower() == cmdLower then
                        applyColorCommandUpdate(v, col, pTint, eTint, cData.phongMult or 1, cData.envMapMult or 1, pBoost, pExp, rBoost, rExp)
                    end
                end
                if sendColorUpdate then
                    sendColorUpdate(cmdLower)
                end
            end
        end

        -- 3. Apply Swaps
        if data.swaps and type(data.swaps) == "table" then
            activeWireSwaps = data.swaps
            for _, holo in ipairs(spawnedHolograms or {}) do
                if isValid(holo) and holo.MeshConfig and holo.MeshConfig.TextureSwaps then
                    for sName, _ in pairs(holo.MeshConfig.TextureSwaps) do
                        local val = activeWireSwaps[sName] or 0
                        holo.ActiveSwaps = holo.ActiveSwaps or {}
                        holo.ActiveSwaps[sName] = (val > 0) or nil
                    end
                    applyHoloSwaps(holo)
                end
            end
            for sName, val in pairs(activeWireSwaps) do
                if player() == owner() then
                    net.start("clientUpdateSwap")
                    net.writeString(sName)
                    net.writeFloat(val)
                    net.send()
                end
            end
        end

        -- 4. Apply Submesh Overrides & Material Decoupling (Executed LAST to preserve custom submesh adjustments)
        if data.submeshes and type(data.submeshes) == "table" then
            for idx, sData in pairs(data.submeshes) do
                local k = tonumber(idx)
                local v = activeConfig and activeConfig.meshes and activeConfig.meshes[k]
                
                if v and type(sData) == "table" then
                    if sData.isIsolatedMaterial == true and not v.isIsolatedMaterial and decoupleSubmeshMaterial then
                        decoupleSubmeshMaterial(k)
                    elseif sData.isIsolatedMaterial ~= true and v.isIsolatedMaterial and relinkSubmeshMaterial then
                        relinkSubmeshMaterial(k)
                    end
                    
                    if sData.color then
                        v.ColorAndAlpha = Color(sData.color.r or 255, sData.color.g or 255, sData.color.b or 255, sData.color.a or 255)
                        if v.mat and v.mat.setFloat and sData.color.a then
                            v.mat:setFloat("$alpha", (sData.color.a or 255) / 255)
                        end
                    end
                    if sData.phongTint then
                        v.ActivePhongTint = Color(sData.phongTint.r or 255, sData.phongTint.g or 255, sData.phongTint.b or 255, 255)
                    end
                    if sData.envMapTint then
                        v.ActiveEnvMapTint = Color(sData.envMapTint.r or 51, sData.envMapTint.g or 51, sData.envMapTint.b or 51, 255)
                    end
                    v.ActivePhongMult = sData.phongMult or v.ActivePhongMult or 1
                    v.ActiveEnvMapMult = sData.envMapMult or v.ActiveEnvMapMult or 1
                    v.PhongBoost = sData.phongBoost or v.PhongBoost or 1
                    v.PhongExp = sData.phongExp or v.PhongExp or 64
                    v.RimLightBoost = sData.rimLightBoost or v.RimLightBoost or 0.25
                    v.RimLightExp = sData.rimLightExp or v.RimLightExp or 4
                    v.hidden = (sData.hidden == true)
                    v.SuppressEngineLighting = (sData.suppressLighting == true)
                    
                    if sData.offsetPos then
                        v.offsetPos = Vector(sData.offsetPos[1] or 0, sData.offsetPos[2] or 0, sData.offsetPos[3] or 0)
                    end
                    if sData.offsetAngle then
                        v.offsetAngle = Angle(sData.offsetAngle[1] or 0, sData.offsetAngle[2] or 0, sData.offsetAngle[3] or 0)
                    end
                    if sData.scale then
                        v.scale = Vector(sData.scale[1] or 1, sData.scale[2] or 1, sData.scale[3] or 1)
                    end
                    
                    for _, holo in ipairs(spawnedHolograms or {}) do
                        if isValid(holo) and holo.MeshConfig == v then
                            updateHoloTransforms(holo)
                            if v.hidden then
                                holo:setScale(Vector(0, 0, 0))
                                holo:setColor(Color(0, 0, 0, 0))
                            else
                                holo:setColor(v.ColorAndAlpha or Color(255))
                                if holo.setMeshMaterial and v.mat then
                                    holo:setMeshMaterial(v.mat)
                                end
                            end
                            holo:suppressEngineLighting(v.SuppressEngineLighting or false)
                        end
                    end
                    applyMeshMaterialOverrides(v)
                    if sendSubmeshUpdate then sendSubmeshUpdate(k) end
                end
            end
        elseif data.submeshOverrides and type(data.submeshOverrides) == "table" then
            for i, ov in pairs(data.submeshOverrides) do
                local v = activeConfig and activeConfig.meshes and activeConfig.meshes[tonumber(i)]
                if v then
                    if ov.color then
                        v.ColorAndAlpha = Color(ov.color[1], ov.color[2], ov.color[3], ov.color[4] or 255)
                    end
                    if ov.phongBoost and v.mat and v.mat.setFloat then
                        v.PhongBoost = ov.phongBoost
                        v.mat:setFloat("$phongboost", ov.phongBoost)
                    end
                    if ov.phongExp and v.mat and v.mat.setFloat then
                        v.PhongExp = ov.phongExp
                        v.mat:setFloat("$phongexponent", ov.phongExp)
                    end
                    if ov.envMapShinyness and v.mat and v.mat.setVector then
                        v.EnvMapShinyness = Vector(ov.envMapShinyness[1], ov.envMapShinyness[2], ov.envMapShinyness[3])
                        v.mat:setVector("$envmaptint", v.EnvMapShinyness)
                    end
                    if ov.offsetPos then
                        v.offsetPos = Vector(ov.offsetPos[1], ov.offsetPos[2], ov.offsetPos[3])
                    end
                    if ov.scale then
                        v.scale = Vector(ov.scale[1], ov.scale[2], ov.scale[3])
                    end
                    if ov.fullbright ~= nil then
                        v.SuppressEngineLighting = ov.fullbright
                    end
                    
                    for _, holo in ipairs(spawnedHolograms or {}) do
                        if isValid(holo) and holo.MeshConfig == v then
                            if ov.color then holo:setColor(v.ColorAndAlpha) end
                            if ov.fullbright ~= nil then holo:suppressEngineLighting(v.SuppressEngineLighting) end
                            updateHoloTransforms(holo)
                        end
                    end
                end
            end
        end

        print(Color(0, 255, 100), "[Mesh Loader] Preset '", Color(255, 255, 255), presetName, Color(0, 255, 100), "' loaded successfully!")
        return true
    end

    local function deletePreset(presetName)
        if not presetName or presetName == "" then return false end
        local safeName = getCarSafeName()
        local path1 = "mesh_presets/" .. safeName .. "/" .. presetName .. ".txt"
        local path2 = "mesh_presets/" .. safeName .. "_" .. presetName .. ".txt"
        local path3 = "mesh_presets/" .. presetName .. ".txt"

        if file.exists(path1) then file.delete(path1) deleted = true end
        if file.exists(path2) then file.delete(path2) deleted = true end

            refreshPresetList()
            return true
        end
        return false
    end

    local function resetSingleChipSettings()
        -- 1. Reset Colors and Shaders back to original config / defaults
        if currentCommandColors then
            for k in pairs(currentCommandColors) do
                currentCommandColors[k] = nil
            end
        end
        if registeredColorCommands then
            for cmdLower in pairs(registeredColorCommands) do
                local def = defaultCommandColors and defaultCommandColors[cmdLower]
                if def then
                    currentCommandColors[cmdLower] = {
                        color = Color(def.color.r, def.color.g, def.color.b, def.color.a or 255),
                        phongTint = def.phongTint and Color(def.phongTint.r, def.phongTint.g, def.phongTint.b, 255) or nil,
                        envMapTint = def.envMapTint and Color(def.envMapTint.r, def.envMapTint.g, def.envMapTint.b, 255) or nil,
                        phongMult = 1,
                        envMapMult = 1,
                        phongBoost = def.phongBoost or 1,
                        phongExp = def.phongExp or 64,
                        rimLightBoost = def.rimLightBoost or 0.25,
                        rimLightExp = def.rimLightExp or 4
                    }
                else
                    currentCommandColors[cmdLower] = {
                        color = Color(255, 255, 255, 255),
                        phongTint = nil,
                        envMapTint = nil,
                        phongMult = 1,
                        envMapMult = 1,
                        phongBoost = 1,
                        phongExp = 64,
                        rimLightBoost = 0.25,
                        rimLightExp = 4
                    }
                end
                
                local st = currentCommandColors[cmdLower]
                for _, v in ipairs(activeConfig.meshes or {}) do
                    if v.ColorCommand and v.ColorCommand:lower() == cmdLower then
                        v.ColorAndAlpha = Color(st.color.r, st.color.g, st.color.b, st.color.a or 255)
                        v.ActivePhongTint = st.phongTint
                        v.ActiveEnvMapTint = st.envMapTint
                        v.ActivePhongMult = 1
                        v.ActiveEnvMapMult = 1
                        v.PhongBoost = st.phongBoost
                        v.PhongExp = st.phongExp
                        v.RimLightBoost = st.rimLightBoost
                        v.RimLightExp = st.rimLightExp
                        applyMeshMaterialOverrides(v)
                    end
                end
                
                if sendColorUpdate then
                    sendColorUpdate(cmdLower)
                end
            end
        end
        
        -- 2. Reset Texture Swaps
        if activeWireSwaps then
            for k in pairs(activeWireSwaps) do
                activeWireSwaps[k] = nil
            end
        end
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) then
                holo.ActiveSwaps = {}
                holo.ActiveSwapData = nil
                if applyHoloSwaps then applyHoloSwaps(holo) end
            end
        end
        
        -- 3. Reset Bodykits & Parts
        if registeredParts then
            for pName in pairs(registeredParts) do
                setPartVariant(pName, 1)
            end
        end
        
        -- 4. Reset Submesh overrides & update transforms
        for _, v in ipairs(activeConfig.meshes or {}) do
            local def = v.DefaultConfig or {}
            v.hidden = false
            v.ColorAndAlpha = def.ColorAndAlpha and Color(def.ColorAndAlpha.r, def.ColorAndAlpha.g, def.ColorAndAlpha.b, def.ColorAndAlpha.a) or (v.OriginalColor and Color(v.OriginalColor.r, v.OriginalColor.g, v.OriginalColor.b, v.OriginalColor.a)) or Color(255, 255, 255, 255)
            v.offsetPos = def.offsetPos and Vector(def.offsetPos.x, def.offsetPos.y, def.offsetPos.z) or nil
            v.offsetAngle = def.offsetAngle and Angle(def.offsetAngle.p, def.offsetAngle.y, def.offsetAngle.r) or nil
            v.scale = def.scale and Vector(def.scale.x, def.scale.y, def.scale.z) or nil
            v.SuppressEngineLighting = (def.SuppressEngineLighting ~= nil) and def.SuppressEngineLighting or false
            v.PhongBoost = def.PhongBoost or v.OriginalPhongBoost or 1
            v.PhongExp = def.PhongExp or v.OriginalPhongExp or 64
            v.RimLightBoost = def.RimLightBoost or v.OriginalRimLightBoost or 0.25
            v.RimLightExp = def.RimLightExp or v.OriginalRimLightExp or 4
            v.PhongTint = def.PhongTint and Vector(def.PhongTint.x, def.PhongTint.y, def.PhongTint.z) or v.OriginalPhong
            v.EnvMapShinyness = def.EnvMapShinyness and Vector(def.EnvMapShinyness.x, def.EnvMapShinyness.y, def.EnvMapShinyness.z) or v.OriginalEnvMap
            v.ActivePhongMult = 1
            v.ActiveEnvMapMult = 1
            v.ActivePhongTint = nil
            v.ActiveEnvMapTint = nil
            applyMeshMaterialOverrides(v)
        end
        
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) then
                if holo.MeshConfig then
                    holo:setColor(holo.MeshConfig.ColorAndAlpha or Color(255, 255, 255, 255))
                    holo:suppressEngineLighting(holo.MeshConfig.SuppressEngineLighting or false)
                end
                if updateHoloTransforms then
                    updateHoloTransforms(holo)
                end
            end
        end
    end

    local function resetAllSettings()
        boothDraggingSlider = nil
        boothSliderInputId = nil
        boothHexInputActive = false
        boothPresetInputActive = false
        if setPresetInputActive then setPresetInputActive(false) end

        return true
    end

    local function safeHasPermission(perm, obj)
        local ok, res = pcall(hasPermission, perm, obj)
        return ok and res == true
    end

    local function canWriteTemp()
        if not file or not file.writeTemp then return false end
        local ok, res = pcall(hasPermission, "file.writeTemp")
        if ok and res == false then return false end
        local ok2, res2 = pcall(hasPermission, "file.temp")
        if ok2 and res2 == false then return false end
        return true
    end

    local function safeExistsTemp(fileName)
        local ok, res = pcall(file.existsTemp, fileName)
        return ok and res == true
    end

    local function safeWriteTemp(fileName, data)
        if not canWriteTemp() then return false end
        local ok, err = pcall(file.writeTemp, fileName, data)
        return ok
    end

    local function safeDeleteTemp(fileName)
        if not canWriteTemp() then return false end
        pcall(file.deleteTemp, fileName)
    end

    local function hashString(str)
        local h = 0
        for i = 1, #str do
            h = (h * 31 + string.byte(str, i)) % 1000000000
        end
        return tostring(h)
    end

    local function vtfTempFileName(url)
        local cleanUrl = url:match("^([^?]+)") or url
        local name = cleanUrl:match("([^/]+)%.[vV][tT][fF]") or cleanUrl:match("([^/]+)%.[^.]+$") or "unnamed"
        return name .. "_" .. hashString(cleanUrl) .. ".vtf"
    end

    local function vtfTempPath(url)
        local cleanUrl = url:match("^([^?]+)") or url
        local name = cleanUrl:match("([^/]+)%.[vV][tT][fF]") or cleanUrl:match("([^/]+)%.[^.]+$") or "unnamed"
        return "../data/sf_filedatatemp/" .. getClientSteamID64() .. "/" .. name .. "_" .. hashString(cleanUrl) .. ".vtf"
    end

    local function shouldDownloadMeshImage(v)
        if v and v.DownloadImages ~= nil then return v.DownloadImages == true end
        if v and v.DownloadImage ~= nil then return v.DownloadImage == true end
        return not not (activeConfig and activeConfig.DownloadImages)
    end

    local function isTextureDownloaded(url, v)
        if type(url) ~= "string" or url == "" then return true end
        if not canWriteTemp() then return true end
        local cleanUrl = url:match("^([^?]+)") or url
        if url:lower():find("%.vtf") then
            local fileName = vtfTempFileName(url)
            return safeExistsTemp(fileName)
        elseif url:sub(1, 4) == "http" then
            if not shouldDownloadMeshImage(v) then return false end
            local lower = cleanUrl:lower()
            local ext = lower:match("%.(png)$") or lower:match("%.(jpg)$") or lower:match("%.(jpeg)$")
            if ext then
                local name = cleanUrl:match("([^/]+)%.[^.]+$") or "unnamed"
                local fileName = name .. "_" .. hashString(cleanUrl) .. "." .. ext
                return safeExistsTemp(fileName)
            end
            return false
        end
        return true
    end

    local function imageTempPath(url, ext)
        local cleanUrl = url:match("^([^?]+)") or url
        local name = cleanUrl:match("([^/]+)%.[^.]+$") or "unnamed"
        return "../data/sf_filedatatemp/" .. getClientSteamID64() .. "/" .. name .. "_" .. hashString(cleanUrl) .. "." .. ext
    end

    local pendingVtfDownloads = {}
    local pendingImageDownloads = {}
    local downloadQueue = {}
    local isPumping = false

    local function pumpDownloadQueue()
        if isPumping then return end
        isPumping = true
        local maxRequests = http.getMaximumRequests()
        while #downloadQueue > 0 and http.getActiveRequests() < maxRequests do
            local job = table.remove(downloadQueue, 1)
            job()
        end
        isPumping = false
        if #downloadQueue > 0 then
            timer.simple(0.1, pumpDownloadQueue)
        end
    end

    local function asyncDownloadVtf(url, callback)
        if not url:lower():find("%.vtf") then
            if callback then callback(nil) end
            return
        end

        local fileName = vtfTempFileName(url)
        local tempPath = vtfTempPath(url)

        if not canWriteTemp() then
            markDownloadCompleted(url)
            markTextureLoaded(url)
            if callback then callback(nil) end
            return
        end

        if safeExistsTemp(fileName) then
            timer.simple(0.1, function()
                markDownloadCompleted(url)
                markTextureLoaded(url)
                if isValid(chip()) and callback then
                    callback(tempPath)
                end
            end)
            timer.simple(1.5, function()
                if isValid(chip()) and callback then
                    callback(tempPath)
                end
            end)
            return
        end

        if pendingVtfDownloads[url] then
            table.insert(pendingVtfDownloads[url], callback)
            return
        end

        pendingVtfDownloads[url] = { callback }

        table.insert(downloadQueue, function()
            safeDeleteTemp(fileName)
            http.get(url,
                function(data)
                    local cbs = pendingVtfDownloads[url] or {}
                    pendingVtfDownloads[url] = nil
                    markDownloadCompleted(url)
                    markTextureLoaded(url)

                    if data and #data > 0 then
                        safeWriteTemp(fileName, data)
                        timer.simple(0.3, function()
                            if isValid(chip()) then
                                for _, cb in ipairs(cbs) do
                                    if cb then cb(tempPath) end
                                end
                            end
                        end)
                        timer.simple(1.5, function()
                            if isValid(chip()) then
                                for _, cb in ipairs(cbs) do
                                    if cb then cb(tempPath) end
                                end
                            end
                        end)
                    else
                        for _, cb in ipairs(cbs) do
                            if cb then cb(nil) end
                        end
                    end
                    pumpDownloadQueue()
                end,
                function(err)
                    local cbs = pendingVtfDownloads[url] or {}
                    pendingVtfDownloads[url] = nil
                    markDownloadCompleted(url)
                    markTextureLoaded(url)
                    for _, cb in ipairs(cbs) do
                        if cb then cb(nil) end
                    end
                    pumpDownloadQueue()
                end
            )
        end)

        pumpDownloadQueue()
    end

    local function queueVtfDownload(url, callback)
        asyncDownloadVtf(url, function(path)
            markTextureLoaded(url)
            if callback then callback(path) end
        end)
    end

    local function asyncDownloadImage(url, callback)
        local cleanUrl = url:match("^([^?]+)") or url
        local lower = cleanUrl:lower()
        local ext = lower:match("%.(png)$") or lower:match("%.(jpg)$") or lower:match("%.(jpeg)$")
        
        if not ext then
            markDownloadCompleted(url)
            markTextureLoaded(url)
            if callback then callback(nil) end
            return
        end
        
        local name = cleanUrl:match("([^/]+)%.[^.]+$") or "unnamed"
        local fileName = name .. "_" .. hashString(cleanUrl) .. "." .. ext
        local tempPath = imageTempPath(url, ext)

        if not canWriteTemp() then
            markDownloadCompleted(url)
            markTextureLoaded(url)
            if callback then callback(nil) end
            return
        end

        if safeExistsTemp(fileName) then
            timer.simple(0.05, function()
                markDownloadCompleted(url)
                markTextureLoaded(url)
                if isValid(chip()) and callback then
                    callback(tempPath)
                end
            end)
            return
        end

        if pendingImageDownloads[url] then
            table.insert(pendingImageDownloads[url], callback)
            return
        end

        pendingImageDownloads[url] = { callback }

        table.insert(downloadQueue, function()
            safeDeleteTemp(fileName)
            http.get(url,
                function(data)
                    local cbs = pendingImageDownloads[url] or {}
                    pendingImageDownloads[url] = nil
                    markDownloadCompleted(url)
                    markTextureLoaded(url)

                    if data and #data > 0 then
                        safeWriteTemp(fileName, data)
                        timer.simple(0.1, function()
                            if isValid(chip()) then
                                for _, cb in ipairs(cbs) do
                                    if cb then cb(tempPath) end
                                end
                            end
                        end)
                    else
                        for _, cb in ipairs(cbs) do
                            if cb then cb(nil) end
                        end
                    end
                    pumpDownloadQueue()
                end,
                function(err)
                    local cbs = pendingImageDownloads[url] or {}
                    pendingImageDownloads[url] = nil
                    markDownloadCompleted(url)
                    markTextureLoaded(url)
                    for _, cb in ipairs(cbs) do
                        if cb then cb(nil) end
                    end
                    pumpDownloadQueue()
                end
            )
        end)

        pumpDownloadQueue()
    end

    local function queueImageDownload(url, callback)
        asyncDownloadImage(url, function(path)
            markTextureLoaded(url)
            if callback then callback(path) end
        end)
    end

    local urlTextureCache = {}

    local function applyUrlTexture(targetMat, paramName, url, callback)
        if not (url and url:sub(1, 4) == "http") then return end
        
        if urlTextureCache[url] and urlTextureCache[url].state == "ready" then
            local tex = urlTextureCache[url].texture
            if tex and targetMat and isValid(chip()) then
                targetMat:setTexture(paramName, tex)
            end
            if callback then callback(tex) end
            return
        end

        if urlTextureCache[url] and urlTextureCache[url].state == "loading" then
            table.insert(urlTextureCache[url].callbacks, function(tex)
                if tex and targetMat and isValid(chip()) then
                    targetMat:setTexture(paramName, tex)
                end
                if callback then callback(tex) end
            end)
            return
        end

        urlTextureCache[url] = {
            state = "loading",
            texture = nil,
            callbacks = {}
        }

        local entry = urlTextureCache[url]
        if targetMat then
            table.insert(entry.callbacks, function(tex)
                if tex and targetMat and isValid(chip()) and targetMat.setTexture then
                    pcall(function() targetMat:setTexture(paramName, tex) end)
                end
                if callback then callback(tex) end
            end)
        elseif callback then
            table.insert(entry.callbacks, callback)
        end

        if safeHasPermission("material.urlcreate", url) then
            if targetMat and targetMat.setTextureURL then
                targetMat:setTextureURL(paramName, url, function(_, _, w, h, layout)
                    markDownloadCompleted(url)
                    markTextureLoaded(url)
                    if not (w and h) then
                        entry.state = "error"
                        for _, cb in ipairs(entry.callbacks) do cb(nil) end
                        entry.callbacks = {}
                        return
                    end
                    local sx, sy = 1024 / w, 1024 / h
                    layout(0, 0, w * sx, h * sy)
                    timer.simple(0.1, function()
                        if isValid(chip()) and targetMat and targetMat.getTexture then
                            local tex = targetMat:getTexture(paramName)
                            entry.texture = tex
                            entry.state = "ready"
                            for _, cb in ipairs(entry.callbacks) do
                                if cb then cb(tex) end
                            end
                            entry.callbacks = {}
                        end
                    end)
                end)
            else
                local okMat, dummyMat = pcall(material.create, "VertexLitGeneric")
                if okMat and dummyMat then
                    dummyMat:setTextureURL("$basetexture", url, function(_, _, w, h, layout)
                        markDownloadCompleted(url)
                        markTextureLoaded(url)
                        if not (w and h) then
                            entry.state = "error"
                            if dummyMat and dummyMat.destroy then pcall(function() dummyMat:destroy() end) end
                            for _, cb in ipairs(entry.callbacks) do cb(nil) end
                            entry.callbacks = {}
                            return
                        end
                        local sx, sy = 1024 / w, 1024 / h
                        layout(0, 0, w * sx, h * sy)
                        timer.simple(0.1, function()
                            if isValid(chip()) and dummyMat then
                                local tex = dummyMat:getTexture("$basetexture")
                                entry.texture = tex
                                entry.state = "ready"
                                if dummyMat.destroy then pcall(function() dummyMat:destroy() end) end
                                for _, cb in ipairs(entry.callbacks) do
                                    if cb then cb(tex) end
                                end
                                entry.callbacks = {}
                            elseif dummyMat and dummyMat.destroy then
                                pcall(function() dummyMat:destroy() end)
                            end
                        end)
                    end)
                else
                    entry.state = "error"
                    for _, cb in ipairs(entry.callbacks) do cb(nil) end
                    entry.callbacks = {}
                end
            end
        else
            entry.state = "error"
            for _, cb in ipairs(entry.callbacks) do cb(nil) end
            entry.callbacks = {}
        end
    end

    local function resolveParent(parentName)
        if not parentName then return nil end
        local val = networkedEntities[parentName]
        if not val then return nil end
        if type(val) == "table" then
            if isValid(val.ent) then
                return val.ent
            elseif val.idx then
                local e = entity(val.idx)
                if isValid(e) then
                    val.ent = e
                    return e
                end
            end
            return nil
        end
        return isValid(val) and val or nil
    end
    fixAnglesHolos = {}

    local function getRestingAngle(wheelEnt, fixEnt, fixP, fixY, fixR, invertAxis)
        local restingWorldAng = fixEnt:getAngles()
        
        if type(fixP) == "number" then restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getRight(), fixP) end
        if type(fixY) == "number" then restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getUp(), fixY) end
        if type(fixR) == "number" then restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getForward(), fixR) end

        if fixP ~= nil and fixP ~= false then
            local R1 = restingWorldAng:getRight()
            local R2 = wheelEnt:getRight()
            if invertAxis then R2 = -R2 end
            local axis = R1:cross(R2)
            local len = axis:getLength()
            if len > 0.0001 then
                axis:normalize()
                local dot = math.clamp(R1:dot(R2), -1, 1)
                restingWorldAng = restingWorldAng:rotateAroundAxis(axis, math.deg(math.acos(dot)))
            elseif R1:dot(R2) < 0 then
                restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getUp(), 180)
            end
        elseif fixY ~= nil and fixY ~= false then
            local U1 = restingWorldAng:getUp()
            local U2 = wheelEnt:getUp()
            if invertAxis then U2 = -U2 end
            local axis = U1:cross(U2)
            local len = axis:getLength()
            if len > 0.0001 then
                axis:normalize()
                local dot = math.clamp(U1:dot(U2), -1, 1)
                restingWorldAng = restingWorldAng:rotateAroundAxis(axis, math.deg(math.acos(dot)))
            elseif U1:dot(U2) < 0 then
                restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getRight(), 180)
            end
        elseif fixR ~= nil and fixR ~= false then
            local F1 = restingWorldAng:getForward()
            local F2 = wheelEnt:getForward()
            if invertAxis then F2 = -F2 end
            local axis = F1:cross(F2)
            local len = axis:getLength()
            if len > 0.0001 then
                axis:normalize()
                local dot = math.clamp(F1:dot(F2), -1, 1)
                restingWorldAng = restingWorldAng:rotateAroundAxis(axis, math.deg(math.acos(dot)))
            elseif F1:dot(F2) < 0 then
                restingWorldAng = restingWorldAng:rotateAroundAxis(restingWorldAng:getUp(), 180)
            end
        else
            restingWorldAng = wheelEnt:getAngles()
        end
        
        return restingWorldAng
    end
    local function toVector(val, defaultVal)
        if type(val) == "Vector" then return val end
        if type(val) == "number" then return Vector(val, val, val) end
        if type(val) == "table" then return Vector(tonumber(val[1] or val.x) or 1, tonumber(val[2] or val.y) or 1, tonumber(val[3] or val.z) or 1) end
        return defaultVal or Vector(0, 0, 0)
    end

    local function toAngle(val, defaultVal)
        if type(val) == "Angle" then return val end
        if type(val) == "table" then return Angle(tonumber(val[1] or val.p or val.pitch) or 0, tonumber(val[2] or val.y or val.yaw) or 0, tonumber(val[3] or val.r or val.roll) or 0) end
        return defaultVal or Angle(0, 0, 0)
    end

    local refreshFixAnglesRefs
    
    local function registerFixAnglesHolo(holo)
        local cfg = holo.MeshConfig
        if not cfg or not cfg.FixAngles then return end
        
        local globalPos = toVector(activeConfig.offsetPos, Vector(0, 0, 0))
        local globalAng = toAngle(activeConfig.offsetAngle, Angle(0, 0, 0))
        local subPos = toVector(cfg.offsetPos, Vector(0, 0, 0))
        local subAng = toAngle(cfg.offsetAngle, Angle(0, 0, 0))
        
        -- Pre-cache all the config values so the render loop does zero table lookups
        local entry = {
            holo = holo,
            parentName = holo.MeshTargetParentName,
            fixTargetName = cfg.AngleFixTarget or "Base",
            offsetPos = globalPos + subPos,
            offsetAng = Angle(globalAng[1] + subAng[1], globalAng[2] + subAng[2], globalAng[3] + subAng[3]),
            fixP = cfg.FixAngles.pitch,
            fixY = cfg.FixAngles.yaw,
            fixR = cfg.FixAngles.roll,
            invertAxis = cfg.FixAngles.invert,
        }
        table.insert(fixAnglesHolos, entry)
        
        -- Resolve references immediately in case networking arrived before the mesh finished downloading
        refreshFixAnglesRefs()
    end
    
    refreshFixAnglesRefs = function()
        for i = #fixAnglesHolos, 1, -1 do
            local e = fixAnglesHolos[i]
            if not isValid(e.holo) then
                table.remove(fixAnglesHolos, i)
            else
                e.parent = resolveParent(e.parentName) or base
                e.fixTarget = (e.fixTargetName ~= "Base" and resolveParent(e.fixTargetName)) or base
            end
        end
    end

    updateHoloTransforms = function(holo)
        if not isValid(holo) then return end
        local v = holo.MeshConfig
        local targetParent = resolveParent(holo.MeshTargetParentName) or base
        
        local activeSwap = holo.ActiveSwapData
        
        local rawPos = (activeSwap and activeSwap.offsetPos) or (v and v.offsetPos) or activeConfig.offsetPos or Vector(0, 0, 0)
        local rawAng = (activeSwap and activeSwap.offsetAngle) or (v and v.offsetAngle) or activeConfig.offsetAngle or Angle(0, 0, 0)
        local rawScale = (activeSwap and activeSwap.scale) or (v and v.scale) or Vector(1, 1, 1)
        
        local finalPos = toVector(rawPos, Vector(0, 0, 0))
        local finalAng = toAngle(rawAng, Angle(0, 0, 0))
        local subScale = toVector(rawScale, Vector(1, 1, 1))
        local globalScale = toVector(activeConfig.globalScale, Vector(1, 1, 1))
        local combinedScale = Vector(subScale[1] * globalScale[1], subScale[2] * globalScale[2], subScale[3] * globalScale[3])
        
        local pName = v and (v.Part or v.part)
        if v and v.hidden then
            combinedScale = Vector(0, 0, 0)
        elseif pName and type(pName) == "string" and pName ~= "" then
            local pLower = pName:lower()
            local activeVar = activePartVariants[pLower] or activePartVariants[pName] or 1
            local thisVar = tonumber(v.Variant or v.variant) or 1
            if thisVar ~= activeVar then
                combinedScale = Vector(0, 0, 0)
            end
        end
        
        holo:setScale(combinedScale)
        
        if v and v.FixAngles then
            if holo:getParent() ~= targetParent then
                holo:setParent(targetParent)
            end
            holo:setPos(targetParent:localToWorld(finalPos))
            
            local fixTargetName = v.AngleFixTarget or "Base"
            local fixTarget = (fixTargetName ~= "Base" and resolveParent(fixTargetName)) or base
            
            local restingWorldAng = getRestingAngle(targetParent, fixTarget, v.FixAngles.pitch, v.FixAngles.yaw, v.FixAngles.roll, v.FixAngles.invert)
            
            local m = Matrix()
            m:setAngles(restingWorldAng)
            m:rotate(finalAng)
            
            holo:setAngles(m:getAngles())
            
            for _, e in ipairs(fixAnglesHolos) do
                if e.holo == holo then
                    e.offsetPos = finalPos
                    e.offsetAng = finalAng
                end
            end
        else
            if holo:getParent() ~= targetParent then
                holo:setParent(targetParent)
            end
            holo:setPos(targetParent:localToWorld(finalPos))
            holo:setAngles(targetParent:localToWorldAngles(finalAng))
        end
    end
    
    -- Render hook (framerate): Compute and apply for frame-perfect visual smoothness
    hook.add("PreDrawOpaqueRenderables", "MeshLoader_AngleFix", function()
        for i = 1, #fixAnglesHolos do
            local e = fixAnglesHolos[i]
            if not isValid(e.holo) then
                table.remove(fixAnglesHolos, i)
            else
                local parent = e.parent
                if not isValid(parent) then parent = base end
                local fix = e.fixTarget
                if not isValid(fix) then fix = base end
                
                -- Force parenting to the stable fixTarget (Chassis/Base) instead of the spinning wheel
                if e.holo:getParent() ~= fix then
                    e.holo:setParent(fix)
                end
                
                -- Smoothly follow the wheel's interpolated position
                e.holo:setPos(parent:localToWorld(e.offsetPos))
                
                -- Calculate the stable resting angle based on chassis orientation + wheel steer/camber
                local ang = getRestingAngle(parent, fix, e.fixP, e.fixY, e.fixR, e.invertAxis)
                
                local m = Matrix()
                m:setAngles(ang)
                m:rotate(e.offsetAng)
                
                e.holo:setAngles(m:getAngles())
            end
        end
    end)

    applyHoloSwaps = function(holo)
        if not isValid(holo) or not holo.MeshConfig then return end
        
        local swapConfig = holo.MeshConfig.TextureSwaps
        if not swapConfig or not holo.ActiveSwaps then return end
        
        local defaultColor = holo.MeshConfig.ColorAndAlpha or Color(255, 255, 255, 255)
        local finalR, finalG, finalB, finalA = defaultColor.r, defaultColor.g, defaultColor.b, defaultColor.a
        local finalHighlight = holo.MeshConfig.SuppressEngineLighting or false
        local textureToSet = nil
        
        -- Priority System: Find the active swap with the highest priority
        local bestPriority = -math.huge
        local bestSwapName = nil
        local bestSwapData = nil
        
        for name, swapData in pairs(swapConfig) do
            if holo.ActiveSwaps[name] then
                local priority = (type(swapData) == "table" and swapData.priority) or 0
                if priority > bestPriority then
                    bestPriority = priority
                    bestSwapName = name
                    bestSwapData = swapData
                end
            end
        end
        
        local activeSwapTable = nil
        if bestSwapName then
            local data = bestSwapData
            activeSwapTable = (type(data) == "table") and data or nil
            
            -- Handle texture
            local texture = type(data) == "table" and data.texture or (type(data) == "string" and data)
            if texture then
                textureToSet = holo.MeshConfig.preloadedSwaps and holo.MeshConfig.preloadedSwaps[bestSwapName]
            end
            
            -- Handle highlight
            if type(data) == "table" and data.highlight ~= nil then
                finalHighlight = data.highlight
            end
            
            -- Handle color calculation
            local targetColor = (type(data) == "table" and data.color) or defaultColor
            finalR, finalG, finalB, finalA = targetColor.r, targetColor.g, targetColor.b, targetColor.a
            
            if type(data) == "table" and data.dynamic then
                local dynamicMin = data.dynamicMin or 0
                local dynamicMax = data.dynamicMax or 1
                local wireVal = activeWireSwaps[bestSwapName] or (data.input and activeWireSwaps[data.input]) or 0
                local mult = 1
                if dynamicMax ~= dynamicMin then
                    mult = math.clamp((wireVal - dynamicMin) / (dynamicMax - dynamicMin), 0, 1)
                end
                finalR = defaultColor.r + (targetColor.r - defaultColor.r) * mult
                finalG = defaultColor.g + (targetColor.g - defaultColor.g) * mult
                finalB = defaultColor.b + (targetColor.b - defaultColor.b) * mult
                finalA = defaultColor.a + (targetColor.a - defaultColor.a) * mult
            end
        end
        
        holo.ActiveSwapData = activeSwapTable
        updateHoloTransforms(holo)
        
        -- Helper to compute flags integer from config table
        local function computeFlags(cfg)
            if not cfg then return 0 end
            local f = 0
            if cfg.Flags and type(cfg.Flags) == "table" then
                for _, flag in ipairs(cfg.Flags) do
                    f = bit.bor(f, tonumber(flag) or 0)
                end
            end
            if cfg.EnableHalfLambert then
                f = bit.bor(f, 134217728)
            end
            if cfg.EnableBaseMapAlphaEnvMask then
                f = bit.bor(f, 1048576)
            end
            if cfg.EnableNormalmapAlphaEnvMapMask then
                f = bit.bor(f, 4194304)
            end
            return f
        end

        -- Apply Texture and Material Properties
        local v = holo.MeshConfig
        local mat = v.mat
        if mat then
            if mat.setTexture then
                mat:setTexture("$basetexture", textureToSet or v.OriginalTexture or "models/debug/debugwhite")
            end
            
            if mat.setInt then
                -- Flags
                local hasSwapFlags = activeSwapTable and (activeSwapTable.Flags ~= nil or activeSwapTable.EnableHalfLambert ~= nil or activeSwapTable.EnableBaseMapAlphaEnvMask ~= nil or activeSwapTable.EnableNormalmapAlphaEnvMapMask ~= nil)
                local finalFlags = hasSwapFlags and computeFlags(activeSwapTable) or computeFlags(v)
                mat:setInt("$flags", finalFlags)
                
                -- BaseMapAlphaPhongMask ($basemapalphaphongmask)
                local bmapPhong = activeSwapTable and (activeSwapTable.EnableBaseMapAlphaPhongMask ~= nil and activeSwapTable.EnableBaseMapAlphaPhongMask or activeSwapTable.BaseMapAlphaPhongMask)
                if bmapPhong == nil then bmapPhong = (v.EnableBaseMapAlphaPhongMask ~= nil and v.EnableBaseMapAlphaPhongMask or v.BaseMapAlphaPhongMask) end
                mat:setInt("$basemapalphaphongmask", (bmapPhong == true or bmapPhong == 1) and 1 or 0)
                
                -- BlendTintByBaseAlpha ($blendtintbybasealpha)
                local blendTint = activeSwapTable and activeSwapTable.BlendTintByBaseAlpha
                if blendTint == nil then blendTint = v.BlendTintByBaseAlpha end
                mat:setInt("$blendtintbybasealpha", (blendTint == true or blendTint == 1) and 1 or 0)
                
                -- Mod2x ($mod2x)
                local mod2x = activeSwapTable and activeSwapTable.Mod2x
                if mod2x == nil then mod2x = v.Mod2x end
                if mod2x ~= nil then mat:setInt("$mod2x", (mod2x == true or mod2x == 1) and 1 or 0) end
                
                -- Phong ($phong)
                local useP = activeSwapTable and (activeSwapTable.UsePhong ~= nil and activeSwapTable.UsePhong or activeSwapTable.Phong)
                if useP == nil then useP = (v.UsePhong ~= nil and v.UsePhong or v.Phong) end
                if useP ~= nil then mat:setInt("$phong", (useP == true or useP == 1) and 1 or 0) end
                
                -- InvertPhongMask ($invertphongmask)
                local invP = activeSwapTable and activeSwapTable.InvertPhongMask
                if invP == nil then invP = v.InvertPhongMask end
                if invP ~= nil then mat:setInt("$invertphongmask", (invP == true or invP == 1) and 1 or 0) end
                
                -- RimLight ($rimlight)
                local useR = activeSwapTable and (activeSwapTable.UseRimLight ~= nil and activeSwapTable.UseRimLight or activeSwapTable.RimLight)
                if useR == nil then useR = (v.UseRimLight ~= nil and v.UseRimLight or v.RimLight) end
                if useR ~= nil then mat:setInt("$rimlight", (useR == true or useR == 1) and 1 or 0) end
                
                -- Detail Blend Mode & Frame

                local df = activeSwapTable and activeSwapTable.DetailFrame or v.DetailFrame

            if mat.setFloat then

                local pb = activeSwapTable and (activeSwapTable.PhongBoost or activeSwapTable.phongBoost) or v.PhongBoost

                local rb = activeSwapTable and (activeSwapTable.RimLightBoost or activeSwapTable.rimLightBoost) or v.RimLightBoost

                -- RimLight Exponent

                -- EnvMap Contrast & Saturation & Fresnel
                local ec = activeSwapTable and (activeSwapTable.EnvMapContrast or activeSwapTable.envMapContrast) or v.EnvMapContrast
                if ec ~= nil then mat:setFloat("$envmapcontrast", tonumber(ec) or 0) end
                
                local es = activeSwapTable and (activeSwapTable.EnvMapSaturation or activeSwapTable.envMapSaturation) or v.EnvMapSaturation
                if es ~= nil then mat:setFloat("$envmapsaturation", tonumber(es) or 0) end
                
                local ef = activeSwapTable and (activeSwapTable.EnvMapFresnel or activeSwapTable.envMapFresnel) or v.EnvMapFresnel
                if ef ~= nil then mat:setFloat("$envmapfresnel", tonumber(ef) or 1) end

                if db ~= nil then mat:setFloat("$detailblendfactor", tonumber(db) or 1) end
                local ds = activeSwapTable and activeSwapTable.DetailScale or v.DetailScale
                if ds ~= nil then mat:setFloat("$detailscale", tonumber(ds) or 4) end

            if mat.setVector then

                local ptVal = activeSwapTable and (activeSwapTable.PhongTint or activeSwapTable.phongTint)

                local etVal = activeSwapTable and (activeSwapTable.EnvMapShinyness or activeSwapTable.envMapShinyness or activeSwapTable.EnvMapTint or activeSwapTable.envMapTint)

                        mat:setVector("$envmaptint", Vector(etVal.r / 255, etVal.g / 255, etVal.b / 255))
                    elseif type(etVal) == "Vector" then
                        mat:setVector("$envmaptint", etVal)
                    end
                elseif v.ActiveEnvMapTint or v.EnvMapShinyness then
                    local et = v.ActiveEnvMapTint or (v.EnvMapShinyness and Color(math.clamp(math.floor(v.EnvMapShinyness[1] * 255), 0, 255), math.clamp(math.floor(v.EnvMapShinyness[2] * 255), 0, 255), math.clamp(math.floor(v.EnvMapShinyness[3] * 255), 0, 255))) or Color(51, 51, 51)
                    local eM = v.ActiveEnvMapMult or 1
                    mat:setVector("$envmaptint", Vector((et.r / 255) * eM, (et.g / 255) * eM, (et.b / 255) * eM))
                end
                
                -- Phong Fresnel Ranges
                local pfr = activeSwapTable and (activeSwapTable.PhongFresnelRange or activeSwapTable.PhongFresnelRanges) or v.PhongFresnelRange
                if pfr ~= nil then mat:setVector("$phongfresnelranges", pfr) end
                
                -- Color2
                local c2 = activeSwapTable and activeSwapTable.Color2 or v.Color2
                if c2 ~= nil then mat:setVector("$color2", c2) end
                
                -- Detail Tint
                local dtint = activeSwapTable and activeSwapTable.DetailTint or v.DetailTint
                if dtint ~= nil then mat:setVector("$detailtint", dtint) end
            end
            
            if mat.setMatrix then
                -- Texture scale matrix ($basetexturetransform)
                local tScale = activeSwapTable and activeSwapTable.TextureScale or v.TextureScale
                if tScale ~= nil then
                    local textureMatrix = Matrix()
                    textureMatrix:setScale(Vector(tScale, tScale, 1))
                    mat:setMatrix("$basetexturetransform", textureMatrix)
                end
                
                -- Detail Texture Transform
                local dtt = activeSwapTable and activeSwapTable.DetailTextureTransform or v.DetailTextureTransform
                if dtt ~= nil then mat:setMatrix("$detailtexturetransform", dtt) end
            end
        end
        
        -- Apply Hologram Filtering
        local lp = activeSwapTable and activeSwapTable.LowPolyTexture
        if lp == nil then lp = v.LowPolyTexture or false end
        holo:setFilterMag(lp)
        holo:setFilterMin(lp)
        
        -- Apply Highlight
        holo:suppressEngineLighting(finalHighlight)
        
        -- Apply Final Color
        holo:setColor(Color(finalR, finalG, finalB, finalA))
    end
    
    local function triggerHoloSwapsForConfig(v)
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) and holo.MeshConfig == v then
                applyHoloSwaps(holo)
            end
        end
    end

    net.receive("sendMeshSwap", function()
        local inputName = net.readString()
        local val = net.readFloat()
        
        activeWireSwaps[inputName] = val
        activeWireSwaps[inputName:lower()] = val
        
        for _, holo in ipairs(spawnedHolograms) do
            if isValid(holo) and holo.MeshConfig and holo.MeshConfig.TextureSwaps then
                local configSwaps = holo.MeshConfig.TextureSwaps
                local anyMatch = false
                
                for swapKey, swapData in pairs(configSwaps) do
                    local targetInput = (type(swapData) == "table" and swapData.input) or swapKey
                    
                    if targetInput:lower() == inputName:lower() or swapKey:lower() == inputName:lower() then
                        holo.ActiveSwaps = holo.ActiveSwaps or {}
                        
                        local isActive = false
                        if type(swapData) == "table" and swapData.condition ~= nil then
                            isActive = (val == swapData.condition)
                        else
                            isActive = (val > 0)
                        end
                        
                        holo.ActiveSwaps[swapKey] = isActive or nil
                        anyMatch = true
                    end
                end
                
                if anyMatch then
                    applyHoloSwaps(holo)
                end
            end
        end
    end)

    local buildThread
    local function doneBuilding(meshName, debugPrints, meshSpawnDelay, showMeshSpawnProgress, meshes, offsetPos, offsetAngle, globalScale, triangles, W, R, G, progressData, texCounts, matCounts)
        local triCount = triangles - mesh.trianglesLeft()
        local namePrefix = "Mesh Loader - " .. triCount .. " tris - " .. meshName
        setName(namePrefix)
        if debugPrints then print(W, "Finished building, using ", G, triCount, W, " tris") end
        if debugPrints then print(W, "Starting to spawn meshes with ", G, meshSpawnDelay, W, " second delay between each...") end
        
        activeConfig.offsetPos = offsetPos
        activeConfig.offsetAngle = offsetAngle

        local spawnQueue = {}
        for meshIndex, v in ipairs(meshes) do
            if v.holoMeshes and #v.holoMeshes > 0 then
                for i, holomesh in ipairs(v.holoMeshes) do
                    table.insert(spawnQueue, {
                        meshIndex = meshIndex,
                        submeshIndex = i,
                        mesh = v,
                        holomesh = holomesh
                    })
                end
            end
        end

        progressData.building = false
        progressData.spawning = true
        progressData.currentStage = "Spawning Meshes"
        progressData.spawnProgress = 0
        progressData.spawnedMeshes = 0
        progressData.totalSpawnItems = #spawnQueue

        local totalToSpawn = #spawnQueue
        local spawned = 0

        local rtCount = type(texCounts) == "table" and (texCounts.rt or 0) or 0
        local localCount = type(texCounts) == "table" and (texCounts.temp or texCounts.localCount or 0) or 0
        local totalT = rtCount + localCount
        
        local texStr = ""
        if localCount > 0 and rtCount > 0 then
            texStr = totalT .. "T (" .. localCount .. " Local, " .. rtCount .. " RT)"
        elseif localCount > 0 then
            texStr = localCount .. "T (" .. localCount .. " Local)"
        elseif rtCount > 0 then
            texStr = rtCount .. "T (" .. rtCount .. " RT)"
        elseif type(texCounts) == "number" then
            texStr = texCounts .. "T"
        else
            texStr = "0T"
        end

        local uMat = type(matCounts) == "table" and (matCounts.unique or 0) or (type(matCounts) == "number" and matCounts or 0)
        local cfgMat = type(matCounts) == "table" and (matCounts.configured or #meshes) or #meshes
        
        local matStr = uMat .. " Mat (" .. cfgMat .. " Slots)"

        local statsTag = " [" .. totalToSpawn .. "M / " .. texStr .. " / " .. matStr .. "]"

        local function spawnNextMesh()
            if #spawnQueue == 0 then
                if debugPrints then print(G, "All meshes spawned successfully!") end
                progressData.spawning = false
                setName(namePrefix .. statsTag .. " - Complete")
                return
            end

            local item = table.remove(spawnQueue, 1)
            local v = item.mesh
            local holomesh = item.holomesh

            spawned = spawned + 1

            progressData.spawnedMeshes = spawned
            progressData.spawnProgress = spawned / totalToSpawn
            -- Track current mesh name for display
            local spawnMeshName = v.mdl:match("[^/]+$") or ("Mesh " .. item.meshIndex)
            spawnMeshName = spawnMeshName:gsub("%%20", " "):gsub("%.obj$", "")
            progressData.currentMeshName = spawnMeshName

            if showMeshSpawnProgress and debugPrints then
                print(W, "Spawning mesh ", G, item.meshIndex, W, " (submesh ", G, item.submeshIndex, W, ") - ",
                      G, spawned, W, "/", G, totalToSpawn, W, " (",
                      G, math.floor(spawned/totalToSpawn*100), W, "%)")
            end

            local assignedParentName = v.Parent
            if type(assignedParentName) ~= "string" then
                assignedParentName = nil
            end

            local holo = hologram.create(base:getPos(), base:getAngles(), "models/Combine_Helicopter/helicopter_bomb01.mdl")
            holo.MeshTargetParentName = assignedParentName
            holo.MeshConfig = v
            table.insert(spawnedHolograms, holo)
            
            -- Apply proper transforms and parenting via central function
            updateHoloTransforms(holo)
            
            -- Register into optimized render-frequency list if needed
            registerFixAnglesHolo(holo)
            
            holo:setFilterMag(v.LowPolyTexture)
            holo:setFilterMin(v.LowPolyTexture)
            holo:setMesh(holomesh)
            holo:setMeshMaterial(v.mat)
            holo:setRenderBounds(Vector(-200), Vector(200))
            holo:setColor(v.ColorAndAlpha or Color(255))

            if not v.SuppressEngineLighting == true then
                holo:suppressEngineLighting(false)
            else
                holo:suppressEngineLighting(v.SuppressEngineLighting)
            end

            -- Track wheel blur holograms
            if v.WheelBlurTexture then
                table.insert(wheelBlurHolos, {
                    holo = holo,
                    cfg = v,
                    parentName = holo.MeshTargetParentName,
                    threshold = v.WheelBlurRPM or 300,
                    isBlurred = false
                })
            end
            
            -- Immediately apply any active swaps that were networked while we were downloading/building
            if v.TextureSwaps then
                holo.ActiveSwaps = {}
                for swapName, swapData in pairs(v.TextureSwaps) do
                    local inputKey = (type(swapData) == "table" and swapData.input) or swapName
                    local val = activeWireSwaps[inputKey] or 0
                    local isActive = false
                    if type(swapData) == "table" and swapData.condition ~= nil then
                        isActive = (val == swapData.condition)
                    else
                        isActive = (val > 0)
                    end
                    if isActive then
                        holo.ActiveSwaps[swapName] = true
                    end
                end
                applyHoloSwaps(holo)
            end

            if #spawnQueue > 0 then
                timer.simple(meshSpawnDelay, spawnNextMesh)
            else
                if debugPrints then print(G, "All meshes spawned successfully!") end
                progressData.spawning = false
                setName(namePrefix .. statsTag .. " - Complete")
                
                -- Reconciliation timer: handles AdvDupe2 timing where wires connect AFTER the chip boots
                -- Periodically re-requests server state AND directly re-parents any mismatched holograms
                local reconcileAttempts = 0
                timer.create("meshParentReconcile", 1, 20, function()
                    if not isValid(chip()) then timer.remove("meshParentReconcile") return end
                    reconcileAttempts = reconcileAttempts + 1
                    
                    -- Re-request all states from server
                    net.start("requestBase")
                    net.send()
                    
                    -- Directly reconcile any holograms whose parent doesn't match networkedEntities
                    local allResolved = true
                    for _, holo in ipairs(spawnedHolograms) do
                        if isValid(holo) and holo.MeshTargetParentName then
                            local expectedParent = resolveParent(holo.MeshTargetParentName)
                            if expectedParent and isValid(expectedParent) then
                                local currentParent = holo:getParent()
                                if currentParent ~= expectedParent then
                                    holo:setParent(expectedParent)
                                    local meshOffsetPos = (holo.MeshConfig and holo.MeshConfig.offsetPos) or activeConfig.offsetPos or offsetPos
                                    local meshOffsetAng = (holo.MeshConfig and holo.MeshConfig.offsetAngle) or activeConfig.offsetAngle or offsetAngle
                                    holo:setPos(expectedParent:localToWorld(meshOffsetPos))
                                    holo:setAngles(expectedParent:localToWorldAngles(meshOffsetAng))
                                end
                            else
                                allResolved = false
                            end
                        end
                    end
                    
                    if allResolved and reconcileAttempts >= 5 then
                        timer.remove("meshParentReconcile")
                    end
                end)
                
                -- Continuous local parent checker to handle PVS / dormancy / late join issues
                -- Runs indefinitely with zero network overhead
                timer.create("meshLocalParentReconcile", 2, 0, function()
                    if not isValid(chip()) then timer.remove("meshLocalParentReconcile") return end
                    
                    -- Refresh refs for any active FixAngles holograms
                    refreshFixAnglesRefs()
                    
                    -- Reconcile normal holograms whose parents don't match the expected parent
                    for _, holo in ipairs(spawnedHolograms) do
                        if isValid(holo) and holo.MeshTargetParentName and not (holo.MeshConfig and holo.MeshConfig.FixAngles) then
                            local expectedParent = resolveParent(holo.MeshTargetParentName)
                            if expectedParent and isValid(expectedParent) then
                                local currentParent = holo:getParent()
                                if currentParent ~= expectedParent then
                                    holo:setParent(expectedParent)
                                    local meshOffsetPos = (holo.MeshConfig and holo.MeshConfig.offsetPos) or activeConfig.offsetPos or offsetPos
                                    local meshOffsetAng = (holo.MeshConfig and holo.MeshConfig.offsetAngle) or activeConfig.offsetAngle or offsetAngle
                                    holo:setPos(expectedParent:localToWorld(meshOffsetPos))
                                    holo:setAngles(expectedParent:localToWorldAngles(meshOffsetAng))
                                end
                            end
                        end
                    end
                end)
            end
        end
        spawnNextMesh()
    end

    local function buildMeshes(meshName, debugPrints, meshSpawnDelay, showMeshSpawnProgress, meshes, offsetPos, offsetAngle, globalScale, progressData, W, R, G)
        setName("Mesh Loader - Building meshes...")
        if debugPrints then print(W, "Starting to build meshes...") end

        progressData.building = true
        progressData.currentStage = "Building Meshes"
        progressData.builtMeshes = 0
        progressData.buildProgress = 0

        local hasMeshData = false
        for _, v in ipairs(meshes) do
            if v.meshData then
                hasMeshData = true
                break
            end
        end

        if not hasMeshData then
            if debugPrints then print(true, W, "No valid mesh data found to build!") end
            progressData.building = false
            return
        end

        local triangles = mesh.trianglesLeft()

        buildThread = coroutine.wrap(function()
            local builtCount = 0
            local materialCache = {}
            local textureCountCache = {}
            local uniqueMaterials = 0

            local function trackTexture(url)
                if type(url) ~= "string" or url == "" or (url:sub(1,4) ~= "http" and not url:find("%.vtf")) then return end
                textureCountCache[url] = true
            end
            
            -- Pre-scan to count total textures and downloads
            for _, v in ipairs(meshes) do
                trackTexture(v.texture)
                trackTexture(v.BumpMapLink)
                trackTexture(v.EnvMapLink)
                trackTexture(v.PhongExponentTexture)
                trackTexture(v.DetailLink)
                trackTexture(v.WheelBlurTexture)
                if v.TextureSwaps then
                    for _, sc in pairs(v.TextureSwaps) do
                        trackTexture(type(sc) == "table" and sc.texture or sc)
                    end
                end
            end
            
            local totalTex = 0
            for _ in pairs(textureCountCache) do totalTex = totalTex + 1 end
            progressData.totalTextures = totalTex
            
            totalDownloads = 0
            completedDownloads = 0
            downloadedUrlsSet = {}
            downloadsNeededSet = {}
            for url in pairs(textureCountCache) do
                if not isTextureDownloaded(url) then
                    totalDownloads = totalDownloads + 1
                    downloadsNeededSet[url] = true
                end
            end
            progressData.totalDownloads = totalDownloads
            progressData.completedDownloads = 0
            
            -- Generate a fingerprint key from all material-defining properties
            local function materialFingerprint(v)
                local parts = {
                    v.texture or "",
                    v.BumpMapLink or "",
                    v.EnvMapLink or "",
                    v.PhongExponentTexture or "",
                    v.CubeMapTexture or "",
                    v.WheelBlurTexture or "",
                    tostring(v.WheelBlurRPM or 0),
                    tostring(v.TextureScale or 1),
                    tostring(v.EnableHalfLambert or false),
                    tostring(v.EnableBaseMapAlphaEnvMask or false),
                    tostring(v.EnableBaseMapAlphaPhongMask or false),
                    tostring(v.EnableNormalmapAlphaEnvMapMask or false),
                    tostring(v.UseEnvMapSpecularReflections or false),
                    tostring(v.UsePhong or 0),
                    tostring(v.InvertPhongMask or 0),
                    tostring(v.UseRimLight or 0),
                    tostring(v.PhongBoost or 0),
                    tostring(v.PhongExp or 0),
                    tostring(v.PhongAlbedoTint or 0),
                    tostring(v.PhongAlbedoBoost or 0),
                    tostring(v.RimLightBoost or 0),
                    tostring(v.RimLightExp or 0),
                    tostring(v.EnvMapContrast or 0),
                    tostring(v.EnvMapSaturation or 0),
                    tostring(v.EnvMapFresnel or 0),
                    tostring(v.LowPolyTexture or false),
                    v.Material or "",
                    tostring(v.UseModTexture or false),
                    v.UseIngameBumpMapp and v.BumpMapTexture or "",
                    v.ColorCommand or "",
                    v.Shader or "",
                    tostring(v.BlendTintByBaseAlpha or false),
                    tostring(v.BlendTintColorOverBase or 0),
                    tostring(v.Mod2x or false),
                    v.Part or v.part or "",
                    tostring(v.Variant or v.variant or 1),
                }
                -- Vectors
                if v.EnvMapShinyness then table.insert(parts, tostring(v.EnvMapShinyness)) end
                if v.PhongTint then table.insert(parts, tostring(v.PhongTint)) end
                if v.PhongFresnelRange then table.insert(parts, tostring(v.PhongFresnelRange)) end
                if v.Color2 then table.insert(parts, tostring(v.Color2)) end
                -- Flags
                if v.Flags then
                    for _, f in ipairs(v.Flags) do table.insert(parts, tostring(f)) end
                end
                
                -- Detail parameters
                table.insert(parts, v.DetailLink or "")
                table.insert(parts, tostring(v.DetailBlendFactor or 1))
                table.insert(parts, tostring(v.DetailBlendMode or 0))
                table.insert(parts, tostring(v.DetailFrame or 0))
                table.insert(parts, tostring(v.DetailScale or 4))
                if v.DetailTextureTransform then table.insert(parts, tostring(v.DetailTextureTransform)) end
                if v.DetailTint then table.insert(parts, tostring(v.DetailTint)) end
                
                return table.concat(parts, "|")
            end

            local function safeCreateMaterial(shaderName)
                local ok, mat = pcall(material.create, shaderName or "VertexLitGeneric")
                if ok and mat and mat.setUndefined then
                    return mat, true
                end
                -- Material limit reached: safely fall back to read-only engine material without stealing/mutating active materials
                local okL, lMat = pcall(material.load, "models/debug/debugwhite")
                return okL and lMat or nil, false
            end

            local function safeSetUndefined(m, slot)
                if m and m.setUndefined then
                    pcall(function() m:setUndefined(slot) end)
                end
            end
            local function safeSetTexture(m, slot, val)
                if m and m.setTexture and val then
                    pcall(function() m:setTexture(slot, val) end)
                end
            end
            
            for k, v in ipairs(meshes) do
                progressData.builtMeshes = builtCount
                progressData.buildProgress = builtCount / #meshes

                -- Track current mesh name for display
                local meshFileName = v.mdl:match("[^/]+$") or ("Mesh " .. k)
                meshFileName = meshFileName:gsub("%%20", " "):gsub("%.obj$", "")
                progressData.currentMeshName = meshFileName

                -- Track current texture name for display
                local texName = v.texture or ""
                if texName:sub(1, 4) == "http" then
                    texName = texName:match("[^/]+$") or texName
                    texName = texName:gsub("%%20", " ")
                end
                progressData.currentTextureName = texName

                if v.UseModTexture then
                    -- Check material cache before creating a new one
                    local fingerprint = materialFingerprint(v)
                    local cachedEntry = materialCache[fingerprint]
                    local isNewCustomMat = false
                    
                    if cachedEntry and cachedEntry.mat and not v.TextureSwaps then
                        -- Reuse existing material (only if NO TextureSwaps are present)
                        v.mat = cachedEntry.mat
                        v.SharedMaterial = cachedEntry.mat
                        v.isIsolatedMaterial = false
                        v.isolatedMat = nil
                        v.OriginalTexture = cachedEntry.originalTexture
                        v.preloadedSwaps = cachedEntry.preloadedSwaps
                        if debugPrints then print(W, "Reusing material for mesh ", G, k) end
                        
                        -- Immediately mark its texture URLs as loaded
                        markTextureLoaded(v.texture)
                        markTextureLoaded(v.BumpMapLink)
                        markTextureLoaded(v.EnvMapLink)
                        markTextureLoaded(v.PhongExponentTexture)
                        markTextureLoaded(v.DetailLink)
                        if v.TextureSwaps then
                            for _, sc in pairs(v.TextureSwaps) do
                                markTextureLoaded(type(sc) == "table" and sc.texture or sc)
                            end
                        end
                    else
                    local shaderName = v.Shader or "VertexLitGeneric"
                    local mat, isCreated = safeCreateMaterial(shaderName)
                    isNewCustomMat = isCreated
                    if not mat then
                        print(R, "[MeshLoader] Critical Error: Failed to allocate material for mesh " .. k)
                        mat = material.load("models/debug/debugwhite")
                    end

                    if isNewCustomMat and mat and mat.setUndefined then
                        mat:setUndefined("$basetexture")
                        mat:setUndefined("$bumpmap")
                        mat:setUndefined("$envmapmask")
                        mat:setUndefined("$phongexponenttexture")
                        mat:setUndefined("$detail")

                        if v.BlendTintByBaseAlpha then
                            mat:setInt("$blendtintbybasealpha", 1)
                            if v.BlendTintColorOverBase then
                                mat:setFloat("$blendtintcoloroverbase", tonumber(v.BlendTintColorOverBase) or 0)
                            end
                        else
                            mat:setInt("$blendtintbybasealpha", 0)
                        end

                        if v.Mod2x ~= nil then
                            mat:setInt("$mod2x", (v.Mod2x == true or v.Mod2x == 1) and 1 or 0)
                        end

                        local combinedFlags = 0
                        if v.EnableHalfLambert then
                            combinedFlags = bit.bor(combinedFlags, 134217728)
                        end

                        if v.EnableBaseMapAlphaEnvMask then
                            combinedFlags = bit.bor(combinedFlags, 1048576)
                            mat:setInt("$basealphaenvmapmask", 1)
                        end

                        if v.EnableBaseMapAlphaPhongMask then
                            mat:setInt("$basemapalphaphongmask", 1)
                        else
                            mat:setInt("$basemapalphaphongmask", 0)
                        end

                        if v.EnableNormalmapAlphaEnvMapMask then
                            combinedFlags = bit.bor(combinedFlags, 4194304)
                            mat:setInt("$normalmapalphaenvmapmask", 1)
                        end

                        for _, flag in ipairs(v.Flags or {}) do
                            combinedFlags = bit.bor(combinedFlags, flag)
                        end
                        if combinedFlags > 0 then
                            mat:setInt("$flags", combinedFlags)
                        end

                        mat:setFloat("$envmapcontrast", v.EnvMapContrast)
                        mat:setFloat("$envmapsaturation", v.EnvMapSaturation)
                        mat:setVector("$envmaptint", v.EnvMapShinyness)
                        mat:setFloat("$envmapfresnel", v.EnvMapFresnel)
                                

                        mat:setInt("$phong", v.UsePhong)
                        mat:setInt("$invertphongmask", v.InvertPhongMask)
                        mat:setVector("$phongtint",v.PhongTint)
                        mat:setFloat("$phongalbedotint",v.PhongAlbedoTint)
                        mat:setFloat("$phongboost", v.PhongBoost)
                        
                        -- Detail Parameters
                        mat:setFloat("$detailblendfactor", v.DetailBlendFactor or 1)
                        mat:setInt("$detailblendmode", v.DetailBlendMode or 0)
                        mat:setInt("$detailframe", v.DetailFrame or 0)
                        mat:setFloat("$detailscale", v.DetailScale or 4)
                        mat:setMatrix("$detailtexturetransform", v.DetailTextureTransform or Matrix())
                        mat:setVector("$detailtint", v.DetailTint or Vector(1, 1, 1))
                        mat:setFloat("$phongexponent", v.PhongExp)
                        mat:setVector("$phongfresnelranges", v.PhongFresnelRange)
                        mat:setFloat("$phongalbedoboost", v.PhongAlbedoBoost)

                        mat:setInt("$rimlight", v.UseRimLight)
                        mat:setFloat("$rimlightexponent", v.RimLightExp)
                        mat:setFloat("$rimlightboost", v.RimLightBoost)

                        mat:setVector("$color2", v.Color2)
                    end

                    if v.UseEnvMapSpecularReflections == true then
                        local cubemapUpdateInterval = 2
                        local previousPosition = chip():getPos()
                        local isInitialSpawn = true
                        local function setCubeMap(force)
                            local currentPosition = chip():getPos()
                            if force or currentPosition ~= previousPosition or isInitialSpawn then
                                local cmName = v.CubeMapTexture or (activeConfig and activeConfig.CubeMapTexture) or "models/shiny"
                                local okM, Nmat = pcall(material.load, cmName)
                                local newEnvMap = okM and Nmat and Nmat.getTexture and Nmat:getTexture("$envmap")
                                if newEnvMap then
                                    cachedResolvedCubemap = newEnvMap
                                    for _, mesh_val in ipairs(meshes) do
                                        if mesh_val.mat and mesh_val.mat.setTexture then
                                            mesh_val.mat:setTexture("$envmap", newEnvMap)
                                        end
                                        if mesh_val.isolatedMat and mesh_val.isolatedMat.setTexture then
                                            mesh_val.isolatedMat:setTexture("$envmap", newEnvMap)
                                        end
                                    end
                                end
                                previousPosition = currentPosition
                                isInitialSpawn = false
                            end
                        end
                        globalSetCubeMap = setCubeMap
                        pcall(timer.simple, 1, setCubeMap)
                        local function updateCubemap()
                            if not isValid(chip()) then return end
                            if not isValid(owner()) then return end
                            setCubeMap()
                            pcall(timer.simple, cubemapUpdateInterval, updateCubemap)
                        end
                        pcall(timer.simple, cubemapUpdateInterval, updateCubemap)
                    end

                    if isNewCustomMat then
                        -- ENV MAP MASK
                        if v.EnvMapLink:lower():find("%.vtf") then
                            safeSetUndefined(mat, "$envmapmask")
                            queueVtfDownload(v.EnvMapLink, function(path)
                                if path then
                                    safeSetTexture(mat, "$envmapmask", path)
                                else
                                    safeSetUndefined(mat, "$envmapmask")
                                end
                            end)
                        elseif v.EnvMapLink:sub(1, 4) == "http" then
                            local isImage = v.EnvMapLink:lower():find("%.png") or v.EnvMapLink:lower():find("%.jpg") or v.EnvMapLink:lower():find("%.jpeg")
                            if shouldDownloadMeshImage(v) and isImage and canWriteTemp() then

    local function downloadMeshes(debugPrints, meshes, buildMeshes, meshName, meshSpawnDelay, showMeshSpawnProgress, offsetPos, offsetAngle, globalScale, downloadStatus, progressData, W, R, G)
        local downloadCount = 0
        local failCount = 0
        local urlToMeshes = {}
        local requestQueue = {}

        downloadStatus.total = #meshes
        downloadStatus.completed = 0
        downloadStatus.failed = {}

        progressData.downloading = true
        progressData.currentStage = "Downloading Meshes"
        progressData.totalMeshes = #meshes
        progressData.downloadedMeshes = 0
        progressData.downloadProgress = 0

        for k, v in ipairs(meshes) do
            if not urlToMeshes[v.mdl] then
                urlToMeshes[v.mdl] = { { idx = k, mesh = v } }
                table.insert(requestQueue, v.mdl)
            else
                table.insert(urlToMeshes[v.mdl], { idx = k, mesh = v })
            end
        end

        local function processQueue()
            progressData.downloadedMeshes = downloadCount
            progressData.downloadProgress = #meshes > 0 and (downloadCount / #meshes) or 1

            local activeRequests = http.getActiveRequests()
            local maxRequests = http.getMaximumRequests()

            while activeRequests < maxRequests and #requestQueue > 0 do
                local url = table.remove(requestQueue, 1)
                local meshList = urlToMeshes[url] or {}
                
                local meshFileName = url:match("[^/]+$") or "Mesh"
                meshFileName = meshFileName:gsub("%%20", " "):gsub("%.obj$", "")
                progressData.currentMeshName = meshFileName
                
                if debugPrints then 
                    print(W, "Downloading mesh file ", G, meshFileName, W, " for ", G, #meshList, W, " mesh entries") 
                end

                http.get(url, function(data)
                    if data[1] == "<" or data[1] == "{" then
                        failCount = failCount + #meshList
                        if debugPrints then print(true, W, "Failed to download mesh ", G, meshFileName, W, ": ", R, "Invalid format") end
                    else
                        for _, entry in ipairs(meshList) do
                            entry.mesh.meshData = data
                        end
                        downloadCount = downloadCount + #meshList
                        if debugPrints then print(W, "Successfully downloaded mesh ", G, meshFileName, W, ", size: ", G, math.round(#data / 1048576, 3) .. "mb") end
                    end
                    processQueue()
                end, function(err)
                    failCount = failCount + #meshList
                    if debugPrints then print(true, W, "HTTP error for mesh ", G, meshFileName, W, ": ", R, err) end
                    processQueue()
                end)
                activeRequests = activeRequests + 1
            end
        end

        timer.create("downloadMeshes", 0.1, 0, function()
            if #requestQueue == 0 and http.getActiveRequests() == 0 then
                timer.remove("downloadMeshes")
                if downloadCount > 0 then
                    if debugPrints then print(W, "All available meshes downloaded: ", G, downloadCount, "/", #meshes) end
                    progressData.downloading = false
                    buildMeshes(meshName, debugPrints, meshSpawnDelay, showMeshSpawnProgress, meshes, offsetPos, offsetAngle, globalScale, progressData, W, R, G)
                else
                    if debugPrints then print(true, W, "Failed to download any meshes!") end
                    progressData.downloading = false
                end
            else
                processQueue()
            end
        end)
    end

    local function initClient(config)
        -- Allow checking version
        local function checkVersionOnSpawn(cfg)
            if not cfg.AllowCheckVersion then return end
            local githubCoreURL = "https://raw.githubusercontent.com/nokillnando/Starfall/main/mesh_loaders_new/mesh_loader_core.txt"
            local githubConfigURL = "https://raw.githubusercontent.com/nokillnando/Starfall/main/mesh_loaders_new/mesh_loader_config.txt"
            local githubUIURL = "https://raw.githubusercontent.com/nokillnando/Starfall/main/mesh_loaders_new/mesh_loader_ui.txt"
            local localVersion = Version
            local localConfigVersion = cfg.ConfigVersion or "2.00"
            local maxRetries = 10
            local W, G, R = cfg.W or Color(255,255,255), cfg.G or Color(0,255,0), cfg.R or Color(255,0,0)

            local function doVersionCheck()
                if not http.canRequest() then
                    maxRetries = maxRetries - 1
                    if maxRetries <= 0 then return end
                    timer.simple(0.5, doVersionCheck)
                    return
                end

                -- Core check
                http.get(githubCoreURL, function(data)
                    if data and data ~= "" then
                        local versionStart = data:find('local Version = "')
                        if versionStart then
                            versionStart = versionStart + 17
                            local versionEnd = data:find('"', versionStart)
                            if versionEnd then
                                local remoteVersion = data:sub(versionStart, versionEnd - 1)
                                if tonumber(remoteVersion) and tonumber(localVersion) and tonumber(remoteVersion) > tonumber(localVersion) then
                                    print(W, "Mesh Loader Core Update available! Local: ", R, localVersion, W, " | Remote: ", G, remoteVersion)
                                else
                                    print(W, "Mesh Loader Core is up to date! Version: ", G, localVersion)
                                end
                            else
                                print(W, "Mesh Loader Core is up to date! (Remote version unreadable) Version: ", G, localVersion)
                            end
                        else
                            print(W, "Mesh Loader Core is up to date! (Remote version missing) Version: ", G, localVersion)
                        end
                    else
                        print(W, "Mesh Loader Core is up to date! (Download failed) Version: ", G, localVersion)
                    end
                end, function(err)
                    print(R, "Mesh Loader Core version check HTTP error: ", W, err)
                end)

                -- Config check (delayed to prevent http burst)
                timer.simple(0.5, function()
                    if not http.canRequest() then return end
                    http.get(githubConfigURL, function(data)
                        if data and data ~= "" then
                            local versionStart = data:find('local ConfigVersion = "')
                            if versionStart then
                                versionStart = versionStart + 23
                                local versionEnd = data:find('"', versionStart)

        -- Sanitize all URLs in meshes before any downloads
        local function sanitizeURL(url)
            if type(url) ~= "string" or url == "" then return url end
            
            -- GitHub: convert browser links to raw content links
            if url:find("github.com") and not url:find("raw.githubusercontent.com") then
                url = string.replace(url, "github.com", "raw.githubusercontent.com")
                url = string.replace(url, "/blob/", "/")
                url = string.replace(url, "/raw/", "/")
            end
            
            -- Dropbox: convert www/browser links to direct download links
            if url:find("dropbox.com") then
                url = string.replace(url, "www.dropbox.com", "dl.dropboxusercontent.com")
                url = string.replace(url, "?dl=0", "")
                url = string.replace(url, "&dl=0", "")
            end
            
            -- Google Drive: convert sharing links to direct download
            -- Handles: https://drive.google.com/file/d/FILE_ID/view?...
            if url:find("drive.google.com/file/d/") then
                local fileId = url:match("drive.google.com/file/d/([^/]+)")
                if fileId then
                    url = "https://drive.google.com/uc?export=download&id=" .. fileId
                end
            end
            -- Handles: https://drive.google.com/open?id=FILE_ID
            if url:find("drive.google.com/open%?id=") then
                local fileId = url:match("id=([^&]+)")
                if fileId then
                    url = "https://drive.google.com/uc?export=download&id=" .. fileId
                end
            end
            
            return url
        end
        
        local function sanitizeMeshURLs(meshes)
            for _, v in ipairs(meshes) do
                v.mdl = sanitizeURL(v.mdl)
                v.texture = sanitizeURL(v.texture)
                if v.BumpMapLink then v.BumpMapLink = sanitizeURL(v.BumpMapLink) end
                if v.EnvMapLink then v.EnvMapLink = sanitizeURL(v.EnvMapLink) end
                if v.PhongExponentTexture then v.PhongExponentTexture = sanitizeURL(v.PhongExponentTexture) end
                if v.DetailLink then v.DetailLink = sanitizeURL(v.DetailLink) end
                
                if v.TextureSwaps and type(v.TextureSwaps) == "table" then
                    local flattened = {}
                    local function extractSwaps(tbl)
                        for k, d in pairs(tbl) do
                            if k == "TextureSwaps" and type(d) == "table" then
                                extractSwaps(d)
                            elseif type(d) == "table" or type(d) == "string" then
                                flattened[k] = d
                            end
                        end
                    end
                    extractSwaps(v.TextureSwaps)
                    v.TextureSwaps = flattened

                    for swapName, swapData in pairs(v.TextureSwaps) do
                        if type(swapData) == "table" and swapData.texture then
                            swapData.texture = sanitizeURL(swapData.texture)
                        elseif type(swapData) == "string" then
                            v.TextureSwaps[swapName] = sanitizeURL(swapData)
                        end
                    end
                end
            end

                    -- Dynamic Material Decoupling / Isolation Helpers
                    local function createSubmeshMaterial(v)
                        local shaderName = v.Shader or "VertexLitGeneric"
                        local ok, mat = pcall(material.create, shaderName)
                        if not ok or not mat or not mat.setUndefined then return nil end

                        mat:setUndefined("$basetexture")
                        mat:setUndefined("$bumpmap")
                        mat:setUndefined("$envmapmask")
                        mat:setUndefined("$phongexponenttexture")
                        mat:setUndefined("$detail")

                        if v.BlendTintByBaseAlpha then
                            mat:setInt("$blendtintbybasealpha", 1)
                            if v.BlendTintColorOverBase then
                                mat:setFloat("$blendtintcoloroverbase", tonumber(v.BlendTintColorOverBase) or 0)
                            end
                        else
                            mat:setInt("$blendtintbybasealpha", 0)
                        end

                        if v.Mod2x ~= nil then
                            mat:setInt("$mod2x", (v.Mod2x == true or v.Mod2x == 1) and 1 or 0)
                        end

                        local combinedFlags = 0
                        if v.EnableHalfLambert then
                            combinedFlags = bit.bor(combinedFlags, 134217728)
                        end

                        if v.EnableBaseMapAlphaEnvMask then
                            combinedFlags = bit.bor(combinedFlags, 1048576)
                            mat:setInt("$basealphaenvmapmask", 1)
                        end

                        if v.EnableBaseMapAlphaPhongMask then
                            mat:setInt("$basemapalphaphongmask", 1)
                        else
                            mat:setInt("$basemapalphaphongmask", 0)
                        end

                        if v.EnableNormalmapAlphaEnvMapMask then
                            combinedFlags = bit.bor(combinedFlags, 4194304)
                            mat:setInt("$normalmapalphaenvmapmask", 1)
                        end

                        for _, flag in ipairs(v.Flags or {}) do
                            combinedFlags = bit.bor(combinedFlags, flag)
                        end
                        if combinedFlags > 0 then
                            mat:setInt("$flags", combinedFlags)
                        end

                        mat:setFloat("$envmapcontrast", tonumber(v.EnvMapContrast) or 0)
                        mat:setFloat("$envmapsaturation", tonumber(v.EnvMapSaturation) or 1)
                        if v.ActiveEnvMapTint then
                            local et = v.ActiveEnvMapTint
                            local eM = v.ActiveEnvMapMult or 1
                            mat:setVector("$envmaptint", Vector((et.r / 255) * eM, (et.g / 255) * eM, (et.b / 255) * eM))
                        elseif v.EnvMapShinyness then
                            local eM = v.ActiveEnvMapMult or 1
                            if type(v.EnvMapShinyness) == "Vector" then
                                mat:setVector("$envmaptint", v.EnvMapShinyness * eM)
                            else
                                mat:setVector("$envmaptint", Vector(0.2, 0.2, 0.2))
                            end
                        else
                            mat:setVector("$envmaptint", Vector(0.2, 0.2, 0.2))
                        end
                        mat:setFloat("$envmapfresnel", tonumber(v.EnvMapFresnel) or 1)

                        local useP = (v.UsePhong ~= nil and v.UsePhong) or (v.Phong ~= nil and v.Phong)
                        if useP == nil then useP = 1 end
                        mat:setInt("$phong", (useP == true or useP == 1) and 1 or 0)
                        mat:setInt("$invertphongmask", (v.InvertPhongMask == true or v.InvertPhongMask == 1) and 1 or 0)

                        if v.ActivePhongTint then
                            local pt = v.ActivePhongTint
                            local pM = v.ActivePhongMult or 1
                            mat:setVector("$phongtint", Vector((pt.r / 255) * pM, (pt.g / 255) * pM, (pt.b / 255) * pM))
                        elseif v.PhongTint and type(v.PhongTint) == "Vector" then
                            local pM = v.ActivePhongMult or 1
                            mat:setVector("$phongtint", v.PhongTint * pM)
                        end
                        mat:setFloat("$phongalbedotint", tonumber(v.PhongAlbedoTint) or 0)
                        mat:setFloat("$phongboost", tonumber(v.PhongBoost) or 1)

                        -- Detail Parameters
                        mat:setFloat("$detailblendfactor", v.DetailBlendFactor or 1)
                        mat:setInt("$detailblendmode", v.DetailBlendMode or 0)
                        mat:setInt("$detailframe", v.DetailFrame or 0)
                        mat:setFloat("$detailscale", v.DetailScale or 4)
                        mat:setMatrix("$detailtexturetransform", v.DetailTextureTransform or Matrix())
                        mat:setVector("$detailtint", v.DetailTint or Vector(1, 1, 1))
                        mat:setFloat("$phongexponent", tonumber(v.PhongExp) or 64)
                        if v.PhongFresnelRange and type(v.PhongFresnelRange) == "Vector" then
                            mat:setVector("$phongfresnelranges", v.PhongFresnelRange)
                        end
                        mat:setFloat("$phongalbedoboost", tonumber(v.PhongAlbedoBoost) or 0)

                        local useR = (v.UseRimLight ~= nil and v.UseRimLight) or (v.RimLight ~= nil and v.RimLight)
                        if useR == nil then useR = (v.RimLightBoost and v.RimLightBoost > 0) and 1 or 0 end
                        mat:setInt("$rimlight", (useR == true or useR == 1) and 1 or 0)
                        mat:setFloat("$rimlightexponent", tonumber(v.RimLightExp) or 4)
                        mat:setFloat("$rimlightboost", tonumber(v.RimLightBoost) or 0.25)

                        if v.Color2 then mat:setVector("$color2", v.Color2) end

                        -- Base Texture
                        if v.OriginalTexture then
                            mat:setTexture("$basetexture", v.OriginalTexture)
                        elseif v.SharedMaterial and v.SharedMaterial.getTexture and v.SharedMaterial:getTexture("$basetexture") then
                            mat:setTexture("$basetexture", v.SharedMaterial:getTexture("$basetexture"))
                        elseif v.texture and v.texture:sub(1, 4) == "http" then
                            applyUrlTexture(mat, "$basetexture", v.texture, function(tex)
                                if tex then v.OriginalTexture = tex end
                            end)
                        elseif v.texture and v.texture ~= "" then
                            mat:setTexture("$basetexture", v.texture)
                        elseif v.Material and v.Material ~= "" then
                            local okM, loadedM = pcall(material.load, v.Material)
                            if okM and loadedM and loadedM.getTexture and loadedM:getTexture("$basetexture") then
                                mat:setTexture("$basetexture", loadedM:getTexture("$basetexture"))
                            end
                        end

                        -- BumpMap
                        if v.BumpMapLink and v.BumpMapLink ~= "" then
                            if v.BumpMapLink:sub(1, 4) == "http" then
                                applyUrlTexture(mat, "$bumpmap", v.BumpMapLink)
                            else
                                mat:setTexture("$bumpmap", v.BumpMapLink)
                            end
                        end

                        -- EnvMapMask
                        if v.EnvMapLink and v.EnvMapLink ~= "" then
                            if v.EnvMapLink:sub(1, 4) == "http" then
                                applyUrlTexture(mat, "$envmapmask", v.EnvMapLink)
                            else
                                mat:setTexture("$envmapmask", v.EnvMapLink)
                            end
                        end

                        -- PhongExponentTexture
                        if v.PhongExponentTexture and v.PhongExponentTexture ~= "" then
                            if v.PhongExponentTexture:sub(1, 4) == "http" then
                                applyUrlTexture(mat, "$phongexponenttexture", v.PhongExponentTexture)
                            else
                                mat:setTexture("$phongexponenttexture", v.PhongExponentTexture)
                            end
                        end

                        -- Detail
                        if v.DetailLink and v.DetailLink ~= "" then
                            if v.DetailLink:sub(1, 4) == "http" then
                                applyUrlTexture(mat, "$detail", v.DetailLink)
                            else
                                mat:setTexture("$detail", v.DetailLink)
                            end
                        end

                        -- Matrix
                        local textureMatrix = Matrix()
                        textureMatrix:setScale(Vector(v.TextureScale or 1, v.TextureScale or 1, 1))
                        mat:setMatrix("$basetexturetransform", textureMatrix)

                        -- Cubemap Assignment
                        local cmName = v.CubeMapTexture or (activeConfig and activeConfig.CubeMapTexture) or "models/shiny"
                        local okMat, Nmat = pcall(material.load, cmName)
                        local newEnvMap = okMat and Nmat and Nmat.getTexture and Nmat:getTexture("$envmap")
                        if newEnvMap then
                            cachedResolvedCubemap = newEnvMap
                            mat:setTexture("$envmap", newEnvMap)
                        elseif cachedResolvedCubemap then
                            mat:setTexture("$envmap", cachedResolvedCubemap)
                        end

                        return mat
                    end

                    decoupleSubmeshMaterial = function(meshIdx)
                        local v = activeConfig and activeConfig.meshes and activeConfig.meshes[meshIdx]
                        if not v then return end
                        if v.isIsolatedMaterial and v.isolatedMat then return end
                        
                        if not v.SharedMaterial then
                            v.SharedMaterial = v.mat
                        end
                        
                        local newMat = createSubmeshMaterial(v)
                        if not newMat then
                            print(R, "[MeshLoader] Error: Failed to create isolated material instance for submesh #" .. meshIdx .. " (Quota exceeded or invalid shader)")
                            return
                        end
                        
                        v.isolatedMat = newMat
                        v.isIsolatedMaterial = true
                        v.mat = newMat
                        
                        -- Update holograms immediately
                        for _, holo in ipairs(spawnedHolograms or {}) do
                            if isValid(holo) and holo.MeshConfig == v then
                                holo:setMeshMaterial(newMat)
                                if v.TextureSwaps then
                                    applyHoloSwaps(holo)
                                end
                            end
                        end
                        
                        -- Apply shader overrides on the new material instance
                        applyMeshMaterialOverrides(v)

                        -- Force update cubemaps across all materials
                        if globalSetCubeMap then
                            globalSetCubeMap(true)
                        end
                        pcall(timer.simple, 0.1, function()
                            if globalSetCubeMap then globalSetCubeMap(true) end
                        end)
                        pcall(timer.simple, 1, function()
                            if globalSetCubeMap then globalSetCubeMap(true) end
                        end)

                        print(G, "[MeshLoader] Decoupled material instance for submesh #" .. meshIdx .. " (1 unique slot allocated)")
                    end

                    relinkSubmeshMaterial = function(meshIdx)
                        local v = activeConfig and activeConfig.meshes and activeConfig.meshes[meshIdx]
                        if not v then return end
                        if not v.isIsolatedMaterial or not v.isolatedMat then return end
                        
                        local oldMat = v.isolatedMat
                        if oldMat and oldMat.destroy then
                            pcall(function() oldMat:destroy() end)
                        end
                        
                        v.mat = v.SharedMaterial or v.mat
                        v.isolatedMat = nil
                        v.isIsolatedMaterial = false
                        
                        for _, holo in ipairs(spawnedHolograms or {}) do
                            if isValid(holo) and holo.MeshConfig == v then
                                holo:setMeshMaterial(v.mat)
                                if v.TextureSwaps then
                                    applyHoloSwaps(holo)
                                end
                            end
                        end
                        
                        applyMeshMaterialOverrides(v)
                        if forceUpdateAllCubemaps then pcall(forceUpdateAllCubemaps) end
                        print(G, "[MeshLoader] Re-linked submesh #" .. meshIdx .. " to shared vehicle material (Slot freed)")

    local function getVehicleEntities(force)
        local now = timer.curtime()
        if not force and cachedVehicleSet and (now - lastVehicleScanTime < 1.5) then
            return cachedVehicleSet, cachedVehicleList, cachedVehicleId
        end
        lastVehicleScanTime = now

        local currentBase = wire.ports.Base
        if not isValid(currentBase) then
            currentBase = chip()
        end
        if not isValid(currentBase) then return {}, {}, 0 end

        local root = getRootEntity(currentBase) or currentBase
        local vehicleSet = {}
        local entityList = {}

        local function addEnt(ent)
            if not ent then return end
            if type(ent) == "number" and ent > 0 then
                local e = (entity and entity(ent)) or (find and find.byIndex and find.byIndex(ent))
                if isValid(e) then ent = e end
            elseif type(ent) == "table" and ent.entIndex then
                local e = (entity and entity(ent:entIndex()))
                if isValid(e) then ent = e end
            end
            if not isValid(ent) then return end
            if not vehicleSet[ent] then
                vehicleSet[ent] = true
                table.insert(entityList, ent)
            end
        end

        addEnt(currentBase)
        addEnt(root)
        addEnt(chip())

        -- Add tracked parent entities from meshes
        for _, parentName in ipairs(trackedParentEntities or {}) do
            addEnt(wire.ports[parentName])
        end

        -- Add Wire Chips array input
        if wire.ports.Chips then
            local chipsIn = wire.ports.Chips
            if type(chipsIn) == "table" then
                for _, cEnt in pairs(chipsIn) do
                    addEnt(cEnt)
                end
            else
                addEnt(chipsIn)
            end
        end

        -- Fast BFS constraint tree traversal (pattern from tire_simulation_core)
        local queue = {}
        for _, ent in ipairs(entityList) do
            table.insert(queue, ent)
        end

        while #queue > 0 do
            local current = table.remove(queue, 1)
            if isValid(current) then
                local parent = current:getParent()
                if isValid(parent) and not vehicleSet[parent] then
                    addEnt(parent)
                    table.insert(queue, parent)
                end
                
                if constraint and constraint.getTable then
                    local constraints = constraint.getTable(current)
                    if constraints then
                        for i = 1, #constraints do
                            local c = constraints[i]
                            local other = (c.Ent1 == current) and c.Ent2 or c.Ent1
                            if isValid(other) and not vehicleSet[other] then
                                addEnt(other)
                                table.insert(queue, other)
                            end
                        end
                    end
                elseif current.getAllConstrained then
                    local constrained = current:getAllConstrained()
                    if type(constrained) == "table" then
                        for _, ent in ipairs(constrained) do
                            if isValid(ent) and not vehicleSet[ent] then
                                addEnt(ent)
                                table.insert(queue, ent)
                            end
                        end
                    end
                end
            end
        end

    local function parseColorArgs(text, cmd)
        local rest = text:sub(#cmd + 1)
        
        -- Split by "|"
        local segments = {}
        local start = 1
        while true do
            local pos = rest:find("|", start)
            if not pos then
                table.insert(segments, rest:sub(start))
                break
            end
            table.insert(segments, rest:sub(start, pos - 1))
            start = pos + 1
        end
        
        -- Helper to parse a color from a string segment
        local function parseColor(str)
            local clean = str:gsub(",", " ")
            local parts = {}
            for word in clean:gmatch("%S+") do
                table.insert(parts, word)
            end
            
            local r, g, b, a
            if #parts >= 3 then
                r = tonumber(parts[1])
                g = tonumber(parts[2])
                b = tonumber(parts[3])
                a = tonumber(parts[4])
            elseif #parts == 1 then
                local hex = parts[1]:gsub("#", "")
                if #hex == 6 or #hex == 8 then
                    r = tonumber(hex:sub(1, 2), 16)
                    g = tonumber(hex:sub(3, 4), 16)
                    b = tonumber(hex:sub(5, 6), 16)
                    if #hex == 8 then
                        a = tonumber(hex:sub(7, 8), 16)
                    end
                end
            end
            
            if r and g and b then
                local maxVal = math.max(r, g, b)
                if maxVal > 0 and maxVal <= 1 then
                    r = r * 255
                    g = g * 255
                    b = b * 255
                    if a and a <= 1 then
                        a = a * 255
                    end
                end
                return Color(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255), a and math.clamp(a, 0, 255) or 255)
            end
            return nil
        end
        
        -- Helper to parse a number (percentage or multiplier) from a string segment
        local function parsePercent(str)
            local clean = str:gsub("%%", ""):match("%S+")
            if clean then
                local num = tonumber(clean)
                if num then
                    if str:find("%%") or num > 2 then
                        return num / 100
                    else
                        return num
                    end
                end
            end
            return nil
        end
        
        local baseColor = parseColor(segments[1] or "")
        if not baseColor then return nil end
        
        local parsedColors = {}
        local parsedMults = {}
        
        for i = 2, #segments do
            local seg = segments[i]
            local col = parseColor(seg)
            if col then
                table.insert(parsedColors, col)
            else
                local pct = parsePercent(seg)
                if pct then
                    table.insert(parsedMults, pct)
                end
            end
        end
        
        -- Resolve tints
        local phongTint = nil
        local envMapTint = nil
        if #parsedColors == 1 then
            phongTint = parsedColors[1]
            envMapTint = parsedColors[1]
        elseif #parsedColors >= 2 then
            phongTint = parsedColors[1]
            envMapTint = parsedColors[2]
        end
        
        -- Resolve multipliers
        local phongMult = 1
        local envMapMult = 1
        if #parsedMults == 1 then
            phongMult = parsedMults[1]
            envMapMult = parsedMults[1]
        elseif #parsedMults >= 2 then
            phongMult = parsedMults[1]
            envMapMult = parsedMults[2]
        end
        
        return baseColor, phongTint, envMapTint, phongMult, envMapMult
    end

                        table.insert(trackedParentEntities, wireInputName)

                end

                            table.insert(inputNames, inputName)
                            table.insert(inputTypes, "normal")
                            usedInputs[inputName] = true
                            table.insert(trackedNormalInputs, inputName)

        hook.add("PlayerSay", "MeshLoaderColorCommands", function(ply, text, team)
            -- Trim leading whitespace using plain Lua bytes check
            local start_idx = 1
            while start_idx <= #text and string.byte(text, start_idx) <= 32 do
                start_idx = start_idx + 1
            end
            if start_idx > #text then return end
            
            -- Extract first word without backtracking regex
            local space_idx = text:find("%s", start_idx)
            local cmd = space_idx and text:sub(start_idx, space_idx - 1) or text:sub(start_idx)
            local cmdLower = cmd:lower()
            
            if cmdLower == "!part" or cmdLower == "/part" then
                if ply ~= owner() then return end
                if isLookingAtVehicle(ply) then
                    local rest = text:sub(start_idx + #cmd):match("^%s*(.-)%s*$")
                    local pName, vNum = rest:match("(%S+)%s+(%S+)")
                    if pName and vNum then
                        local pLower = pName:lower()
                        local varInt = tonumber(vNum)
                        if not varInt then
                            local s = vNum:lower()
                            if s == "0" or s == "none" or s == "off" or s == "remove" then varInt = 0
                            else varInt = 1 end
                        end
                        serverStoredParts[pLower] = varInt
                        net.start("syncMeshPart")
                        net.writeString(pName)
                        net.writeInt(varInt, 8)
                        net.send(find.allPlayers())
                        return ""
                    end
                end
            elseif cmdLower == "!swap" or cmdLower == "/swap" then
                if ply ~= owner() then return end
                if isLookingAtVehicle(ply) then
                    local rest = text:sub(start_idx + #cmd):match("^%s*(.-)%s*$")
                    local sName, sVal = rest:match("(%S+)%s*(%S*)")
                    if sName and sName ~= "" then
                        local foundKey = nil
                        for swKey in pairs(serverStoredSwaps or {}) do
                            if swKey:lower() == sName:lower() then foundKey = swKey break end
                        end
                        if not foundKey and config and config.meshes then
                            for _, mData in ipairs(config.meshes) do
                                if mData.TextureSwaps then
                                    for swKey, swDef in pairs(mData.TextureSwaps) do
                                        local inName = (type(swDef) == "table" and swDef.input) or swKey
                                        if inName:lower() == sName:lower() or swKey:lower() == sName:lower() then
                                            foundKey = inName
                                            break
                                        end
                                    end
                                end
                                if foundKey then break end
                            end
                        end
                        if foundKey then
                            local cur = serverStoredSwaps[foundKey] or 0
                            local val = tonumber(sVal)
                            if not val then
                                local s = sVal:lower()
                                if s == "on" or s == "1" or s == "true" then val = 1
                                elseif s == "off" or s == "0" or s == "false" then val = 0
                                else val = (cur > 0) and 0 or 1 end
                            end
                            serverStoredSwaps[foundKey] = val
                            net.start("sendMeshSwap")
                            net.writeString(foundKey)
                            net.writeFloat(val)
                            net.send(find.allPlayers())
                            return ""
                        end
                    end
                end
            elseif registeredCommands[cmdLower] then
                if ply ~= owner() then return end
                local actualCmd = registeredCommands[cmdLower]
                if isLookingAtVehicle(ply) then
                    local cleanText = text:sub(start_idx)
                    local baseColor, phongTint, envMapTint, phongMult, envMapMult = parseColorArgs(cleanText, actualCmd)
                    if baseColor then
                        local prevSt = serverStoredColors[cmdLower] or {}
                        local pBoost = prevSt.phongBoost or 1
                        local pExp = prevSt.phongExp or 64
                        local rBoost = prevSt.rimLightBoost or 0.25
                        local rExp = prevSt.rimLightExp or 4

                            phongMult = phongMult,
                            envMapMult = envMapMult,
                            phongBoost = pBoost,
                            phongExp = pExp,
                            rimLightBoost = rBoost,
                            rimLightExp = rExp
                        }
                        
                        -- Broadcast update to everyone
                        net.start("syncMeshColor")
                        net.writeString(actualCmd)
                        net.writeUInt(baseColor.r, 8)
                        net.writeUInt(baseColor.g, 8)
                        net.writeUInt(baseColor.b, 8)
                        net.writeUInt(baseColor.a, 8)
                        
                        if phongTint then
                            net.writeBool(true)
                            net.writeUInt(phongTint.r, 8)
                            net.writeUInt(phongTint.g, 8)
                            net.writeUInt(phongTint.b, 8)
                        else
                            net.writeBool(false)
                        end
                        
                        if envMapTint then
                            net.writeBool(true)
                            net.writeUInt(envMapTint.r, 8)
                            net.writeUInt(envMapTint.g, 8)
                            net.writeUInt(envMapTint.b, 8)
                        else
                            net.writeBool(false)
                        end
                        
                        net.writeFloat(phongMult)
                        net.writeFloat(envMapMult)
                        net.writeFloat(pBoost)
                        net.writeUInt(math.clamp(math.floor(pExp), 1, 2048), 12)
                        net.writeFloat(rBoost)
                        net.writeUInt(math.clamp(math.floor(rExp), 1, 2048), 12)
                        net.send(find.allPlayers())
                        
                        return ""

    hook.add("tick", "monitorBase", function()
        local now = timer.curtime()
        if now < nextMonitor then return end
        
        nextMonitor = now + serverUpdateRate

        if net.getBytesLeft() < 64 then return end

        local currentBase = wire.ports.Base
        local currentChips = wire.ports.Chips
        
        local chipsChanged = false
        if type(currentChips) ~= type(lastChips) then
            chipsChanged = true
        elseif type(currentChips) == "table" then
            if #currentChips ~= (lastChipsCount or 0) then
                chipsChanged = true
            else
                for i = 1, #currentChips do
                    if currentChips[i] ~= lastChips[i] then
                        chipsChanged = true
                        break
                    end
                end
            end
        elseif currentChips ~= lastChips then
            chipsChanged = true
        end

        if currentBase ~= lastBase or chipsChanged then
            lastBase = currentBase
            if type(currentChips) == "table" then
                lastChips = {}
                lastChipsCount = #currentChips
                for i, v in ipairs(currentChips) do lastChips[i] = v end
            else
                lastChips = currentChips
                lastChipsCount = 0
            end
            net.start("sendBase")
            net.writeEntity(isValid(currentBase) and currentBase or chip())
            net.send(find.allPlayers())
            sendVehicleConstrainedEntities(nil, true)
            syncModuleInfo()
        end
        
        for _, parentName in ipairs(trackedParentEntities) do
            if net.getBytesLeft() < 64 then break end
            local currentEnt = wire.ports[parentName]
            if currentEnt ~= lastParentEntities[parentName] then
                lastParentEntities[parentName] = currentEnt
                
                local entToSend = isValid(currentEnt) and currentEnt or chip()
                net.start("sendMeshParent")
                net.writeString(parentName)
                net.writeEntity(entToSend)
                net.writeUInt(entToSend:entIndex(), 16)
                net.send(find.allPlayers())
            end
        end
        
        for _, inputName in ipairs(trackedNormalInputs) do
            if net.getBytesLeft() < 64 then break end
            local currentVal = wire.ports[inputName] or 0
            local lastVal = lastNormalInputs[inputName] or 0
            
            local shouldUpdate = false
            if currentVal ~= lastVal then
                local currentIsInt = math.abs(currentVal - math.floor(currentVal + 0.5)) < 0.0001
                local lastIsInt = math.abs(lastVal - math.floor(lastVal + 0.5)) < 0.0001
                
                if currentIsInt or lastIsInt then
                    shouldUpdate = true
                elseif math.abs(currentVal - lastVal) >= 0.01 then
                    shouldUpdate = true
                end
            end
            
            if shouldUpdate then
                lastNormalInputs[inputName] = currentVal
                
                if inputName:sub(1, 5) == "Part_" then
                    local pName = inputName:sub(6)
                    local vInt = math.floor(currentVal + 0.5)
                    serverStoredParts[pName:lower()] = vInt
                    net.start("syncMeshPart")
                    net.writeString(pName)
                    net.writeInt(vInt, 8)
                    net.send(find.allPlayers())
                else
                    net.start("sendMeshSwap")
                    net.writeString(inputName)
                    net.writeFloat(currentVal)
                    net.send(find.allPlayers())
                end
            end
        end

        local lastVehiclePeriodicSync = lastVehiclePeriodicSync or 0
        if (now - lastVehiclePeriodicSync >= 2.5) then
            lastVehiclePeriodicSync = now
            sendVehicleConstrainedEntities(nil, false)
        end
    end)

    _G.MeshLoaderCore = {
        mirror = mirror,
        initServer = initServer,
        Version = Version
    }
end
hook.add("removed", "stopStreams", function()
    if CLIENT then
        if isValid(chip()) and isValid(chip().ProgressHolo) then
            chip().ProgressHolo:remove()
        end
        if isValid(chip()) and chip().ProgressRTMat and chip().ProgressRTMat.destroy then
            pcall(function() chip().ProgressRTMat:destroy() end)
            chip().ProgressRTMat = nil
        end
        if isValid(chip()) then
            local cId = chip():entIndex()
            pcall(function()
                if file.exists("mesh_presets/module_" .. cId .. ".txt") then
                    file.delete("mesh_presets/module_" .. cId .. ".txt")
                end
            end)
            if _G.MeshLoaderRegistry then
                _G.MeshLoaderRegistry[cId] = nil
            end
        end
        if activeConfig and activeConfig.meshes then
            for _, m in ipairs(activeConfig.meshes) do
                if m.isolatedMat and m.isolatedMat.destroy then
                    pcall(function() m.isolatedMat:destroy() end)
                    m.isolatedMat = nil
                end
            end
        end
        if materialCache then
            for _, entry in pairs(materialCache) do
                if entry and entry.mat and entry.mat.destroy then
                    pcall(function() entry.mat:destroy() end)
                end
            end
            materialCache = {}
        end
    end
    if net.isStreaming() then
        net.cancelStream()
    end
end)
